    83→    83→/** manual word-wrap; returns lines */
    84→    84→export function wrapText(ctx: SKRSContext2D, text: string, maxWidth: number, font: string): string[] {
    85→    85→  ctx.font = font;
    86→    86→  const words = text.split(/\s+/).filter((w) => w.length > 0);
    87→    87→  const lines: string[] = [];
    88→    88→  let cur = "";
    89→    89→  for (const wd of words) {
    90→    90→    const test = cur ? `${cur} ${wd}` : wd;
    91→    91→    if (ctx.measureText(test).width <= maxWidth || !cur) {
    92→    92→      cur = test;
    93→    93→    } else {
    94→    94→      lines.push(cur);
    95→    95→      cur = wd;
    96→    96→    }
    97→    97→  }
    98→    98→  if (cur) lines.push(cur);
    99→    99→  return lines.length ? lines : [""];
   100→   100→}

   102→   102→/** draw wrapped text; returns the y coordinate BELOW the last line */
   103→   103→export function fillWrapped(
   104→   104→  ctx: SKRSContext2D,
   105→   105→  text: string,
   106→   106→  x: number,
   107→   107→  y: number,
   108→   108→  maxWidth: number,
   109→   109→  font: string,
   110→   110→  color: string,
   111→   111→  opts?: { align?: CanvasTextAlign; lineHeight?: number; maxLines?: number },
   112→   112→): number {
   113→   113→  const lh = opts?.lineHeight ?? 30;
   114→   114→  ctx.font = font;
   115→   115→  let lines = wrapText(ctx, text, maxWidth, font);
   116→   116→  if (opts?.maxLines && lines.length > opts.maxLines) {
   117→   117→    lines = lines.slice(0, opts.maxLines);
   118→   118→    lines[lines.length - 1] = `${lines[lines.length - 1].replace(/[.,]$/, "")}...`;
   119→   119→  }
   120→   120→  ctx.fillStyle = color;
   121→   121→  ctx.textAlign = opts?.align ?? "left";
   122→   122→  ctx.textBaseline = "alphabetic";
   123→   123→  let yy = y;
   124→   124→  for (const line of lines) {
   125→   125→    ctx.fillText(line, x, yy);
   126→   126→    yy += lh;
   127→   127→  }
   128→   128→  ctx.textAlign = "left";
   129→   129→  return yy - lh;
   130→   130→}

   159→   159→    ctx.fillRect(x + w - i - thick, y + i + 4, thick, h - 2 * i - 8);
   160→   160→  }

   163→   163→/** pixel segmented progress bar (0..1) */
   164→   164→export function segmentedBar(
   165→   165→  ctx: SKRSContext2D,
   166→   166→  x: number,
   167→   167→  y: number,
   168→   168→  w: number,
   169→   169→  h: number,
   170→   170→  frac: number,
   171→   171→  color: string,
   172→   172→): void {
   173→   173→  const period = 16;
   174→   174→  const nSeg = Math.max(1, Math.floor((w + 4) / period));
   175→   175→  const filled = Math.round(nSeg * Math.min(1, Math.max(0, frac)));
   176→   176→  ctx.fillStyle = "rgba(255,255,255,0.07)";
   177→   177→  ctx.fillRect(x, y, w, h);
   178→   178→  ctx.fillStyle = "rgba(0,0,0,0.45)";
   179→   179→  ctx.fillRect(x, y + h, w, 3);
   180→   180→  for (let i = 0; i < nSeg; i++) {
   181→   181→    ctx.fillStyle = i < filled ? color : "rgba(255,255,255,0.10)";
   182→   182→    ctx.fillRect(x + i * period, y, period - 4, h);
   183→   183→    if (i < filled) {
   184→   184→      ctx.fillStyle = "rgba(255,255,255,0.35)";
   185→   185→      ctx.fillRect(x + i * period, y, period - 4, 3);
   186→   186→    }
   187→   187→  }
   188→   188→}

   192→   192→interface ChipOpts {
   193→   193→  font: string;
   194→   194→  fg: string;
   195→   195→  bg?: string;
   196→   196→  border?: string;
   197→   197→  padX?: number;

   200→   200→
   201→   201→class Painter {
   202→   202→  ctx: SKRSContext2D;
   203→   203→  w: number;
   204→   204→  h: number;
   205→   205→  rng: Rng;
   206→   206→  latin = 0;
   207→   207→
   208→   208→  constructor(ctx: SKRSContext2D, w: number, h: number, rng: Rng) {
   209→   209→    this.ctx = ctx;
   210→   210→    this.w = w;
   211→   211→    this.h = h;
   212→   212→    this.rng = rng;
   213→   213→  }
   214→   214→
   215→   215→  /** latinize + count changes */
   216→   216→  l(s: string): string {
   217→   217→    const r = latinizeCount(s);
   218→   218→    this.latin += r.changes;
   219→   219→    return r.text;
   220→   220→  }
   221→   221→
   222→   222→  measure(s: string, font: string): number {
   223→   223→    this.ctx.font = font;
   224→   224→    return this.ctx.measureText(latinize(s)).width;
   225→   225→  }
   226→   226→
   227→   227→  text(
   228→   228→    s: string,
   229→   229→    x: number,
   230→   230→    y: number,
   231→   231→    font: string,
   232→   232→    color: string,
   233→   233→    align: CanvasTextAlign = "left",
   234→   234→    baseline: CanvasTextBaseline = "alphabetic",
   235→   235→  ): void {
   236→   236→    const t = this.l(s);
   237→   237→    this.ctx.font = font;
   238→   238→    this.ctx.fillStyle = color;
   239→   239→    this.ctx.textAlign = align;
   240→   240→    this.ctx.textBaseline = baseline;
   241→   241→    this.ctx.fillText(t, x, y);
   242→   242→    this.ctx.textAlign = "left";
   243→   243→    this.ctx.textBaseline = "alphabetic";
   244→   244→  }
   245→   245→
   246→   246→  /** letter-spaced text (arcade title style); counted once */
   247→   247→  textSpaced(
   248→   248→    s: string,
   249→   249→    x: number,
   250→   250→    y: number,
   251→   251→    font: string,
   252→   252→    color: string,
   253→   253→    spacing: number,
   254→   254→    align: "left" | "center" | "right" = "left",
   255→   255→  ): void {
   256→   256→    const t = this.l(s);
   257→   257→    this.ctx.font = font;
   258→   258→    let total = 0;
   259→   259→    for (const ch of t) total += this.ctx.measureText(ch).width + spacing;
   260→   260→    total -= spacing;
   261→   261→    let cx = x;
   262→   262→    if (align === "center") cx = x - total / 2;
   263→   263→    if (align === "right") cx = x - total;
   264→   264→    this.ctx.fillStyle = color;
   265→   265→    this.ctx.textAlign = "left";
   266→   266→    this.ctx.textBaseline = "alphabetic";
   267→   267→    for (const ch of t) {
   268→   268→      this.ctx.fillText(ch, cx, y);
   269→   269→      cx += this.ctx.measureText(ch).width + spacing;
   270→   270→    }
   271→   271→  }
   272→   272→
   273→   273→  /** wrapped text with latin accounting; returns bottom y of last line */
   274→   274→  wrapped(
   275→   275→    s: string,
   276→   276→    x: number,
   277→   277→    y: number,
   278→   278→    maxW: number,
   279→   279→    font: string,
   280→   280→    color: string,
   281→   281→    lh = 30,
   282→   282→    maxLines?: number,
   283→   283→    align: CanvasTextAlign = "left",
   284→   284→  ): number {
   285→   285→    const t = this.l(s);
   286→   286→    return fillWrapped(this.ctx, t, x, y, maxW, font, color, {
   287→   287→      lineHeight: lh,
   288→   288→      maxLines,
   289→   289→      align,
   290→   290→    });
   291→   291→  }
   292→   292→
   293→   293→  /** small labeled chip; returns total width */
   294→   294→  chip(text: string, x: number, y: number, o: ChipOpts): number {
   295→   295→    const t = this.l(text);
   296→   296→    this.ctx.font = o.font;
   297→   297→    const tw = this.ctx.measureText(t).width;
   298→   298→    const padX = o.padX ?? 12;
   299→   299→    const h = o.h ?? 36;
   300→   300→    const w = Math.ceil(tw) + padX * 2;
   301→   301→    if (o.bg) {
   302→   302→      this.ctx.fillStyle = o.bg;
   303→   303→      this.ctx.fillRect(x, y, w, h);
   304→   304→    }
   305→   305→    if (o.border) {
   306→   306→      this.ctx.fillStyle = o.border;
   307→   307→      this.ctx.fillRect(x, y, w, 2);
   308→   308→      this.ctx.fillRect(x, y + h - 2, w, 2);
   309→   309→      this.ctx.fillRect(x, y, 2, h);
   310→   310→      this.ctx.fillRect(x + w - 2, y, 2, h);
   311→   311→    }
   312→   312→    this.ctx.fillStyle = o.fg;
   313→   313→    this.ctx.textAlign = "left";
   314→   314→    this.ctx.textBaseline = "middle";
   315→   315→    this.ctx.fillText(t, x + padX, y + h / 2 + 1);
   316→   316→    this.ctx.textBaseline = "alphabetic";
   317→   317→    return w;
   318→   318→  }
   319→   319→
   320→   320→  /** centered chip; returns width */
   321→   321→  chipCentered(text: string, cx: number, y: number, o: ChipOpts): number {
   322→   322→    this.ctx.font = o.font;
   323→   323→    const tw = this.ctx.measureText(this.l(text)).width;
   324→   324→    const w = Math.ceil(tw) + (o.padX ?? 12) * 2;
   325→   325→    this.chip(text, Math.round(cx - w / 2), y, o);
   326→   326→    return w;
   327→   327→  }
   328→   328→
   329→   329→  /** right-aligned chip; x is the RIGHT edge; returns width */
   330→   330→  chipRight(text: string, right: number, y: number, o: ChipOpts): number {
   331→   331→    this.ctx.font = o.font;
   332→   332→    const tw = this.ctx.measureText(this.l(text)).width;
   333→   333→    const w = Math.ceil(tw) + (o.padX ?? 12) * 2;
   334→   334→    this.chip(text, right - w, y, o);
   335→   335→    return w;
   336→   336→  }
   337→   337→
   338→   338→  panel(x: number, y: number, w: number, h: number, fill: string = PANEL): void {
   339→   339→    this.ctx.fillStyle = fill;
   340→   340→    this.ctx.fillRect(x, y, w, h);
   341→   341→  }
   342→   342→
   343→   343→  /** truncate with ellipsis to fit width */
   344→   344→  fit(s: string, maxW: number, font: string): string {
   345→   345→    let t = this.l(s);
   346→   346→    this.ctx.font = font;
   347→   347→    if (this.ctx.measureText(t).width <= maxW) return t;
   348→   348→    while (t.length > 1 && this.ctx.measureText(`${t}...`).width > maxW) {
   349→   349→      t = t.slice(0, -1);
   350→   350→    }
   351→   351→    return `${t.trimEnd()}...`;
   352→   352→  }
   353→   353→}
