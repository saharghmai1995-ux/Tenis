    83→/** manual word-wrap; returns lines */
    84→export function wrapText(ctx: SKRSContext2D, text: string, maxWidth: number, font: string): string[] {
    85→  ctx.font = font;
    86→  const words = text.split(/\s+/).filter((w) => w.length > 0);
    87→  const lines: string[] = [];
    88→  let cur = "";
    89→  for (const wd of words) {
    90→    const test = cur ? `${cur} ${wd}` : wd;
    91→    if (ctx.measureText(test).width <= maxWidth || !cur) {
    92→      cur = test;
    93→    } else {
    94→      lines.push(cur);
    95→      cur = wd;
    96→    }
    97→  }
    98→  if (cur) lines.push(cur);
    99→  return lines.length ? lines : [""];
   100→}

   102→/** draw wrapped text; returns the y coordinate BELOW the last line */
   103→export function fillWrapped(
   104→  ctx: SKRSContext2D,
   105→  text: string,
   106→  x: number,
   107→  y: number,
   108→  maxWidth: number,
   109→  font: string,
   110→  color: string,
   111→  opts?: { align?: CanvasTextAlign; lineHeight?: number; maxLines?: number },
   112→): number {
   113→  const lh = opts?.lineHeight ?? 30;
   114→  ctx.font = font;
   115→  let lines = wrapText(ctx, text, maxWidth, font);
   116→  if (opts?.maxLines && lines.length > opts.maxLines) {
   117→    lines = lines.slice(0, opts.maxLines);
   118→    lines[lines.length - 1] = `${lines[lines.length - 1].replace(/[.,]$/, "")}...`;
   119→  }
   120→  ctx.fillStyle = color;
   121→  ctx.textAlign = opts?.align ?? "left";
   122→  ctx.textBaseline = "alphabetic";
   123→  let yy = y;
   124→  for (const line of lines) {
   125→    ctx.fillText(line, x, yy);
   126→    yy += lh;
   127→  }
   128→  ctx.textAlign = "left";
   129→  return yy - lh;
   130→}

   132→/** classic 16-bit dialog border: double line with corner notches */
   133→export function drawPixelBorder(
   134→  ctx: SKRSContext2D,
   135→  x: number,
   136→  y: number,
   137→  w: number,
   138→  h: number,
   139→  color: string,
   140→  thick = 2,
   141→): void {
   142→  ctx.fillStyle = color;
   143→  // outer line, notched corners
   144→  ctx.fillRect(x + 4, y, w - 8, thick);
   145→  ctx.fillRect(x + 4, y + h - thick, w - 8, thick);
   146→  ctx.fillRect(x, y + 4, thick, h - 8);
   147→  ctx.fillRect(x + w - thick, y + 4, thick, h - 8);
   148→  // corner step pixels
   149→  ctx.fillRect(x + 2, y + 2, thick, thick);
   150→  ctx.fillRect(x + w - 2 - thick, y + 2, thick, thick);
   151→  ctx.fillRect(x + 2, y + h - 2 - thick, thick, thick);
   152→  ctx.fillRect(x + w - 2 - thick, y + h - 2 - thick, thick, thick);
   153→  // inner line
   154→  const i = 7;
   155→  if (w > 2 * i + 10 && h > 2 * i + 10) {
   156→    ctx.fillRect(x + i + 4, y + i, w - 2 * i - 8, thick);
   157→    ctx.fillRect(x + i + 4, y + h - i - thick, w - 2 * i - 8, thick);
   158→    ctx.fillRect(x + i, y + i + 4, thick, h - 2 * i - 8);
   159→    ctx.fillRect(x + w - i - thick, y + i + 4, thick, h - 2 * i - 8);
   160→  }
   161→}

   163→/** pixel segmented progress bar (0..1) */
   164→export function segmentedBar(
   165→  ctx: SKRSContext2D,
   166→  x: number,
   167→  y: number,
   168→  w: number,
   169→  h: number,
   170→  frac: number,
   171→  color: string,
   172→): void {
   173→  const period = 16;
   174→  const nSeg = Math.max(1, Math.floor((w + 4) / period));
   175→  const filled = Math.round(nSeg * Math.min(1, Math.max(0, frac)));
   176→  ctx.fillStyle = "rgba(255,255,255,0.07)";
   177→  ctx.fillRect(x, y, w, h);
   178→  ctx.fillStyle = "rgba(0,0,0,0.45)";
   179→  ctx.fillRect(x, y + h, w, 3);
   180→  for (let i = 0; i < nSeg; i++) {
   181→    ctx.fillStyle = i < filled ? color : "rgba(255,255,255,0.10)";
   182→    ctx.fillRect(x + i * period, y, period - 4, h);
   183→    if (i < filled) {
   184→      ctx.fillStyle = "rgba(255,255,255,0.35)";
   185→      ctx.fillRect(x + i * period, y, period - 4, 3);
   186→    }
   187→  }
   188→}

   192→interface ChipOpts {
   193→  font: string;
   194→  fg: string;
   195→  bg?: string;
   196→  border?: string;
   197→  padX?: number;
   198→  h?: number;
   199→}
   200→
   201→class Painter {
   202→  ctx: SKRSContext2D;
   203→  w: number;
   204→  h: number;
   205→  rng: Rng;
   206→  latin = 0;
   207→
   208→  constructor(ctx: SKRSContext2D, w: number, h: number, rng: Rng) {
   209→    this.ctx = ctx;
   210→    this.w = w;
   211→    this.h = h;
   212→    this.rng = rng;
   213→  }
   214→
   215→  /** latinize + count changes */
   216→  l(s: string): string {
   217→    const r = latinizeCount(s);
   218→    this.latin += r.changes;
   219→    return r.text;
   220→  }
   221→
   222→  measure(s: string, font: string): number {
   223→    this.ctx.font = font;
   224→    return this.ctx.measureText(latinize(s)).width;
   225→  }
   226→
   227→  text(
   228→    s: string,
   229→    x: number,
   230→    y: number,
   231→    font: string,
   232→    color: string,
   233→    align: CanvasTextAlign = "left",
   234→    baseline: CanvasTextBaseline = "alphabetic",
   235→  ): void {
   236→    const t = this.l(s);
   237→    this.ctx.font = font;
   238→    this.ctx.fillStyle = color;
   239→    this.ctx.textAlign = align;
   240→    this.ctx.textBaseline = baseline;
   241→    this.ctx.fillText(t, x, y);
   242→    this.ctx.textAlign = "left";
   243→    this.ctx.textBaseline = "alphabetic";
   244→  }
   245→
   246→  /** letter-spaced text (arcade title style); counted once */
   247→  textSpaced(
   248→    s: string,
   249→    x: number,
   250→    y: number,
   251→    font: string,
   252→    color: string,
   253→    spacing: number,
   254→    align: "left" | "center" | "right" = "left",
   255→  ): void {
   256→    const t = this.l(s);
   257→    this.ctx.font = font;
   258→    let total = 0;
   259→    for (const ch of t) total += this.ctx.measureText(ch).width + spacing;
   260→    total -= spacing;
   261→    let cx = x;
   262→    if (align === "center") cx = x - total / 2;
   263→    if (align === "right") cx = x - total;
   264→    this.ctx.fillStyle = color;
   265→    this.ctx.textAlign = "left";
   266→    this.ctx.textBaseline = "alphabetic";
   267→    for (const ch of t) {
   268→      this.ctx.fillText(ch, cx, y);
   269→      cx += this.ctx.measureText(ch).width + spacing;
   270→    }
   271→  }
   272→
   273→  /** wrapped text with latin accounting; returns bottom y of last line */
   274→  wrapped(
   275→    s: string,
   276→    x: number,
   277→    y: number,
   278→    maxW: number,
   279→    font: string,
   280→    color: string,
   281→    lh = 30,
   282→    maxLines?: number,
   283→    align: CanvasTextAlign = "left",
   284→  ): number {
   285→    const t = this.l(s);
   286→    return fillWrapped(this.ctx, t, x, y, maxW, font, color, {
   287→      lineHeight: lh,
   288→      maxLines,
   289→      align,
   290→    });
   291→  }
   292→
   293→  /** small labeled chip; returns total width */
   294→  chip(text: string, x: number, y: number, o: ChipOpts): number {
   295→    const t = this.l(text);
   296→    this.ctx.font = o.font;
   297→    const tw = this.ctx.measureText(t).width;
   298→    const padX = o.padX ?? 12;
   299→    const h = o.h ?? 36;
   300→    const w = Math.ceil(tw) + padX * 2;
   301→    if (o.bg) {
   302→      this.ctx.fillStyle = o.bg;
   303→      this.ctx.fillRect(x, y, w, h);
   304→    }
   305→    if (o.border) {
   306→      this.ctx.fillStyle = o.border;
   307→      this.ctx.fillRect(x, y, w, 2);
   308→      this.ctx.fillRect(x, y + h - 2, w, 2);
   309→      this.ctx.fillRect(x, y, 2, h);
   310→      this.ctx.fillRect(x + w - 2, y, 2, h);
   311→    }
   312→    this.ctx.fillStyle = o.fg;
   313→    this.ctx.textAlign = "left";
   314→    this.ctx.textBaseline = "middle";
   315→    this.ctx.fillText(t, x + padX, y + h / 2 + 1);
   316→    this.ctx.textBaseline = "alphabetic";
   317→    return w;
   318→  }
   319→
   320→  /** centered chip; returns width */
   321→  chipCentered(text: string, cx: number, y: number, o: ChipOpts): number {
   322→    this.ctx.font = o.font;
   323→    const tw = this.ctx.measureText(this.l(text)).width;
   324→    const w = Math.ceil(tw) + (o.padX ?? 12) * 2;
   325→    this.chip(text, Math.round(cx - w / 2), y, o);
   326→    return w;
   327→  }
   328→
   329→  /** right-aligned chip; x is the RIGHT edge; returns width */
   330→  chipRight(text: string, right: number, y: number, o: ChipOpts): number {
   331→    this.ctx.font = o.font;
   332→    const tw = this.ctx.measureText(this.l(text)).width;
   333→    const w = Math.ceil(tw) + (o.padX ?? 12) * 2;
   334→    this.chip(text, right - w, y, o);
   335→    return w;
   336→  }
   337→
   338→  panel(x: number, y: number, w: number, h: number, fill: string = PANEL): void {
   339→    this.ctx.fillStyle = fill;
   340→    this.ctx.fillRect(x, y, w, h);
   341→  }
   342→
   343→  /** truncate with ellipsis to fit width */
   344→  fit(s: string, maxW: number, font: string): string {
   345→    let t = this.l(s);
   346→    this.ctx.font = font;
   347→    if (this.ctx.measureText(t).width <= maxW) return t;
   348→    while (t.length > 1 && this.ctx.measureText(`${t}...`).width > maxW) {
   349→      t = t.slice(0, -1);
   350→    }
   351→    return `${t.trimEnd()}...`;
   352→  }
   353→}

   369→function drawHud(P: Painter, req: PosterRequest): void {
   370→  const { ctx, w } = P;
   371→  ctx.fillStyle = "rgba(5,6,14,0.92)";
   372→  ctx.fillRect(0, 0, w, 72);
   373→
   374→  // left 16-BIT badge (gold pixel border)
   375→  drawPixelBorder(ctx, 24, 14, 118, 46, GOLD);
   376→  P.text("16-BIT", 83, 44, fMono(21), GOLD, "center");
   377→
   378→  // center title (uppercase mono, white/gold, letter-spaced, auto-fit)
   379→  const title = latinize(req.title).toUpperCase();
   380→  const maxW = w - 430;
   381→  let size = title.length > 18 ? 32 : 40;
   382→  let font = fMono(size);
   383→  while (P.measure(title, font) + (title.length - 1) * 2 > maxW && size > 22) {
   384→    size -= 2;
   385→    font = fMono(size);
   386→  }
   387→  const baseline = size >= 36 ? 48 : 45;
   388→  // drop shadow then main text
   389→  P.textSpaced(title, w / 2 + 3, baseline + 3, font, "rgba(255,93,93,0.55)", 2, "center");
   390→  P.textSpaced(title, w / 2, baseline, font, CREAM, 2, "center");
   391→
   392→  // right: INSERT COIN + coin sprite + stage label
   393→  const stage = req.stage ?? `ST ${String(KINDS.indexOf(req.kind) + 1).padStart(2, "0")}`;
   394→  const coinFont = fMono(20);
   395→  const coinTextW = P.measure("INSERT COIN", coinFont);
   396→  P.text("INSERT COIN", w - 24, 34, coinFont, GOLD, "right");
   397→  P.text("_", w - 22, 34, fMono(20), "rgba(255,209,102,0.45)", "left");
   398→  drawSprite(ctx, SPRITES.COIN, w - 24 - coinTextW - 48, 12, 3);
   399→  P.text(stage, w - 24, 58, fMono(15), GREY, "right");
   400→
   401→  // neon magenta divider
   402→  ctx.fillStyle = MAGENTA;
   403→  ctx.fillRect(0, 72, w, 3);
   404→  ctx.fillStyle = "rgba(255,168,236,0.8)";
   405→  ctx.fillRect(0, 72, w, 1);
   406→}

   410→function drawFooter(P: Painter, req: PosterRequest, seed: number): void {
   411→  const { ctx, w, h } = P;
   412→  const y0 = h - 64;
   413→  ctx.fillStyle = "rgba(5,6,14,0.92)";
   414→  ctx.fillRect(0, y0, w, 64);
   415→  ctx.fillStyle = "rgba(77,214,255,0.45)";
   416→  ctx.fillRect(0, y0, w, 2);
   417→  ctx.fillStyle = "rgba(77,214,255,0.12)";
   418→  ctx.fillRect(0, y0 + 2, w, 1);
   419→
   420→  // lives: 3 hearts
   421→  for (let i = 0; i < 3; i++) {
   422→    drawSprite(ctx, SPRITES.HEART, 24 + i * 38, y0 + 16, 3);
   423→  }
   424→  const rng = makeRng(seed ^ 0xc01a);
   425→  const coins = 50 + (hashString(`${seed}`) % 450);
   426→  P.text(`COINS x${coins}`, 148, y0 + 30, fMono(20), GOLD);
   427→  P.text("CREDIT 01", 148, y0 + 54, fMono(17), CYAN);
   428→
   429→  // right: lock hash + footer note
   430→  const hash =
   431→    req.hash ??
   432→    `${(hashString(latinize(req.title)) >>> 0).toString(16)}${(hashString(req.kind) >>> 0).toString(16)}`.slice(0, 10).toUpperCase();
   433→  P.text(`LOCK #${hash}`, w - 24, y0 + 30, fMono(18), GOLD, "right");
   434→  if (req.footerNote) {
   435→    P.text(P.fit(req.footerNote, 520, fMono(16)), w - 24, y0 + 54, fMono(16), GREY, "right");
   436→  }
   437→}

   442→  WIN: { color: GREEN, sprite: SPRITES.CHECK },
   443→  LOSS: { color: RED, sprite: SPRITES.CROSS },

   449→function statusChip(P: Painter, cx: number, cy: number, status: string, big = false): number {
   450→  const st = STATUS_STYLE[status] ?? STATUS_STYLE.INFO;
   451→  const font = big ? fMono(22) : fMono(16);
   452→  const padX = big ? 14 : 10;
   453→  const h = big ? 48 : 36;
   454→  const spriteW = st.sprite ? (big ? 30 : 22) : 0;
   455→  P.ctx.font = font;
   456→  const tw = P.ctx.measureText(latinize(status)).width;
   457→  const w = Math.ceil(tw) + padX * 2 + spriteW + (st.sprite ? 8 : 0);
   458→  const x = Math.round(cx - w / 2);
   459→  const y = Math.round(cy - h / 2);
   460→  P.ctx.fillStyle = `${st.color}22`;
   461→  P.ctx.fillRect(x, y, w, h);
   462→  P.ctx.fillStyle = st.color;
   463→  P.ctx.fillRect(x, y, w, 2);
   464→  P.ctx.fillRect(x, y + h - 2, w, 2);
   465→  P.ctx.fillRect(x, y, 2, h);
   466→  P.ctx.fillRect(x + w - 2, y, 2, h);
   467→  let ix = x + padX;
   468→  if (st.sprite) {
   469→    drawSprite(P.ctx, st.sprite, ix, Math.round(cy - (big ? 15 : 11)), big ? 3 : 2, { "#": st.color });
   470→    ix += spriteW + 8;
   471→  }
   472→  P.ctx.fillStyle = st.color;
   473→  P.ctx.textAlign = "left";
   474→  P.ctx.textBaseline = "middle";
   475→  P.ctx.fillText(latinize(status), ix, cy + 1);
   476→  P.ctx.textBaseline = "alphabetic";
   477→  return w;
   478→}

   482→function drawPlayerPlate(
   483→  P: Painter,
   484→  x: number,
   485→  y: number,
   486→  w: number,
   487→  h: number,
   488→  name: string,
   489→  full: string,
   490→  rank: number | undefined,
   491→  flag: string | undefined,
   492→  accent: string,
   493→  fav = false,
   494→  seed?: number,
   495→): void {
   496→  P.panel(x, y, w, h, PANEL);
   497→  drawPixelBorder(P.ctx, x, y, w, h, accent);
   498→  if (rank !== undefined) {
   499→    P.chip(`#${rank}`, x + 18, y + 16, { font: fMono(20), fg: GOLD, border: GOLD, bg: "rgba(255,209,102,0.10)", h: 34, padX: 10 });
   500→  }
   501→  if (seed !== undefined) {
   502→    const seedX = x + 18 + (rank !== undefined ? 76 : 0);
   503→    P.chip(`S${seed}`, seedX, y + 16, { font: fMono(20), fg: CYAN, border: CYAN, bg: "rgba(77,214,255,0.10)", h: 34, padX: 10 });
   504→  }
   505→  if (flag) {
   506→    P.chipRight(flag, x + w - 18, y + 16, { font: fMono(18), fg: GREY, border: "#3d4654", bg: "rgba(61,70,84,0.3)", h: 34, padX: 10 });
   507→  }
   508→  const shortFont = fMono(40);

  1374→function drawChannelChip(P: Painter, cx: number, cy: number, channel: string): number {
  1375→  const color = CHANNEL_COLOR[channel] ?? GREY;
  1376→  const font = fMono(18);
  1377→  P.ctx.font = font;
  1378→  const tw = P.ctx.measureText(channel).width;
  1379→  const w = Math.ceil(tw) + 24;
  1380→  const h = 40;
  1381→  const x = Math.round(cx - w / 2);
  1382→  const y = Math.round(cy - h / 2);
  1383→  P.ctx.fillStyle = `${color}22`;
  1384→  P.ctx.fillRect(x, y, w, h);
  1385→  P.ctx.fillStyle = color;
  1386→  P.ctx.fillRect(x, y, w, 2);
  1387→  P.ctx.fillRect(x, y + h - 2, w, 2);
  1388→  P.ctx.fillRect(x, y, 2, h);
  1389→  P.ctx.fillRect(x + w - 2, y, 2, h);
  1390→  P.ctx.fillStyle = color;
  1391→  P.ctx.textAlign = "center";
  1392→  P.ctx.textBaseline = "middle";
  1393→  P.ctx.fillText(latinize(channel), cx, cy + 1);
  1394→  P.ctx.textBaseline = "alphabetic";
  1395→  return w;
  1396→}

  1779→export function renderPoster(req: PosterRequest): RenderResult {
  1780→  const kind = req.kind;
  1781→  const w = 1080;
  1782→  const h = kind === "BRIEFING" ? 1920 : 1440;
  1783→  try {
  1784→    registerFonts();
  1785→    const canvas = createCanvas(w, h);
  1786→    const ctx = canvas.getContext("2d");
  1787→    ctx.imageSmoothingEnabled = false;
  1788→
  1789→    const seed = hashString(`${latinize(req.title)}|${kind}|${req.seedSalt ?? 0}`);
  1790→    const rng = makeRng(seed ^ 0xf00ba11);
  1791→
  1792→    const P = new Painter(ctx, w, h, rng);
  1793→
  1794→    // 1. background
  1795→    drawBackground(ctx, req.theme, seed, w, h);
  1796→
  1797→    // 2. kind layout
  1798→    switch (kind) {
  1799→      case "FLASHCARD":
  1800→      case "FREECARD":
  1801→        drawFlashcard(P, req);
  1802→        break;
  1803→      case "BRIEFING":
  1804→        drawBriefing(P, req);
  1805→        break;
  1806→      case "AGGREGATE":
  1807→        drawAggregate(P, req);
  1808→        break;
  1809→      case "WEEKLY":
  1810→        drawWeekly(P, req);
  1811→        break;
  1812→      case "NEWS":
  1813→        drawNews(P, req);
  1814→        break;
  1815→      case "ACADEMY":
  1816→        drawAcademy(P, req);
  1817→        break;
  1818→      case "BANNER":
  1819→        drawBanner(P, req);
  1820→        break;
  1821→      case "FTRESULTS":
  1822→        drawFtResults(P, req);
  1823→        break;
  1824→      case "DAILYPROGRAM":
  1825→        drawDailyProgram(P, req);
  1826→        break;
  1827→    }
  1828→
  1829→    // 3. chrome
  1830→    drawHud(P, req);
  1831→    drawFooter(P, req, seed);
  1832→
  1833→    // 4. CRT overlay last
  1834→    drawCrt(ctx, w, h, 72);
  1835→
  1836→    const buf = canvas.toBuffer("image/png");
  1837→    return {
  1838→      ok: true,
  1839→      kind,
  1840→      pngBase64: Buffer.from(buf).toString("base64"),
  1841→      width: w,
  1842→      height: h,
  1843→      latinized: P.latin,
  1844→    };
  1845→  } catch (e) {
  1846→    const msg = e instanceof Error ? e.message : String(e);
  1847→    return { ok: false, kind, pngBase64: "", width: 0, height: 0, latinized: 0, error: msg };
  1848→  }
  1849→}
