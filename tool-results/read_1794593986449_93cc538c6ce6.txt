    88→// ─── v4.0 Persian date/time + countdown (تاریخ و ساعت دقیق شروع) ───────────
    89→
    90→const faDate = (iso: string): string => {
    91→  const d = new Date(iso);
    92→  if (Number.isNaN(d.getTime())) return "";
    93→  return new Intl.DateTimeFormat("fa-IR", {
    94→    weekday: "long",
    95→    day: "numeric",
    96→    month: "long",
    97→  }).format(d);
    98→};
    99→
   100→const faTime = (iso: string): string => {
   101→  const d = new Date(iso);
   102→  if (Number.isNaN(d.getTime())) return "";
   103→  return new Intl.DateTimeFormat("fa-IR", {
   104→    hour: "2-digit",
   105→    minute: "2-digit",
   106→  }).format(d);
   107→};
   108→
   109→function countdownLabel(iso: string, now: number): string {
   110→  const t = new Date(iso).getTime();
   111→  if (!Number.isFinite(t)) return "";
   112→  const diff = t - now;
   113→  if (diff <= 0) return "شروع‌شده / در جریان";
   114→  const mins = Math.floor(diff / 60000);
   115→  const secs = Math.floor((diff % 60000) / 1000);
   116→  if (mins >= 60) {
   117→    return `شروع تا ${Math.floor(mins / 60)} ساعت و ${mins % 60} دقیقه دیگر`;
   118→  }
   119→  return `شروع تا ${mins}:${String(secs).padStart(2, "0")} دیگر`;
   120→}

   122→function MatchClock({ item }: { item: LineItem }) {
   123→  const [now, setNow] = useState(() => Date.now());
   124→  useEffect(() => {
   125→    const t = setInterval(() => setNow(Date.now()), 1000);
   126→    return () => clearInterval(t);
   127→  }, []);
   128→  if (!item.startTime) return <span className="text-[11px] text-muted-foreground">{item.timing.label}</span>;
   129→  return (
   130→    <span className="inline-flex flex-wrap items-center gap-x-2 gap-y-0.5 text-[11px] text-white/75">
   131→      <CalendarClock className="h-3.5 w-3.5 text-white/50" />
   132→      <span className="font-bold text-white/90">
   133→        {faDate(item.startTime)} — ساعت {faTime(item.startTime)}
   134→      </span>
   135→      {item.state === "pre" && (
   136→        <span className="mono-num rounded-md border border-amber-300/30 bg-amber-300/10 px-1.5 py-0.5 text-[10px] text-amber-200" dir="ltr">
   137→          {countdownLabel(item.startTime, now)}
   138→        </span>
   139→      )}
   140→      {item.state === "in" && (
   141→        <span className="rounded-md border border-rose-400/30 bg-rose-400/10 px-1.5 py-0.5 text-[10px] font-bold text-rose-200">
   142→          در حال بازی
   143→        </span>
   144→      )}
   145→    </span>
   146→  );
   147→}

   320→function espnUrlOf(eventKey?: string, espnUrl?: string): string {
   321→  if (espnUrl) return espnUrl;
   322→  if (eventKey && /^\d+$/.test(eventKey)) {
   323→    return `https://www.espn.com/tennis/match/_/gameId/${eventKey}`;
   324→  }
   325→  return "https://www.espn.com/tennis/";
   326→}

   328→function HashVerifyChip({
   329→  hash,
   330→  eventKey,
   331→  espnUrl,
   332→  matchLabel,
   333→}: {
   334→  hash: string;
   335→  eventKey?: string;
   336→  espnUrl?: string;
   337→  matchLabel: string;
   338→}) {
   339→  const [open, setOpen] = useState(false);
   340→  const [copied, setCopied] = useState(false);
   341→  const link = espnUrlOf(eventKey, espnUrl);
   342→
   343→  const copyHash = async () => {
   344→    try {
   345→      await navigator.clipboard.writeText(hash);
   346→      setCopied(true);
   347→      setTimeout(() => setCopied(false), 1600);
   348→    } catch {
   349→      /* clipboard unavailable */
   350→    }
   351→  };
   352→
   353→  return (
   354→    <>
   355→      <Tooltip>
   356→        <TooltipTrigger asChild>
   357→          <button
   358→            type="button"
   359→            onClick={() => setOpen(true)}
   360→            aria-label="راستی‌آزمایی هش SHA-256"
   361→            className="mono-num inline-flex items-center gap-1 rounded-lg border border-cyan-400/30 bg-cyan-400/[0.07] px-2 py-1 text-[10px] text-cyan-300 transition-colors hover:bg-cyan-400/15"
   362→            dir="ltr"
   363→          >
   364→            <Lock className="h-3 w-3" />
   365→            {hash.slice(0, 10).toUpperCase()}
   366→          </button>
   367→        </TooltipTrigger>
   368→        <TooltipContent className="text-[10px]">
   369→          قفل SHA-256 پیش از مسابقه — کلیک برای راستی‌آزمایی
   370→        </TooltipContent>
   371→      </Tooltip>
   372→
   373→      <Dialog open={open} onOpenChange={setOpen}>
   374→        <DialogContent
   375→          onClick={(e) => e.stopPropagation()}
   376→          className="max-w-lg border-white/10 bg-[oklch(0.16_0.01_270)]"

   378→          <DialogHeader>

   658→// ─── Near-gate strip: live 3-archetype candidacy for unapproved matches ────
   659→
   660→function NearGateStrip({ matches }: { matches: AnalyzedMatch[] }) {
   661→  const candidates = useMemo(() => {
   662→    return matches
   663→      .filter((m) => m.state !== "post" && !m.veto.vetoed)
   664→      .slice(0, 40)
   665→      .map((m) => {
   666→        const fav = m.favorite;
   667→        const hold = fav === "A" ? m.markov.gA : m.markov.gB;
   668→        const sFav = fav === "A" ? m.markov.setWinA : 1 - m.markov.setWinA;
   669→        const best = [...(m.archEvals ?? [])]
   670→          .filter((e) => e.eligible)
   671→          .sort((a, b) => a.pairLoss - b.pairLoss)[0];
   672→        const anyWindow = (m.archEvals ?? []).some((e) => e.eligible);
   673→        return { m, hold, sFav, best, anyWindow };
   674→      })
   675→      .filter((c) => !c.best)
   676→      .sort((a, b) => a.sFav - b.sFav)
   677→      .slice(0, 4);
   678→  }, [matches]);
   679→
   680→  if (candidates.length === 0) return null;
   681→  return (
   682→    <div className="rounded-2xl border border-white/10 bg-white/[0.03] p-4">
   683→      <div className="flex items-center gap-2 text-xs font-bold text-white/80">
   684→        <ShieldAlert className="h-3.5 w-3.5 text-amber-300" />
   685→        کاندیدهای زیر ذره‌بین سوپروایزر (ارزیابی زنده ۳ الگو)
   686→      </div>
   687→      <div className="mt-3 grid gap-2 sm:grid-cols-2">
   688→        {candidates.map(({ m, hold, sFav }) => (
   689→          <div key={m.eventKey} className="rounded-xl border border-white/5 bg-white/[0.03] p-2.5">
   690→            <div className="flex items-center justify-between text-xs">
   691→              <span className="font-bold">
   692→                <SurfaceDot surface={m.surface} />{" "}
   693→                {m.favorite === "A" ? m.playerA.name : m.playerB.name}
   694→                <span className="text-muted-foreground"> × {m.favorite === "A" ? m.playerB.name : m.playerA.name}</span>
   695→              </span>
   696→              <span className="mono-num text-[10px] text-muted-foreground">{m.tournament.slice(0, 16)}</span>
   697→            </div>
   698→            <div className="mt-2 space-y-1 text-[10px] text-muted-foreground">
   699→              <div className="flex items-center gap-2">
   700→                <span className="w-24">S_fav</span>
   701→                <div className="h-1.5 flex-1 rounded-full bg-white/10">
   702→                  <div
   703→                    className="h-full rounded-full bg-cyan-400"
   704→                    style={{ width: `${Math.min(100, (sFav / 0.9) * 100)}%` }}
   705→                  />
   706→                </div>
   707→                <span className="mono-num">{(sFav * 100).toFixed(0)}%</span>
   708→              </div>
   709→              <div className="flex items-center gap-2">
   710→                <span className="w-24">Hold fav</span>
   711→                <div className="h-1.5 flex-1 rounded-full bg-white/10">
   712→                  <div
   713→                    className={`h-full rounded-full ${hold >= 0.84 ? "bg-emerald-400" : "bg-amber-400"}`}
   714→                    style={{ width: `${Math.min(100, (hold / 0.95) * 100)}%` }}
   715→                  />
   716→                </div>
   717→                <span className={`mono-num ${hold >= 0.84 ? "text-emerald-300" : ""}`}>{Math.round(hold * 100)}%</span>
   718→              </div>
   719→              <p className="pt-0.5 leading-4 text-muted-foreground/80">
   720→                خارج از پنجره الگوها — پنجره رقابتی{" "}
   721→                <span className="mono-num">54% ≤ S_A &lt; 64%</span>
   722→              </p>
   723→            </div>
   724→          </div>
   725→        ))}
   726→      </div>
   727→    </div>
   728→  );
   729→}
