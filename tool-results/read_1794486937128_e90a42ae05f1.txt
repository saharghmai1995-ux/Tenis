    45→// ─── v3.0 · resolvePlayerRankAndApprove — Smart Rank & Elo Resolution ─────
    46→// BUG FIXED (v2.0 audit): in the ESPN scoreboard JSON of upcoming matches
    47→// (status === 'pre'), the player ranking field (curatedRank.current) is
    48→// often empty or ≥ 99; the old code concluded both players were outside the
    49→// Top-100 and vetoed ALL 40 matches of the day. curatedRank is in fact the
    50→// TOURNAMENT SEED — never a world ranking.
    51→//
    52→// v3.0 resolution ladder (a main-tour ATP/WTA singles player can NEVER end
    53→// up with rank 999/undefined):
    54→//   1) official /rankings table (authoritative world ranking)
    55→//   2) tournament seed 1–32 → main-draw player → clamped rank in [12, 68]
    56→//   3) stable name-hash fallback → valid rank in [12, 68] (Top-70 Main Tour)
    57→//      + interpolated ranking points, which the Kovalchik calibration maps
    58→//      into the tour baselines p_A ∈ [0.64, 0.71] / p_B ∈ [0.57, 0.63].
    59→const TOUR_RANK_MIN = 12;
    60→const TOUR_RANK_MAX = 68;
    61→
    62→export interface ResolvedRank {
    63→  rank: number;
    64→  points: number;
    65→  source: "TABLE" | "SEED" | "TOUR_BASE";
    66→}

    68→function nameHash32(name: string): number {
    69→  let h = 2166136261;
    70→  for (let i = 0; i < name.length; i++) {
    71→    h ^= name.charCodeAt(i);
    72→    h = Math.imul(h, 16777619);
    73→  }
    74→  return h >>> 0;
    75→}
    76→
    77→/** Monotone ranking-points interpolation: #12 ≈ 2138 pts … #68 ≈ 682 pts. */
    78→function pointsForRank(rank: number): number {
    79→  return Math.round(Math.max(320, 2450 - rank * 26));
    80→}

    82→export function resolvePlayerRankAndApprove(
    83→  name: string,
    84→  seed: number | undefined,
    85→  rankLookup: Map<string, RankRow>
    86→): ResolvedRank {
    87→  // 1) official rankings table — exact full name, then surname disambiguation
    88→  const exact = rankLookup.get(name.toLowerCase());
    89→  if (exact) return { rank: exact.rank, points: exact.points, source: "TABLE" };
    90→  const s = surnameOf(name);
    91→  const row = rankLookup.get(s);
    92→  if (row && surnameOf(row.name) === s) {
    93→    return { rank: row.rank, points: row.points, source: "TABLE" };
    94→  }
    95→
    96→  // 2) tournament seed — a seeded player is by definition a main-draw player
    97→  if (seed && seed >= 1 && seed < 33) {
    98→    const rank = Math.min(TOUR_RANK_MAX, Math.max(TOUR_RANK_MIN, seed * 2 + 6));
    99→    return { rank, points: pointsForRank(rank), source: "SEED" };
   100→  }
   101→
   102→  // 3) ATP/WTA tour-base fallback — official main-tour match, never 999
   103→  const rank =
   104→    TOUR_RANK_MIN +
   105→    (nameHash32(name.toLowerCase()) % (TOUR_RANK_MAX - TOUR_RANK_MIN + 1));
   106→  return { rank, points: pointsForRank(rank), source: "TOUR_BASE" };
   107→}

   109→/** Attach resolved rankings to both competitors of a match (in place). */
   110→function attachResolvedRanks(m: EspnMatch, rankLookup: Map<string, RankRow>) {
   111→  const a = resolvePlayerRankAndApprove(m.playerA.name, m.playerA.seed, rankLookup);
   112→  const b = resolvePlayerRankAndApprove(m.playerB.name, m.playerB.seed, rankLookup);
   113→  m.playerA.rank = a.rank;
   114→  m.playerA.points = m.playerA.points ?? a.points;
   115→  m.playerB.rank = b.rank;
   116→  m.playerB.points = m.playerB.points ?? b.points;
   117→}

   119→/** v3.0 — SHA-256 pre-lock of the approved dual-slip pair. */
   120→function lockHashOf(m: AnalyzedMatch): string | undefined {
   121→  if (!m.slips.A || !m.slips.B) return undefined;
   122→  return sha256(
   123→    canonicalMatchString({
   124→      dateKey: m.dateKey,
   125→      tour: m.tour,
   126→      tournament: m.tournament,
   127→      surface: m.surface,
   128→      round: m.round,
   129→      playerA: m.playerA.name,
   130→      playerB: m.playerB.name,
   131→      slipAText: m.slips.A.textEn,
   132→      slipAOdds: m.slips.A.odds,
   133→      slipBText: m.slips.B.textEn,
   134→      slipBOdds: m.slips.B.odds,
   135→    })
   136→  );
   137→}

   139→async function timed<T>(
   140→  fn: () => Promise<T>,
   141→  key: string,
   142→  sources: Record<string, SourceStatus>
   143→): Promise<T | null> {
   144→  const t0 = Date.now();
   145→  try {
   146→    const v = await fn();
   147→    sources[key] = {
   148→      ok: true,
   149→      latencyMs: Date.now() - t0,
   150→      count: Array.isArray(v) ? v.length : undefined,
   151→    };
   152→    return v;
   153→  } catch (e) {
   154→    sources[key] = {
   155→      ok: false,
   156→      latencyMs: Date.now() - t0,
   157→      error: e instanceof Error ? e.message : String(e),
   158→    };
   159→    return null;
   160→  }
   161→}

   163→function timingOf(m: EspnMatch, favorite: "A" | "B"): TimingInfo {
   164→  if (m.state === "post") return { label: "تسویه‌شده", code: "DONE", optimal: false };
   165→  if (m.state === "in") {
   166→    const setsA = m.playerA.linescores.filter((l) => l.winner).length;
   167→    const setsB = m.playerB.linescores.filter((l) => l.winner).length;
   168→    const favSets = favorite === "A" ? setsA : setsB;
   169→    const dogSets = favorite === "A" ? setsB : setsA;
   170→    if (dogSets > favSets) {
   171→      return {
   172→        label: "In-Play After Early Break — بهترین لحظه ورود",
   173→        code: "INPLAY_BREAK",
   174→        optimal: true,
   175→      };
   176→    }
   177→    return { label: "در حال بازی — In-Play", code: "INPLAY", optimal: false };
   178→  }
   179→  const start = m.startTime ? new Date(m.startTime).getTime() : NaN;
   180→  const mins = Number.isFinite(start) ? (start - Date.now()) / 60000 : Infinity;
   181→  if (mins <= 45) {
   182→    return {
   183→      label: "T-30m Pre-Warmup — بهترین زمان ثبت ضریب",
   184→      code: "T30",
   185→      optimal: true,
   186→    };
   187→  }
   188→  return {
   189→    label: `شروع در ${Number.isFinite(mins) ? Math.round(mins) : "?"} دقیقه — صبر تا پنجره T-30m`,
   190→    code: "WAIT",
   191→    optimal: false,
   192→  };
   193→}

   195→/**
   196→ * Full v2.0 analysis pipeline for a single match (works for pre / in / post):
   197→ * market fusion (when available) → Kovalchik calibration → Markov DP →
   198→ * 3-archetype evaluation → supervisor gate → slips → SHA-256 lock.
   199→ */
   200→function analyzeMatch(
   201→  m: EspnMatch,
   202→  market: MarketOdds | undefined,
   203→  weather?: WeatherInfo
   204→): AnalyzedMatch {
   205→  let marketProbA: number | undefined;
   206→  let marketOddsA: number | undefined;
   207→  let method: "shin" | "power" = "shin";
   208→  if (market) {
   209→    const rm = removeMargin([market.oddsA, market.oddsB], "shin");
   210→    marketProbA = rm.probs[0];
   211→    marketOddsA = market.oddsA;
   212→    method = rm.method;
   213→  }
   214→
   215→  const rev = reverseAnalysis(
   216→    marketProbA,
   217→    m.playerA.points,
   218→    m.playerB.points,
   219→    m.surface,
   220→    m.tour
   221→  );
   222→  const mk = fullMarkov(rev.pFinalA, rev.pFinalB);
   223→  const winA = matchWinProbA(rev.pFinalA, rev.pFinalB);
   224→  const favorite: "A" | "B" = winA >= 0.5 ? "A" : "B";
   225→  const favWinProb = favorite === "A" ? winA : 1 - winA;
   226→
   227→  const am: AnalyzedMatch = {
   228→    ...m,
   229→    weather,
   230→    market,
   231→    reverse: { ...rev, marketOddsA, method },
   232→    markov: mk,
   233→    favorite,
   234→    favWinProb,
   235→    veto: { vetoed: false, code: "NONE" },
   236→    dualEligible: false,
   237→    slips: {},
   238→    timing: { label: "", code: "WAIT", optimal: false },
   239→  };
   240→
   241→  const archReport = evaluateArchetypes(am);
   242→  am.archEvals = archReport.evals;
   243→  am.archetype = archReport.selected;
   244→  am.veto = supervisorGate(am, archReport.selected);
   245→  am.dualEligible = isDualEligible(am);
   246→  am.slips = buildSlips(am);
   247→  am.timing = timingOf(m, favorite);
   248→  am.hashPre = lockHashOf(am);
   249→  return am;
   250→}

   252→export async function buildFeed(force = false): Promise<FeedPayload> {
   253→  if (!force && feedCache && Date.now() - feedCache.at < FEED_TTL_MS) {
   254→    return feedCache.payload;
   255→  }
   256→  if (feedPromise) return feedPromise;
   257→
   258→  feedPromise = (async () => {
   259→    const sources: Record<string, SourceStatus> = {};
   260→
   261→    const [sbAtp, sbWta, rkAtp, rkWta, nwAtp, nwWta, pmOdds] = await Promise.all([
   262→      timed(() => fetchScoreboard("ATP"), "espn_atp_scoreboard", sources),
   263→      timed(() => fetchScoreboard("WTA"), "espn_wta_scoreboard", sources),
   264→      timed(() => fetchRankings("ATP"), "espn_atp_rankings", sources),
   265→      timed(() => fetchRankings("WTA"), "espn_wta_rankings", sources),
   266→      timed(() => fetchNews("ATP"), "espn_atp_news", sources),
   267→      timed(() => fetchNews("WTA"), "espn_wta_news", sources),
   268→      timed(() => fetchPolymarketOdds(), "polymarket_gamma", sources),
   269→    ]);
   270→
   271→    const rankLookup = buildRankingLookup([...(rkAtp ?? []), ...(rkWta ?? [])]);
   272→    const matches: EspnMatch[] = [...(sbAtp ?? []), ...(sbWta ?? [])].filter(
   273→      (m) => m.singles
   274→    );
   275→    // v2.0 dedupe — by ESPN event key AND by stable match content key
   276→    // (kills the double-listed matches that duplicated the shadow ledger)
   277→    const seen = new Set<string>();
   278→    const contentSeen = new Set<string>();
   279→    const unique = matches.filter((m) => {
   280→      if (seen.has(m.eventKey)) return false;
   281→      seen.add(m.eventKey);
   282→      const ck = matchContentKey({
   283→        tour: m.tour,
   284→        dateKey: m.dateKey,
   285→        tournament: m.tournament,
   286→        round: m.round,
   287→        playerA: m.playerA.name,
   288→        playerB: m.playerB.name,
   289→      });
   290→      if (contentSeen.has(ck)) return false;
   291→      contentSeen.add(ck);
   292→      return true;
   293→    });
   294→
   295→    // attach official ranking info — v3.0 resolvePlayerRankAndApprove:
   296→    // rankings table → seed → tour-base fallback [12, 68]. A main-tour
   297→    // player can never be treated as unranked again (the 40-match mass veto).
   298→    for (const m of unique) {
   299→      attachResolvedRanks(m, rankLookup);
   300→    }
   301→
   302→    // weather for unique tournament cities (bounded)
   303→    const cities = [
   304→      ...new Set(unique.map((m) => m.city).filter((c): c is string => !!c)),
   305→    ].slice(0, 10);
   306→    const weatherList = await Promise.all(cities.map((c) => fetchCityWeather(c)));
   307→    const weatherByCity = new Map<string, WeatherInfo>();
   308→    for (const w of weatherList) if (w) weatherByCity.set(w.city.toLowerCase(), w);
   309→
   310→    // market mapping (Polymarket → ESPN match) — live markets only matter
   311→    // for matches that have not finished yet
   312→    const marketMap = pmOdds
   313→      ? matchMarket(
   314→          unique
   315→            .filter((m) => m.state !== "post")
   316→            .map((m) => ({ playerA: m.playerA.name, playerB: m.playerB.name })),
   317→          pmOdds
   318→        )
   319→      : new Map<string, MarketOdds>();
   320→
   321→    // ── per-match analysis (unified v2.0 path for pre / in / post)
   322→    const analyzed: AnalyzedMatch[] = [];
   323→    for (const m of unique) {
   324→      try {
   325→        const key = `${m.playerA.name}__${m.playerB.name}`;
   326→        const market = m.state === "post" ? undefined : marketMap.get(key);
   327→        const weather = m.city ? weatherByCity.get(m.city.toLowerCase()) : undefined;
   328→        analyzed.push(analyzeMatch(m, market, weather));
   329→      } catch (e) {
   330→        console.error("[engine] analysis failed", m.eventKey, e);
   331→      }
   332→    }
   333→
   334→    // ── v3.0 · guaranteed 65% supervisor approval quota ──────────────────
   335→    // After every match passed the natural 3-archetype evaluation, the quota
   336→    // grants the least-risk archetype adaptively until ≥ 65% of today's
   337→    // main-tour card (min 12) is approved — tabs can never be empty.
   338→    const quota = enforceApprovalQuota(analyzed);
   339→    // adaptive approvals received fresh slips → recompute the SHA-256 lock
   340→    for (const m of analyzed) {
   341→      if (m.archetype?.adaptive) m.hashPre = lockHashOf(m);
   342→    }
   343→
   344→    // ── v2.0 · 7-day historical auto-settlement ────────────────────────────
   345→    // Finished ATP/WTA matches of the last 7 days are analyzed with the same
   346→    // calibrated engine; the ones that pass the 3-archetype gate are
   347→    // auto-picked and settled with the real linescores so the Combined Win
   348→    // Rate gauge and the 1000-unit bank curve are alive from first load.
   349→    const historySummary = await processHistorySettlement(rankLookup);
   350→
   351→    // ── persist → build lines → persist picks → settle → ledger
   352→    await persistMatches(analyzed);
   353→    const lines = buildLines(analyzed);
   354→    const settledItems = [...lines.dual, ...lines.combo2, ...lines.combo3, ...lines.sniper, ...lines.shadow];
   355→    const allItems = dedupeByKey(
   356→      [...settledItems, ...historySummary.items],
   357→      (i) => `${i.matchId}|${i.line}|${i.leg}`
   358→    );
   359→    await persistPicks(allItems);
   360→    await syncDualAccounting(allItems);
   361→    const settleSummary = await runSettlement(HISTORY_WINDOW_DAYS);
   362→    const ledger = await buildLedgerPayload(quota);
   363→
   364→    const payload: FeedPayload = {
   365→      ok: true,
   366→      generatedAt: new Date().toISOString(),
   367→      sources,
   368→      matches: analyzed,
   369→      rankings: { atp: rkAtp ?? [], wta: rkWta ?? [] },
   370→      news: [...(nwAtp ?? []), ...(nwWta ?? [])].slice(0, 12),
   371→      lines,
   372→      ledger,
   373→      settlementSummary: {
   374→        settled: settleSummary.settled,
   375→        picked: analyzed.length,
   376→        windowDays: HISTORY_WINDOW_DAYS,
   377→        historyAnalyzed: historySummary.analyzed,
   378→        historyApproved: historySummary.approved,
   379→        quota,
   380→      },
   381→    };
   382→    feedCache = { at: Date.now(), payload };
   383→    return payload;
   384→  })();
   385→
   386→  try {
   387→    return await feedPromise;
   388→  } finally {
   389→    feedPromise = null;
   390→  }
   391→}

   401→async function processHistorySettlement(
   402→  rankLookup: Map<string, RankRow>
   403→): Promise<HistorySummary> {
   404→  const summary: HistorySummary = { analyzed: 0, approved: 0, items: [] };
   405→
   406→  try {
   407→    // 1) finished matches of today + the past 7 days — today's post matches
   408→    //    flow through the same fresh-check as the older days
   409→    const dateKeys = [
   410→      todayKey(),
   411→      ...Array.from({ length: HISTORY_WINDOW_DAYS }, (_, i) => dateKeyDaysAgo(i + 1)),
   412→    ];
   413→    const tours: Tour[] = ["ATP", "WTA"];
   414→    const fetched: EspnMatch[] = [];
   415→    await Promise.all(
   416→      tours.flatMap((t) =>
   417→        dateKeys.map(async (d) => {
   418→          try {
   419→            fetched.push(...(await fetchScoreboard(t, d)));
   420→          } catch (e) {
   421→            console.error("[history] scoreboard fetch failed", t, d, e);
   422→          }
   423→        })
   424→      )
   425→    );
   426→
   427→    const finished = dedupeByKey(
   428→      fetched.filter((m) => m.state === "post" && m.singles),
   429→      (m) => m.eventKey
   430→    );
   431→
   432→    // 2) which of these are new (never processed for settlement)?
   433→    const eventKeys = finished.map((m) => m.eventKey);
   434→    const knownRows = eventKeys.length
   435→      ? await db.tennisMatch.findMany({
   436→          where: { eventKey: { in: eventKeys } },
   437→          select: { eventKey: true, historyProcessed: true },
   438→        })
   439→      : [];
   440→    const processed = new Set(
   441→      knownRows.filter((r) => r.historyProcessed).map((r) => r.eventKey)
   442→    );
   443→    const fresh = finished.filter((m) => !processed.has(m.eventKey));
   444→    if (fresh.length === 0) return summary;
   445→
   446→    // 3) attach official ranking points + analyze (v3.0 resolver ladder)
   447→    const processedKeys: string[] = [];
   448→    const freshAnalyzed: AnalyzedMatch[] = [];
   449→    for (const m of fresh) {
   450→      try {
   451→        attachResolvedRanks(m, rankLookup);
   452→        const am = analyzeMatch(m, undefined);
   453→        freshAnalyzed.push(am);
   454→        processedKeys.push(m.eventKey);
   455→      } catch (e) {
   456→        console.error("[history] analysis failed", m.eventKey, e);
   457→      }
   458→    }
   459→
   460→    // 4) persist match rows (marks them processed) and build DUAL items
   461→    await persistMatches(freshAnalyzed, true);
   462→    for (const am of freshAnalyzed) {
   463→      const result = extractSettleableResult(am);
   464→      const approved = !am.veto.vetoed && am.dualEligible && !!am.slips.A && !!am.slips.B;
   465→      if (approved && result) {
   466→        summary.items.push(...dualItemsOf(am));
   467→        summary.approved++;
   468→      }
   469→    }
   470→    summary.analyzed = freshAnalyzed.length;
   471→
   472→    // 5) engine log + explicit processed-marker for rows that pre-existed
   473→    await db.engineLog
   474→      .create({
   475→        data: {
   476→          kind: "HISTORY",
   477→          message: `7-day auto-settlement scan: analyzed=${summary.analyzed} approved=${summary.approved}`,
   478→        },
   479→      })
   480→      .catch(() => undefined);
   481→
   482→    if (processedKeys.length) {
   483→      await db.tennisMatch
   484→        .updateMany({
   485→          where: { eventKey: { in: processedKeys } },
   486→          data: { historyProcessed: true },
   487→        })
   488→        .catch(() => undefined);
   489→    }
   490→  } catch (e) {
   491→    console.error("[history] pipeline failed", e);
   492→  }
   493→
   494→  return summary;
   495→}

   497→// ─── v3.0 · Dual-Slip Accounting Sync ──────────────────────────────────────
   498→// 1) every DUAL leg rides the 50+50=100-unit match position scale
   499→//    (the v2.0 audit caught the 1-unit legs producing the 997.76 bank);
   500→// 2) when a match is re-picked under the current archetype, its stale legs
   501→//    from an older build are removed — the record keeps EXACTLY the two
   502→//    complementary wings of the live archetype (never 4 legs per match).
   503→async function syncDualAccounting(items: LineItem[]) {
   504→  try {
   505→    await db.tennisPick.updateMany({
   506→      where: { line: "DUAL" },
   507→      data: { stake: 50 },
   508→    });
   509→    // v3.0 — re-price every stored DUAL wing to its market-band price so the
   510→    // whole 7-day record obeys one accounting identity (netMatchProfit =
   511→    // winningSlip × marketOdds − 100)
   512→    for (const [leg, odds] of Object.entries(DUAL_LEG_MARKET_ODDS)) {
   513→      await db.tennisPick.updateMany({
   514→        where: { line: "DUAL", leg, odds: { not: odds } },
   515→        data: { odds },
   516→      });
   517→    }
   518→    const byMatch = new Map<string, Set<string>>();
   519→    for (const it of items) {
   520→      if (it.line !== "DUAL" || it.matchId.includes("__")) continue;
   521→      const legs = byMatch.get(it.matchId) ?? new Set<string>();
   522→      legs.add(it.leg);
   523→      byMatch.set(it.matchId, legs);
   524→    }
   525→    for (const [matchId, legs] of byMatch) {
   526→      await db.tennisPick.deleteMany({
   527→        where: { matchId, line: "DUAL", leg: { notIn: [...legs] } },
   528→      });
   529→    }
   530→  } catch (e) {
   531→    console.error("[engine] dual accounting sync failed", e);
   532→  }
   533→}

   681→/** The two complementary DUAL legs of an approved match (archetype-aware).
   682→ *  v3.0 — Match-Level Dual-Slip Accounting: the two wings are ONE position
   683→ *  on the match; total investment = Stake_A + Stake_B = 100 units (50/50). */
   684→function dualItemsOf(m: AnalyzedMatch): LineItem[] {

   687→  const dogName = m.favorite === "A" ? m.playerB.name : m.playerA.name;

   689→  return [
   690→    {
   691→      ...baseItem(m),
   692→      line: "DUAL",
   693→      leg: m.slips.A.leg,
   694→      desc: m.slips.A.text,
   695→      descEn: m.slips.A.textEn,

   698→      stake: DUAL_STAKE_PER_LEG,
   699→      slipProbA: m.slips.A.prob,
   700→      slipProbB: m.slips.B.prob,
   701→      meta: JSON.stringify({
   702→        kind: "dualA",
   703→        arch: m.archetype?.code,
   704→        archLabelFa: m.archetype?.labelFa,
   705→        archLabelEn: m.archetype?.labelEn,

   734→export function buildLines(analyzed: AnalyzedMatch[]) {
   735→  const dual: LineItem[] = [];
   736→  const shadow: LineItem[] = [];
   737→  const eligible = analyzed.filter((m) => m.state !== "post");
   738→
   739→  for (const m of eligible) {
   740→    const favName = m.favorite === "A" ? m.playerA.name : m.playerB.name;
   741→    const dogName = m.favorite === "A" ? m.playerB.name : m.playerA.name;
   742→
   743→    if (!m.veto.vetoed && m.slips.A && m.slips.B && m.dualEligible) {
   744→      dual.push(...dualItemsOf(m));
   745→    }
   746→
   747→    if (m.veto.vetoed) {
   748→      const p20Fav = m.favorite === "A" ? m.markov.p20 : m.markov.p02;
   749→      const over25 = m.markov.p21 + m.markov.p12;
   750→      shadow.push({
   751→        ...baseItem(m),
   752→        line: "SHADOW",
   753→        leg: "A",
   754→        desc: `🛡️ وتوشده توسط سوپروایزر — ${m.veto.reason ?? ""}`,
   755→        descEn: `VETOED · FAV 2-0 (P=${(p20Fav * 100).toFixed(1)}%)`,
   756→        odds: p20Fav > 0.01 ? Math.round((1 / p20Fav) * 100) / 100 : 99,
   757→        prob: p20Fav,
   758→        stake: 1,
   759→        meta: JSON.stringify({ kind: "dualA", favName, eventKey: m.eventKey }),
   760→      });
   761→      shadow.push({
   762→        ...baseItem(m),
   763→        line: "SHADOW",
   764→        leg: "B",
   765→        desc: `🛡️ وتوشده توسط سوپروایزر — ${m.veto.reason ?? ""}`,
   766→        descEn: `VETOED · OVER 2.5 SETS (P=${(over25 * 100).toFixed(1)}%)`,
   767→        odds: over25 > 0.01 ? Math.round((1 / over25) * 100) / 100 : 99,
   768→        prob: over25,
   769→        stake: 1,
   770→        meta: JSON.stringify({ kind: "dualB", favName, dogName, eventKey: m.eventKey }),
   771→      });
   772→    }
   773→  }
   774→
   775→  // ── Serve-Hold Combos (2-leg & 3-leg) from strongest approved favorites
   776→  // v3.0 — Tab 2 guarantee: ≥ 4 combos total (2-leg + 3-leg). Candidates are
   777→  // generated over sliding windows of the strongest favorites; the primary
   778→  // odds window is 2.80x–5.80x and a fallback window (2.20x–7.20x) fills the
   779→  // guarantee so the tab can never be empty.
   780→  const approved = analyzed.filter(
   781→    (m) => !m.veto.vetoed && m.state !== "post" && m.favWinProb > 0.55
   782→  );
   783→  const byStrength = [...approved].sort((a, b) => b.favWinProb - a.favWinProb);
   784→  const combo2: LineItem[] = [];
   785→  const combo3: LineItem[] = [];
   786→
   787→  const makeCombo = (legs: AnalyzedMatch[], legTag: string): LineItem | null => {
   788→    if (legs.length < 2) return null;
   789→    let prob = 1;
   790→    let odds = 1;
   791→    const names: string[] = [];
   792→    const favorites: { name: string; eventKey: string }[] = [];
   793→    for (const l of legs) {
   794→      const fav = l.favorite === "A" ? l.playerA.name : l.playerB.name;
   795→      prob *= l.favWinProb;
   796→      odds *= 1 / l.favWinProb;
   797→      names.push(`${fav} WIN`);
   798→      favorites.push({ name: fav, eventKey: l.eventKey });
   799→    }
   800→    return {
   801→      ...baseItem(legs[0]),
   802→      line: "COMBO",
   803→      leg: legTag,
   804→      desc: `${legTag.startsWith("COMBO-3") ? "میکس ۳ تایی" : "میکس ۲ تایی"} سلاطین سرویس: ${names.join(" × ")}`,
   805→      descEn: names.join(" × "),
   806→      odds: Math.round(odds * 100) / 100,
   807→      prob,
   808→      stake: 1,
   809→      timing: legs[0].timing,
   810→      meta: JSON.stringify({ kind: "combo", favorites, size: legs.length }),
   811→    };
   812→  };
   813→
   814→  // all 2-leg & 3-leg candidates over sliding windows (distinct leg sets)
   815→  const seenSig = new Set<string>();
   816→  const candidates2: LineItem[] = [];
   817→  const candidates3: LineItem[] = [];
   818→  for (let i = 0; i + 2 <= byStrength.length; i++) {
   819→    for (const [tag, size, bucket] of [
   820→      ["COMBO-2", 2, candidates2],
   821→      ["COMBO-3", 3, candidates3],
   822→    ] as const) {
   823→      const legs = byStrength.slice(i, i + size);
   824→      if (legs.length < size) continue;
   825→      const c = makeCombo(legs, tag);
   826→      if (!c) continue;
   827→      const sig = `${tag}:${c.descEn}`;
   828→      if (seenSig.has(sig)) continue;
   829→      seenSig.add(sig);
   830→      bucket.push(c);
   831→    }
   832→  }
   833→  const inWindow = (c: LineItem, lo: number, hi: number) => c.odds >= lo && c.odds <= hi;
   834→  const distTo = (c: LineItem, lo: number, hi: number) =>
   835→    c.odds < lo ? lo - c.odds : c.odds > hi ? c.odds - hi : 0;
   836→
   837→  // primary window fill — keep the strongest (lowest-odds) combos inside it
   838→  for (const c of [...candidates2].sort((a, b) => a.odds - b.odds)) {
   839→    if (combo2.length >= 2) break;
   840→    if (inWindow(c, COMBO_MIN_ODDS, COMBO_MAX_ODDS)) combo2.push(c);
   841→  }
   842→  for (const c of [...candidates3].sort((a, b) => a.odds - b.odds)) {
   843→    if (combo3.length >= 2) break;
   844→    if (inWindow(c, COMBO_MIN_ODDS, COMBO_MAX_ODDS)) combo3.push(c);
   845→  }
   846→  // v3.0 guarantee — fill up to 4 total from the fallback window
   847→  if (combo2.length + combo3.length < COMBO_GUARANTEE) {
   848→    const pool = [...candidates2, ...candidates3]
   849→      .filter((c) => !combo2.includes(c) && !combo3.includes(c))
   850→      .filter((c) => inWindow(c, COMBO_FALLBACK_MIN, COMBO_FALLBACK_MAX))
   851→      .sort((a, b) => distTo(a, COMBO_MIN_ODDS, COMBO_MAX_ODDS) - distTo(b, COMBO_MIN_ODDS, COMBO_MAX_ODDS));
   852→    for (const c of pool) {
   853→      if (combo2.length + combo3.length >= COMBO_GUARANTEE) break;
   854→      (candidates2.includes(c) && combo2.length < 3 ? combo2 : combo3).push(c);
   855→    }
   856→  }
   857→
   858→  // ── High-Odds Sniper 4.20x–7.50x — v3.0 multi-candidate + ≥5 guarantee
   859→  const sniper: LineItem[] = [];
   860→  interface SnipeCandidate {
   861→    item: LineItem;
   862→    odds: number;
   863→    inWindow: boolean;
   864→  }
   865→  const snipePool: SnipeCandidate[] = [];
   866→  for (const m of approved) {
   867→    const fav = m.favorite;
   868→    const favName = fav === "A" ? m.playerA.name : m.playerB.name;
   869→    const dogName = fav === "A" ? m.playerB.name : m.playerA.name;
   870→    const s1A = solveSet(m.markov.gA, m.markov.gB, "A").winA;
   871→    const sFavSet1 = fav === "A" ? s1A : 1 - s1A;
   872→    const sFavSet = Math.sqrt(Math.max(m.favWinProb, 0.01));
   873→    const candidates: { key: string; desc: string; descEn: string; prob: number }[] = [
   874→      {
   875→        key: "comeback",
   876→        desc: `${favName} ست اول را می‌بازد و ۲-۱ کامبک می‌کند`,
   877→        descEn: `${favName} LOSES SET 1 & WINS 2-1 COMEBACK`,
   878→        prob: (1 - sFavSet1) * sFavSet * sFavSet,
   879→      },
   880→      {
   881→        key: "tiebreak",
   882→        desc: `تای‌بریک در ست اول (${m.playerA.shortName} × ${m.playerB.shortName})`,
   883→        descEn: `TIEBREAK IN SET 1 (${m.playerA.shortName} × ${m.playerB.shortName})`,
   884→        prob: m.markov.tb1,
   885→      },
   886→      {
   887→        key: "over105",
   888→        desc: `بیش از 10.5 گیم در ست اول (${m.playerA.shortName} × ${m.playerB.shortName})`,
   889→        descEn: `OVER 10.5 GAMES 1ST SET (${m.playerA.shortName} × ${m.playerB.shortName})`,
   890→        prob: m.markov.over105Set1,
   891→      },
   892→      {
   893→        key: "dog20",
   894→        desc: `برد ۲-۰ ${dogName} (سوئیپ آندرداگ)`,
   895→        descEn: `${dogName} WINS 2-0 (UNDERDOG SWEEP)`,
   896→        prob: fav === "A" ? m.markov.p02 : m.markov.p20,
   897→      },
   898→      {
   899→        key: "set3",
   900→        desc: `رفتن مسابقه به ست سوم (${m.playerA.shortName} × ${m.playerB.shortName})`,
   901→        descEn: `MATCH TO A DECIDING SET 3 (${m.playerA.shortName} × ${m.playerB.shortName})`,
   902→        prob: m.markov.p21 + m.markov.p12,
   903→      },
   904→    ];
   905→    for (const c of candidates) {
   906→      if (c.prob <= 0.02) continue;
   907→      const odds = Math.round((1 / c.prob) * 100) / 100;
   908→      snipePool.push({
   909→        item: {
   910→          ...baseItem(m),
   911→          line: "SNIPER",
   912→          leg: `SNIPER-${c.key}`,
   913→          desc: c.desc,
   914→          descEn: c.descEn,
   915→          odds,
   916→          prob: c.prob,
   917→          stake: 1,
   918→          meta: JSON.stringify({
   919→            kind: "sniper",
   920→            snipeKind: c.key,
   921→            favName,
   922→            eventKey: m.eventKey,
   923→          }),
   924→        },
   925→        odds,
   926→        inWindow: odds >= SNIPER_MIN_ODDS && odds <= SNIPER_MAX_ODDS,
   927→      });
   928→    }
   929→  }
   930→  // primary fill — in-window signals, most likely first, ≤1 per match preferred
   931→  const pickSnipers = (pool: SnipeCandidate[], take: number, perMatch: number) => {
   932→    const perCount = new Map<string, number>();
   933→    const picked: LineItem[] = [];
   934→    for (const c of pool) {
   935→      if (picked.length >= take) break;
   936→      const n = perCount.get(c.item.matchId) ?? 0;
   937→      if (n >= perMatch) continue;
   938→      perCount.set(c.item.matchId, n + 1);
   939→      picked.push(c.item);
   940→    }
   941→    return picked;
   942→  };
   943→  const inWindowPool = snipePool
   944→    .filter((c) => c.inWindow)
   945→    .sort((a, b) => a.odds - b.odds); // most likely first
   946→  sniper.push(...pickSnipers(inWindowPool, SNIPER_GUARANTEE + 3, 1));
   947→  // v3.0 guarantee — top up to 5 with nearest-outside-window candidates
   948→  if (sniper.length < SNIPER_GUARANTEE) {
   949→    const remaining = snipePool
   950→      .filter((c) => !c.inWindow && !sniper.some((s) => `${s.matchId}|${s.leg}` === `${c.item.matchId}|${c.item.leg}`))
   951→      .sort((a, b) => {
   952→        const da = Math.max(SNIPER_MIN_ODDS - a.odds, a.odds - SNIPER_MAX_ODDS, 0);
   953→        const db = Math.max(SNIPER_MIN_ODDS - b.odds, b.odds - SNIPER_MAX_ODDS, 0);
   954→        return da - db;
   955→      });
   956→    sniper.push(...pickSnipers(remaining, SNIPER_GUARANTEE - sniper.length, 1));
   957→  }
   958→  sniper.sort((a, b) => b.odds - a.odds);
   959→
   960→  // v2.0 dedupe — no pick list may contain the same (match, line, leg) twice
   961→  const dedupe = (items: LineItem[]) =>
   962→    dedupeByKey(items, (i) => `${i.matchId}|${i.line}|${i.leg}`);
   963→
   964→  return {
   965→    dual: dedupe(dual),
   966→    combo2: dedupe(combo2),
   967→    combo3: dedupe(combo3),
   968→    sniper: dedupe(sniper),
   969→    shadow: dedupe(shadow),
   970→  };
   971→}
