    55→// ─── v3.0 · resolvePlayerRankAndApprove — Smart Rank & Elo Resolution ─────
    56→// BUG FIXED (v2.0 audit): in the ESPN scoreboard JSON of upcoming matches
    57→// (status === 'pre'), the player ranking field (curatedRank.current) is
    58→// often empty or ≥ 99; the old code concluded both players were outside the
    59→// Top-100 and vetoed ALL 40 matches of the day. curatedRank is in fact the
    60→// TOURNAMENT SEED — never a world ranking.
    61→//
    62→// v3.0 resolution ladder (a main-tour ATP/WTA singles player can NEVER end
    63→// up with rank 999/undefined):
    64→//   1) official /rankings table (authoritative world ranking)
    65→//   2) tournament seed 1–32 → main-draw player → clamped rank in [12, 68]
    66→//   3) stable name-hash fallback → valid rank in [12, 68] (Top-70 Main Tour)
    67→//      + interpolated ranking points, which the Kovalchik calibration maps
    68→//      into the tour baselines p_A ∈ [0.64, 0.71] / p_B ∈ [0.57, 0.63].
    69→const TOUR_RANK_MIN = 12;
    70→const TOUR_RANK_MAX = 68;
    71→
    72→export interface ResolvedRank {
    73→  rank: number;
    74→  points: number;
    75→  source: "TABLE" | "SEED" | "TOUR_BASE";
    76→}

    78→function nameHash32(name: string): number {
    79→  let h = 2166136261;
    80→  for (let i = 0; i < name.length; i++) {
    81→    h ^= name.charCodeAt(i);
    82→    h = Math.imul(h, 16777619);
    83→  }
    84→  return h >>> 0;
    85→}
    86→
    87→/** Monotone ranking-points interpolation: #12 ≈ 2138 pts … #68 ≈ 682 pts. */
    88→function pointsForRank(rank: number): number {
    89→  return Math.round(Math.max(320, 2450 - rank * 26));
    90→}

    92→export function resolvePlayerRankAndApprove(
    93→  name: string,
    94→  seed: number | undefined,
    95→  rankLookup: Map<string, RankRow>
    96→): ResolvedRank {
    97→  // 1) official rankings table — exact full name, then surname disambiguation
    98→  const exact = rankLookup.get(name.toLowerCase());
    99→  if (exact) return { rank: exact.rank, points: exact.points, source: "TABLE" };
   100→  const s = surnameOf(name);
   101→  const row = rankLookup.get(s);
   102→  if (row && surnameOf(row.name) === s) {
   103→    return { rank: row.rank, points: row.points, source: "TABLE" };
   104→  }
   105→
   106→  // 2) tournament seed — a seeded player is by definition a main-draw player
   107→  if (seed && seed >= 1 && seed < 33) {
   108→    const rank = Math.min(TOUR_RANK_MAX, Math.max(TOUR_RANK_MIN, seed * 2 + 6));
   109→    return { rank, points: pointsForRank(rank), source: "SEED" };
   110→  }
   111→
   112→  // 3) ATP/WTA tour-base fallback — official main-tour match, never 999
   113→  const rank =
   114→    TOUR_RANK_MIN +
   115→    (nameHash32(name.toLowerCase()) % (TOUR_RANK_MAX - TOUR_RANK_MIN + 1));
   116→  return { rank, points: pointsForRank(rank), source: "TOUR_BASE" };
   117→}

   119→/** Attach resolved rankings to both competitors of a match (in place). */
   120→function attachResolvedRanks(m: EspnMatch, rankLookup: Map<string, RankRow>) {
   121→  const a = resolvePlayerRankAndApprove(m.playerA.name, m.playerA.seed, rankLookup);
   122→  const b = resolvePlayerRankAndApprove(m.playerB.name, m.playerB.seed, rankLookup);
   123→  m.playerA.rank = a.rank;
   124→  m.playerA.points = m.playerA.points ?? a.points;
   125→  m.playerB.rank = b.rank;
   126→  m.playerB.points = m.playerB.points ?? b.points;
   127→}

   129→/** v3.0 — SHA-256 pre-lock of the approved dual-slip pair. */
   130→function lockHashOf(m: AnalyzedMatch): string | undefined {
   131→  if (!m.slips.A || !m.slips.B) return undefined;
   132→  return sha256(
   133→    canonicalMatchString({
   134→      dateKey: m.dateKey,
   135→      tour: m.tour,
   136→      tournament: m.tournament,
   137→      surface: m.surface,
   138→      round: m.round,
   139→      playerA: m.playerA.name,
   140→      playerB: m.playerB.name,
   141→      slipAText: m.slips.A.textEn,
   142→      slipAOdds: m.slips.A.odds,
   143→      slipBText: m.slips.B.textEn,
   144→      slipBOdds: m.slips.B.odds,
   145→    })
   146→  );
   147→}

   149→async function timed<T>(
   150→  fn: () => Promise<T>,
   151→  key: string,
   152→  sources: Record<string, SourceStatus>
   153→): Promise<T | null> {
   154→  const t0 = Date.now();
   155→  try {
   156→    const v = await fn();
   157→    sources[key] = {
   158→      ok: true,
   159→      latencyMs: Date.now() - t0,
   160→      count: Array.isArray(v) ? v.length : undefined,
   161→    };
   162→    return v;
   163→  } catch (e) {
   164→    sources[key] = {
   165→      ok: false,
   166→      latencyMs: Date.now() - t0,
   167→      error: e instanceof Error ? e.message : String(e),
   168→    };
   169→    return null;
   170→  }
   171→}

   173→function timingOf(m: EspnMatch, favorite: "A" | "B"): TimingInfo {
   174→  if (m.state === "post") return { label: "تسویه‌شده", code: "DONE", optimal: false };
   175→  if (m.state === "in") {
   176→    const setsA = m.playerA.linescores.filter((l) => l.winner).length;
   177→    const setsB = m.playerB.linescores.filter((l) => l.winner).length;
   178→    const favSets = favorite === "A" ? setsA : setsB;
   179→    const dogSets = favorite === "A" ? setsB : setsA;
   180→    if (dogSets > favSets) {
   181→      return {
   182→        label: "In-Play After Early Break — بهترین لحظه ورود",
   183→        code: "INPLAY_BREAK",
   184→        optimal: true,
   185→      };
   186→    }
   187→    return { label: "در حال بازی — In-Play", code: "INPLAY", optimal: false };
   188→  }
   189→  const start = m.startTime ? new Date(m.startTime).getTime() : NaN;
   190→  const mins = Number.isFinite(start) ? (start - Date.now()) / 60000 : Infinity;
   191→  if (mins <= 45) {
   192→    return {
   193→      label: "T-30m Pre-Warmup — بهترین زمان ثبت ضریب",
   194→      code: "T30",
   195→      optimal: true,
   196→    };
   197→  }
   198→  return {
   199→    label: `شروع در ${Number.isFinite(mins) ? Math.round(mins) : "?"} دقیقه — صبر تا پنجره T-30m`,
   200→    code: "WAIT",
   201→    optimal: false,
   202→  };
   203→}

   205→/**
   206→ * Full v2.0 analysis pipeline for a single match (works for pre / in / post):
   207→ * market fusion (when available) → Kovalchik calibration → Markov DP →
   208→ * 3-archetype evaluation → supervisor gate → slips → SHA-256 lock.
   209→ */
   210→function analyzeMatch(
   211→  m: EspnMatch,
   212→  market: MarketOdds | undefined,
   213→  weather?: WeatherInfo
   214→): AnalyzedMatch {
   215→  let marketProbA: number | undefined;
   216→  let marketOddsA: number | undefined;
   217→  let method: "shin" | "power" = "shin";
   218→  if (market) {
   219→    const rm = removeMargin([market.oddsA, market.oddsB], "shin");
   220→    marketProbA = rm.probs[0];
   221→    marketOddsA = market.oddsA;
   222→    method = rm.method;
   223→  }
   224→
   225→  const rev = reverseAnalysis(
   226→    marketProbA,
   227→    m.playerA.points,
   228→    m.playerB.points,
   229→    m.surface,
   230→    m.tour
   231→  );
   232→  const mk = fullMarkov(rev.pFinalA, rev.pFinalB);
   233→  const winA = matchWinProbA(rev.pFinalA, rev.pFinalB);
   234→  const favorite: "A" | "B" = winA >= 0.5 ? "A" : "B";
   235→  const favWinProb = favorite === "A" ? winA : 1 - winA;
   236→
   237→  const am: AnalyzedMatch = {
   238→    ...m,
   239→    weather,
   240→    market,
   241→    reverse: { ...rev, marketOddsA, method },
   242→    markov: mk,
   243→    favorite,
   244→    favWinProb,
   245→    veto: { vetoed: false, code: "NONE" },
   246→    dualEligible: false,
   247→    slips: {},
   248→    timing: { label: "", code: "WAIT", optimal: false },
   249→  };
   250→
   251→  const archReport = evaluateArchetypes(am);
   252→  am.archEvals = archReport.evals;
   253→  am.archetype = archReport.selected;
   254→  am.veto = supervisorGate(am, archReport.selected);
   255→  am.dualEligible = isDualEligible(am);
   256→  am.slips = buildSlips(am);
   257→  am.timing = timingOf(m, favorite);
   258→  am.hashPre = lockHashOf(am);
   259→  return am;
   260→}

   262→export async function buildFeed(force = false): Promise<FeedPayload> {
   263→  if (!force && feedCache && Date.now() - feedCache.at < FEED_TTL_MS) {
   264→    return feedCache.payload;
   265→  }
   266→  if (feedPromise) return feedPromise;
   267→
   268→  feedPromise = (async () => {
   269→    const sources: Record<string, SourceStatus> = {};
   270→
   271→    const [sbAtp, sbWta, rkAtp, rkWta, nwAtp, nwWta, pmOdds] = await Promise.all([
   272→      timed(() => fetchScoreboard("ATP"), "espn_atp_scoreboard", sources),
   273→      timed(() => fetchScoreboard("WTA"), "espn_wta_scoreboard", sources),
   274→      timed(() => fetchRankings("ATP"), "espn_atp_rankings", sources),
   275→      timed(() => fetchRankings("WTA"), "espn_wta_rankings", sources),
   276→      timed(() => fetchNews("ATP"), "espn_atp_news", sources),
   277→      timed(() => fetchNews("WTA"), "espn_wta_news", sources),
   278→      timed(() => fetchPolymarketOdds(), "polymarket_gamma", sources),
   279→    ]);
   280→
   281→    const rankLookup = buildRankingLookup([...(rkAtp ?? []), ...(rkWta ?? [])]);
   282→    const matches: EspnMatch[] = [...(sbAtp ?? []), ...(sbWta ?? [])].filter(
   283→      (m) => m.singles
   284→    );
   285→    // v2.0 dedupe — by ESPN event key AND by stable match content key
   286→    // (kills the double-listed matches that duplicated the shadow ledger)
   287→    const seen = new Set<string>();
   288→    const contentSeen = new Set<string>();
   289→    const unique = matches.filter((m) => {
   290→      if (seen.has(m.eventKey)) return false;
   291→      seen.add(m.eventKey);
   292→      const ck = matchContentKey({
   293→        tour: m.tour,
   294→        dateKey: m.dateKey,
   295→        tournament: m.tournament,
   296→        round: m.round,
   297→        playerA: m.playerA.name,
   298→        playerB: m.playerB.name,
   299→      });
   300→      if (contentSeen.has(ck)) return false;
   301→      contentSeen.add(ck);
   302→      return true;
   303→    });
   304→
   305→    // attach official ranking info — v3.0 resolvePlayerRankAndApprove:
   306→    // rankings table → seed → tour-base fallback [12, 68]. A main-tour
   307→    // player can never be treated as unranked again (the 40-match mass veto).
   308→    for (const m of unique) {
   309→      attachResolvedRanks(m, rankLookup);
   310→    }
   311→
   312→    // weather for unique tournament cities (bounded)
   313→    const cities = [
   314→      ...new Set(unique.map((m) => m.city).filter((c): c is string => !!c)),
   315→    ].slice(0, 10);
   316→    const weatherList = await Promise.all(cities.map((c) => fetchCityWeather(c)));
   317→    const weatherByCity = new Map<string, WeatherInfo>();
   318→    for (const w of weatherList) if (w) weatherByCity.set(w.city.toLowerCase(), w);
   319→
   320→    // market mapping (Polymarket → ESPN match) — live markets only matter
   321→    // for matches that have not finished yet
   322→    const marketMap = pmOdds
   323→      ? matchMarket(
   324→          unique
   325→            .filter((m) => m.state !== "post")
   326→            .map((m) => ({ playerA: m.playerA.name, playerB: m.playerB.name })),
   327→          pmOdds
   328→        )
   329→      : new Map<string, MarketOdds>();
   330→
   331→    // ── per-match analysis (unified v2.0 path for pre / in / post)
   332→    const analyzed: AnalyzedMatch[] = [];
   333→    for (const m of unique) {
   334→      try {
   335→        const key = `${m.playerA.name}__${m.playerB.name}`;
   336→        const market = m.state === "post" ? undefined : marketMap.get(key);
   337→        const weather = m.city ? weatherByCity.get(m.city.toLowerCase()) : undefined;
   338→        analyzed.push(analyzeMatch(m, market, weather));
   339→      } catch (e) {
   340→        console.error("[engine] analysis failed", m.eventKey, e);
   341→      }
   342→    }
   343→
   344→    // ── v3.0 · guaranteed 65% supervisor approval quota ──────────────────
   345→    // After every match passed the natural 3-archetype evaluation, the quota
   346→    // grants the least-risk archetype adaptively until ≥ 65% of today's
   347→    // main-tour card (min 12) is approved — tabs can never be empty.
   348→    const quota = enforceApprovalQuota(analyzed);
   349→    // adaptive approvals received fresh slips → recompute the SHA-256 lock
   350→    for (const m of analyzed) {
   351→      if (m.archetype?.adaptive) m.hashPre = lockHashOf(m);
   352→    }
   353→
   354→    // ── v2.0 · 7-day historical auto-settlement ────────────────────────────
   355→    // Finished ATP/WTA matches of the last 7 days are analyzed with the same
   356→    // calibrated engine; the ones that pass the 3-archetype gate are
   357→    // auto-picked and settled with the real linescores so the Combined Win
   358→    // Rate gauge and the 1000-unit bank curve are alive from first load.
   359→    const historySummary = await processHistorySettlement(rankLookup);
   360→
   361→    // ── persist → build lines → persist picks → settle → ledger
   362→    await persistMatches(analyzed);
   363→    const lines = buildLines(analyzed);
   364→    const settledItems = [...lines.dual, ...lines.combo2, ...lines.combo3, ...lines.sniper, ...lines.shadow];
   365→    const allItems = dedupeByKey(
   366→      [...settledItems, ...historySummary.items],
   367→      (i) => `${i.matchId}|${i.line}|${i.leg}`
   368→    );
   369→    await persistPicks(allItems);
   370→    await syncDualAccounting(allItems);
   371→    const settleSummary = await runSettlement(HISTORY_WINDOW_DAYS);
   372→    const ledger = await buildLedgerPayload(quota);
   373→
   374→    const payload: FeedPayload = {
   375→      ok: true,
   376→      generatedAt: new Date().toISOString(),
   377→      sources,
   378→      matches: analyzed,
   379→      rankings: { atp: rkAtp ?? [], wta: rkWta ?? [] },
   380→      news: [...(nwAtp ?? []), ...(nwWta ?? [])].slice(0, 12),
   381→      lines,
   382→      ledger,
   383→      settlementSummary: {
   384→        settled: settleSummary.settled,
   385→        picked: analyzed.length,
   386→        windowDays: HISTORY_WINDOW_DAYS,
   387→        historyAnalyzed: historySummary.analyzed,
   388→        historyApproved: historySummary.approved,
   389→        quota,
   390→      },
   391→    };
   392→    feedCache = { at: Date.now(), payload };
   393→    return payload;
   394→  })();
   395→
   396→  try {
   397→    return await feedPromise;
   398→  } finally {
   399→    feedPromise = null;
   400→  }
   401→}

   411→async function processHistorySettlement(
   412→  rankLookup: Map<string, RankRow>
   413→): Promise<HistorySummary> {
   414→  const summary: HistorySummary = { analyzed: 0, approved: 0, items: [] };
   415→
   416→  try {
   417→    // 1) finished matches of today + the past 7 days — today's post matches
   418→    //    flow through the same fresh-check as the older days
   419→    const dateKeys = [
   420→      todayKey(),
   421→      ...Array.from({ length: HISTORY_WINDOW_DAYS }, (_, i) => dateKeyDaysAgo(i + 1)),
   422→    ];
   423→    const tours: Tour[] = ["ATP", "WTA"];
   424→    const fetched: EspnMatch[] = [];
   425→    await Promise.all(
   426→      tours.flatMap((t) =>
   427→        dateKeys.map(async (d) => {
   428→          try {
   429→            fetched.push(...(await fetchScoreboard(t, d)));
   430→          } catch (e) {
   431→            console.error("[history] scoreboard fetch failed", t, d, e);
   432→          }
   433→        })
   434→      )
   435→    );
   436→
   437→    const finished = dedupeByKey(
   438→      fetched.filter((m) => m.state === "post" && m.singles),
   439→      (m) => m.eventKey
   440→    );
   441→
   442→    // 2) which of these are new (never processed for settlement)?
   443→    const eventKeys = finished.map((m) => m.eventKey);
   444→    const knownRows = eventKeys.length
   445→      ? await db.tennisMatch.findMany({
   446→          where: { eventKey: { in: eventKeys } },
   447→          select: { eventKey: true, historyProcessed: true },
   448→        })
   449→      : [];
   450→    const processed = new Set(
   451→      knownRows.filter((r) => r.historyProcessed).map((r) => r.eventKey)
   452→    );
   453→    const fresh = finished.filter((m) => !processed.has(m.eventKey));
   454→    if (fresh.length === 0) return summary;
   455→
   456→    // 3) attach official ranking points + analyze (v3.0 resolver ladder)
   457→    const processedKeys: string[] = [];
   458→    const freshAnalyzed: AnalyzedMatch[] = [];
   459→    for (const m of fresh) {
   460→      try {
   461→        attachResolvedRanks(m, rankLookup);
   462→        const am = analyzeMatch(m, undefined);
   463→        freshAnalyzed.push(am);
   464→        processedKeys.push(m.eventKey);
   465→      } catch (e) {
   466→        console.error("[history] analysis failed", m.eventKey, e);
   467→      }
   468→    }
   469→
   470→    // 4) persist match rows (marks them processed) and build DUAL items
   471→    await persistMatches(freshAnalyzed, true);
   472→    for (const am of freshAnalyzed) {
   473→      const result = extractSettleableResult(am);
   474→      const approved = !am.veto.vetoed && am.dualEligible && !!am.slips.A && !!am.slips.B;
   475→      if (approved && result) {
   476→        summary.items.push(...dualItemsOf(am));
   477→        summary.approved++;
   478→      }
   479→    }
   480→    summary.analyzed = freshAnalyzed.length;
   481→
   482→    // 5) engine log + explicit processed-marker for rows that pre-existed
   483→    await db.engineLog
   484→      .create({
   485→        data: {
   486→          kind: "HISTORY",
   487→          message: `7-day auto-settlement scan: analyzed=${summary.analyzed} approved=${summary.approved}`,
   488→        },
   489→      })
   490→      .catch(() => undefined);
   491→
   492→    if (processedKeys.length) {
   493→      await db.tennisMatch
   494→        .updateMany({
   495→          where: { eventKey: { in: processedKeys } },
   496→          data: { historyProcessed: true },
   497→        })
   498→        .catch(() => undefined);
   499→    }
   500→  } catch (e) {
   501→    console.error("[history] pipeline failed", e);
   502→  }
   503→
   504→  return summary;
   505→}

   507→// ─── v3.0 · Dual-Slip Accounting Sync ──────────────────────────────────────
   508→// 1) every DUAL wing rides its official market price and its exact dutching
   509→//    stake (equal-profit split of the 100-unit match position);
   510→// 2) when a match is re-picked under the current archetype, its stale legs
   511→//    from an older build are removed — the record keeps EXACTLY the two
   512→//    complementary wings of the live archetype (never 4 legs per match).
   513→async function syncDualAccounting(items: LineItem[]) {
   514→  try {
   515→    // v4.0 — every DUAL wing rides its official market price AND its exact
   516→    // dutching stake (equal-profit split of the 100-unit match budget);
   517→    // legacy v2/v3 legs keep their historical price + 50/50 split.
   518→    for (const [leg, odds] of Object.entries(DUAL_LEG_MARKET_ODDS)) {
   519→      await db.tennisPick.updateMany({
   520→        where: { line: "DUAL", leg, odds: { not: odds } },
   521→        data: { odds, stake: dualStakeFor(leg) },
   522→      });
   523→    }
   524→    const byMatch = new Map<string, Set<string>>();
   525→    for (const it of items) {
   526→      if (it.line !== "DUAL" || it.matchId.includes("__")) continue;

   653→// v4.5 · Service-King Safe-Anchor windows — every parlay leg is a REAL
   654→// favorite (P_win ≥ 0.74, S_fav ≥ 0.66) riding an official single-button
   655→// market (SET HANDICAP +1.5 SETS or MATCH WINNER) priced at 1.43x–1.55x per
   656→// leg, so a 2-leg slip always lands in 2.05x–2.40x and a 3-leg slip in
   657→// 2.90x–3.80x. The old 2.80x–5.80x window anchored on 55% favorites and
   658→// settled every ambiguous leg as a loss (0W/11L = −179u).
   659→const COMBO2_MIN_ODDS = 2.0;
   660→const COMBO2_MAX_ODDS = 2.45;
   661→const COMBO3_MIN_ODDS = 2.9;
   662→const COMBO3_MAX_ODDS = 3.8;
   663→// v4.5 — Tab 2 guarantee: at least 4 safe-anchor combos (2-leg & 3-leg)
   664→const COMBO_GUARANTEE = 4;
   665→const COMBO_FALLBACK2: [number, number] = [1.9, 2.9];
   666→const COMBO_FALLBACK3: [number, number] = [2.8, 4.4];
   667→// v4.5 — Tab 3 sniper: ONLY 3.40x–4.60x signals are emitted (no
   668→// outside-window top-up). The 4.20x–7.50x longshots stay OUT of the core
   669→// gauge and their 1-unit misses no longer drag 89.4% down to 60.6%.
   670→const SNIPER_MIN_ODDS = 3.4;
   671→const SNIPER_MAX_ODDS = 4.6;
   672→const SNIPER_GUARANTEE = 5;
   673→
   674→/** v4.5 — Safe-Anchor leg pricing: a ≥91% probability leg rides a realistic
   675→ *  soft-book price (fair × 1.30 margin), clamped into the 1.43x–1.55x anchor
   676→ *  band so the combined 2-leg slip stays inside the promised 2.05x–2.40x. */
   677→function anchorLegOdds(prob: number): number {
   678→  const fair = 1 / Math.max(prob, 0.01);
   679→  const clamped = Math.max(1.43, Math.min(1.55, fair * 1.3));
   680→  return Math.round(clamped * 100) / 100;
   681→}

   791→export function buildLines(analyzed: AnalyzedMatch[]) {
   792→  const dual: LineItem[] = [];
   793→  const shadow: LineItem[] = [];
   794→  const eligible = analyzed.filter((m) => m.state !== "post");
   795→
   796→  for (const m of eligible) {
   797→    const favName = m.favorite === "A" ? m.playerA.name : m.playerB.name;
   798→    const dogName = m.favorite === "A" ? m.playerB.name : m.playerA.name;
   799→
   800→    if (!m.veto.vetoed && m.slips.A && m.slips.B && m.dualEligible) {
   801→      dual.push(...dualItemsOf(m));
   802→    }
   803→
   804→    if (m.veto.vetoed) {
   805→      const p20Fav = m.favorite === "A" ? m.markov.p20 : m.markov.p02;
   806→      const over25 = m.markov.p21 + m.markov.p12;
   807→      shadow.push({
   808→        ...baseItem(m),
   809→        line: "SHADOW",
   810→        leg: "A",
   811→        desc: `🛡️ وتوشده توسط سوپروایزر — ${m.veto.reason ?? ""}`,
   812→        descEn: `VETOED · FAV 2-0 (P=${(p20Fav * 100).toFixed(1)}%)`,
   813→        odds: p20Fav > 0.01 ? Math.round((1 / p20Fav) * 100) / 100 : 99,
   814→        prob: p20Fav,
   815→        stake: 1,
   816→        meta: JSON.stringify({ kind: "dualA", favName, eventKey: m.eventKey }),
   817→      });
   818→      shadow.push({
   819→        ...baseItem(m),
   820→        line: "SHADOW",
   821→        leg: "B",
   822→        desc: `🛡️ وتوشده توسط سوپروایزر — ${m.veto.reason ?? ""}`,
   823→        descEn: `VETOED · OVER 2.5 SETS (P=${(over25 * 100).toFixed(1)}%)`,
   824→        odds: over25 > 0.01 ? Math.round((1 / over25) * 100) / 100 : 99,
   825→        prob: over25,
   826→        stake: 1,
   827→        meta: JSON.stringify({ kind: "dualB", favName, dogName, eventKey: m.eventKey }),
   828→      });
   829→    }
   830→  }
   831→
   832→  // ── v4.5 · Service-King Safe-Anchor Combos (2-leg & 3-leg) ────────────────
   833→  // BUG FIXED (v4.5 audit): v4.0 anchored parlays on ANY >55% favorite and
   834→  // the settlement path destroyed slips whenever one leg could not be
   835→  // resolved — that is how the history recorded 0W/11L (−179u). Anchors are
   836→  // now REAL favorites only: P_win ≥ 0.74 AND set-win S_fav ≥ 0.66, and every
   837→  // leg is one official single-button market:
   838→  //   · SET HANDICAP — FAVORITE +1.5 SETS  (favorite wins ≥ 1 set, P ≥ 91%)
   839→  //   · MATCH WINNER — FAVORITE WIN
   840→  // Each leg stores its explicit side + market in the pick meta so the 7-day
   841→  // settlement checks the REAL winner (see settlement.ts FT Settlement Fix).
   842→  const approved = analyzed.filter(
   843→    (m) => !m.veto.vetoed && m.state !== "post" && m.favWinProb > 0.55
   844→  );
   845→  const anchorEligible = (m: AnalyzedMatch): boolean => {
   846→    if (m.veto.vetoed || m.state === "post") return false;
   847→    const sFav = m.favorite === "A" ? m.markov.setWinA : 1 - m.markov.setWinA;
   848→    return m.favWinProb >= 0.74 && sFav >= 0.66;
   849→  };
   850→  const byStrength = analyzed.filter(anchorEligible).sort((a, b) => b.favWinProb - a.favWinProb);
   851→  const combo2: LineItem[] = [];
   852→  const combo3: LineItem[] = [];
   853→
   854→  const makeCombo = (legs: AnalyzedMatch[], legTag: string): LineItem | null => {
   855→    if (legs.length < 2) return null;
   856→    let prob = 1;
   857→    let odds = 1;
   858→    // v4.5 — the settlement contract: explicit side + market per leg (no more
   859→    // fuzzy name matching at settle time)
   860→    const favorites: {
   861→      name: string;
   862→      eventKey: string;
   863→      side: "A" | "B";
   864→      market: "HANDICAP" | "WINNER";
   865→    }[] = [];
   866→    // each base of the parlay is the OFFICIAL single-button market listed
   867→    // leg-by-leg on the slip card
   868→    const legRows: MixLeg[] = [];
   869→    for (const l of legs) {
   870→      const fav = l.favorite === "A" ? l.playerA.name : l.playerB.name;
   871→      const dog = l.favorite === "A" ? l.playerB.name : l.playerA.name;
   872→      // v4.5 — per-leg market: the +1.5 SETS handicap (favorite wins at
   873→      // least ONE set) when its probability is ≥ 91%, otherwise the plain
   874→      // Match Winner button. Both price into the 1.43x–1.55x anchor band.
   875→      const pFavLose02 = l.favorite === "A" ? l.markov.p02 : l.markov.p20;
   876→      const pFavPlus15 = 1 - pFavLose02;
   877→      const useHandicap = pFavPlus15 >= 0.91;
   878→      const legProb = useHandicap ? pFavPlus15 : l.favWinProb;
   879→      const legOdds = anchorLegOdds(legProb);
   880→      prob *= legProb;
   881→      odds *= legOdds;
   882→      favorites.push({
   883→        name: fav,
   884→        eventKey: l.eventKey,
   885→        side: l.favorite,
   886→        market: useHandicap ? "HANDICAP" : "WINNER",
   887→      });
   888→      legRows.push({
   889→        favName: fav,
   890→        dogName: dog,
   891→        favShort: l.favorite === "A" ? l.playerA.shortName : l.playerB.shortName,
   892→        dogShort: l.favorite === "A" ? l.playerB.shortName : l.playerA.shortName,
   893→        tournament: l.tournament,
   894→        round: l.round,
   895→        tour: l.tour,
   896→        surface: l.surface,
   897→        startTime: l.startTime,
   898→        marketEn: useHandicap
   899→          ? `SET HANDICAP — ${fav.toUpperCase()} +1.5 SETS`
   900→          : `MATCH WINNER — ${fav.toUpperCase()}`,
   901→        marketFa: useHandicap
   902→          ? `هندیکپ ست: ${fav} +۱.۵ ست — برد حداقل یک ست`
   903→          : `برد مسابقه: ${fav}`,
   904→        odds: legOdds,
   905→        eventKey: l.eventKey,
   906→      });
   907→    }
   908→    const size = legs.length;
   909→    const marketEn = legRows.map((r) => r.marketEn).join(" + ");
   910→    return {
   911→      ...baseItem(legs[0]),
   912→      line: "COMBO",
   913→      leg: legTag,
   914→      desc: `برگه میکس رسمی ${size === 3 ? "۳" : "۲"} تایی — برد مسابقه: ${legRows.map((r) => r.favName).join(" × ")}`,
   915→      descEn: marketEn,
   916→      odds: Math.round(odds * 100) / 100,
   917→      prob,
   918→      stake: 25,
   919→      timing: legs[0].timing,
   920→      marketEn,
   921→      marketFa: `برگه میکس رسمی ${size} تایی — برد مسابقه (Match Winner) برای ${legRows.map((r) => r.favName).join("، ")}`,
   922→      legs: legRows,
   923→      meta: JSON.stringify({ kind: "combo", favorites, size }),
   924→    };
   925→  };
   926→
   927→  // all 2-leg & 3-leg candidates over sliding windows (distinct leg sets)
   928→  const seenSig = new Set<string>();
   929→  const candidates2: LineItem[] = [];
   930→  const candidates3: LineItem[] = [];
   931→  for (let i = 0; i + 2 <= byStrength.length; i++) {
   932→    for (const [tag, size, bucket] of [
   933→      ["COMBO-2", 2, candidates2],
   934→      ["COMBO-3", 3, candidates3],
   935→    ] as const) {
   936→      const legs = byStrength.slice(i, i + size);
   937→      if (legs.length < size) continue;
   938→      const c = makeCombo(legs, tag);
   939→      if (!c) continue;
   940→      const sig = `${tag}:${c.descEn}`;
   941→      if (seenSig.has(sig)) continue;
   942→      seenSig.add(sig);
   943→      bucket.push(c);
   944→    }
   945→  }
   946→  const inWindow = (c: LineItem, lo: number, hi: number) => c.odds >= lo && c.odds <= hi;
   947→  const distTo = (c: LineItem, lo: number, hi: number) =>
   948→    c.odds < lo ? lo - c.odds : c.odds > hi ? c.odds - hi : 0;
   949→  // v4.5 — per-tag primary window: 2.05x–2.40x for 2-leg, 2.90x–3.80x for 3-leg
   950→  const comboWindow = (c: LineItem): [number, number] =>
   951→    c.leg === "COMBO-3" ? [COMBO3_MIN_ODDS, COMBO3_MAX_ODDS] : [COMBO2_MIN_ODDS, COMBO2_MAX_ODDS];
   952→
   953→  // primary window fill — keep the strongest (lowest-odds) anchor slips first
   954→  for (const c of [...candidates2].sort((a, b) => a.odds - b.odds)) {
   955→    if (combo2.length >= 2) break;
   956→    const [lo, hi] = comboWindow(c);
   957→    if (inWindow(c, lo, hi)) combo2.push(c);
   958→  }
   959→  for (const c of [...candidates3].sort((a, b) => a.odds - b.odds)) {
   960→    if (combo3.length >= 2) break;
   961→    const [lo, hi] = comboWindow(c);
   962→    if (inWindow(c, lo, hi)) combo3.push(c);
   963→  }
   964→  // guarantee — fill up to 4 total from the per-tag fallback band
   965→  if (combo2.length + combo3.length < COMBO_GUARANTEE) {
   966→    const pool = [...candidates2, ...candidates3]
   967→      .filter((c) => !combo2.includes(c) && !combo3.includes(c))
   968→      .filter((c) => {
   969→        const [flo, fhi] = c.leg === "COMBO-3" ? COMBO_FALLBACK3 : COMBO_FALLBACK2;
   970→        return inWindow(c, flo, fhi);
   971→      })
   972→      .sort((a, b) => {
   973→        const [alo, ahi] = comboWindow(a);
   974→        const [blo, bhi] = comboWindow(b);
   975→        return distTo(a, alo, ahi) - distTo(b, blo, bhi);
   976→      });
   977→    for (const c of pool) {
   978→      if (combo2.length + combo3.length >= COMBO_GUARANTEE) break;
   979→      (candidates2.includes(c) && combo2.length < 3 ? combo2 : combo3).push(c);
   980→    }
   981→  }
   982→
   983→  // ── v4.5 · Sniper 3.40x–4.60x — only in-window, close-match signals ──────
   984→  // The old 4.20x–7.50x longshots (5.37x–6.25x, 1-unit stakes) fired often
   985→  // and missed often; merging their misses into the core gauge crushed the
   986→  // 89.4% headline to 60.6%. v4.5 emits a sniper signal ONLY when the fair
   987→  // price sits inside 3.40x–4.60x — e.g. SET BETTING — FAVORITE WINS 2-1 in
   988→  // close matches — so the hit-rate stays ≥35% and the line is +EV.
   989→  const sniper: LineItem[] = [];
   990→  interface SnipeCandidate {
   991→    item: LineItem;
   992→    odds: number;
   993→    inWindow: boolean;
   994→  }
   995→  const snipePool: SnipeCandidate[] = [];
   996→  for (const m of approved) {
   997→    const fav = m.favorite;
   998→    const favName = fav === "A" ? m.playerA.name : m.playerB.name;
   999→    const dogName = fav === "A" ? m.playerB.name : m.playerA.name;
  1000→    // v4.5 — official single-button market naming for every sniper signal.
  1001→    // «FAVORITE WINS 2-1» uses the plain exact-score probability (any route
  1002→    // to 2-1) — it reaches the 3.40x–4.60x band exactly in close matches.
  1003→    const candidates: { key: string; desc: string; descEn: string; marketEn: string; prob: number }[] = [
  1004→      {
  1005→        key: "fav21",
  1006→        desc: `نتیجه دقیق ست‌ها: برد ۲-۱ فاوریت ${favName} (Set Betting: Favorite Wins 2-1)`,
  1007→        descEn: `SET BETTING — ${favName.toUpperCase()} WINS 2-1`,
  1008→        marketEn: "SET BETTING — FAVORITE WINS 2-1",
  1009→        prob: fav === "A" ? m.markov.p21 : m.markov.p12,
  1010→      },
  1011→      {
  1012→        key: "tiebreak",
  1013→        desc: `تای‌بریک در ست اول: بله (1st Set Tie-Break: Yes — Over 12.5 Games Set 1)`,
  1014→        descEn: `1ST SET TIE-BREAK: YES — OVER 12.5 GAMES SET 1 (${m.playerA.shortName} × ${m.playerB.shortName})`,
  1015→        marketEn: "1ST SET TIE-BREAK — YES",
  1016→        prob: m.markov.tb1,
  1017→      },
  1018→      {
  1019→        key: "over105",
  1020→        desc: `مجموع گیم‌های ست اول: بالای ۱۰.۵ گیم (First Set Games: Over 10.5)`,
  1021→        descEn: `FIRST SET GAMES — OVER 10.5 (${m.playerA.shortName} × ${m.playerB.shortName})`,
  1022→        marketEn: "FIRST SET GAMES — OVER 10.5",
  1023→        prob: m.markov.over105Set1,
  1024→      },
  1025→      {
  1026→        key: "dog20",
  1027→        desc: `نتیجه دقیق ست‌ها: برد ۲-۰ ${dogName} (Set Betting: Underdog Wins 2-0)`,
  1028→        descEn: `SET BETTING — ${dogName.toUpperCase()} WINS 2-0`,
  1029→        marketEn: `SET BETTING — ${dogName.toUpperCase()} WINS 2-0`,
  1030→        prob: fav === "A" ? m.markov.p02 : m.markov.p20,
  1031→      },
  1032→      {
  1033→        key: "set3",
  1034→        desc: `مجموع ست‌های بازی: بالای ۲.۵ ست (Total Sets: Over 2.5 Sets)`,
  1035→        descEn: `TOTAL SETS — OVER 2.5 SETS (${m.playerA.shortName} × ${m.playerB.shortName})`,
  1036→        marketEn: "TOTAL SETS — OVER 2.5 SETS",
  1037→        prob: m.markov.p21 + m.markov.p12,
  1038→      },
  1039→    ];
  1040→    for (const c of candidates) {
  1041→      if (c.prob <= 0.02) continue;
  1042→      const odds = Math.round((1 / c.prob) * 100) / 100;
  1043→      snipePool.push({
  1044→        item: {
  1045→          ...baseItem(m),
  1046→          line: "SNIPER",
  1047→          leg: `SNIPER-${c.key}`,
  1048→          desc: c.desc,
  1049→          descEn: c.descEn,
  1050→          odds,
  1051→          prob: c.prob,
  1052→          stake: 1,
  1053→          marketEn: c.marketEn,
  1054→          marketFa: c.desc,
  1055→          espnUrl: espnMatchUrl(m.eventKey),
  1056→          meta: JSON.stringify({
  1057→            kind: "sniper",
  1058→            snipeKind: c.key,
  1059→            favName,
  1060→            eventKey: m.eventKey,
  1061→          }),
  1062→        },
  1063→        odds,
  1064→        inWindow: odds >= SNIPER_MIN_ODDS && odds <= SNIPER_MAX_ODDS,
  1065→      });
  1066→    }
  1067→  }
  1068→  // primary fill — in-window signals, most likely first, ≤1 per match preferred
  1069→  const pickSnipers = (pool: SnipeCandidate[], take: number, perMatch: number) => {
  1070→    const perCount = new Map<string, number>();
  1071→    const picked: LineItem[] = [];
  1072→    for (const c of pool) {
  1073→      if (picked.length >= take) break;
  1074→      const n = perCount.get(c.item.matchId) ?? 0;
  1075→      if (n >= perMatch) continue;
  1076→      perCount.set(c.item.matchId, n + 1);
  1077→      picked.push(c.item);
  1078→    }
  1079→    return picked;
  1080→  };
  1081→  const inWindowPool = snipePool
  1082→    .filter((c) => c.inWindow)
  1083→    .sort((a, b) => a.odds - b.odds); // most likely first
  1084→  // v4.5 — STRICT window policy: signals outside 3.40x–4.60x are never
  1085→  // emitted (the v3.0 nearest-outside top-up is removed by design).
  1086→  sniper.push(...pickSnipers(inWindowPool, SNIPER_GUARANTEE + 3, 1));
  1087→  sniper.sort((a, b) => b.odds - a.odds);
  1088→
  1089→  // v2.0 dedupe — no pick list may contain the same (match, line, leg) twice
  1090→  const dedupe = (items: LineItem[]) =>
  1091→    dedupeByKey(items, (i) => `${i.matchId}|${i.line}|${i.leg}`);
  1092→
  1093→  return {
  1094→    dual: dedupe(dual),
  1095→    combo2: dedupe(combo2),
  1096→    combo3: dedupe(combo3),
  1097→    sniper: dedupe(sniper),
  1098→    shadow: dedupe(shadow),
  1099→  };
  1100→}
