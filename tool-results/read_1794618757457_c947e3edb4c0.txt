    63→    63→type Field = { wildcard: boolean; any: Set<number>; step?: number };
    64→    64→
    65→    65→function parseField(raw: string, min: number, max: number): Field {
    66→    66→  const any = new Set<number>();
    67→    67→  let wildcard = false;
    68→    68→  let step: number | undefined;
    69→    69→  for (const chunk of raw.split(",")) {
    70→    70→    if (chunk === "*") {
    71→    71→      wildcard = true;
    72→    72→      for (let i = min; i <= max; i++) any.add(i);
    73→    73→    } else if (chunk.startsWith("*/")) {
    74→    74→      wildcard = true;
    75→    75→      step = Number(chunk.slice(2));
    76→    76→      for (let i = min; i <= max; i += step) any.add(i);
    77→    77→    } else if (chunk.includes("-")) {
    78→    78→      const [a, b] = chunk.split("-").map(Number);
    79→    79→      for (let i = a; i <= b; i++) any.add(i);
    80→    80→    } else {
    81→    81→      any.add(Number(chunk));
    82→    82→    }
    83→    83→  }
    84→    84→  return { wildcard, any, step };
    85→    85→}

    87→    87→interface Cron {
    88→    88→  sec: Field; min: Field; hour: Field; dom: Field; mon: Field; dow: Field;
    89→    89→}

    91→    91→function parseCron(expr: string): Cron {
    92→    92→  const [sec, min, hour, dom, mon, dow] = expr.trim().split(/\s+/);
    93→    93→  return {
    94→    94→    sec: parseField(sec, 0, 59),
    95→    95→    min: parseField(min, 0, 59),
    96→    96→    hour: parseField(hour, 0, 23),
    97→    97→    dom: parseField(dom, 1, 31),
    98→    98→    mon: parseField(mon, 1, 12),
    99→    99→    dow: parseField(dow, 0, 6),
   100→   100→  };
   101→   101→}

   103→   103→function cronMatches(c: Cron, f: ReturnType<typeof tehranFields>): boolean {
   104→   104→  return (
   105→   105→    c.sec.any.has(f.s) &&
   106→   106→    c.min.any.has(f.min) &&
   107→   107→    c.hour.any.has(f.h) &&
   108→   108→    c.mon.any.has(f.mon) &&
   109→   109→    (c.dom.wildcard || c.dow.wildcard
   110→   110→      ? c.dom.any.has(f.dom) || c.dow.any.has(f.dow)
   111→   111→      : c.dom.any.has(f.dom) && c.dow.any.has(f.dow))
   112→   112→  );
   113→   113→}

   115→   115→/** next run strictly after `after` (ms epoch) for a Tehran-anchored cron */
   116→   116→export function nextCronRun(expr: string, after = Date.now()): number {
   117→   117→  const c = parseCron(expr);
   118→   118→  // iterate the next 2 days at second resolution starting from `after + 1s`
   119→   119→  let t = after + 1000;
   120→   120→  const limit = after + 2 * 86_400_000;
   121→   121→  while (t <= limit) {
   122→   122→    if (cronMatches(c, tehranFields(t))) return t;
   123→   123→    // jump smartly: skip to next minute boundary when seconds won't match soon
   124→   124→    const f = tehranFields(t);
   125→   125→    if (!c.hour.any.has(f.h) || !c.min.any.has(f.min)) {
   126→   126→      t += (59 - f.s) * 1000 + 1000; // finish current minute
   127→   127→      if (!c.hour.any.has(f.h)) t += (59 - f.min) * 60_000; // finish current hour
   128→   128→    } else {
   129→   129→      t += 1000;
   130→   130→    }
   131→   131→  }
   132→   132→  return after + 60_000;
   133→   133→}

   790→   790→async function runJob(jobId: JobId, reason: "cron" | "manual" = "cron"): Promise<PostOutcome> {
   791→   791→  if (running.has(jobId)) return { status: "SKIPPED", error: "already running" };
   792→   792→  running.add(jobId);
   793→   793→  try {
   794→   794→    const out = await handlers[jobId]();
   795→   795→    store.recordRun(jobId, out.status, { error: out.error, messageId: out.messageId });
   796→   796→    return out;
   797→   797→  } catch (e) {
   798→   798→    const error = e instanceof Error ? e.message : String(e);
   799→   799→    console.error(`[job:${jobId}] ${reason} error:`, error);
   800→   800→    store.recordRun(jobId, "ERROR", { error });
   801→   801→    return { status: "ERROR", error };
   802→   802→  } finally {
   803→   803→    running.delete(jobId);
   804→   804→  }
   805→   805→}

   807→   807→function scheduleAll(): void {
   808→   808→  for (const job of JOB_SPECS) {
   809→   809→    if (job.cron) {
   810→   810→      nextRunAt.set(job.id, nextCronRun(job.cron));
   811→   811→    } else if (job.everySeconds) {
   812→   812→      nextRunAt.set(job.id, Date.now() + (job.everySeconds * 1000));
   813→   813→    }
   814→   814→  }
   815→   815→}

   817→   817→function tick(): void {
   818→   818→  const now = Date.now();
   819→   819→  for (const job of JOB_SPECS) {
   820→   820→    const next = nextRunAt.get(job.id);
   821→   821→    if (next === undefined || now < next) continue;
   822→   822→    if (job.cron) {
   823→   823→      nextRunAt.set(job.id, nextCronRun(job.cron, now));
   824→   824→    } else if (job.everySeconds) {
   825→   825→      nextRunAt.set(job.id, now + job.everySeconds * 1000);
   826→   826→    }
   827→   827→    void runJob(job.id);
   828→   828→  }
   829→   829→}

   870→   870→      const isChannelish = ch.type === "channel" || ch.type === "supergroup";
   871→   871→      if (isChannelish && ch.id) {
   872→   872→        store.kvSet(`chatSeen:${ch.id}`, JSON.stringify(ch));
   873→   873→      }
   874→   874→      if (isChannelish && !isPublic && ch.id) {

   880→   880→        }
   881→   881→      }
   882→   882→    }
   883→   883→  } catch {

   885→   885→  }
