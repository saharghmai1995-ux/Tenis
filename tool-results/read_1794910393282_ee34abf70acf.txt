    49→/** Tehran calendar day key (YYYYMMDD) of an ISO timestamp */
    50→function tehranDayKey(iso: string): string {
    51→  const [d, m, y] = TEHRAN_DATE_FMT.format(new Date(iso)).split("/");
    52→  return `${y}${m}${d}`;
    53→}

    55→/** Iranian week key (weeks start SATURDAY in Tehran) + label */
    56→function tehranWeekKey(iso: string): { key: string; label: string } {
    57→  const [dStr, mStr, yStr] = TEHRAN_DATE_FMT.format(new Date(iso)).split("/");
    58→  const d = new Date(Number(yStr), Number(mStr) - 1, Number(dStr));
    59→  // JS dow: 6=Saturday → shift so Saturday = 0 (Sat=0, Sun=1, … Fri=6)
    60→  const offset = (d.getDay() + 1) % 7;
    61→  d.setDate(d.getDate() - offset); // back to the Saturday week-start
    62→  const [sd, sm, sy] = TEHRAN_DATE_FMT.format(d).split("/");
    63→  return { key: `${sy}${sm}${sd}`, label: `هفته ${sd}/${sm}` };
    64→}

    71→function bucketize(
    72→  points: BankPoint[],
    73→  keyOf: (iso: string) => { key: string; label: string }
    74→): BankPeriodBucket[] {
    75→  const order: string[] = [];
    76→  const map = new Map<string, BankPeriodBucket>();
    77→  for (const p of points) {
    78→    if (!p.at || p.index === 0) continue; // skip the START point
    79→    const { key, label } = keyOf(p.at);
    80→    let b = map.get(key);
    81→    if (!b) {
    82→      b = { period: key, label, pnl: 0, wins: 0, losses: 0, matches: 0, winRate: 0, bankEnd: 0, staked: 0, roi: 0 };
    83→      map.set(key, b);
    84→      order.push(key);
    85→    }
    86→    b.pnl += p.delta ?? 0;
    87→    b.matches += 1;
    88→    if (p.status === "WON") b.wins += 1;
    89→    else if (p.status === "LOST") b.losses += 1;
    90→    b.bankEnd = p.bank;
    91→  }
    92→  return order.map((k) => {
    93→    const b = map.get(k)!;
    94→    b.pnl = Math.round(b.pnl * 100) / 100;
    95→    const decided = b.wins + b.losses;
    96→    b.winRate = decided > 0 ? b.wins / decided : 0;
    97→    b.roi = b.staked > 0 ? b.pnl / b.staked : b.pnl > 0 ? 1 : b.pnl < 0 ? -1 : 0;
    98→    return b;
    99→  });
   100→}

   104→function buildBankHistory(bankCurve: BankPoint[], chronological: LedgerMatchRow[]): BankHistory {
   105→  const daily = bucketize(bankCurve, (iso) => {
   106→    const key = tehranDayKey(iso);
   107→    return { key, label: `${key.slice(6, 8)}/${key.slice(4, 6)}` };
   108→  });
   109→  // stake totals per daily bucket from chronological rows
   110→  const dayKeyOfRow = (r: LedgerMatchRow) => tehranDayKey(r.settledAt ?? r.createdAt);
   111→  const dayStakes = new Map<string, number>();
   112→  for (const r of chronological) {
   113→    const k = dayKeyOfRow(r);
   114→    dayStakes.set(k, (dayStakes.get(k) ?? 0) + r.stake);
   115→  }
   116→  for (const b of daily) b.staked = Math.round((dayStakes.get(b.period) ?? 0) * 100) / 100;
   117→
   118→  const weekly = bucketize(bankCurve, (iso) => tehranWeekKey(iso));
   119→  const weekStakes = new Map<string, number>();
   120→  for (const r of chronological) {
   121→    const { key } = tehranWeekKey(r.settledAt ?? r.createdAt);
   122→    weekStakes.set(key, (weekStakes.get(key) ?? 0) + r.stake);
   123→  }
   124→  for (const b of weekly) b.staked = Math.round((weekStakes.get(b.period) ?? 0) * 100) / 100;
   125→
   126→  const monthly = bucketize(bankCurve, (iso) => {
   127→    const key = tehranDayKey(iso);
   128→    const mk = `${key.slice(0, 6)}`;
   129→    return { key: mk, label: `${FA_MONTHS[Number(key.slice(4, 6)) - 1] ?? mk} ${key.slice(0, 4)}` };
   130→  });
   131→  const monthStakes = new Map<string, number>();
   132→  for (const r of chronological) {
   133→    const k = tehranDayKey(r.settledAt ?? r.createdAt).slice(0, 6);
   134→    monthStakes.set(k, (monthStakes.get(k) ?? 0) + r.stake);
   135→  }
   136→  for (const b of monthly) b.staked = Math.round((monthStakes.get(b.period) ?? 0) * 100) / 100;
   137→
   138→  const decidedDaily = daily.filter((b) => b.matches > 0);
   139→  const bestDay = [...decidedDaily].sort((a, b) => b.pnl - a.pnl)[0];
   140→  const worstDay = [...decidedDaily].sort((a, b) => a.pnl - b.pnl)[0];
   141→
   142→  const todayKey = tehranDayKey(new Date().toISOString());
   143→  const todayRows = chronological.filter((r) => dayKeyOfRow(r) === todayKey);
   144→  const todayWins = todayRows.filter((r) => r.status === "WON").length;
   145→  const todayLosses = todayRows.filter((r) => r.status === "LOST").length;
   146→  const todayMatches = todayRows.length;
   147→
   148→  // peak & max drawdown over the full point series
   149→  let peak: { bank: number; at: string } = { bank: BANK_START, at: bankCurve[0]?.at ?? new Date().toISOString() };
   150→  let peakBank = BANK_START;
   151→  let maxDd = { units: 0, pct: 0, from: "", to: "" };
   152→  let ddPeakAt = bankCurve[0]?.at ?? new Date().toISOString();
   153→  for (const p of bankCurve) {
   154→    if (p.bank >= peakBank) {
   155→      peakBank = p.bank;
   156→      peak = { bank: p.bank, at: p.at ?? ddPeakAt };
   157→      ddPeakAt = p.at ?? ddPeakAt;
   158→    } else {
   159→      const ddUnits = Math.round((peakBank - p.bank) * 100) / 100;
   160→      const ddPct = peakBank > 0 ? Math.round((ddUnits / peakBank) * 10000) / 100 : 0;
   161→      if (ddUnits > maxDd.units) {
   162→        maxDd = { units: ddUnits, pct: ddPct, from: ddPeakAt, to: p.at ?? ddPeakAt };
   163→      }
   164→    }
   165→  }
   166→
   167→  return {
   168→    daily,
   169→    weekly,
   170→    monthly,
   171→    bestDay,
   172→    worstDay,
   173→    today: {
   174→      matches: todayMatches,
   175→      wins: todayWins,
   176→      losses: todayLosses,
   177→      pnl: Math.round(todayRows.reduce((s, r) => s + r.profit, 0) * 100) / 100,
   178→      winRate: todayWins + todayLosses > 0 ? todayWins / (todayWins + todayLosses) : 0,
   179→    },
   180→    peak,
   181→    maxDrawdown: maxDd,
   182→  };
   183→}

   316→/**
   317→ * v5.5 — the 100-unit stake law (master prompt §1, mandatory in ledger.ts).
   318→ * Idempotent migration that rewrites every legacy stake in the database:
   319→ *   1) single-leg cards (BANKER 60u / mix 25u / accumulator 25u / shadow 1u)
   320→ *      → exactly STANDARD_MATCH_STAKE (100u);
   321→ *   2) multi-wing positions (DUAL / REGCOMBO / CLUSTER / SNIPER) → every
   322→ *      match group rescaled proportionally so the wings sum EXACTLY to 100u
   323→ *      (silver-era 50u halves and legacy 50/50 splits included);
   324→ *   3) 1u / 2u naked sniper stakes of the bleeding era are gone by the same
   325→ *      rescale — no ledger row can ever carry them again.
   326→ * Wing profits are re-derived from (odds × stake) at read time, so the
   327→ * accounting identity stays intact after the rewrite.
   328→ */
   329→export async function normalizeStakeUnits(): Promise<void> {
   330→  try {
   331→    await db.tennisPick.updateMany({
   332→      where: {
   333→        line: { in: ["BANKER", "SINGLE", "COMBO", "ACCUMULATOR", "SHADOW"] },
   334→        stake: { not: STANDARD_MATCH_STAKE },
   335→      },
   336→      data: { stake: STANDARD_MATCH_STAKE },
   337→    });
   338→
   339→    const legs = await db.tennisPick.findMany({
   340→      where: { line: { in: ["DUAL", "REGCOMBO", "CLUSTER", "SNIPER"] } },
   341→      select: { id: true, matchId: true, line: true, stake: true },
   342→    });
   343→    const groups = new Map<string, { id: string; stake: number }[]>();
   344→    for (const l of legs) {
   345→      const key = `${l.line}|${l.matchId}`;
   346→      const arr = groups.get(key) ?? [];
   347→      arr.push({ id: l.id, stake: l.stake });
   348→      groups.set(key, arr);
   349→    }
   350→    for (const group of groups.values()) {
   351→      if (group.length === 0) continue;
   352→      const sum = group.reduce((s, g) => s + g.stake, 0);
   353→      if (sum <= 0) continue;
   354→      const scaled = group.map((g) => ({
   355→        id: g.id,
   356→        stake: Math.round((g.stake / sum) * STANDARD_MATCH_STAKE * 10) / 10,
   357→      }));
   358→      const drift =
   359→        Math.round((STANDARD_MATCH_STAKE - scaled.reduce((s, g) => s + g.stake, 0)) * 10) / 10;
   360→      const biggest = scaled.reduce((a, b) => (b.stake > a.stake ? b : a));
   361→      biggest.stake = Math.round((biggest.stake + drift) * 10) / 10;
   362→      for (const g of scaled) {
   363→        const before = group.find((x) => x.id === g.id)!.stake;
   364→        if (Math.abs(g.stake - before) < 0.05) continue;
   365→        await db.tennisPick.update({ where: { id: g.id }, data: { stake: g.stake } });
   366→      }
   367→    }
   368→  } catch (e) {
   369→    console.error("[ledger] 100-unit stake normalization failed", e);
   370→  }
   371→}

   373→export async function buildLedgerPayload(
   374→  quota?: QuotaTelemetry
   375→): Promise<LedgerPayload> {
   376→  const picks = (await db.tennisPick.findMany({
   377→    include: { match: true },
   378→    orderBy: { createdAt: "asc" },
   379→  })) as unknown as PickWithMatch[];
   380→
   381→  // de-id dupes defensively
   382→  const byId = new Map<string, PickWithMatch>();
   383→  for (const p of picks) if (!byId.has(p.id)) byId.set(p.id, p);
   384→  const all = [...byId.values()];
   385→
   386→  // ── v3.0 · group the DUAL hedge into ONE match row ────────────────────────
   387→  interface DualAgg {
   388→    matchId: string;
   389→    legs: LedgerLeg[];
   390→    match: PickWithMatch["match"];
   391→    meta: PickMeta;
   392→    stake: number;
   393→    anyPending: boolean;
   394→    winner?: LedgerLeg;
   395→    settledAt: string;
   396→    createdAt: string;
   397→  }
   398→  const dual = new Map<string, DualAgg>();
   399→  const cluster = new Map<DualAgg["matchId"], DualAgg>();
   400→  // v5.5 — REGCOMBO (main + cover) and the packaged SNIPER position
   401→  // (main + capital-protection cover) also group into ONE match row so the
   402→  // SHA-256 record keeps exactly one row per match across ALL regimes.
   403→  const regCombo = new Map<DualAgg["matchId"], DualAgg>();
   404→  const sniperAgg = new Map<DualAgg["matchId"], DualAgg>();
   405→  const singles: { p: PickWithMatch; leg: LedgerLeg }[] = [];
   406→
   407→  const aggFor = (map: Map<string, DualAgg>, p: PickWithMatch): DualAgg => {
   408→    let agg = map.get(p.matchId);
   409→    if (!agg) {
   410→      agg = {
   411→        matchId: p.matchId,
   412→        legs: [],
   413→        match: p.match,
   414→        meta: parseMeta(p.meta),
   415→        stake: 0,
   416→        anyPending: false,
   417→        settledAt: (p.settledAt ?? p.createdAt).toISOString(),
   418→        createdAt: p.createdAt.toISOString(),
   419→      };
   420→      map.set(p.matchId, agg);
   421→    }
   422→    return agg;
   423→  };
   424→
   425→  const pushAgg = (map: Map<string, DualAgg>, p: PickWithMatch, leg: LedgerLeg) => {
   426→    const agg = aggFor(map, p);
   427→    agg.legs.push(leg);
   428→    agg.stake += leg.stake;
   429→    if (leg.status === "PENDING") agg.anyPending = true;
   430→    if (leg.status === "WON") agg.winner = leg;
   431→    const t = (p.settledAt ?? p.createdAt).toISOString();
   432→    if (t > agg.settledAt) agg.settledAt = t;
   433→  };
   434→
   435→  for (const p of all) {
   436→    const leg = buildLeg(p);
   437→    if (p.line === "DUAL") {
   438→      const agg = aggFor(dual, p);
   439→      // prefer the meta of wing A (carries the archetype lock)
   440→      if (leg.leg.startsWith("A") && !agg.meta.arch && parseMeta(p.meta).arch) {
   441→        agg.meta = parseMeta(p.meta);
   442→      }
   443→      pushAgg(dual, p, leg);
   444→      dual.set(p.matchId, agg);
   445→    } else if (p.line === "REGCOMBO") {
   446→      pushAgg(regCombo, p, leg);
   447→    } else if (p.line === "SNIPER") {
   448→      pushAgg(sniperAgg, p, leg);
   449→    } else if (p.line === "CLUSTER") {
   450→      pushAgg(cluster, p, leg);
   451→    } else if (p.line === "COMBO" || p.line === "BANKER" || p.line === "ACCUMULATOR" || p.line === "SINGLE") {
   452→      // v4.5 — VOID rows (buggy-era settlements) stay out of the official record
   453→      // v7.0 — the DEEP SINGLE line is a single-leg 100-unit position
   454→      if (p.status !== "VOID") singles.push({ p, leg });
   455→    }
   456→    // SHADOW rows stay in the disjoint shadow ledger — never in the record
   457→  }
   458→
   459→  // v5.0 — one shared mapper for match-level multi-wing positions (DUAL 2
   460→  // wings, CLUSTER 3 dutching scenarios)
   461→  const aggRowOf = (a: DualAgg, line: ProductionLine): LedgerMatchRow => {
   462→    const decided = !a.anyPending && a.legs.every((l) => l.status !== "PENDING");
   463→    const status: LedgerMatchRow["status"] = !decided
   464→      ? "PENDING"
   465→      : a.winner
   466→        ? "WON"
   467→        : "LOST";
   468→    // doc formula: netMatchProfit = Σ(winningSlipStake × winningSlipOdds) − total stake
   469→    const wonReturn = a.legs
   470→      .filter((l) => l.status === "WON")
   471→      .reduce((s, l) => s + l.odds * l.stake, 0);
   472→    const profit = !decided ? 0 : wonReturn > 0 ? wonReturn - a.stake : -a.stake;
   473→    const archCode = a.meta.arch;
   474→    const fallback = archCode ? ARCH_FALLBACK[archCode] : undefined;
   475→    return {
   476→      id: `${line}:${a.matchId}`,
   477→      matchId: a.matchId,
   478→      line,
   479→      matchLabel: `${a.match.playerA} × ${a.match.playerB}`,
   480→      tournament: a.match.tournament,
   481→      tour: a.match.tour as Tour,
   482→      surface: a.match.surface as Surface,
   483→      round: a.match.round,
   484→      archetype:
   485→        archCode && fallback
   486→          ? {
   487→              code: archCode as "ARCH1" | "ARCH2" | "ARCH3",
   488→              labelFa: a.meta.archLabelFa ?? fallback.labelFa,
   489→              labelEn: a.meta.archLabelEn ?? fallback.labelEn,
   490→              coverage: a.meta.archCoverage ?? 0,
   491→              adaptive: a.meta.archAdaptive,
   492→            }
   493→          : undefined,
   494→      legs: a.legs.sort((x, y) => x.label.localeCompare(y.label)),
   495→      winnerLeg: a.winner?.leg,
   496→      status,
   497→      stake: a.stake,
   498→      profit: Math.round(profit * 100) / 100,
   499→      result: a.match.resultCode ?? undefined,
   500→      setsA: a.match.resultSetsA ?? undefined,
   501→      setsB: a.match.resultSetsB ?? undefined,
   502→      hash: a.match.hashPre ?? undefined,
   503→      espnUrl: espnUrlOf(a.match.eventKey),
   504→      settledAt: decided ? a.settledAt : undefined,
   505→      createdAt: a.createdAt,
   506→    };
   507→  };
   508→
   509→  const dualRows: LedgerMatchRow[] = [...dual.values()].map((a) => aggRowOf(a, "DUAL"));
   510→  const clusterRows: LedgerMatchRow[] = [...cluster.values()].map((a) => aggRowOf(a, "CLUSTER"));
   511→  const regComboRows: LedgerMatchRow[] = [...regCombo.values()].map((a) => aggRowOf(a, "REGCOMBO"));
   512→  const sniperRows: LedgerMatchRow[] = [...sniperAgg.values()].map((a) => aggRowOf(a, "SNIPER"));
   513→
   514→  const singleRows: LedgerMatchRow[] = singles.map(({ p, leg }) => {
   515→    const decided = leg.status === "WON" || leg.status === "LOST";
   516→    return {
   517→      id: `${p.line}:${p.id}`,
   518→      matchId: p.matchId,
   519→      line: p.line as ProductionLine,
   520→      matchLabel: `${p.match.playerA} × ${p.match.playerB}`,
   521→      tournament: p.match.tournament,
   522→      tour: p.match.tour as Tour,
   523→      surface: p.match.surface as Surface,
   524→      round: p.match.round,
   525→      legs: [leg],
   526→      winnerLeg: leg.status === "WON" ? leg.leg : undefined,
   527→      status: decided ? (leg.status as "WON" | "LOST") : "PENDING",
   528→      stake: leg.stake,
   529→      profit: Math.round(leg.profit * 100) / 100,
   530→      result: p.match.resultCode ?? undefined,
   531→      setsA: p.match.resultSetsA ?? undefined,
   532→      setsB: p.match.resultSetsB ?? undefined,
   533→      hash: p.match.hashPre ?? undefined,
   534→      espnUrl: espnUrlOf(p.match.eventKey),
   535→      settledAt: (p.settledAt ?? undefined)?.toISOString(),
   536→      createdAt: p.createdAt.toISOString(),
   537→    };
   538→  });
   539→
   540→  // latest first, bounded
   541→  const rows = [...dualRows, ...clusterRows, ...regComboRows, ...sniperRows, ...singleRows]
   542→    .sort(
   543→      (a, b) =>
   544→        (b.settledAt ?? b.createdAt).localeCompare(a.settledAt ?? a.createdAt)
   545→    )
   546→    .slice(0, 80);
   547→
   548→  // ── disjoint line stats (DUAL at MATCH level, others per pick) ────────────
   549→  const lineStatsOf = (line: ProductionLine, rs: LedgerMatchRow[]): LineStats => {
   550→    const won = rs.filter((r) => r.status === "WON").length;
   551→    const lost = rs.filter((r) => r.status === "LOST").length;
   552→    const pending = rs.filter((r) => r.status === "PENDING").length;
   553→    const profit = rs.reduce((s, r) => s + r.profit, 0);
   554→    const stake = rs.reduce((s, r) => s + r.stake, 0);
   555→    const decided = won + lost;
   556→    return {
   557→      line,
   558→      picks: rs.length,
   559→      won,
   560→      lost,
   561→      pending,
   562→      winRate: decided > 0 ? won / decided : 0,
   563→      profit: Math.round(profit * 100) / 100,
   564→      roi: stake > 0 ? profit / stake : 0,
   565→    };
   566→  };
   567→
   568→  const groups: LineStats[] = [
   569→    lineStatsOf("DUAL", dualRows),
   570→    lineStatsOf("BANKER", singleRows.filter((r) => r.line === "BANKER")),
   571→    lineStatsOf("SINGLE", singleRows.filter((r) => r.line === "SINGLE")),
   572→    lineStatsOf("REGCOMBO", regComboRows),
   573→    lineStatsOf("CLUSTER", clusterRows),
   574→    lineStatsOf("ACCUMULATOR", singleRows.filter((r) => r.line === "ACCUMULATOR")),
   575→    lineStatsOf("COMBO", singleRows.filter((r) => r.line === "COMBO")),
   576→    lineStatsOf("SNIPER", sniperRows),
   577→    lineStatsOf(
   578→      "SHADOW",
   579→      all
   580→        .filter((p) => p.line === "SHADOW")
   581→        .map((p) => ({
   582→          id: `SHADOW:${p.id}`,
   583→          matchId: p.matchId,
   584→          line: "SHADOW" as ProductionLine,
   585→          matchLabel: `${p.match.playerA} × ${p.match.playerB}`,
   586→          tournament: p.match.tournament,
   587→          tour: p.match.tour as Tour,
   588→          surface: p.match.surface as Surface,
   589→          legs: [buildLeg(p)],
   590→          status: (p.status === "WON" ? "WON" : p.status === "LOST" ? "LOST" : "PENDING") as
   591→            | "PENDING"
   592→            | "WON"
   593→            | "LOST",
   594→          stake: p.stake,
   595→          profit: Math.round(wingProfitOf(p.status, p.odds, p.stake) * 100) / 100,
   596→          createdAt: p.createdAt.toISOString(),
   597→        }))
   598→    ),
   599→  ];
   600→
   601→  // ── combined stats (official record only — shadow excluded, match level) ──
   602→  const official = [...dualRows, ...clusterRows, ...regComboRows, ...sniperRows, ...singleRows];
   603→  const decidedRows = official.filter((r) => r.status !== "PENDING");
   604→  const combinedWonCount = decidedRows.filter((r) => r.status === "WON").length;
   605→  const combinedProfit = decidedRows.reduce((s, r) => s + r.profit, 0);
   606→  const combinedStake = decidedRows.reduce((s, r) => s + r.stake, 0);
   607→
   608→  // v4.5 — Core System Win Rate: the headline gauge shows the CORE lines only
   609→  // (complementary golden pairs + safe service-king mixes). The high-odds
   610→  // sniper line keeps its own disjoint stats and is never merged into the
   611→  // core number — its 1-unit misses used to drag 89.4% down to 60.6%.
   612→  const coreRows = [
   613→    ...dualRows,
   614→    ...singleRows.filter(
   615→      (r) =>
   616→        r.line === "COMBO" ||
   617→        r.line === "ACCUMULATOR" ||
   618→        r.line === "BANKER" ||
   619→        // v7.0 — the DEEP SINGLE line is a core production line
   620→        r.line === "SINGLE"
   621→    ),
   622→    ...regComboRows,
   623→    ...clusterRows,
   624→  ];
   625→  const coreDecided = coreRows.filter((r) => r.status !== "PENDING");
   626→  const coreWon = coreDecided.filter((r) => r.status === "WON").length;
   627→  // v4.5 — stake-weighted alternative: Σ won stake / Σ decided stake over the
   628→  // whole official record
   629→  const wonStake = decidedRows
   630→    .filter((r) => r.status === "WON")
   631→    .reduce((s, r) => s + r.stake, 0);
   632→
   633→  // bank curve over decided rows (chronological, match-level P&L for DUAL)
   634→  // v6.0 — every point now carries its settle timestamp, outcome and delta so
   635→  // the dashboard can bucket the curve into daily/weekly/monthly analytics.
   636→  const chronological = [...decidedRows].sort(
   637→    (a, b) => (a.settledAt ?? a.createdAt).localeCompare(b.settledAt ?? b.createdAt)
   638→  );
   639→  const bankCurve: BankPoint[] = [{ index: 0, bank: BANK_START, label: "START" }];
   640→  let bank = BANK_START;
   641→  chronological.forEach((r, i) => {
   642→    bank += r.profit;
   643→    bankCurve.push({
   644→      index: i + 1,
   645→      bank: Math.round(bank * 100) / 100,
   646→      label: `${r.matchLabel} · ${r.status}`,
   647→      at: r.settledAt ?? r.createdAt,
   648→      status: r.status === "WON" || r.status === "LOST" ? r.status : undefined,
   649→      delta: Math.round(r.profit * 100) / 100,
   650→    });
   651→  });
   652→
   653→  // ── v6.0 · Professional bank-growth analytics (Tehran-anchored buckets) ──
   654→  const bankHistory = buildBankHistory(bankCurve, chronological);
   655→
   656→  // current streak — one result event per match (DUAL collapsed)
   657→  const resultEvents = chronological.map((r) => ({
   658→    status: r.status as "WON" | "LOST",
   659→    at: r.settledAt ?? r.createdAt,
   660→  }));
   661→  let streak = 0;
   662→  for (let i = resultEvents.length - 1; i >= 0; i--) {
   663→    const s = resultEvents[i].status;
   664→    if (i === resultEvents.length - 1) {
   665→      streak = s === "WON" ? 1 : -1;
   666→    } else {
   667→      if (s === "WON" && streak > 0) streak++;
   668→      else if (s === "LOST" && streak < 0) streak--;
   669→      else break;
   670→    }
   671→  }
   672→
   673→  const bestWin = [...official]
   674→    .filter((r) => r.status === "WON")
   675→    .sort((a, b) => b.profit - a.profit)[0];
   676→
   677→  // ── v5.0 · Live Bankroll Audit Inspector ──────────────────────────────────
   678→  // the algebraic accounting box rendered above the SHA-256 record table:
   679→  //   bankStart + Σ dual wins − Σ dual losses + combo + cluster + banker +
   680→  //   regCombo + sniper + accumulator = current bank (every user can sum it
   681→  //   in 3s)
   682→  // v5.6 FIX (master prompt 1877.1u): cluster/sniper profits were computed
   683→  // from `singleRows` which NEVER contains CLUSTER/SNIPER rows (they live in
   684→  // their own aggregated arrays) — the audit printed 0.00 for both and hid a
   685→  // −876.72 residual in `otherProfit`. Every term now comes from its own row
   686→  // set, REGCOMBO is an explicit term, and the identity is exact by
   687→  // construction (otherProfit ≡ 0: right side == real sum of left side).
   688→  const lineProfit = (line: ProductionLine, rs: LedgerMatchRow[]) =>
   689→    Math.round(
   690→      rs.filter((r) => r.line === line).reduce((s, r) => s + r.profit, 0) * 100
   691→    ) / 100;
   692→  const dualWonProfit =
   693→    Math.round(dualRows.filter((r) => r.status === "WON").reduce((s, r) => s + r.profit, 0) * 100) / 100;
   694→  const dualLostStake =
   695→    Math.round(dualRows.filter((r) => r.status === "LOST").reduce((s, r) => s + r.profit, 0) * 100) / 100;
   696→  const comboProfit = lineProfit("COMBO", singleRows);
   697→  const clusterProfit = lineProfit("CLUSTER", clusterRows);
   698→  const bankerProfit = lineProfit("BANKER", singleRows);
   699→  // v7.0 — the DEEP SINGLE line is an explicit audit term
   700→  const singleProfit = lineProfit("SINGLE", singleRows);
   701→  const regComboProfit = lineProfit("REGCOMBO", regComboRows);
   702→  const sniperProfit = lineProfit("SNIPER", sniperRows);
   703→  const accumulatorProfit = lineProfit("ACCUMULATOR", singleRows);
   704→  const bankAudit: BankAudit = {
   705→    bankStart: BANK_START,
   706→    dualWonProfit,
   707→    dualLostStake,
   708→    comboProfit,
   709→    clusterProfit,
   710→    bankerProfit,
   711→    singleProfit,
   712→    regComboProfit,
   713→    sniperProfit,
   714→    accumulatorProfit,
   715→    // exact-by-construction identity: bankStart + every explicit term = bank
   716→    otherProfit:
   717→      Math.round(
   718→        (bank -
   719→          (BANK_START +
   720→            dualWonProfit +
   721→            dualLostStake +
   722→            comboProfit +
   723→            clusterProfit +
   724→            bankerProfit +
   725→            singleProfit +
   726→            regComboProfit +
   727→            sniperProfit +
   728→            accumulatorProfit)) *
   729→          100
   730→      ) / 100,
   731→    bank: Math.round(bank * 100) / 100,
   732→    wonCount: combinedWonCount,
   733→    lostCount: decidedRows.length - combinedWonCount,
   734→    auditHash: "",
   735→  };
   736→  bankAudit.auditHash = sha256(
   737→    `${bankAudit.bankStart}|${bankAudit.dualWonProfit}|${bankAudit.dualLostStake}|${bankAudit.comboProfit}|${bankAudit.clusterProfit}|${bankAudit.bankerProfit}|${bankAudit.singleProfit}|${bankAudit.regComboProfit}|${bankAudit.sniperProfit}|${bankAudit.accumulatorProfit}|${bankAudit.otherProfit}|=${bankAudit.bank}`
   738→  );
   739→
   740→  return {
   741→    stats: {
   742→      combinedWinRate: decidedRows.length > 0 ? combinedWonCount / decidedRows.length : 0,
   743→      // v5.0 — core system win rate (DUAL + BANKER + COMBO + CLUSTER; sniper excluded)
   744→      coreWinRate: coreDecided.length > 0 ? coreWon / coreDecided.length : 0,
   745→      // v4.5 — stake-weighted win rate over the official record
   746→      stakeWeightedWinRate: combinedStake > 0 ? wonStake / combinedStake : 0,
   747→      bank: Math.round(bank * 100) / 100,
   748→      bankStart: BANK_START,

   750→      roi: combinedStake > 0 ? combinedProfit / combinedStake : 0,
   751→      picks: official.length,
   752→      settled: decidedRows.length,
   753→      streak,
   754→      bestWin,
   755→      // ─── v5.0 · Live Bankroll Audit Inspector — exact algebraic identity ────
   756→      // bank = bankStart + Σ(profit of WON positions) + Σ(profit of LOST positions)
   757→      winsGross:

   759→          decidedRows.filter((r) => r.status === "WON").reduce((s, r) => s + r.profit, 0) * 100
   760→        ) / 100,
   761→      lossesGross:

   765→      winsCount: combinedWonCount,
   766→      lossesCount: decidedRows.length - combinedWonCount,
   767→    },
   768→    groups,

   773→    bankAudit,
