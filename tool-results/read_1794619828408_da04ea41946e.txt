   212→   212→/** v4.0 — equal-profit dutching stakes from the two leg prices (sum = 100). */
   213→   213→export function dutchStakes(
   214→   214→  oddsA: number,
   215→   215→  oddsB: number
   216→   216→): { stakeA: number; stakeB: number } {
   217→   217→  const invA = 1 / Math.max(oddsA, 1.01);
   218→   218→  const invB = 1 / Math.max(oddsB, 1.01);
   219→   219→  const stakeA = Math.round(((100 * invA) / (invA + invB)) * 10) / 10;
   220→   220→  const stakeB = Math.round((100 - stakeA) * 10) / 10;
   221→   221→  return { stakeA, stakeB };
   222→   222→}

   244→   244→/**
   245→   245→ * Evaluates the 3 official atomic archetypes and selects the eligible one
   246→   246→ * with the lowest exact simultaneous-loss probability. If none qualifies the
   247→   247→ * match is vetoed to the shadow ledger (no forced approvals).
   248→   248→ */
   249→   249→export function evaluateArchetypes(m: AnalyzedMatch): ArchetypeReport {
   250→   250→  const a = m.markov.arch;
   251→   251→  const sFav = m.favorite === "A" ? m.markov.setWinA : 1 - m.markov.setWinA;
   252→   252→  const holdFav = m.favorite === "A" ? m.markov.gA : m.markov.gB;
   253→   253→  const holdDog = m.favorite === "A" ? m.markov.gB : m.markov.gA;
   254→   254→
   255→   255→  const mkEval = (
   256→   256→    code: "ARCH1" | "ARCH2" | "ARCH3",
   257→   257→    pairLoss: number,
   258→   258→    windowOk: boolean,
   259→   259→    windowWhy: string
   260→   260→  ): ArchetypeEval => {
   261→   261→    const max =
   262→   262→      code === "ARCH1"
   263→   263→        ? ARCH1_MAX_PAIR_LOSS
   264→   264→        : code === "ARCH2"
   265→   265→          ? ARCH2_MAX_PAIR_LOSS
   266→   266→          : ARCH3_MAX_PAIR_LOSS;
   267→   267→    const riskOk = pairLoss <= max;
   268→   268→    const eligible = windowOk && riskOk;
   269→   269→    return {
   270→   270→      code,
   271→   271→      labelFa: ARCH_LABELS[code].fa,
   272→   272→      labelEn: ARCH_LABELS[code].en,
   273→   273→      coverage: 1 - pairLoss,
   274→   274→      pairLoss,
   275→   275→      eligible,
   276→   276→      whyNot: eligible
   277→   277→        ? undefined
   278→   278→        : !windowOk
   279→   279→          ? `${windowWhy} (${code === "ARCH2" ? "پنجره الگو ۲: 54% ≤ S_A < 64%" : code === "ARCH1" ? "پنجره الگو ۱: S_fav ≥ 64%" : "پنجره الگو ۳: Hold ≥ 84%"})`
   280→   280→          : `ریسک باخت همزمان ${(pairLoss * 100).toFixed(1)}% بالاتر از سقف ${(max * 100).toFixed(1)}%`,
   281→   281→    };
   282→   282→  };
   283→   283→
   284→   284→  const arch1 = mkEval(
   285→   285→    "ARCH1",
   286→   286→    a.pairLoss1,
   287→   287→    sFav >= ARCH1_MIN_S,
   288→   288→    `S_fav = ${(sFav * 100).toFixed(1)}% زیر آستانه تسلط 64%`
   289→   289→  );
   290→   290→  const arch2 = mkEval(
   291→   291→    "ARCH2",
   292→   292→    a.pairLoss2,
   293→   293→    m.markov.setWinA >= ARCH2_MIN_S && m.markov.setWinA < ARCH2_MAX_S,
   294→   294→    m.markov.setWinA < ARCH2_MIN_S
   295→   295→      ? `S_A = ${(m.markov.setWinA * 100).toFixed(1)}% زیر کف پنجره رقابتی 54%`
   296→   296→      : `S_A = ${(m.markov.setWinA * 100).toFixed(1)}% بالاتر از سقف پنجره رقابتی 64%`
   297→   297→  );
   298→   298→  const arch3 = mkEval(
   299→   299→    "ARCH3",
   300→   300→    a.pairLoss3,
   301→   301→    holdFav >= ARCH3_MIN_HOLD_FAV && holdDog >= ARCH3_MIN_HOLD_DOG,
   302→   302→    `Hold فاوریت ${(holdFav * 100).toFixed(1)}% / حریف ${(holdDog * 100).toFixed(1)}% — زیر آستانه دوئل سرویس (84%/82%)`
   303→   303→  );
   304→   304→
   305→   305→  const evals = [arch1, arch2, arch3];
   306→   306→  const eligible = evals.filter((e) => e.eligible);
   307→   307→  if (eligible.length === 0) return { evals };
   308→   308→
   309→   309→  // most robust coverage wins the tie
   310→   310→  eligible.sort((x, y) => x.pairLoss - y.pairLoss);
   311→   311→  const win = eligible[0];
   312→   312→  return {
   313→   313→    evals,
   314→   314→    selected: {
   315→   315→      code: win.code,
   316→   316→      labelFa: win.labelFa,
   317→   317→      labelEn: win.labelEn,
   318→   318→      coverage: win.coverage,
   319→   319→      pairLoss: win.pairLoss,
   320→   320→    },
   321→   321→  };
   322→   322→}

   325→   325→export function supervisorGate(m: AnalyzedMatch, selected?: ArchetypeInfo): VetoInfo {
   326→   326→  const rankA = m.playerA.rank;
   327→   327→  const rankB = m.playerB.rank;
   328→   328→  const badRankA = !rankA || rankA > TOP100_LIMIT;
   329→   329→  const badRankB = !rankB || rankB > TOP100_LIMIT;
   330→   330→  if (badRankA || badRankB) {
   331→   331→    return {
   332→   332→      vetoed: true,
   333→   333→      code: "TOP100",
   334→   334→      reason: `فیلتر تاپ-۱۰۰ سوپروایزر — ${badRankA ? `${m.playerA.name}${rankA ? ` (#${rankA})` : " (بدون رتبه)"}` : ""}${badRankA && badRankB ? " و " : ""}${badRankB ? `${m.playerB.name}${rankB ? ` (#${rankB})` : " (بدون رتبه)"}` : ""} خارج از دامنه دقت مدل (Kovalchik 2016)`,
   335→   335→    };
   336→   336→  }
   337→   337→
   338→   338→  if (!selected) {
   339→   339→    const sFav = m.favorite === "A" ? m.markov.setWinA : 1 - m.markov.setWinA;
   340→   340→    const best = [...(m.archEvals ?? [])].sort((x, y) => x.pairLoss - y.pairLoss)[0];
   341→   341→    const bestTxt = best
   342→   342→      ? `کم‌ریسک‌ترین الگو (${best.labelFa}) ریسک ${(best.pairLoss * 100).toFixed(1)}% دارد`
   343→   343→      : "هیچ الگویی ارزیابی نشد";
   344→   344→    return {
   345→   345→      vetoed: true,
   346→   346→      code: "P02_RISK",
   347→   347→      reason: `هیچ الگوی مکملی صلاحیت ندارد — S_fav = ${(sFav * 100).toFixed(1)}% خارج از پنجره الگوها و ${bestTxt}؛ انتقال به دفتر سایه`,
   348→   348→    };
   349→   349→  }
   350→   350→
   351→   351→  return { vetoed: false, code: "NONE" };
   352→   352→}

   510→   510→    pool.length,
   511→   511→    Math.max(QUOTA_MIN, Math.ceil(pool.length * QUOTA_RATIO))

   541→   541→          labelEn: ARCH_LABELS.ARCH1.en,
   542→   542→          coverage: 1 - m.mc.p02,

   548→   548→        m.slips = {};
   549→   549→      }
   550→   550→      approvedCount++;
   551→   551→      telemetry.adaptive++;
   552→   552→    }
   553→   553→  }
   554→   554→
   555→   555→  telemetry.approved = approvedCount;

   598→   598→/** Non-linear Sharpe score = mean(P&L) / std(P&L) across the 10k sims. */
   599→   599→export function sharpeOf(pnl: number[]): number {
   600→   600→  const n = pnl.length;
   601→   601→  if (n === 0) return 0;
   602→   602→  const mean = pnl.reduce((a, b) => a + b, 0) / n;
   603→   603→  const varr = pnl.reduce((a, b) => a + (b - mean) * (b - mean), 0) / n;
   604→   604→  const std = Math.sqrt(varr);
   605→   605→  return std < 1e-9 ? 0 : mean / std;
   606→   606→}

   666→   666→// ─── v5.0 — 3-way dutching for the CLUSTER regime (sum = 100) ──────────────
   667→   667→export function clusterStakes(odds1: number, odds2: number, odds3: number): [number, number, number] {
   668→   668→  const inv = [1 / Math.max(odds1, 1.01), 1 / Math.max(odds2, 1.01), 1 / Math.max(odds3, 1.01)];
   669→   669→  const s = inv[0] + inv[1] + inv[2];
   670→   670→  const raw = inv.map((x) => (100 * x) / s);
   671→   671→  const rounded = raw.map((x) => Math.round(x * 10) / 10);
   672→   672→  const drift = Math.round((100 - (rounded[0] + rounded[1] + rounded[2])) * 10) / 10;
   673→   673→  rounded[1] = Math.round((rounded[1] + drift) * 10) / 10;
   674→   674→  return [rounded[0], rounded[1], rounded[2]];
   675→   675→}

   678→   678→export function monteCarloScore(route: RegimeRoute, mc: MonteCarloSummary): number {
   679→   679→  const cov = Math.max(0, Math.min(1, route.coverage));
   680→   680→  const sh = Math.max(0, Math.min(1, route.sharpe / 0.6));
   681→   681→  const bp = Math.max(0, Math.min(1, mc.bankerProb));
   682→   682→  return Math.round(100 * (0.4 * cov + 0.3 * sh + 0.3 * bp));
   683→   683→}

  1134→  1134→/** Per-unit betting Sharpe of a single-leg position: EV / σ of the P&L. */
  1135→  1135→export function bettingSharpe(prob: number, odds: number): number {
  1136→  1136→  const ev = prob * odds - 1;
  1137→  1137→  const secondMoment = prob * Math.pow(odds - 1, 2) + (1 - prob);
  1138→  1138→  const variance = secondMoment - ev * ev;
  1139→  1139→  return variance > 1e-9 ? ev / Math.sqrt(variance) : 0;
  1140→  1140→}
  1141→  1141→
  1142→  1142→/** Equal-profit dutching stakes for N legs (sum = total). */
  1143→  1143→function dutchN(oddsList: number[], total = 100): number[] {
  1144→  1144→  const invs = oddsList.map((o) => 1 / Math.max(o, 1.01));
  1145→  1145→  const s = invs.reduce((a, b) => a + b, 0);
  1146→  1146→  return invs.map((i) => Math.round(((total * i) / s) * 10) / 10);
  1147→  1147→}
  1148→  1148→
  1149→  1149→/**
  1150→  1150→ * Per-unit Sharpe of a dutched multi-leg position over disjoint outcomes:
  1151→  1151→ * winProbs[i] → leg i pays; else the whole position loses.
  1152→  1152→ */
  1153→  1153→function dutchedSharpe(winProbs: number[], oddsList: number[]): { sharpe: number; ev: number; stakes: number[] } {
  1154→  1154→  const stakes = dutchN(oddsList, 100);
  1155→  1155→  const payoffs = stakes.map((st, i) => (st * oddsList[i] - 100) / 100); // per-unit net
  1156→  1156→  const pLose = Math.max(0, 1 - winProbs.reduce((a, b) => a + b, 0));
  1157→  1157→  const ev = winProbs.reduce((s, p, i) => s + p * payoffs[i], 0) - pLose * 1;
  1158→  1158→  const secondMoment =
  1159→  1159→    winProbs.reduce((s, p, i) => s + p * payoffs[i] * payoffs[i], 0) + pLose;
  1160→  1160→  const variance = secondMoment - ev * ev;
  1161→  1161→  return {
  1162→  1162→    sharpe: variance > 1e-9 ? ev / Math.sqrt(variance) : 0,
  1163→  1163→    ev,
  1164→  1164→    stakes,
  1165→  1165→  };
  1166→  1166→}
