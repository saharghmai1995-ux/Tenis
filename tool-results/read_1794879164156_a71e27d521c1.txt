    64→    64→type Field = { wildcard: boolean; any: Set<number>; step?: number };
    65→    65→
    66→    66→function parseField(raw: string, min: number, max: number): Field {
    67→    67→  const any = new Set<number>();
    68→    68→  let wildcard = false;
    69→    69→  let step: number | undefined;
    70→    70→  for (const chunk of raw.split(",")) {
    71→    71→    if (chunk === "*") {
    72→    72→      wildcard = true;
    73→    73→      for (let i = min; i <= max; i++) any.add(i);
    74→    74→    } else if (chunk.startsWith("*/")) {
    75→    75→      wildcard = true;
    76→    76→      step = Number(chunk.slice(2));
    77→    77→      for (let i = min; i <= max; i += step) any.add(i);
    78→    78→    } else if (chunk.includes("-")) {
    79→    79→      const [a, b] = chunk.split("-").map(Number);
    80→    80→      for (let i = a; i <= b; i++) any.add(i);
    81→    81→    } else {
    82→    82→      any.add(Number(chunk));
    83→    83→    }
    84→    84→  }
    85→    85→  return { wildcard, any, step };
    86→    86→}

    88→    88→interface Cron {
    89→    89→  sec: Field; min: Field; hour: Field; dom: Field; mon: Field; dow: Field;
    90→    90→}

    92→    92→function parseCron(expr: string): Cron {
    93→    93→  const [sec, min, hour, dom, mon, dow] = expr.trim().split(/\s+/);
    94→    94→  return {
    95→    95→    sec: parseField(sec, 0, 59),
    96→    96→    min: parseField(min, 0, 59),
    97→    97→    hour: parseField(hour, 0, 23),
    98→    98→    dom: parseField(dom, 1, 31),
    99→    99→    mon: parseField(mon, 1, 12),
   100→   100→    dow: parseField(dow, 0, 6),
   101→   101→  };
   102→   102→}

   104→   104→function cronMatches(c: Cron, f: ReturnType<typeof tehranFields>): boolean {
   105→   105→  return (
   106→   106→    c.sec.any.has(f.s) &&
   107→   107→    c.min.any.has(f.min) &&
   108→   108→    c.hour.any.has(f.h) &&
   109→   109→    c.mon.any.has(f.mon) &&
   110→   110→    (c.dom.wildcard || c.dow.wildcard
   111→   111→      ? c.dom.any.has(f.dom) || c.dow.any.has(f.dow)
   112→   112→      : c.dom.any.has(f.dom) && c.dow.any.has(f.dow))
   113→   113→  );
   114→   114→}

   116→   116→/** next run strictly after `after` (ms epoch) for a Tehran-anchored cron */
   117→   117→export function nextCronRun(expr: string, after = Date.now()): number {
   118→   118→  const c = parseCron(expr);
   119→   119→  // iterate the next 2 days at second resolution starting from `after + 1s`
   120→   120→  let t = after + 1000;
   121→   121→  const limit = after + 2 * 86_400_000;
   122→   122→  while (t <= limit) {
   123→   123→    if (cronMatches(c, tehranFields(t))) return t;
   124→   124→    // jump smartly: skip to next minute boundary when seconds won't match soon
   125→   125→    const f = tehranFields(t);
   126→   126→    if (!c.hour.any.has(f.h) || !c.min.any.has(f.min)) {
   127→   127→      t += (59 - f.s) * 1000 + 1000; // finish current minute
   128→   128→      if (!c.hour.any.has(f.h)) t += (59 - f.min) * 60_000; // finish current hour
   129→   129→    } else {
   130→   130→      t += 1000;
   131→   131→    }
   132→   132→  }
   133→   133→  return after + 60_000;
   134→   134→}

   824→   824→async function runJob(jobId: JobId, reason: "cron" | "manual" = "cron"): Promise<PostOutcome> {
   825→   825→  if (running.has(jobId)) return { status: "SKIPPED", error: "already running" };
   826→   826→  running.add(jobId);
   827→   827→  try {
   828→   828→    const out = await handlers[jobId]();
   829→   829→    store.recordRun(jobId, out.status, { error: out.error, messageId: out.messageId });
   830→   830→    return out;
   831→   831→  } catch (e) {
   832→   832→    const error = e instanceof Error ? e.message : String(e);
   833→   833→    console.error(`[job:${jobId}] ${reason} error:`, error);
   834→   834→    store.recordRun(jobId, "ERROR", { error });
   835→   835→    return { status: "ERROR", error };
   836→   836→  } finally {
   837→   837→    running.delete(jobId);
   838→   838→  }
   839→   839→}

   841→   841→function scheduleAll(): void {
   842→   842→  for (const job of JOB_SPECS) {
   843→   843→    if (job.cron) {
   844→   844→      nextRunAt.set(job.id, nextCronRun(job.cron));

   847→   847→    }
   848→   848→  }

   851→   851→function tick(): void {
   852→   852→  const now = Date.now();
   853→   853→  for (const job of JOB_SPECS) {
   854→   854→    const next = nextRunAt.get(job.id);
   855→   855→    if (next === undefined || now < next) continue;
   856→   856→    if (job.cron) {
   857→   857→      nextRunAt.set(job.id, nextCronRun(job.cron, now));
   858→   858→    } else if (job.everySeconds) {
   859→   859→      nextRunAt.set(job.id, now + job.everySeconds * 1000);
   860→   860→    }
   861→   861→    void runJob(job.id);
   862→   862→  }
   863→   863→}

   898→   898→/** capture the VIP channel id from any chat the bot can see.
   899→   899→ *  .env PRIORITY LAW: when TELEGRAM_VIP_CHAT_ID is explicitly configured it
   900→   900→ *  is the single source of truth — discovery NEVER overwrites it (the old
   901→   901→ *  bug captured the PUBLIC channel as VIP whenever publicChatId was numeric,
   902→   902→ *  because `publicChatId.includes(username)` can never match a numeric id). */
   903→   903→async function discoverVipLoop(): Promise<void> {
   904→   904→  if (CONFIG.vipChatId) {
   905→   905→    // explicit director config wins — keep the title fresh, never the id
   906→   906→    try {
   907→   907→      for (const ch of await telegram.discoverChats()) {
   908→   908→        if (String(ch.id) === CONFIG.vipChatId && ch.title) {
   909→   909→          store.kvSet("vipChatTitle", ch.title);
   910→   910→        }
   911→   911→      }
   912→   912→    } catch {
   913→   913→      /* best-effort */
   914→   914→    }
   915→   915→    return;
   916→   916→  }
   917→   917→  try {
   918→   918→    const chats = await telegram.discoverChats();
   919→   919→    for (const ch of chats) {
   920→   920→      const isPublic =
   921→   921→        (!!ch.username && CONFIG.publicChatId.includes(ch.username)) ||
   922→   922→        String(ch.id) === CONFIG.publicChatId;
   923→   923→      const isChannelish = ch.type === "channel" || ch.type === "supergroup";
   924→   924→      if (isChannelish && ch.id) {
   925→   925→        store.kvSet(`chatSeen:${ch.id}`, JSON.stringify(ch));
   926→   926→      }
   927→   927→      if (isChannelish && !isPublic && ch.id) {
   928→   928→        const idStr = String(ch.id);
   929→   929→        if (vipChatId() !== idStr) {
   930→   930→          store.kvSet("vipChatId", idStr);
   931→   931→          store.kvSet("vipChatTitle", ch.title ?? "");
   932→   932→          console.log(`[discover] VIP chat captured: ${idStr} (${ch.title ?? ch.type})`);
   933→   933→        }
   934→   934→      }
   935→   935→    }
   936→   936→  } catch {
   937→   937→    /* discovery is best-effort */
   938→   938→  }
   939→   939→}
