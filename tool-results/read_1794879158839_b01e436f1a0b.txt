    40→export function tehranFields(ms = Date.now()): {
    41→  y: number; mon: number; dom: number; h: number; min: number; s: number; dow: number;
    42→} {
    43→  const parts = partsFmt.formatToParts(new Date(ms));
    44→  const get = (t: string) => Number(parts.find((p) => p.type === t)?.value ?? 0);
    45→  const wd = dowFmt.format(new Date(ms)).toUpperCase();
    46→  return {
    47→    y: get("year"),
    48→    mon: get("month"),
    49→    dom: get("day"),
    50→    h: get("hour") % 24,
    51→    min: get("minute"),
    52→    s: get("second"),
    53→    dow: { SUN: 0, MON: 1, TUE: 2, WED: 3, THU: 4, FRI: 5, SAT: 6 }[wd] ?? 0,
    54→  };
    55→}

    62→// ─── 6-field cron matcher ──────────────────────────────────────────────────

    64→type Field = { wildcard: boolean; any: Set<number>; step?: number };
    65→
    66→function parseField(raw: string, min: number, max: number): Field {
    67→  const any = new Set<number>();
    68→  let wildcard = false;
    69→  let step: number | undefined;
    70→  for (const chunk of raw.split(",")) {
    71→    if (chunk === "*") {
    72→      wildcard = true;
    73→      for (let i = min; i <= max; i++) any.add(i);
    74→    } else if (chunk.startsWith("*/")) {
    75→      wildcard = true;
    76→      step = Number(chunk.slice(2));
    77→      for (let i = min; i <= max; i += step) any.add(i);
    78→    } else if (chunk.includes("-")) {
    79→      const [a, b] = chunk.split("-").map(Number);
    80→      for (let i = a; i <= b; i++) any.add(i);
    81→    } else {
    82→      any.add(Number(chunk));
    83→    }
    84→  }
    85→  return { wildcard, any, step };
    86→}
    87→
    88→interface Cron {
    89→  sec: Field; min: Field; hour: Field; dom: Field; mon: Field; dow: Field;
    90→}
    91→
    92→function parseCron(expr: string): Cron {
    93→  const [sec, min, hour, dom, mon, dow] = expr.trim().split(/\s+/);
    94→  return {
    95→    sec: parseField(sec, 0, 59),
    96→    min: parseField(min, 0, 59),
    97→    hour: parseField(hour, 0, 23),
    98→    dom: parseField(dom, 1, 31),
    99→    mon: parseField(mon, 1, 12),
   100→    dow: parseField(dow, 0, 6),
   101→  };
   102→}
   103→
   104→function cronMatches(c: Cron, f: ReturnType<typeof tehranFields>): boolean {
   105→  return (
   106→    c.sec.any.has(f.s) &&
   107→    c.min.any.has(f.min) &&
   108→    c.hour.any.has(f.h) &&
   109→    c.mon.any.has(f.mon) &&
   110→    (c.dom.wildcard || c.dow.wildcard
   111→      ? c.dom.any.has(f.dom) || c.dow.any.has(f.dow)
   112→      : c.dom.any.has(f.dom) && c.dow.any.has(f.dow))
   113→  );
   114→}

   116→/** next run strictly after `after` (ms epoch) for a Tehran-anchored cron */
   117→export function nextCronRun(expr: string, after = Date.now()): number {
   118→  const c = parseCron(expr);
   119→  // iterate the next 2 days at second resolution starting from `after + 1s`
   120→  let t = after + 1000;
   121→  const limit = after + 2 * 86_400_000;
   122→  while (t <= limit) {
   123→    if (cronMatches(c, tehranFields(t))) return t;
   124→    // jump smartly: skip to next minute boundary when seconds won't match soon
   125→    const f = tehranFields(t);
   126→    if (!c.hour.any.has(f.h) || !c.min.any.has(f.min)) {
   127→      t += (59 - f.s) * 1000 + 1000; // finish current minute
   128→      if (!c.hour.any.has(f.h)) t += (59 - f.min) * 60_000; // finish current hour
   129→    } else {
   130→      t += 1000;
   131→    }
   132→  }
   133→  return after + 60_000;
   134→}

   181→  jobId: JobId,
   182→  target: string,
   183→  kind: PosterKind,
   184→  caption: string,
   185→  png: Buffer
   186→): Promise<PostOutcome> {

   198→    console.log(`[job:${jobId}] DRY-RUN ${kind} → ${target} (${Math.round(png.length / 1024)}KB)`);
   199→    return { status: "DRY-RUN" };
   200→  }
   201→  const chatId = target === "vip" ? vipChatId() : CONFIG.publicChatId;
   202→  if (!chatId) {
   203→    store.logPost({
   204→      at: new Date().toISOString(),
   205→      jobId,
   206→      target,
   207→      posterKind: kind,
   208→      mode: "LIVE",
   209→      caption,
   210→      status: "SKIPPED",
   211→      error: "chat id not configured",
   212→    });
   213→    return { status: "SKIPPED", error: "chat id not configured" };
   214→  }
   215→  const res = await telegram.sendPhoto(chatId, png, caption);
   216→  store.logPost({
   217→    at: new Date().toISOString(),
   218→    jobId,
   219→    target,
   220→    posterKind: kind,
   221→    mode: "LIVE",
   222→    messageId: res.messageId,
   223→    caption,
   224→    status: res.ok ? "SENT" : "ERROR",
   225→    error: res.description,
   226→  });
   227→  console.log(
   228→    `[job:${jobId}] ${res.ok ? "SENT" : "ERROR"} ${kind} → ${target}${res.messageId ? ` #${res.messageId}` : ""}${res.description ? ` (${res.description})` : ""}`
   229→  );
   230→  return { status: res.ok ? "SENT" : "ERROR", error: res.description, messageId: res.messageId };
   231→}

   824→async function runJob(jobId: JobId, reason: "cron" | "manual" = "cron"): Promise<PostOutcome> {
   825→  if (running.has(jobId)) return { status: "SKIPPED", error: "already running" };
   826→  running.add(jobId);
   827→  try {
   828→    const out = await handlers[jobId]();
   829→    store.recordRun(jobId, out.status, { error: out.error, messageId: out.messageId });
   830→    return out;
   831→  } catch (e) {
   832→    const error = e instanceof Error ? e.message : String(e);
   833→    console.error(`[job:${jobId}] ${reason} error:`, error);
   834→    store.recordRun(jobId, "ERROR", { error });
   835→    return { status: "ERROR", error };
   836→  } finally {
   837→    running.delete(jobId);
   838→  }
   839→}

   841→function scheduleAll(): void {
   842→  for (const job of JOB_SPECS) {
   843→    if (job.cron) {
   844→      nextRunAt.set(job.id, nextCronRun(job.cron));
   845→    } else if (job.everySeconds) {
   846→      nextRunAt.set(job.id, Date.now() + (job.everySeconds * 1000));
   847→    }
   848→  }
   849→}
   850→
   851→function tick(): void {
   852→  const now = Date.now();
   853→  for (const job of JOB_SPECS) {
   854→    const next = nextRunAt.get(job.id);
   855→    if (next === undefined || now < next) continue;
   856→    if (job.cron) {
   857→      nextRunAt.set(job.id, nextCronRun(job.cron, now));
   858→    } else if (job.everySeconds) {
   859→      nextRunAt.set(job.id, now + job.everySeconds * 1000);
   860→    }
   861→    void runJob(job.id);
   862→  }
   863→}

   865→export function startScheduler(): void {
   866→  registerFonts();
   867→  restoreLive();
   868→  void refreshMe().catch(() => null);
   869→  scheduleAll();
   870→  // warm immediately so the first flashcard is never cold
   871→  void dataHub.warm().then((ok) => {
   872→    if (ok) store.kvSet("lastFeedWarmMs", String(Date.now()));
   873→  });
   874→  setInterval(tick, 1000);
   875→  // engine keep-warm loop — the core must never go cold
   876→  setInterval(() => {
   877→    void ensureWarm();
   878→  }, 5 * 60_000);
   879→  // tennis radar keep-warm (news/live/fitness cache stays hot)
   880→  setInterval(() => {
   881→    void fetchTennisRadar().catch(() => null);
   882→  }, 4 * 60_000);
   883→  // VIP channel auto-discovery — the moment the bot becomes admin of (or
   884→  // receives any update from) the VIP channel, its numeric id is captured.
   885→  setInterval(() => {
   886→    void discoverVipLoop();
   887→  }, 30_000);
   888→  console.log(
   889→    `[scheduler] started - ${JOB_SPECS.length} jobs - live=${isLive()} - tz=${TZ}`
   890→  );
   891→  for (const job of JOB_SPECS) {
   892→    const next = nextRunAt.get(job.id);
   893→    console.log(`[scheduler] ${job.id.padEnd(18)} next=${next ? new Date(next).toISOString() : "-"}`);
   894→  }
   895→  void discoverVipLoop();
   896→}

   898→/** capture the VIP channel id from any chat the bot can see.
   899→ *  .env PRIORITY LAW: when TELEGRAM_VIP_CHAT_ID is explicitly configured it
   900→ *  is the single source of truth — discovery NEVER overwrites it (the old
   901→ *  bug captured the PUBLIC channel as VIP whenever publicChatId was numeric,
   902→ *  because `publicChatId.includes(username)` can never match a numeric id). */
   903→async function discoverVipLoop(): Promise<void> {
   904→  if (CONFIG.vipChatId) {
   905→    // explicit director config wins — keep the title fresh, never the id
   906→    try {
   907→      for (const ch of await telegram.discoverChats()) {
   908→        if (String(ch.id) === CONFIG.vipChatId && ch.title) {
   909→          store.kvSet("vipChatTitle", ch.title);
   910→        }
   911→      }
   912→    } catch {
   913→      /* best-effort */
   914→    }
   915→    return;
   916→  }
   917→  try {
   918→    const chats = await telegram.discoverChats();
   919→    for (const ch of chats) {
   920→      const isPublic =
   921→        (!!ch.username && CONFIG.publicChatId.includes(ch.username)) ||
   922→        String(ch.id) === CONFIG.publicChatId;
   923→      const isChannelish = ch.type === "channel" || ch.type === "supergroup";
   924→      if (isChannelish && ch.id) {
   925→        store.kvSet(`chatSeen:${ch.id}`, JSON.stringify(ch));
   926→      }
   927→      if (isChannelish && !isPublic && ch.id) {
   928→        const idStr = String(ch.id);
   929→        if (vipChatId() !== idStr) {
   930→          store.kvSet("vipChatId", idStr);
   931→          store.kvSet("vipChatTitle", ch.title ?? "");
   932→          console.log(`[discover] VIP chat captured: ${idStr} (${ch.title ?? ch.type})`);
   933→        }
   934→      }
   935→    }
   936→  } catch {
   937→    /* discovery is best-effort */
   938→  }
   939→}

   954→      ...(j as JobSpec),

   962→      runs: st?.runs ?? 0,
   963→    } as JobSpec & JobRunState;
   964→  });
   965→  // program table joined with live job state (dashboard Daily Program panel)
   966→  const jobsById = new Map(jobs.map((j) => [j.id, j]));
   967→  const nowF = tehranFields();

   970→    const j = jobsById.get(p.jobId);

   972→      p.min >= 0 && p.min <= 22 * 60 ? nowMin >= p.min : p.time === ":15" ? nowF.min >= 15 : true;

   974→      time: p.time,
   975→      fa: p.fa,
   976→      en: p.en,
   977→      target: p.target,
