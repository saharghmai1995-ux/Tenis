    39→export function tehranFields(ms = Date.now()): {
    40→  y: number; mon: number; dom: number; h: number; min: number; s: number; dow: number;
    41→} {
    42→  const parts = partsFmt.formatToParts(new Date(ms));
    43→  const get = (t: string) => Number(parts.find((p) => p.type === t)?.value ?? 0);
    44→  const wd = dowFmt.format(new Date(ms)).toUpperCase();
    45→  return {
    46→    y: get("year"),
    47→    mon: get("month"),
    48→    dom: get("day"),
    49→    h: get("hour") % 24,
    50→    min: get("minute"),
    51→    s: get("second"),
    52→    dow: { SUN: 0, MON: 1, TUE: 2, WED: 3, THU: 4, FRI: 5, SAT: 6 }[wd] ?? 0,
    53→  };
    54→}

    61→// ─── 6-field cron matcher ──────────────────────────────────────────────────

    63→type Field = { wildcard: boolean; any: Set<number>; step?: number };
    64→
    65→function parseField(raw: string, min: number, max: number): Field {
    66→  const any = new Set<number>();
    67→  let wildcard = false;
    68→  let step: number | undefined;
    69→  for (const chunk of raw.split(",")) {
    70→    if (chunk === "*") {
    71→      wildcard = true;
    72→      for (let i = min; i <= max; i++) any.add(i);
    73→    } else if (chunk.startsWith("*/")) {
    74→      wildcard = true;
    75→      step = Number(chunk.slice(2));
    76→      for (let i = min; i <= max; i += step) any.add(i);
    77→    } else if (chunk.includes("-")) {
    78→      const [a, b] = chunk.split("-").map(Number);
    79→      for (let i = a; i <= b; i++) any.add(i);
    80→    } else {
    81→      any.add(Number(chunk));
    82→    }
    83→  }
    84→  return { wildcard, any, step };
    85→}
    86→
    87→interface Cron {
    88→  sec: Field; min: Field; hour: Field; dom: Field; mon: Field; dow: Field;
    89→}
    90→
    91→function parseCron(expr: string): Cron {
    92→  const [sec, min, hour, dom, mon, dow] = expr.trim().split(/\s+/);
    93→  return {
    94→    sec: parseField(sec, 0, 59),
    95→    min: parseField(min, 0, 59),
    96→    hour: parseField(hour, 0, 23),
    97→    dom: parseField(dom, 1, 31),
    98→    mon: parseField(mon, 1, 12),
    99→    dow: parseField(dow, 0, 6),
   100→  };
   101→}
   102→
   103→function cronMatches(c: Cron, f: ReturnType<typeof tehranFields>): boolean {
   104→  return (
   105→    c.sec.any.has(f.s) &&
   106→    c.min.any.has(f.min) &&
   107→    c.hour.any.has(f.h) &&
   108→    c.mon.any.has(f.mon) &&
   109→    (c.dom.wildcard || c.dow.wildcard
   110→      ? c.dom.any.has(f.dom) || c.dow.any.has(f.dow)
   111→      : c.dom.any.has(f.dom) && c.dow.any.has(f.dow))
   112→  );
   113→}

   115→/** next run strictly after `after` (ms epoch) for a Tehran-anchored cron */
   116→export function nextCronRun(expr: string, after = Date.now()): number {
   117→  const c = parseCron(expr);
   118→  // iterate the next 2 days at second resolution starting from `after + 1s`
   119→  let t = after + 1000;
   120→  const limit = after + 2 * 86_400_000;
   121→  while (t <= limit) {
   122→    if (cronMatches(c, tehranFields(t))) return t;
   123→    // jump smartly: skip to next minute boundary when seconds won't match soon
   124→    const f = tehranFields(t);
   125→    if (!c.hour.any.has(f.h) || !c.min.any.has(f.min)) {
   126→      t += (59 - f.s) * 1000 + 1000; // finish current minute
   127→      if (!c.hour.any.has(f.h)) t += (59 - f.min) * 60_000; // finish current hour
   128→    } else {
   129→      t += 1000;
   130→    }
   131→  }
   132→  return after + 60_000;
   133→}

   179→async function deliver(
   180→  jobId: JobId,
   181→  target: string,
   182→  kind: PosterKind,
   183→  caption: string,
   184→  png: Buffer
   185→): Promise<PostOutcome> {
   186→  saveOutbox(jobId, png);
   187→  if (!isLive()) {
   188→    store.logPost({
   189→      at: new Date().toISOString(),
   190→      jobId,
   191→      target,
   192→      posterKind: kind,
   193→      mode: "DRY-RUN",
   194→      caption,
   195→      status: "DRY-RUN",
   196→    });
   197→    console.log(`[job:${jobId}] DRY-RUN ${kind} → ${target} (${Math.round(png.length / 1024)}KB)`);
   198→    return { status: "DRY-RUN" };
   199→  }
   200→  const chatId = target === "vip" ? vipChatId() : CONFIG.publicChatId;
   201→  if (!chatId) {
   202→    store.logPost({
   203→      at: new Date().toISOString(),
   204→      jobId,
   205→      target,
   206→      posterKind: kind,
   207→      mode: "LIVE",
   208→      caption,
   209→      status: "SKIPPED",
   210→      error: "chat id not configured",
   211→    });
   212→    return { status: "SKIPPED", error: "chat id not configured" };
   213→  }
   214→  const res = await telegram.sendPhoto(chatId, png, caption);
   215→  store.logPost({
   216→    at: new Date().toISOString(),
   217→    jobId,
   218→    target,
   219→    posterKind: kind,
   220→    mode: "LIVE",
   221→    messageId: res.messageId,
   222→    caption,
   223→    status: res.ok ? "SENT" : "ERROR",
   224→    error: res.description,
   225→  });
   226→  console.log(
   227→    `[job:${jobId}] ${res.ok ? "SENT" : "ERROR"} ${kind} → ${target}${res.messageId ? ` #${res.messageId}` : ""}${res.description ? ` (${res.description})` : ""}`
   228→  );
   229→  return { status: res.ok ? "SENT" : "ERROR", error: res.description, messageId: res.messageId };
   230→}

   612→
   613→async function runJob(jobId: JobId, reason: "cron" | "manual" = "cron"): Promise<PostOutcome> {
   614→  if (running.has(jobId)) return { status: "SKIPPED", error: "already running" };
   615→  running.add(jobId);
   616→  try {
   617→    const out = await handlers[jobId]();
   618→    store.recordRun(jobId, out.status, { error: out.error, messageId: out.messageId });
   619→    return out;
   620→  } catch (e) {
   621→    const error = e instanceof Error ? e.message : String(e);
   622→    console.error(`[job:${jobId}] ${reason} error:`, error);
   623→    store.recordRun(jobId, "ERROR", { error });
   624→    return { status: "ERROR", error };
   625→  } finally {
   626→    running.delete(jobId);
   627→  }
   628→}

   630→function scheduleAll(): void {
   631→  for (const job of JOB_SPECS) {
   632→    if (job.cron) {
   633→      nextRunAt.set(job.id, nextCronRun(job.cron));
   634→    } else if (job.everySeconds) {
   635→      nextRunAt.set(job.id, Date.now() + (job.everySeconds * 1000));
   636→    }
   637→  }
   638→}
   639→
   640→function tick(): void {
   641→  const now = Date.now();
   642→  for (const job of JOB_SPECS) {
   643→    const next = nextRunAt.get(job.id);
   644→    if (next === undefined || now < next) continue;
   645→    if (job.cron) {
   646→      nextRunAt.set(job.id, nextCronRun(job.cron, now));
   647→    } else if (job.everySeconds) {
   648→      nextRunAt.set(job.id, now + job.everySeconds * 1000);
   649→    }
   650→    void runJob(job.id);
   651→  }
   652→}

   654→export function startScheduler(): void {
   655→  registerFonts();
   656→  restoreLive();
   657→  scheduleAll();
   658→  // warm immediately so the first flashcard is never cold
   659→  void dataHub.warm().then((ok) => {
   660→    if (ok) store.kvSet("lastFeedWarmMs", String(Date.now()));
   661→  });
   662→  setInterval(tick, 1000);
   663→  // engine keep-warm loop — the core must never go cold
   664→  setInterval(() => {
   665→    void ensureWarm();
   666→  }, 5 * 60_000);
   667→  // tennis radar keep-warm (news/live/fitness cache stays hot)
   668→  setInterval(() => {
   669→    void fetchTennisRadar().catch(() => null);
   670→  }, 4 * 60_000);
   671→  // VIP channel auto-discovery — the moment the bot becomes admin of (or
   672→  // receives any update from) the VIP channel, its numeric id is captured.
   673→  setInterval(() => {
   674→    void discoverVipLoop();
   675→  }, 30_000);
   676→  console.log(
   677→    `[scheduler] started - ${JOB_SPECS.length} jobs - live=${isLive()} - tz=${TZ}`
   678→  );
   679→  for (const job of JOB_SPECS) {
   680→    const next = nextRunAt.get(job.id);
   681→    console.log(`[scheduler] ${job.id.padEnd(18)} next=${next ? new Date(next).toISOString() : "-"}`);
   682→  }
   683→  void discoverVipLoop();
   684→}

   686→/** capture the VIP channel id from any chat the bot can see */
   687→async function discoverVipLoop(): Promise<void> {
   688→  try {
   689→    const chats = await telegram.discoverChats();
   690→    for (const ch of chats) {
   691→      const isPublic = !!ch.username && CONFIG.publicChatId.includes(ch.username);
   692→      const isChannelish = ch.type === "channel" || ch.type === "supergroup";
   693→      if (isChannelish && ch.id) {
   694→        store.kvSet(`chatSeen:${ch.id}`, JSON.stringify(ch));
   695→      }
   696→      if (isChannelish && !isPublic && ch.id) {
   697→        const idStr = String(ch.id);
   698→        if (vipChatId() !== idStr) {
   699→          store.kvSet("vipChatId", idStr);
   700→          store.kvSet("vipChatTitle", ch.title ?? "");
   701→          console.log(`[discover] VIP chat captured: ${idStr} (${ch.title ?? ch.type})`);
   702→        }
   703→      }
   704→    }
   705→  } catch {
   706→    /* discovery is best-effort */
   707→  }
   708→}
