   101→const faDate = (iso: string): string => {
   102→  const d = new Date(iso);
   103→  if (Number.isNaN(d.getTime())) return "";
   104→  return new Intl.DateTimeFormat("fa-IR", {
   105→    weekday: "long",
   106→    day: "numeric",
   107→    month: "long",
   108→  }).format(d);
   109→};
   110→
   111→const faTime = (iso: string): string => {
   112→  const d = new Date(iso);
   113→  if (Number.isNaN(d.getTime())) return "";
   114→  return new Intl.DateTimeFormat("fa-IR", {
   115→    hour: "2-digit",
   116→    minute: "2-digit",
   117→  }).format(d);
   118→};
   119→
   120→function countdownLabel(iso: string, now: number): string {
   121→  const t = new Date(iso).getTime();
   122→  if (!Number.isFinite(t)) return "";
   123→  const diff = t - now;
   124→  if (diff <= 0) return "شروع‌شده / در جریان";
   125→  const mins = Math.floor(diff / 60000);
   126→  const secs = Math.floor((diff % 60000) / 1000);
   127→  if (mins >= 60) {
   128→    return `شروع تا ${Math.floor(mins / 60)} ساعت و ${mins % 60} دقیقه دیگر`;
   129→  }
   130→  return `شروع تا ${mins}:${String(secs).padStart(2, "0")} دیگر`;
   131→}

  1106→function NearGateStrip({ matches }: { matches: AnalyzedMatch[] }) {

  1109→      .filter((m) => m.state !== "post" && !m.veto.vetoed)
