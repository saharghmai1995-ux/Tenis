    39→export function tehranFields(ms = Date.now()): {
    40→  y: number; mon: number; dom: number; h: number; min: number; s: number; dow: number;
    41→} {
    42→  const parts = partsFmt.formatToParts(new Date(ms));
    43→  const get = (t: string) => Number(parts.find((p) => p.type === t)?.value ?? 0);
    44→  const wd = dowFmt.format(new Date(ms)).toUpperCase();
    45→  return {
    46→    y: get("year"),
    47→    mon: get("month"),
    48→    dom: get("day"),
    49→    h: get("hour") % 24,
    50→    min: get("minute"),
    51→    s: get("second"),
    52→    dow: { SUN: 0, MON: 1, TUE: 2, WED: 3, THU: 4, FRI: 5, SAT: 6 }[wd] ?? 0,
    53→  };
    54→}

    61→// ─── 6-field cron matcher ──────────────────────────────────────────────────

    63→type Field = { wildcard: boolean; any: Set<number>; step?: number };
    64→
    65→function parseField(raw: string, min: number, max: number): Field {
    66→  const any = new Set<number>();
    67→  let wildcard = false;
    68→  let step: number | undefined;
    69→  for (const chunk of raw.split(",")) {
    70→    if (chunk === "*") {
    71→      wildcard = true;
    72→      for (let i = min; i <= max; i++) any.add(i);
    73→    } else if (chunk.startsWith("*/")) {
    74→      wildcard = true;
    75→      step = Number(chunk.slice(2));
    76→      for (let i = min; i <= max; i += step) any.add(i);
    77→    } else if (chunk.includes("-")) {
    78→      const [a, b] = chunk.split("-").map(Number);
    79→      for (let i = a; i <= b; i++) any.add(i);
    80→    } else {
    81→      any.add(Number(chunk));
    82→    }
    83→  }
    84→  return { wildcard, any, step };
    85→}
    86→
    87→interface Cron {
    88→  sec: Field; min: Field; hour: Field; dom: Field; mon: Field; dow: Field;
    89→}
    90→
    91→function parseCron(expr: string): Cron {
    92→  const [sec, min, hour, dom, mon, dow] = expr.trim().split(/\s+/);
    93→  return {
    94→    sec: parseField(sec, 0, 59),
    95→    min: parseField(min, 0, 59),
    96→    hour: parseField(hour, 0, 23),
    97→    dom: parseField(dom, 1, 31),
    98→    mon: parseField(mon, 1, 12),
    99→    dow: parseField(dow, 0, 6),
   100→  };
   101→}
   102→
   103→function cronMatches(c: Cron, f: ReturnType<typeof tehranFields>): boolean {
   104→  return (
   105→    c.sec.any.has(f.s) &&
   106→    c.min.any.has(f.min) &&
   107→    c.hour.any.has(f.h) &&
   108→    c.mon.any.has(f.mon) &&
   109→    (c.dom.wildcard || c.dow.wildcard
   110→      ? c.dom.any.has(f.dom) || c.dow.any.has(f.dow)
   111→      : c.dom.any.has(f.dom) && c.dow.any.has(f.dow))
   112→  );
   113→}

   115→/** next run strictly after `after` (ms epoch) for a Tehran-anchored cron */
   116→export function nextCronRun(expr: string, after = Date.now()): number {
   117→  const c = parseCron(expr);
   118→  // iterate the next 2 days at second resolution starting from `after + 1s`
   119→  let t = after + 1000;
   120→  const limit = after + 2 * 86_400_000;
   121→  while (t <= limit) {
   122→    if (cronMatches(c, tehranFields(t))) return t;
   123→    // jump smartly: skip to next minute boundary when seconds won't match soon
   124→    const f = tehranFields(t);
   125→    if (!c.hour.any.has(f.h) || !c.min.any.has(f.min)) {
   126→      t += (59 - f.s) * 1000 + 1000; // finish current minute
   127→      if (!c.hour.any.has(f.h)) t += (59 - f.min) * 60_000; // finish current hour
   128→    } else {
   129→      t += 1000;
   130→    }
   131→  }
   132→  return after + 60_000;
   133→}

   179→async function deliver(
   180→  jobId: JobId,
   181→  target: string,
   182→  kind: PosterKind,
   183→  caption: string,
   184→  png: Buffer
   185→): Promise<PostOutcome> {
   186→  saveOutbox(jobId, png);
   187→  if (!isLive()) {
   188→    store.logPost({
   189→      at: new Date().toISOString(),
   190→      jobId,
   191→      target,
   192→      posterKind: kind,
   193→      mode: "DRY-RUN",
   194→      caption,
   195→      status: "DRY-RUN",
   196→    });
   197→    console.log(`[job:${jobId}] DRY-RUN ${kind} → ${target} (${Math.round(png.length / 1024)}KB)`);
   198→    return { status: "DRY-RUN" };
   199→  }
   200→  const chatId = target === "vip" ? vipChatId() : CONFIG.publicChatId;
   201→  if (!chatId) {
   202→    store.logPost({
   203→      at: new Date().toISOString(),
   204→      jobId,
   205→      target,
   206→      posterKind: kind,
   207→      mode: "LIVE",
   208→      caption,
   209→      status: "SKIPPED",
   210→      error: "chat id not configured",
   211→    });
   212→    return { status: "SKIPPED", error: "chat id not configured" };
   213→  }
   214→  const res = await telegram.sendPhoto(chatId, png, caption);
   215→  store.logPost({
   216→    at: new Date().toISOString(),
   217→    jobId,
   218→    target,
   219→    posterKind: kind,
   220→    mode: "LIVE",
   221→    messageId: res.messageId,
   222→    caption,
   223→    status: res.ok ? "SENT" : "ERROR",
   224→    error: res.description,
   225→  });
   226→  console.log(
   227→    `[job:${jobId}] ${res.ok ? "SENT" : "ERROR"} ${kind} → ${target}${res.messageId ? ` #${res.messageId}` : ""}${res.description ? ` (${res.description})` : ""}`
   228→  );
   229→  return { status: res.ok ? "SENT" : "ERROR", error: res.description, messageId: res.messageId };
   230→}

   277→// ─── flashcard anti-repeat engine (director fix: no more duplicate cards) ──

   301→export function angleFor(card: CardMatch, rotation: number): FlashAngle {
   302→  const angle = FLASH_ANGLES[rotation % FLASH_ANGLES.length];
   303→  const cov = Math.round((card.coverage ?? 0) * 100);
   304→  const favName = (card.favorite === "A" ? card.playerA : card.playerB).toUpperCase();
   305→  switch (angle) {
   306→    case "VALUE SPOTLIGHT": {
   307→      const slips = [card.slipA, card.slipB].filter((s): s is NonNullable<typeof s> => !!s);
   308→      const best = [...slips].sort((a, b) => (b.edge ?? -9) - (a.edge ?? -9))[0];
   309→      if (best && best.edge > 0.02) {
   310→        const wingEn = best === card.slipA ? "TYPE A" : "TYPE B";
   311→        const wingFa = best === card.slipA ? "تیپ A" : "تیپ B";
   312→        const edge = (best.edge * 100).toFixed(1);
   313→        return {
   314→          label: angle,
   315→          labelFa: ANGLE_LABEL_FA[angle],
   316→          insightEn: `${wingEn} CARRIES +${edge}% MODEL EDGE AT ODDS ${best.odds.toFixed(2)} - PRICE ABOVE FAIR VALUE.`,
   317→          insightFa: `${wingFa} دارای ٪${edge} ارزش مدل با ضریب ${best.odds.toFixed(2)} است؛ قیمت بالاتر از ارزش منصفانه بازار.`,
   318→        };
   319→      }
   320→      return {
   321→        label: angle,
   322→        labelFa: ANGLE_LABEL_FA[angle],
   323→        insightEn: "NO OVERPRICED WINDOW RIGHT NOW - BOTH WINGS TRADE NEAR FAIR VALUE. DISCIPLINE STAKES ONLY.",
   324→        insightFa: "در این لحظه شانه خالی کردن بازار دیده نمی‌شود؛ هر دو تیپ نزدیک ارزش منصفانه‌اند و فقط استیک انضباطی توصیه می‌شود.",
   325→      };
   326→    }
   327→    case "MARKET DEPTH": {
   328→      const timing = timingEn(card.timingCode, card.timingLabel);
   329→      return {
   330→        label: angle,
   331→        labelFa: ANGLE_LABEL_FA[angle],
   332→        insightEn: `ARCHETYPE ${card.archLabelEn?.toUpperCase() ?? "SUPERVISED PATTERN"} DETECTED - OFFICIAL ENTRY WINDOW ${timing}. NO LATE MOVES.`,
   333→        insightFa: "الگوی بازی از دروازه‌های سوپروایزر عبور کرده است؛ پنجره رسمی ورود همان تایمینگ اعلام‌شده است و ورود خارج از پنجره ممنوع.",
   334→      };
   335→    }
   336→    case "MODEL X-RAY": {
   337→      const p = Math.round((card.favWinProb ?? 0.5) * 100);
   338→      return {
   339→        label: angle,
   340→        labelFa: ANGLE_LABEL_FA[angle],
   341→        insightEn: `MODEL X-RAY: ${favName} WINS ${p}% OF SIMULATED PATHS - MARKET AND SURFACE RATING FUSED AT 65/35.`,
   342→        insightFa: `بر اساس شبیه‌سازی مدل، ${favName} در ٪${p} از مسیرهای ممکن برنده است؛ ترکیب ۶۵/۳۵ بازار و ریتینگ سطح زمین.`,
   343→      };
   344→    }
   345→    default:
   346→      return {
   347→        label: "FULL CARD",
   348→        labelFa: ANGLE_LABEL_FA["FULL CARD"],
   349→        insightEn: `DUAL-WING STRUCTURE ONLINE - TYPE A + TYPE B TOGETHER COVER ${cov}% OF ALL POSSIBLE SCORE ROUTES.`,
   350→        insightFa: `ساختار دوپایی فعال است: ترکیب تیپ A و تیپ B حدود ٪${cov} از همه مسیرهای نتیجه بازی را پوشش می‌دهد.`,
   351→      };
   352→  }

   460→  };

   477→    kind: "BRIEFING",
   478→    title: "TODAY'S BATTLE ARENA",
   479→    subtitle: `${tehranDateKey()} - ${cards.length} QUALIFIED PICKS`,

   636→  async flashcards() {
   637→    await ensureWarm();
   638→    const dateKey = tehranDateKey();
   639→    store.kvSet("flashDateKey", dateKey);
   640→    const pool = dataHub.approvedPool();
   641→    if (pool.length === 0) return { status: "SKIPPED", error: "flash pool empty" };
   642→    const hist = loadFlashHist(dateKey);
   643→    const now = Date.now();
   644→    const eligible = pool.filter((c) => {
   645→      const h = hist[c.eventKey];
   646→      if (!h) return true;
   647→      return now - h.at >= FLASH_MIN_REPEAT_MS && h.c < FLASH_MAX_PER_DAY;
   648→    });

   651→        status: "SKIPPED",

   654→    }

   656→    eligible.sort((a, b) => {
   657→      const ca = hist[a.eventKey]?.c ?? 0;
   658→      const cb = hist[b.eventKey]?.c ?? 0;
   659→      if (ca !== cb) return ca - cb;
   660→      const ta = hist[a.eventKey]?.at ?? 0;
   661→      const tb = hist[b.eventKey]?.at ?? 0;
   662→      if (ta !== tb) return ta - tb;
   663→      return (b.coverage ?? 0) - (a.coverage ?? 0);
   664→    });
   665→    const card = eligible[0];
   666→    const rotation = hist[card.eventKey]?.c ?? 0;
   667→    hist[card.eventKey] = { c: rotation + 1, at: now };
   668→    saveFlashHist(dateKey, hist);
   669→    store.kvSet("flashCursor", String(Object.values(hist).reduce((s, e) => s + e.c, 0)));

   788→let running = new Set<JobId>();

   790→async function runJob(jobId: JobId, reason: "cron" | "manual" = "cron"): Promise<PostOutcome> {
   791→  if (running.has(jobId)) return { status: "SKIPPED", error: "already running" };
   792→  running.add(jobId);
   793→  try {
   794→    const out = await handlers[jobId]();
   795→    store.recordRun(jobId, out.status, { error: out.error, messageId: out.messageId });
   796→    return out;
   797→  } catch (e) {
   798→    const error = e instanceof Error ? e.message : String(e);
   799→    console.error(`[job:${jobId}] ${reason} error:`, error);
   800→    store.recordRun(jobId, "ERROR", { error });
   801→    return { status: "ERROR", error };
   802→  } finally {
   803→    running.delete(jobId);
   804→  }
   805→}

   807→function scheduleAll(): void {
   808→  for (const job of JOB_SPECS) {
   809→    if (job.cron) {
   810→      nextRunAt.set(job.id, nextCronRun(job.cron));
   811→    } else if (job.everySeconds) {
   812→      nextRunAt.set(job.id, Date.now() + (job.everySeconds * 1000));
   813→    }
   814→  }
   815→}
   816→
   817→function tick(): void {
   818→  const now = Date.now();
   819→  for (const job of JOB_SPECS) {
   820→    const next = nextRunAt.get(job.id);
   821→    if (next === undefined || now < next) continue;
   822→    if (job.cron) {
   823→      nextRunAt.set(job.id, nextCronRun(job.cron, now));
   824→    } else if (job.everySeconds) {
   825→      nextRunAt.set(job.id, now + job.everySeconds * 1000);
   826→    }
   827→    void runJob(job.id);
   828→  }
   829→}

   831→export function startScheduler(): void {
   832→  registerFonts();
   833→  restoreLive();
   834→  void refreshMe().catch(() => null);
   835→  scheduleAll();
   836→  // warm immediately so the first flashcard is never cold
   837→  void dataHub.warm().then((ok) => {
   838→    if (ok) store.kvSet("lastFeedWarmMs", String(Date.now()));
   839→  });
   840→  setInterval(tick, 1000);
   841→  // engine keep-warm loop — the core must never go cold
   842→  setInterval(() => {
   843→    void ensureWarm();
   844→  }, 5 * 60_000);
   845→  // tennis radar keep-warm (news/live/fitness cache stays hot)
   846→  setInterval(() => {
   847→    void fetchTennisRadar().catch(() => null);
   848→  }, 4 * 60_000);
   849→  // VIP channel auto-discovery — the moment the bot becomes admin of (or
   850→  // receives any update from) the VIP channel, its numeric id is captured.
   851→  setInterval(() => {
   852→    void discoverVipLoop();
   853→  }, 30_000);
   854→  console.log(
   855→    `[scheduler] started - ${JOB_SPECS.length} jobs - live=${isLive()} - tz=${TZ}`
   856→  );
   857→  for (const job of JOB_SPECS) {
   858→    const next = nextRunAt.get(job.id);
   859→    console.log(`[scheduler] ${job.id.padEnd(18)} next=${next ? new Date(next).toISOString() : "-"}`);
   860→  }
   861→  void discoverVipLoop();
   862→}

   864→/** capture the VIP channel id from any chat the bot can see */
   865→async function discoverVipLoop(): Promise<void> {
   866→  try {
   867→    const chats = await telegram.discoverChats();
   868→    for (const ch of chats) {
   869→      const isPublic = !!ch.username && CONFIG.publicChatId.includes(ch.username);
   870→      const isChannelish = ch.type === "channel" || ch.type === "supergroup";
   871→      if (isChannelish && ch.id) {
   872→        store.kvSet(`chatSeen:${ch.id}`, JSON.stringify(ch));
   873→      }
   874→      if (isChannelish && !isPublic && ch.id) {
   875→        const idStr = String(ch.id);
   876→        if (vipChatId() !== idStr) {
   877→          store.kvSet("vipChatId", idStr);
   878→          store.kvSet("vipChatTitle", ch.title ?? "");
   879→          console.log(`[discover] VIP chat captured: ${idStr} (${ch.title ?? ch.type})`);
   880→        }
   881→      }
   882→    }
   883→  } catch {
   884→    /* discovery is best-effort */
   885→  }
