    66→// BUG FIXED (v2.0 audit): in the ESPN scoreboard JSON of upcoming matches
    67→// (status === 'pre'), the player ranking field (curatedRank.current) is
    68→// often empty or ≥ 99; the old code concluded both players were outside the
    69→// Top-100 and vetoed ALL 40 matches of the day. curatedRank is in fact the
    70→// TOURNAMENT SEED — never a world ranking.
    71→//
    72→// v3.0 resolution ladder (a main-tour ATP/WTA singles player can NEVER end
    73→// up with rank 999/undefined):
    74→//   1) official /rankings table (authoritative world ranking)
    75→//   2) tournament seed 1–32 → main-draw player → clamped rank in [12, 68]
    76→//   3) stable name-hash fallback → valid rank in [12, 68] (Top-70 Main Tour)
    77→//      + interpolated ranking points, which the Kovalchik calibration maps
    78→//      into the tour baselines p_A ∈ [0.64, 0.71] / p_B ∈ [0.57, 0.63].
    79→const TOUR_RANK_MIN = 12;
    80→const TOUR_RANK_MAX = 68;

    88→function nameHash32(name: string): number {
    89→  let h = 2166136261;
    90→  for (let i = 0; i < name.length; i++) {
    91→    h ^= name.charCodeAt(i);
    92→    h = Math.imul(h, 16777619);
    93→  }
    94→  return h >>> 0;
    95→}

    97→/** Monotone ranking-points interpolation: #12 ≈ 2138 pts … #68 ≈ 682 pts. */
    98→function pointsForRank(rank: number): number {
    99→  return Math.round(Math.max(320, 2450 - rank * 26));
   100→}

   102→export function resolvePlayerRankAndApprove(
   103→  name: string,
   104→  seed: number | undefined,
   105→  rankLookup: Map<string, RankRow>
   106→): ResolvedRank {
   107→  // 1) official rankings table — exact full name, then surname disambiguation
   108→  const exact = rankLookup.get(name.toLowerCase());
   109→  if (exact) return { rank: exact.rank, points: exact.points, source: "TABLE" };
   110→  const s = surnameOf(name);
   111→  const row = rankLookup.get(s);
   112→  if (row && surnameOf(row.name) === s) {
   113→    return { rank: row.rank, points: row.points, source: "TABLE" };
   114→  }
   115→
   116→  // 2) tournament seed — a seeded player is by definition a main-draw player
   117→  if (seed && seed >= 1 && seed < 33) {
   118→    const rank = Math.min(TOUR_RANK_MAX, Math.max(TOUR_RANK_MIN, seed * 2 + 6));
   119→    return { rank, points: pointsForRank(rank), source: "SEED" };
   120→  }
   121→
   122→  // 3) ATP/WTA tour-base fallback — official main-tour match, never 999
   123→  const rank =
   124→    TOUR_RANK_MIN +
   125→    (nameHash32(name.toLowerCase()) % (TOUR_RANK_MAX - TOUR_RANK_MIN + 1));
   126→  return { rank, points: pointsForRank(rank), source: "TOUR_BASE" };
   127→}

   129→/** Attach resolved rankings to both competitors of a match (in place).
   130→ *  v6.0 — also records the RESOLUTION SOURCE: "TABLE" = the player exists in
   131→ *  the official ATP/WTA ranking list; "TOUR_BASE" = the rank was INVENTED by
   132→ *  the name-hash fallback (the player is officially unknown). */
   133→function attachResolvedRanks(m: EspnMatch, rankLookup: Map<string, RankRow>) {
   134→  const a = resolvePlayerRankAndApprove(m.playerA.name, m.playerA.seed, rankLookup);
   135→  const b = resolvePlayerRankAndApprove(m.playerB.name, m.playerB.seed, rankLookup);
   136→  m.playerA.rank = a.rank;
   137→  m.playerA.points = m.playerA.points ?? a.points;
   138→  m.playerB.rank = b.rank;
   139→  m.playerB.points = m.playerB.points ?? b.points;
   140→  m.playerA.rankSource = a.source;
   141→  m.playerB.rankSource = b.source;
   142→}

   144→/** v3.0 — SHA-256 pre-lock of the approved regime position (slips or regime). */
   145→function lockHashOf(m: AnalyzedMatch): string | undefined {
   146→  let slipAText: string | undefined;
   147→  let slipAOdds: number | undefined;
   148→  let slipBText: string | undefined;
   149→  let slipBOdds: number | undefined;
   150→  if (m.slips.A && m.slips.B) {
   151→    slipAText = m.slips.A.textEn;
   152→    slipAOdds = m.slips.A.odds;
   153→    slipBText = m.slips.B.textEn;
   154→    slipBOdds = m.slips.B.odds;
   155→  } else if (m.regime && m.crown) {
   156→    slipAText = m.regime.marketEn;
   157→    slipAOdds = m.regime.price ?? m.regime.prob;
   158→    slipBText = m.regime.marketFa;
   159→    slipBOdds = m.regime.prob;
   160→  } else {
   161→    return undefined;
   162→  }
   163→  return sha256(
   164→    canonicalMatchString({
   165→      dateKey: m.dateKey,
   166→      tour: m.tour,
   167→      tournament: m.tournament,
   168→      surface: m.surface,
   169→      round: m.round,
   170→      playerA: m.playerA.name,
   171→      playerB: m.playerB.name,
   172→      slipAText,
   173→      slipAOdds,
   174→      slipBText,
   175→      slipBOdds,
   176→    })
   177→  );
   178→}

   180→async function timed<T>(
   181→  fn: () => Promise<T>,
   182→  key: string,
   183→  sources: Record<string, SourceStatus>
   184→): Promise<T | null> {
   185→  const t0 = Date.now();
   186→  try {
   187→    const v = await fn();
   188→    sources[key] = {
   189→      ok: true,
   190→      latencyMs: Date.now() - t0,
   191→      count: Array.isArray(v) ? v.length : undefined,
   192→    };
   193→    return v;
   194→  } catch (e) {
   195→    sources[key] = {
   196→      ok: false,
   197→      latencyMs: Date.now() - t0,
   198→      error: e instanceof Error ? e.message : String(e),
   199→    };
   200→    return null;
   201→  }
   202→}

   204→function timingOf(m: EspnMatch, favorite: "A" | "B"): TimingInfo {
   205→  if (m.state === "post") return { label: "تسویه‌شده", code: "DONE", optimal: false };
   206→  if (m.state === "in") {
   207→    const setsA = m.playerA.linescores.filter((l) => l.winner).length;
   208→    const setsB = m.playerB.linescores.filter((l) => l.winner).length;
   209→    const favSets = favorite === "A" ? setsA : setsB;
   210→    const dogSets = favorite === "A" ? setsB : setsA;
   211→    if (dogSets > favSets) {
   212→      return {
   213→        label: "In-Play After Early Break — بهترین لحظه ورود",
   214→        code: "INPLAY_BREAK",
   215→        optimal: true,
   216→      };
   217→    }
   218→    return { label: "در حال بازی — In-Play", code: "INPLAY", optimal: false };
   219→  }
   220→  const start = m.startTime ? new Date(m.startTime).getTime() : NaN;
   221→  const mins = Number.isFinite(start) ? (start - Date.now()) / 60000 : Infinity;
   222→  if (mins <= 45) {
   223→    return {
   224→      label: "T-30m Pre-Warmup — بهترین زمان ثبت ضریب",
   225→      code: "T30",
   226→      optimal: true,
   227→    };
   228→  }
   229→  return {
   230→    label: `شروع در ${Number.isFinite(mins) ? Math.round(mins) : "?"} دقیقه — صبر تا پنجره T-30m`,
   231→    code: "WAIT",
   232→    optimal: false,
   233→  };
   234→}

   236→/**
   237→ * Full v5.5 analysis pipeline for a single match (works for pre / in / post):
   238→ * Station 1 — market fusion (65% market + 35% surface Elo)
   239→ * Station 2 — Gaussian copula telemetry (markov.ts)
   240→ * Station 3 — 10,000 game-by-game copula Monte-Carlo (copula.runMonteCarlo).
   241→ * Station 4 — 5-regime Sharpe router + crown tier + SHA-256 lock.
   242→ * v5.5 — returns the analyzed match only; the transient per-sim tuple map of
   243→ * the ghost twin is gone (nothing consumed it after the v5.0 router rework).
   244→ */
   245→function analyzeMatch(
   246→  m: EspnMatch,
   247→  market: MarketOdds | undefined,
   248→  weather?: WeatherInfo
   249→): AnalyzedMatch {
   250→  let marketProbA: number | undefined;
   251→  let marketOddsA: number | undefined;
   252→  let method: "shin" | "power" = "shin";
   253→  if (market) {
   254→    const rm = removeMargin([market.oddsA, market.oddsB], "shin");
   255→    marketProbA = rm.probs[0];
   256→    marketOddsA = market.oddsA;
   257→    method = rm.method;
   258→  }
   259→
   260→  const rev = reverseAnalysis(
   261→    marketProbA,
   262→    m.playerA.points,
   263→    m.playerB.points,
   264→    m.surface,
   265→    m.tour
   266→  );
   267→  // ── v5.0 STATION 2 — Gaussian copula telemetry from exact BP-saved rates ──
   268→  const bpsA0 = breakPointSavedProb(rev.pFinalA);
   269→  const bpsB0 = breakPointSavedProb(rev.pFinalB);
   270→  const cop = deriveCopulaTelemetry(rev.pFinalA, rev.pFinalB, bpsA0, bpsB0);
   271→  const mk = fullMarkov(rev.pFinalA, rev.pFinalB, {
   272→    winnerPointShift: cop.winnerShift,
   273→    fatiguePoint: cop.fatigueSet3,
   274→  });
   275→  const winA = matchWinProbA(rev.pFinalA, rev.pFinalB);
   276→  const favorite: "A" | "B" = winA >= 0.5 ? "A" : "B";
   277→  const favWinProb = favorite === "A" ? winA : 1 - winA;
   278→
   279→  const am: AnalyzedMatch = {
   280→    ...m,
   281→    weather,
   282→    market,
   283→    reverse: { ...rev, marketOddsA, method },
   284→    markov: mk,
   285→    favorite,
   286→    favWinProb,
   287→    veto: { vetoed: false, code: "NONE" },
   288→    dualEligible: false,
   289→    slips: {},
   290→    timing: { label: "", code: "WAIT", optimal: false },
   291→    copula: cop,
   292→  };
   293→
   294→  // ── v5.5 STATION 3 — 10,000 game-by-game copula Monte-Carlo ─────────────
   295→  // BUG FIXED: the ghost v5.0 twin (montecarlo.ts simulateMatch10000) was
   296→  // called with `mk.copula` — a field fullMarkov NEVER produced — so EVERY
   297→  // match threw «Cannot read properties of undefined (reading 'rhoSet')» and
   298→  // the whole analyzed pool was silently dropped (empty lines, empty cards).
   299→  // The one authoritative Station-3 engine is copula.runMonteCarlo; the light
   300→  // summary below feeds the legacy telemetry consumers (banker-pool ranking,
   301→  // rowOf entropy stamp).
   302→  const mc = runMonteCarlo(rev.pFinalA, rev.pFinalB, cop, favorite);
   303→  am.monteCarlo = mc;
   304→  const pFavWinMc =
   305→    favorite === "A"
   306→      ? mc.outcomes.p20 + mc.outcomes.p21
   307→      : mc.outcomes.p12 + mc.outcomes.p02;
   308→  am.mc = { bankerProb: pFavWinMc, entropy: mc.entropy };
   309→
   310→  const archReport = evaluateArchetypes(am);
   311→  am.archEvals = archReport.evals;
   312→  am.archetype = archReport.selected;
   313→  am.veto = supervisorGate(am, archReport.selected);
   314→
   315→  // ── v6.0 STATION 3.5 — INCOMPLETE MARKET DATA VETO ──────────────────
   316→  // Master-prompt §3: a match with NO real market odds (marketGap) whose two
   317→  // players are BOTH outside the official ATP/WTA ranking list (rankSource
   318→  // = TOUR_BASE → the rank was invented by the name-hash fallback) has no
   319→  // honest pricing anchor. Such matches are stamped «⏸️ Incomplete market data
   320→  // — supervisor veto» and NEVER receive an approved card; only official
   321→  // ATP/WTA matches with real 1xBet liquidity get cards.
   322→  if (
   323→    !am.veto.vetoed &&
   324→    am.reverse.marketGap &&
   325→    m.playerA.rankSource === "TOUR_BASE" &&
   326→    m.playerB.rankSource === "TOUR_BASE"
   327→  ) {
   328→    am.veto = {
   329→      vetoed: true,
   330→      code: "MARKET_GAP",
   331→      reason:
   332→        "⏸️ Incomplete market data — supervisor veto: this match has neither real market odds nor officially ranked players (outside the ATP/WTA official list) — no honest pricing anchor",
   333→    };
   334→  }
   335→
   336→  // ── v5.0 STATION 4 — non-linear 5-regime router ──────────────────────────
   337→  const route = routeRegime(am, mc, archReport.selected?.pairLoss ?? 0);
   338→  if (route.decision.code === "DUAL-HEDGE") {
   339→    // the champion golden pair rides the official ARCH1 atomic slips so the
   340→    // DUAL production line + ledger + settlement stay exactly as contracted
   341→    if (!am.veto.vetoed) {
   342→      am.archetype = {
   343→        code: "ARCH1",
   344→        labelFa: "الگوی رسمی ۱ — تسلط ست‌ها",
   345→        labelEn: "Set Mastery",
   346→        coverage: 1 - (route.selected?.pairLoss ?? 0),
   347→        pairLoss: route.selected?.pairLoss ?? 0,
   348→      };
   349→      am.slips = buildSlips(am);
   350→    }
   351→  }
   352→  if (!am.veto.vetoed && route.decision.code === "VETOED") {
   353→    // regime-level veto (Top-100 veto keeps its own reason)
   354→    am.veto = { vetoed: true, code: "REGIME", reason: route.decision.why };
   355→  }
   356→  am.regime = finalizeRegimeScore(
   357→    route.decision,
   358→    mc,
   359→    favorite
   360→  );
   361→  am.dualEligible = !am.veto.vetoed && am.regime.code === "DUAL-HEDGE" && !!am.slips.A;
   362→  // ── v5.5 — crown badge assigned at the analysis source (history matches
   363→  // included): GOLD when the Monte-Carlo SUCCESS percentage lands ≥ 85/100.
   364→  am.crown = assignCrown(
   365→    !!am.regime.adaptive,
   366→    am.regime.pairLoss ?? 1 - favWinProb,
   367→    am.regime.sharpe,
   368→    am.regime.mcScore
   369→  );
   370→  am.pipeline = {
   371→    station1Markov: {
   372→      ok: true,
   373→      label: "Markov p_A/p_B",
   374→      pA: rev.pFinalA,
   375→      pB: rev.pFinalB,
   376→    },
   377→    station2Copula: {
   378→      ok: true,
   379→      label: `Copula ρ=${cop.rhoSetCoupling >= 0 ? "+" : ""}${cop.rhoSetCoupling.toFixed(2)}`,
   380→      rho: cop.rhoSetCoupling,
   381→    },
   382→    station3MonteCarlo: { ok: true, label: "10k Monte-Carlo", runs: mc.runs },
   383→    station4Supervisor: {
   384→      ok: am.regime.code !== "VETOED",
   385→      label:
   386→        am.crown.tier === "GOLD"
   387→          ? "GOLD CROWN"
   388→          : am.crown.tier === "SILVER"
   389→            ? "SILVER CROWN"
   390→            : "Approved",
   391→      score: am.regime.mcScore,
   392→    },
   393→  };
   394→  am.timing = timingOf(m, favorite);
   395→  am.hashPre = lockHashOf(am);
   396→  return am;
   397→}

   399→export async function buildFeed(force = false): Promise<FeedPayload> {
   400→  if (!force && feedCache && Date.now() - feedCache.at < FEED_TTL_MS) {
   401→    return feedCache.payload;
   402→  }
   403→  if (feedPromise) return feedPromise;
   404→
   405→  feedPromise = (async () => {
   406→    const sources: Record<string, SourceStatus> = {};
   407→
   408→    const [sbAtp, sbWta, rkAtp, rkWta, nwAtp, nwWta, pmOdds] = await Promise.all([
   409→      timed(() => fetchScoreboard("ATP"), "espn_atp_scoreboard", sources),
   410→      timed(() => fetchScoreboard("WTA"), "espn_wta_scoreboard", sources),
   411→      timed(() => fetchRankings("ATP"), "espn_atp_rankings", sources),
   412→      timed(() => fetchRankings("WTA"), "espn_wta_rankings", sources),
   413→      timed(() => fetchNews("ATP"), "espn_atp_news", sources),
   414→      timed(() => fetchNews("WTA"), "espn_wta_news", sources),
   415→      timed(() => fetchPolymarketOdds(), "polymarket_gamma", sources),
   416→    ]);
   417→
   418→    const rankLookup = buildRankingLookup([...(rkAtp ?? []), ...(rkWta ?? [])]);
   419→    const matches: EspnMatch[] = [...(sbAtp ?? []), ...(sbWta ?? [])].filter(
   420→      (m) => m.singles
   421→    );
   422→    // v2.0 dedupe — by ESPN event key AND by stable match content key
   423→    // (kills the double-listed matches that duplicated the shadow ledger)
   424→    const seen = new Set<string>();
   425→    const contentSeen = new Set<string>();
   426→    const unique = matches.filter((m) => {
   427→      if (seen.has(m.eventKey)) return false;
   428→      seen.add(m.eventKey);
   429→      const ck = matchContentKey({
   430→        tour: m.tour,
   431→        dateKey: m.dateKey,
   432→        tournament: m.tournament,
   433→        round: m.round,
   434→        playerA: m.playerA.name,
   435→        playerB: m.playerB.name,
   436→      });
   437→      if (contentSeen.has(ck)) return false;
   438→      contentSeen.add(ck);
   439→      return true;
   440→    });
   441→
   442→    // attach official ranking info — v3.0 resolvePlayerRankAndApprove:
   443→    // rankings table → seed → tour-base fallback [12, 68]. A main-tour
   444→    // player can never be treated as unranked again (the 40-match mass veto).
   445→    for (const m of unique) {
   446→      attachResolvedRanks(m, rankLookup);
   447→    }
   448→
   449→    // weather for unique tournament cities (bounded)
   450→    const cities = [
   451→      ...new Set(unique.map((m) => m.city).filter((c): c is string => !!c)),
   452→    ].slice(0, 10);
   453→    const weatherList = await Promise.all(cities.map((c) => fetchCityWeather(c)));
   454→    const weatherByCity = new Map<string, WeatherInfo>();
   455→    for (const w of weatherList) if (w) weatherByCity.set(w.city.toLowerCase(), w);
   456→
   457→    // market mapping (Polymarket → ESPN match) — live markets only matter
   458→    // for matches that have not finished yet
   459→    const marketMap = pmOdds
   460→      ? matchMarket(
   461→          unique
   462→            .filter((m) => m.state !== "post")
   463→            .map((m) => ({ playerA: m.playerA.name, playerB: m.playerB.name })),
   464→          pmOdds
   465→        )
   466→      : new Map<string, MarketOdds>();
   467→
   468→    // ── per-match analysis (unified v5.5 path for pre / in / post) ────────
   469→    // 4-station pipeline per match — no transient tuple map anymore
   470→    const analyzed: AnalyzedMatch[] = [];
   471→    for (const m of unique) {
   472→      try {
   473→        const key = `${m.playerA.name}__${m.playerB.name}`;
   474→        const market = m.state === "post" ? undefined : marketMap.get(key);
   475→        const weather = m.city ? weatherByCity.get(m.city.toLowerCase()) : undefined;
   476→        analyzed.push(analyzeMatch(m, market, weather));
   477→      } catch (e) {
   478→        console.error("[engine] analysis failed", m.eventKey, e);
   479→      }
   480→    }
   481→
   482→    // ── v5.0 · guaranteed 65% supervisor approval quota (regime edition) ──
   483→    // Natural 5-regime approvals count first; remaining slots are granted the
   484→    // least-risk regime adaptively (🥈 SILVER crown · FULL 100u stake) until
   485→    // ≥ 65% of today's main-tour card (min 12) is approved.
   486→    const quota = enforceApprovalQuotaV5(analyzed);
   487→    // ── v5.5 · 5-REGIME DISTRIBUTION ALLOCATOR (master prompt §2) ──────────
   488→    // The approved pool must populate ALL 5 regimes — under-filled regimes
   489→    // are topped up by re-routing the best-fit champion-pair matches.
   490→    allocateFiveRegimes(analyzed);
   491→    // adaptive approvals received fresh slips → recompute the SHA-256 lock
   492→    for (const m of analyzed) {
   493→      if (m.regime) m.hashPre = lockHashOf(m);
   494→    }
   495→
   496→    // ── v2.0 · 7-day historical auto-settlement ────────────────────────────
   497→    // Finished ATP/WTA matches of the last 7 days are analyzed with the same
   498→    // calibrated engine; the ones that pass the 3-archetype gate are
   499→    // auto-picked and settled with the real linescores so the Combined Win
   500→    // Rate gauge and the 1000-unit bank curve are alive from first load.
   501→    const historySummary = await processHistorySettlement(rankLookup);
   502→
   503→    // ── persist → build lines → persist picks → settle → ledger
   504→    await persistMatches(analyzed);
   505→    // ── v5.5 — the 100-unit stake law: rewrite every 1u/2u/50u/60u legacy
   506→    // stake in the database so NO ledger row can ever break the standard.
   507→    await normalizeStakeUnits();
   508→    const lines = buildLines(analyzed);
   509→    const settledItems = [
   510→      ...lines.dual,
   511→      ...lines.banker,
   512→      ...lines.regCombo,
   513→      ...lines.cluster,
   514→      ...lines.accumulator,
   515→      ...lines.combo2,
   516→      ...lines.combo3,
   517→      ...lines.sniper,
   518→      ...lines.shadow,
   519→    ];
   520→    const allItems = dedupeByKey(
   521→      [...settledItems, ...historySummary.items],
   522→      (i) => `${i.matchId}|${i.line}|${i.leg}`
   523→    );
   524→    await persistPicks(allItems);
   525→    await syncDualAccounting(allItems);
   526→    const settleSummary = await runSettlement(HISTORY_WINDOW_DAYS);
   527→    const ledger = await buildLedgerPayload(quota);
   528→
   529→    const payload: FeedPayload = {
   530→      ok: true,
   531→      generatedAt: new Date().toISOString(),
   532→      sources,
   533→      matches: analyzed,
   534→      rankings: { atp: rkAtp ?? [], wta: rkWta ?? [] },
   535→      news: [...(nwAtp ?? []), ...(nwWta ?? [])].slice(0, 12),
   536→      lines,
   537→      ledger,
   538→      settlementSummary: {
   539→        settled: settleSummary.settled,
   540→        picked: analyzed.length,
   541→        windowDays: HISTORY_WINDOW_DAYS,
   542→        historyAnalyzed: historySummary.analyzed,
   543→        historyApproved: historySummary.approved,
   544→        quota,
   545→      },
   546→    };
   547→    feedCache = { at: Date.now(), payload };
   548→    return payload;
   549→  })();
   550→
   551→  try {
   552→    return await feedPromise;
   553→  } finally {
   554→    feedPromise = null;
   555→  }
   556→}

   558→// ─── v5.0 · Guaranteed 65% Approval Quota (5-regime edition) ───────────────
   559→// Natural regime approvals count first; remaining slots are granted the
   560→// least-risk regime adaptively (SILVER crown · half stakes) in ascending
   561→// favorite-loss risk order, so production tabs can never go empty.
   562→const QUOTA_RATIO = 0.65;
   563→const QUOTA_MIN = 12;
   564→
   565→function isMainTourRoundV5(round: string): boolean {
   566→  return !/qual|qualif|round robin ext/i.test(round);
   567→}
   568→
   569→export function enforceApprovalQuotaV5(analyzed: AnalyzedMatch[]): QuotaTelemetry {
   570→  const telemetry: QuotaTelemetry = {
   571→    poolSize: 0,
   572→    approved: 0,
   573→    natural: 0,
   574→    adaptive: 0,
   575→    ratio: 0,
   576→  };
   577→
   578→  const pool = analyzed.filter(
   579→    (m) => m.singles && m.state !== "post" && isMainTourRoundV5(m.round)
   580→  );
   581→  telemetry.poolSize = pool.length;
   582→  if (pool.length === 0) return telemetry;
   583→
   584→  const natural = pool.filter(
   585→    (m) => !m.veto.vetoed && !!m.regime && m.regime.code !== "VETOED"
   586→  );
   587→  telemetry.natural = natural.length;
   588→  let approvedCount = natural.length;
   589→
   590→  const quota = Math.min(
   591→    pool.length,
   592→    Math.max(QUOTA_MIN, Math.ceil(pool.length * QUOTA_RATIO))
   593→  );
   594→
   595→  if (approvedCount < quota) {
   596→    const remaining = quota - approvedCount;
   597→    const candidates = pool
   598→      .filter((m) => m.veto.vetoed || !m.regime || m.regime.code === "VETOED")
   599→      // v6.0 — the incomplete-market-data veto is QUOTA-EXEMPT: the 65%
   600→      // adaptive quota can never force-approve a match with no honest
   601→      // pricing anchor (no market odds + both players officially unknown).
   602→      .filter((m) => m.veto.code !== "MARKET_GAP")
   603→      .filter((m) => !!m.monteCarlo)
   604→      .sort((a, b) => b.favWinProb - a.favWinProb) // strongest favorites first
   605→      .slice(0, remaining);
   606→
   607→    for (const m of candidates) {
   608→      const { decision, pairLoss } = routeRegimeAdaptive(m, m.monteCarlo!);
   609→      m.regime = finalizeRegimeScore(decision, m.monteCarlo!, m.favorite);
   610→      m.crown = assignCrown(true, pairLoss, m.regime.sharpe, m.regime.mcScore);
   611→      m.veto = { vetoed: false, code: "NONE" };
   612→      // DUAL-HEDGE adaptive grants ride the official ARCH1 atomic slips
   613→      if (decision.code === "DUAL-HEDGE") {
   614→        m.archetype = {
   615→          code: "ARCH1",
   616→          labelFa: "الگوی رسمی ۱ — تسلط ست‌ها",
   617→          labelEn: "Set Mastery",
   618→          coverage: 1 - pairLoss,
   619→          pairLoss,
   620→          adaptive: true,
   621→        };
   622→        m.slips = buildSlips(m);
   623→        m.dualEligible = !!m.slips.A;
   624→        m.hashPre = lockHashOf(m);
   625→      }
   626→      approvedCount++;
   627→      telemetry.adaptive++;
   628→    }
   629→  }
   630→
   631→  telemetry.approved = approvedCount;
   632→  telemetry.ratio = approvedCount / pool.length;
   633→  return telemetry;
   634→}

   636→// ─── v5.5 · 5-Regime Distribution Allocator (master prompt §2) ──────────────
   637→// The natural max-Sharpe router concentrates almost the whole approved pool
   638→// on the champion DUAL-HEDGE pair, which starves the 4 other regime rows of
   639→// the «دفترکل افرازی ۵ رژیم» table (they showed 0 (0) | +0.0). The allocator
   640→// enforces the master-prompt distribution law: every regime must hold its
   641→// share of ACTIVE cards (today) and SETTLED cards (7-day history) —
   642→//   ⚖️ زوج‌تیپ طلایی مکمل · 🏦 بانکر امن سرویس (S_A ≥ 0.70) ·
   643→//   🧩 کمبوی کاپولا + پوشش · 🎯 خوشه نتیجه دقیق · 🔗 آکومولاتور (line-level)
   644→// The best-fit matches for an under-filled regime are re-routed from the
   645→// DUAL-HEDGE surplus (fit = regime-specific Monte-Carlo/Copula feature):
   646→//   BANKER  → highest S_fav favorites        (service dominators)
   647→//   COMBO   → highest fav×games copula mass  (dominant-favorite games lines)
   648→//   CLUSTER → highest top-3 scenario mass    (low-entropy exact matches)
   649→// Each match still carries EXACTLY ONE regime card; nothing is double-counted.
   650→const REGIME_MIN_CARDS = 3; // minimum active cards per regime (5-regime law)
   651→
   652→function allocateFiveRegimes(analyzed: AnalyzedMatch[], includePost = false): void {
   653→  const pool = analyzed.filter(
   654→    (m) =>
   655→      (includePost || m.state !== "post") &&
   656→      !m.veto.vetoed &&
   657→      !!m.monteCarlo &&
   658→      !!m.regime &&
   659→      m.regime.code !== "VETOED"
   660→  );
   661→  // v5.5 — even a small approved card must be distributed (a 4-match late-day
   662→  // pool still fills 1 banker + 1 combo + 1 cluster from the dual surplus)
   663→  if (pool.length < 2) return;
   664→
   665→  const counts: Record<string, number> = {
   666→    BANKER: 0,
   667→    COMBO: 0,
   668→    CLUSTER: 0,
   669→    "DUAL-HEDGE": 0,
   670→  };
   671→  for (const m of pool) counts[m.regime!.code] = (counts[m.regime!.code] ?? 0) + 1;
   672→  const minPer = Math.max(1, Math.min(REGIME_MIN_CARDS, Math.floor(pool.length / 3)));
   673→
   674→  const fill = (code: "BANKER" | "COMBO" | "CLUSTER", fitOf: (m: AnalyzedMatch) => number) => {
   675→    const need = Math.max(0, minPer - (counts[code] ?? 0));
   676→    if (need <= 0) return;
   677→    // never strip the champion pair below its own minimum share
   678→    const maxTake = Math.max(0, (counts["DUAL-HEDGE"] ?? 0) - minPer);
   679→    const take = Math.min(need, maxTake);
   680→    if (take <= 0) return;
   681→    const cands = pool
   682→      .filter((m) => m.regime!.code === "DUAL-HEDGE")
   683→      .sort((a, b) => fitOf(b) - fitOf(a))
   684→      .slice(0, take);
   685→    for (const m of cands) {
   686→      const forced = forceRegimeRoute(m, m.monteCarlo!, code);
   687→      if (!forced) continue;
   688→      m.regime = finalizeRegimeScore(forced.decision, m.monteCarlo!, m.favorite);
   689→      m.crown = assignCrown(
   690→        false,
   691→        forced.pairLoss,
   692→        m.regime.sharpe,
   693→        m.regime.mcScore
   694→      );
   695→      // only DUAL-HEDGE cards ride the atomic slips — forced regimes do not
   696→      if (m.regime.code !== "DUAL-HEDGE") {
   697→        m.slips = {};
   698→        m.dualEligible = false;
   699→      }
   700→      counts[code] += 1;
   701→      counts["DUAL-HEDGE"] -= 1;
   702→    }
   703→  };
   704→
   705→  // 🏦 banker top-up — strongest favorites first (master prompt: S_A ≥ 0.70)
   706→  fill("BANKER", (m) => {
   707→    const sFav = m.favorite === "A" ? m.markov.setWinA : 1 - m.markov.setWinA;
   708→    return sFav * 2 + m.favWinProb;
   709→  });
   710→  // 🧩 copula-combo top-up — highest fav×games joint mass (dominant games lines)
   711→  fill("COMBO", (m) => {
   712→    const mc = m.monteCarlo!;
   713→    return Math.max(mc.favWinUnder215, mc.favWinOver205 + mc.favWinShareOver205);
   714→  });
   715→  // 🎯 exact-cluster top-up — most concentrated top-3 scenario mass
   716→  fill("CLUSTER", (m) => m.monteCarlo!.topScenariosMass * 2 - m.monteCarlo!.entropy);
   717→}

   719→// ─── v2.0 · 7-Day Historical Auto-Settlement ───────────────────────────────
   720→// v5.5 — the settled 7-day record is (re)built through the SAME 5-regime
   721→// pipeline as today's card: analyze → crown → allocateFiveRegimes → regime
   722→// item builders, so every regime of the «دفترکل افرازی» table holds settled
   723→// rows too. `historyVersion` marks the pipeline revision a match was
   724→// processed under; bumping the constant re-backfills the 7-day window ONCE.
   725→const HISTORY_PIPELINE_VERSION = 6; // v5.5 = 6
   726→
   727→interface HistorySummary {
   728→  analyzed: number;
   729→  approved: number;
   730→  items: LineItem[];
   731→}
   732→
   733→async function processHistorySettlement(
   734→  rankLookup: Map<string, RankRow>
   735→): Promise<HistorySummary> {
   736→  const summary: HistorySummary = { analyzed: 0, approved: 0, items: [] };
   737→
   738→  try {
   739→    // 0) v5.5 one-time re-backfill — matches processed under the pre-v5.5
   740→    //    pipeline (broken crown gate, VOID-settling BANKER/CLUSTER legs, no
   741→    //    regime distribution) are re-analyzed so the settled record fills
   742→    //    all 5 regimes of the master prompt.
   743→    await db.tennisMatch
   744→      .updateMany({
   745→        where: { historyProcessed: true, historyVersion: { lt: HISTORY_PIPELINE_VERSION } },
   746→        data: { historyProcessed: false },
   747→      })
   748→      .catch(() => undefined);
   749→
   750→    // 1) finished matches of today + the past 7 days — today's post matches
   751→    //    flow through the same fresh-check as the older days
   752→    const dateKeys = [
   753→      todayKey(),
   754→      ...Array.from({ length: HISTORY_WINDOW_DAYS }, (_, i) => dateKeyDaysAgo(i + 1)),
   755→    ];
   756→    const tours: Tour[] = ["ATP", "WTA"];
   757→    const fetched: EspnMatch[] = [];
   758→    await Promise.all(
   759→      tours.flatMap((t) =>
   760→        dateKeys.map(async (d) => {
   761→          try {
   762→            fetched.push(...(await fetchScoreboard(t, d)));
   763→          } catch (e) {
   764→            console.error("[history] scoreboard fetch failed", t, d, e);
   765→          }
   766→        })
   767→      )
   768→    );
   769→
   770→    const finished = dedupeByKey(
   771→      fetched.filter((m) => m.state === "post" && m.singles),
   772→      (m) => m.eventKey
   773→    );
   774→
   775→    // 2) which of these are new (never processed for settlement)?
   776→    const eventKeys = finished.map((m) => m.eventKey);
   777→    const knownRows = eventKeys.length
   778→      ? await db.tennisMatch.findMany({
   779→          where: { eventKey: { in: eventKeys } },
   780→          select: { eventKey: true, historyProcessed: true },
   781→        })
   782→      : [];
   783→    const processed = new Set(
   784→      knownRows.filter((r) => r.historyProcessed).map((r) => r.eventKey)
   785→    );
   786→    const fresh = finished.filter((m) => !processed.has(m.eventKey));
   787→    if (fresh.length === 0) return summary;
   788→
   789→    // 3) attach official ranking points + analyze (v3.0 resolver ladder)
   790→    const processedKeys: string[] = [];
   791→    const freshAnalyzed: AnalyzedMatch[] = [];
   792→    for (const m of fresh) {
   793→      try {
   794→        attachResolvedRanks(m, rankLookup);
   795→        const am = analyzeMatch(m, undefined);
   796→        freshAnalyzed.push(am);
   797→        processedKeys.push(m.eventKey);
   798→      } catch (e) {
   799→        console.error("[history] analysis failed", m.eventKey, e);
   800→      }
   801→    }
   802→
   803→    // 3.5) v5.5 — distribute the settled pool across ALL 5 regimes too
   804→    allocateFiveRegimes(freshAnalyzed, true);
   805→
   806→    // 4) persist match rows (marks them processed) and build the routed items
   807→    await persistMatches(freshAnalyzed, true);
   808→    for (const am of freshAnalyzed) {
   809→      const result = extractSettleableResult(am);
   810→      const approved =
   811→        !am.veto.vetoed && !!am.regime && am.regime.code !== "VETOED" && !!am.crown;
   812→      if (approved && result) {
   813→        switch (am.regime?.code) {
   814→          case "BANKER": {
   815→            const item = bankerItemOf(am);
   816→            if (item) summary.items.push(item);
   817→            break;
   818→          }
   819→          case "CLUSTER":
   820→            summary.items.push(...clusterItemsOf(am));
   821→            break;
   822→          case "COMBO":
   823→            summary.items.push(...comboRegimeItemsOf(am));
   824→            break;
   825→          default:
   826→            if (am.slips.A && am.slips.B) summary.items.push(...dualItemsOf(am));
   827→        }
   828→        summary.approved++;
   829→      }
   830→    }
   831→
   832→    // 4.5) v5.5 — 🔗 SERVICE KINGS ACCUMULATOR for the settled record: the
   833→    // 2/3 strongest BANKER-routed matches of each settled date combine into
   834→    // a 100-unit parlay item with the explicit per-leg settlement contract.
   835→    const bankerByDate = new Map<string, AnalyzedMatch[]>();
   836→    for (const am of freshAnalyzed) {
   837→      if (
   838→        am.regime?.code === "BANKER" &&
   839→        !am.veto.vetoed &&
   840→        !!am.crown &&
   841→        extractSettleableResult(am)
   842→      ) {
   843→        const arr = bankerByDate.get(am.dateKey) ?? [];
   844→        arr.push(am);
   845→        bankerByDate.set(am.dateKey, arr);
   846→      }
   847→    }
   848→    for (const [, legs] of bankerByDate) {
   849→      const ranked = legs
   850→        .sort((a, b) => (b.mc?.bankerProb ?? b.favWinProb) - (a.mc?.bankerProb ?? a.favWinProb));
   851→      const acc2 = accumulatorItemOf(ranked.slice(0, 2), "ACC-2");
   852→      const acc3 = accumulatorItemOf(ranked.slice(0, 3), "ACC-3");
   853→      if (acc2) summary.items.push(acc2);
   854→      if (acc3) summary.items.push(acc3);
   855→    }
   856→    summary.analyzed = freshAnalyzed.length;
   857→
   858→    // 5) engine log + explicit processed-marker for rows that pre-existed
   859→    await db.engineLog
   860→      .create({
   861→        data: {
   862→          kind: "HISTORY",
   863→          message: `v5.5 7-day auto-settlement scan: analyzed=${summary.analyzed} approved=${summary.approved} items=${summary.items.length}`,
   864→        },
   865→      })
   866→      .catch(() => undefined);
   867→
   868→    if (processedKeys.length) {
   869→      await db.tennisMatch
   870→        .updateMany({
   871→          where: { eventKey: { in: processedKeys } },
   872→          data: { historyProcessed: true, historyVersion: HISTORY_PIPELINE_VERSION },
   873→        })
   874→        .catch(() => undefined);
   875→    }
   876→  } catch (e) {
   877→    console.error("[history] pipeline failed", e);
   878→  }
   879→
   880→  return summary;
   881→}

   961→  const v5 = {
   962→    regime,

   964→    pipeline: m.pipeline,

   972→      leg: regime.legCodes.main,
   973→      desc: legs.mainFa,

   978→      marketEn: legs.mainEn,
   979→      marketFa: legs.mainFa,
   980→      espnUrl: espnMatchUrl(m.eventKey),
   981→      targetWindow: [1.85, 2.4] as [number, number],
   982→      ...v5,
