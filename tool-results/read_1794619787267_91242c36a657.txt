   212→/** v4.0 — equal-profit dutching stakes from the two leg prices (sum = 100). */
   213→export function dutchStakes(
   214→  oddsA: number,
   215→  oddsB: number
   216→): { stakeA: number; stakeB: number } {
   217→  const invA = 1 / Math.max(oddsA, 1.01);
   218→  const invB = 1 / Math.max(oddsB, 1.01);
   219→  const stakeA = Math.round(((100 * invA) / (invA + invB)) * 10) / 10;
   220→  const stakeB = Math.round((100 - stakeA) * 10) / 10;
   221→  return { stakeA, stakeB };
   222→}
   223→
   224→/** v4.0 — dutching stake (units) of a single DUAL wing inside its 100-unit match position. */
   225→export function dualStakeFor(leg: string): number {
   226→  const pair = LEG_PAIR[leg];
   227→  if (!pair) return 50;
   228→  if (LEGACY_LEGS.has(leg)) return 50; // historical accounting identity
   229→  const oddsLeg = DUAL_LEG_MARKET_ODDS[leg] ?? 2;
   230→  const oddsPair = DUAL_LEG_MARKET_ODDS[pair] ?? 2;
   231→  const { stakeA, stakeB } = dutchStakes(oddsLeg, oddsPair);
   232→  return LEG_IS_A.has(leg) ? stakeA : stakeB;
   233→}

   244→/**
   245→ * Evaluates the 3 official atomic archetypes and selects the eligible one
   246→ * with the lowest exact simultaneous-loss probability. If none qualifies the
   247→ * match is vetoed to the shadow ledger (no forced approvals).
   248→ */
   249→export function evaluateArchetypes(m: AnalyzedMatch): ArchetypeReport {
   250→  const a = m.markov.arch;
   251→  const sFav = m.favorite === "A" ? m.markov.setWinA : 1 - m.markov.setWinA;
   252→  const holdFav = m.favorite === "A" ? m.markov.gA : m.markov.gB;
   253→  const holdDog = m.favorite === "A" ? m.markov.gB : m.markov.gA;
   254→
   255→  const mkEval = (
   256→    code: "ARCH1" | "ARCH2" | "ARCH3",
   257→    pairLoss: number,
   258→    windowOk: boolean,
   259→    windowWhy: string
   260→  ): ArchetypeEval => {
   261→    const max =
   262→      code === "ARCH1"
   263→        ? ARCH1_MAX_PAIR_LOSS
   264→        : code === "ARCH2"
   265→          ? ARCH2_MAX_PAIR_LOSS
   266→          : ARCH3_MAX_PAIR_LOSS;
   267→    const riskOk = pairLoss <= max;
   268→    const eligible = windowOk && riskOk;
   269→    return {
   270→      code,
   271→      labelFa: ARCH_LABELS[code].fa,
   272→      labelEn: ARCH_LABELS[code].en,
   273→      coverage: 1 - pairLoss,
   274→      pairLoss,
   275→      eligible,
   276→      whyNot: eligible
   277→        ? undefined
   278→        : !windowOk
   279→          ? `${windowWhy} (${code === "ARCH2" ? "پنجره الگو ۲: 54% ≤ S_A < 64%" : code === "ARCH1" ? "پنجره الگو ۱: S_fav ≥ 64%" : "پنجره الگو ۳: Hold ≥ 84%"})`
   280→          : `ریسک باخت همزمان ${(pairLoss * 100).toFixed(1)}% بالاتر از سقف ${(max * 100).toFixed(1)}%`,
   281→    };
   282→  };
   283→
   284→  const arch1 = mkEval(
   285→    "ARCH1",
   286→    a.pairLoss1,
   287→    sFav >= ARCH1_MIN_S,
   288→    `S_fav = ${(sFav * 100).toFixed(1)}% زیر آستانه تسلط 64%`
   289→  );
   290→  const arch2 = mkEval(
   291→    "ARCH2",
   292→    a.pairLoss2,
   293→    m.markov.setWinA >= ARCH2_MIN_S && m.markov.setWinA < ARCH2_MAX_S,
   294→    m.markov.setWinA < ARCH2_MIN_S
   295→      ? `S_A = ${(m.markov.setWinA * 100).toFixed(1)}% زیر کف پنجره رقابتی 54%`
   296→      : `S_A = ${(m.markov.setWinA * 100).toFixed(1)}% بالاتر از سقف پنجره رقابتی 64%`
   297→  );
   298→  const arch3 = mkEval(
   299→    "ARCH3",
   300→    a.pairLoss3,
   301→    holdFav >= ARCH3_MIN_HOLD_FAV && holdDog >= ARCH3_MIN_HOLD_DOG,
   302→    `Hold فاوریت ${(holdFav * 100).toFixed(1)}% / حریف ${(holdDog * 100).toFixed(1)}% — زیر آستانه دوئل سرویس (84%/82%)`
   303→  );
   304→
   305→  const evals = [arch1, arch2, arch3];
   306→  const eligible = evals.filter((e) => e.eligible);
   307→  if (eligible.length === 0) return { evals };
   308→
   309→  // most robust coverage wins the tie
   310→  eligible.sort((x, y) => x.pairLoss - y.pairLoss);
   311→  const win = eligible[0];
   312→  return {
   313→    evals,
   314→    selected: {
   315→      code: win.code,
   316→      labelFa: win.labelFa,
   317→      labelEn: win.labelEn,
   318→      coverage: win.coverage,
   319→      pairLoss: win.pairLoss,
   320→    },
   321→  };
   322→}

   324→/** Hard veto gate: Top-100 domain check + 3-archetype approval check. */
   325→export function supervisorGate(m: AnalyzedMatch, selected?: ArchetypeInfo): VetoInfo {
   326→  const rankA = m.playerA.rank;
   327→  const rankB = m.playerB.rank;
   328→  const badRankA = !rankA || rankA > TOP100_LIMIT;
   329→  const badRankB = !rankB || rankB > TOP100_LIMIT;
   330→  if (badRankA || badRankB) {
   331→    return {
   332→      vetoed: true,
   333→      code: "TOP100",
   334→      reason: `فیلتر تاپ-۱۰۰ سوپروایزر — ${badRankA ? `${m.playerA.name}${rankA ? ` (#${rankA})` : " (بدون رتبه)"}` : ""}${badRankA && badRankB ? " و " : ""}${badRankB ? `${m.playerB.name}${rankB ? ` (#${rankB})` : " (بدون رتبه)"}` : ""} خارج از دامنه دقت مدل (Kovalchik 2016)`,
   335→    };
   336→  }
   337→
   338→  if (!selected) {
   339→    const sFav = m.favorite === "A" ? m.markov.setWinA : 1 - m.markov.setWinA;
   340→    const best = [...(m.archEvals ?? [])].sort((x, y) => x.pairLoss - y.pairLoss)[0];
   341→    const bestTxt = best
   342→      ? `کم‌ریسک‌ترین الگو (${best.labelFa}) ریسک ${(best.pairLoss * 100).toFixed(1)}% دارد`
   343→      : "هیچ الگویی ارزیابی نشد";
   344→    return {
   345→      vetoed: true,
   346→      code: "P02_RISK",
   347→      reason: `هیچ الگوی مکملی صلاحیت ندارد — S_fav = ${(sFav * 100).toFixed(1)}% خارج از پنجره الگوها و ${bestTxt}؛ انتقال به دفتر سایه`,
   348→    };
   349→  }
   350→
   351→  return { vetoed: false, code: "NONE" };
   352→}

   459→// ─── v3.0 · Guaranteed 65% Approval Quota ──────────────────────────────────
   460→// The v2.0 audit found ALL 40 matches of the day vetoed (0 approved / 40
   461→// vetoed) because missing scoreboard ranks starved the Top-100 gate. The
   462→// rank fix (resolvePlayerRankAndApprove in the engine) removes the starvation
   463→// cause; this quota guarantees the OUTPUT side: ≥ 65% of today's main-tour

   467→export interface QuotaTelemetry {
   468→  poolSize: number;
   469→  approved: number;
   470→  natural: number;
   471→  adaptive: number;
   472→  ratio: number;
   473→}

   475→const QUOTA_RATIO = 0.65;
   476→const QUOTA_MIN = 12;
   477→
   478→const isMainTourRound = (round: string): boolean => !/qual|qualif|round robin ext/i.test(round);
   479→
   480→/**
   481→ * Enforces the ≥65% approval quota on today's (pre/in) main-tour singles
   482→ * card. v5.0 — NATURAL approvals are 5-regime router routes (Station 4);
   483→ * remaining quota slots are filled with an ADAPTIVE max-Sharpe regime grant
   484→ * stamped as a SILVER crown (half stake). Mutates the matches in place.
   485→ * `simsByEvent` carries the raw 10k MC tuples of each match (transient).
   486→ */
   487→export function enforceApprovalQuota(
   488→  analyzed: AnalyzedMatch[],
   489→  simsByEvent: Map<string, McSim[]>
   490→): QuotaTelemetry {
   491→  const telemetry: QuotaTelemetry = {
   492→    poolSize: 0,
   493→    approved: 0,
   494→    natural: 0,
   495→    adaptive: 0,
   496→    ratio: 0,
   497→  };
   498→
   499→  const pool = analyzed.filter(
   500→    (m) => m.singles && m.state !== "post" && isMainTourRound(m.round)
   501→  );
   502→  telemetry.poolSize = pool.length;
   503→  if (pool.length === 0) return telemetry;
   504→
   505→  const natural = pool.filter((m) => !m.veto.vetoed && !!m.regime && !!m.crown);
   506→  telemetry.natural = natural.length;
   507→  let approvedCount = natural.length;
   508→
   509→  const quota = Math.min(
   510→    pool.length,
   511→    Math.max(QUOTA_MIN, Math.ceil(pool.length * QUOTA_RATIO))
   512→  );
   513→
   514→  if (approvedCount < quota) {
   515→    const remaining = quota - approvedCount;
   516→    const candidates = pool
   517→      .filter((m) => !m.regime && !!m.mc)
   518→      .map((m) => {
   519→        const evals = [...(m.archEvals ?? [])].sort((a, b) => a.pairLoss - b.pairLoss);
   520→        return { m, best: evals[0], sims: simsByEvent.get(m.eventKey) ?? [] };
   521→      })
   522→      .filter((c) => c.best)
   523→      .sort((a, b) => (a.best!.pairLoss - b.best!.pairLoss))
   524→      .slice(0, remaining);
   525→
   526→    for (const { m, sims } of candidates) {
   527→      if (!m.mc) continue;
   528→      // adaptive grant — least-risk regime by 10k Monte-Carlo Sharpe
   529→      const route = adaptiveRegimeGrant(m, m.mc, sims);
   530→      if (!route) continue;
   531→      m.regime = route;
   532→      m.crown = crownTierOf(route, m.mc, true);
   533→      m.veto = { vetoed: false, code: "NONE" };
   534→      m.dualEligible = true;
   535→      // DUAL_HEDGE grants keep an ARCH1 archetype so the legacy slips,
   536→      // dutching stakes and settlement legs stay in one consistent shape
   537→      if (route.code === "DUAL_HEDGE") {
   538→        m.archetype = {
   539→          code: "ARCH1",
   540→          labelFa: ARCH_LABELS.ARCH1.fa,
   541→          labelEn: ARCH_LABELS.ARCH1.en,
   542→          coverage: 1 - m.mc.p02,
   543→          pairLoss: m.mc.p02,
   544→          adaptive: true,
   545→        };
   546→        m.slips = buildSlips(m);
   547→      } else {
   548→        m.slips = {};
   549→      }
   550→      approvedCount++;
   551→      telemetry.adaptive++;
   552→    }
   553→  }
   554→
   555→  telemetry.approved = approvedCount;
   556→  telemetry.ratio = approvedCount / pool.length;
   557→  return telemetry;
   558→}

   590→function simFavFrame(s: McSim, fav: "A" | "B") {
   591→  const flip = fav === "B";
   592→  const favWon = flip ? s.code === 2 || s.code === 3 : s.code === 0 || s.code === 1;
   593→  const favWon20 = flip ? s.code === 3 : s.code === 0;

   598→/** Non-linear Sharpe score = mean(P&L) / std(P&L) across the 10k sims. */
   599→export function sharpeOf(pnl: number[]): number {
   600→  const n = pnl.length;
   601→  if (n === 0) return 0;
   602→  const mean = pnl.reduce((a, b) => a + b, 0) / n;
   603→  const varr = pnl.reduce((a, b) => a + (b - mean) * (b - mean), 0) / n;
   604→  const std = Math.sqrt(varr);
   605→  return std < 1e-9 ? 0 : mean / std;
   606→}

   608→// ─── regime P&L distributions over the Monte-Carlo sims ────────────────────
   609→function regimePnL(
   610→  code: RegimeCode,
   611→  sims: McSim[],
   612→  fav: "A" | "B",
   613→  variant?: "UNDER" | "OVER"
   614→): number[] {
   615→  const { stakeA, stakeB } = dutchStakes(
   616→    DUAL_LEG_MARKET_ODDS["SB-F20"],
   617→    DUAL_LEG_MARKET_ODDS["TS-O25"]
   618→  );
   619→  return sims.map((s) => {
   620→    const f = simFavFrame(s, fav);
   621→    switch (code) {
   622→      case "BANKER": {
   623→        // MW-FAV @ 1.48 / SH-F15 @ 1.62 — 60u direct stake
   624→        const odds = variant === "UNDER" ? DUAL_LEG_MARKET_ODDS["SH-F15"] : DUAL_LEG_MARKET_ODDS["MW-FAV"];
   625→        const won = variant === "UNDER" ? f.favWon20 : f.favWon;
   626→        return won ? 60 * (odds - 1) : -60;
   627→      }
   628→      case "DUAL_HEDGE": {
   629→        // SB-F20 @ 1.97 ∪ TS-O25 @ 2.68 — equal-profit dutch (champion pattern)
   630→        const pnlA = f.favWon20 ? stakeA * (DUAL_LEG_MARKET_ODDS["SB-F20"] - 1) : -stakeA;
   631→        const pnlB = f.threeSets ? stakeB * (DUAL_LEG_MARKET_ODDS["TS-O25"] - 1) : -stakeB;
   632→        return pnlA + pnlB;
   633→      }
   634→      case "COMBO": {
   635→        const mainOdds = variant === "UNDER" ? DUAL_LEG_MARKET_ODDS["RC-FWU215"] : DUAL_LEG_MARKET_ODDS["RC-FWO205"];
   636→        const coverOdds = variant === "UNDER" ? DUAL_LEG_MARKET_ODDS["RC-O215"] : DUAL_LEG_MARKET_ODDS["RC-U205"];
   637→        const { stakeA: sm, stakeB: sc } = dutchStakes(mainOdds, coverOdds);
   638→        const mainWon =
   639→          variant === "UNDER"
   640→            ? f.favWon && f.total <= 21
   641→            : f.favWon && f.total >= 21;
   642→        const coverWon = variant === "UNDER" ? f.total >= 22 : f.total <= 20;
   643→        return (mainWon ? sm * (mainOdds - 1) : -sm) + (coverWon ? sc * (coverOdds - 1) : -sc);
   644→      }
   645→      case "CLUSTER": {
   646→        const [s1, s2, s3] = clusterStakes(
   647→          DUAL_LEG_MARKET_ODDS["CL-F20U195"],
   648→          DUAL_LEG_MARKET_ODDS["CL-F20G20"],
   649→          DUAL_LEG_MARKET_ODDS["CL-F21"]
   650→        );
   651→        const w1 = f.favWon20 && f.total <= 19;
   652→        const w2 = f.favWon20 && f.total >= 20 && f.total <= 22;
   653→        const w3 = f.favWon && f.threeSets;
   654→        let pnl = 0;
   655→        if (w1) pnl += s1 * (DUAL_LEG_MARKET_ODDS["CL-F20U195"] - 1); else pnl -= s1;
   656→        if (w2) pnl += s2 * (DUAL_LEG_MARKET_ODDS["CL-F20G20"] - 1); else pnl -= s2;
   657→        if (w3) pnl += s3 * (DUAL_LEG_MARKET_ODDS["CL-F21"] - 1); else pnl -= s3;
   658→        return pnl;
   659→      }
   660→      default:
   661→        return 0;
   662→    }
   663→  });
   664→}

   666→// ─── v5.0 — 3-way dutching for the CLUSTER regime (sum = 100) ──────────────
   667→export function clusterStakes(odds1: number, odds2: number, odds3: number): [number, number, number] {
   668→  const inv = [1 / Math.max(odds1, 1.01), 1 / Math.max(odds2, 1.01), 1 / Math.max(odds3, 1.01)];
   669→  const s = inv[0] + inv[1] + inv[2];
   670→  const raw = inv.map((x) => (100 * x) / s);
   671→  const rounded = raw.map((x) => Math.round(x * 10) / 10);
   672→  const drift = Math.round((100 - (rounded[0] + rounded[1] + rounded[2])) * 10) / 10;
   673→  rounded[1] = Math.round((rounded[1] + drift) * 10) / 10;
   674→  return [rounded[0], rounded[1], rounded[2]];
   675→}

   677→// ─── 10k Monte-Carlo score (0..100) ────────────────────────────────────────
   678→export function monteCarloScore(route: RegimeRoute, mc: MonteCarloSummary): number {
   679→  const cov = Math.max(0, Math.min(1, route.coverage));
   680→  const sh = Math.max(0, Math.min(1, route.sharpe / 0.6));
   681→  const bp = Math.max(0, Math.min(1, mc.bankerProb));
   682→  return Math.round(100 * (0.4 * cov + 0.3 * sh + 0.3 * bp));
   683→}

  1134→/** Per-unit betting Sharpe of a single-leg position: EV / σ of the P&L. */
  1135→export function bettingSharpe(prob: number, odds: number): number {
  1136→  const ev = prob * odds - 1;
  1137→  const secondMoment = prob * Math.pow(odds - 1, 2) + (1 - prob);
  1138→  const variance = secondMoment - ev * ev;
  1139→  return variance > 1e-9 ? ev / Math.sqrt(variance) : 0;
  1140→}
  1141→
  1142→/** Equal-profit dutching stakes for N legs (sum = total). */
  1143→function dutchN(oddsList: number[], total = 100): number[] {
  1144→  const invs = oddsList.map((o) => 1 / Math.max(o, 1.01));
  1145→  const s = invs.reduce((a, b) => a + b, 0);
  1146→  return invs.map((i) => Math.round(((total * i) / s) * 10) / 10);
  1147→}
  1148→
  1149→/**
  1150→ * Per-unit Sharpe of a dutched multi-leg position over disjoint outcomes:
  1151→ * winProbs[i] → leg i pays; else the whole position loses.
  1152→ */
  1153→function dutchedSharpe(winProbs: number[], oddsList: number[]): { sharpe: number; ev: number; stakes: number[] } {
  1154→  const stakes = dutchN(oddsList, 100);
  1155→  const payoffs = stakes.map((st, i) => (st * oddsList[i] - 100) / 100); // per-unit net
  1156→  const pLose = Math.max(0, 1 - winProbs.reduce((a, b) => a + b, 0));
  1157→  const ev = winProbs.reduce((s, p, i) => s + p * payoffs[i], 0) - pLose * 1;
  1158→  const secondMoment =
  1159→    winProbs.reduce((s, p, i) => s + p * payoffs[i] * payoffs[i], 0) + pLose;
  1160→  const variance = secondMoment - ev * ev;
  1161→  return {
  1162→    sharpe: variance > 1e-9 ? ev / Math.sqrt(variance) : 0,
  1163→    ev,
  1164→    stakes,
  1165→  };
  1166→}

  1223→ * the match is vetoed to the shadow ledger.

  1225→export function routeRegime(
  1226→  m: AnalyzedMatch,
  1227→  mc: MonteCarloResult,
  1228→  _archPairLoss1: number
  1229→): RegimeRoute {
  1230→  const ctx = regimeContext(m, mc);
  1231→  const favName = favNameOf(m).toUpperCase();
  1232→  const favNameRaw = favNameOf(m);
  1233→  const evals: RegimeEval[] = [];
  1234→  const candidates: { decision: RegimeDecision; sharpe: number; pairLoss: number }[] = [];
  1235→
  1236→  // ── 1 · 🏦 TENNIS BANKER ─────────────────────────────────────────────────
  1237→  {
  1238→    const windowOk = ctx.sFav >= BANKER_MIN_S && ctx.holdFav >= BANKER_MIN_HOLD;
  1239→    const pairLoss = 1 - ctx.pFavWin;
  1240→    const cands: { marketEn: string; marketFa: string; prob: number; window: [number, number] }[] = [
  1241→      {
  1242→        marketEn: `MATCH WINNER — ${favName}`,
  1243→        marketFa: `برد مسابقه: ${favNameRaw}`,
  1244→        prob: ctx.pFavWin,
  1245→        window: [1.38, 1.65],
  1246→      },
  1247→      {
  1248→        marketEn: `SET HANDICAP — ${favName} −1.5 SETS`,
  1249→        marketFa: `هندیکپ ست: ${favNameRaw} ۲-۰`,
  1250→        prob: ctx.pFav20,
  1251→        window: [1.38, 1.65],
  1252→      },
  1253→    ];
  1254→    let bestSharpe = -Infinity;
  1255→    let best: (typeof cands)[number] | null = null;
  1256→    let bestPrice = 1.38;
  1257→    for (const c of cands) {
  1258→      const price = clampPrice(Math.max(c.window[0], fairPrice(c.prob, BOOK_MARGIN_MAIN)), c.window[0], c.window[1]);
  1259→      const sh = bettingSharpe(c.prob, price);
  1260→      if (sh > bestSharpe) {
