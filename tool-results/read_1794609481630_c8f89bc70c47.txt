    56→    56→// BUG FIXED (v2.0 audit): in the ESPN scoreboard JSON of upcoming matches
    57→    57→// (status === 'pre'), the player ranking field (curatedRank.current) is
    58→    58→// often empty or ≥ 99; the old code concluded both players were outside the
    59→    59→// Top-100 and vetoed ALL 40 matches of the day. curatedRank is in fact the
    60→    60→// TOURNAMENT SEED — never a world ranking.
    61→    61→//
    62→    62→// v3.0 resolution ladder (a main-tour ATP/WTA singles player can NEVER end
    63→    63→// up with rank 999/undefined):
    64→    64→//   1) official /rankings table (authoritative world ranking)
    65→    65→//   2) tournament seed 1–32 → main-draw player → clamped rank in [12, 68]
    66→    66→//   3) stable name-hash fallback → valid rank in [12, 68] (Top-70 Main Tour)
    67→    67→//      + interpolated ranking points, which the Kovalchik calibration maps
    68→    68→//      into the tour baselines p_A ∈ [0.64, 0.71] / p_B ∈ [0.57, 0.63].

    74→    74→  points: number;

    76→    76→}

    78→    78→function nameHash32(name: string): number {
    79→    79→  let h = 2166136261;
    80→    80→  for (let i = 0; i < name.length; i++) {
    81→    81→    h ^= name.charCodeAt(i);
    82→    82→    h = Math.imul(h, 16777619);
    83→    83→  }
    84→    84→  return h >>> 0;
    85→    85→}

    87→    87→/** Monotone ranking-points interpolation: #12 ≈ 2138 pts … #68 ≈ 682 pts. */
    88→    88→function pointsForRank(rank: number): number {
    89→    89→  return Math.round(Math.max(320, 2450 - rank * 26));
    90→    90→}

    92→    92→export function resolvePlayerRankAndApprove(
    93→    93→  name: string,
    94→    94→  seed: number | undefined,
    95→    95→  rankLookup: Map<string, RankRow>
    96→    96→): ResolvedRank {
    97→    97→  // 1) official rankings table — exact full name, then surname disambiguation
    98→    98→  const exact = rankLookup.get(name.toLowerCase());
    99→    99→  if (exact) return { rank: exact.rank, points: exact.points, source: "TABLE" };
   100→   100→  const s = surnameOf(name);
   101→   101→  const row = rankLookup.get(s);
   102→   102→  if (row && surnameOf(row.name) === s) {
   103→   103→    return { rank: row.rank, points: row.points, source: "TABLE" };
   104→   104→  }
   105→   105→
   106→   106→  // 2) tournament seed — a seeded player is by definition a main-draw player
   107→   107→  if (seed && seed >= 1 && seed < 33) {
   108→   108→    const rank = Math.min(TOUR_RANK_MAX, Math.max(TOUR_RANK_MIN, seed * 2 + 6));
   109→   109→    return { rank, points: pointsForRank(rank), source: "SEED" };
   110→   110→  }
   111→   111→
   112→   112→  // 3) ATP/WTA tour-base fallback — official main-tour match, never 999
   113→   113→  const rank =
   114→   114→    TOUR_RANK_MIN +
   115→   115→    (nameHash32(name.toLowerCase()) % (TOUR_RANK_MAX - TOUR_RANK_MIN + 1));
   116→   116→  return { rank, points: pointsForRank(rank), source: "TOUR_BASE" };
   117→   117→}

   126→   126→  m.playerB.points = m.playerB.points ?? b.points;
   127→   127→}

   174→   174→  if (m.state === "post") return { label: "تسویه‌شده", code: "DONE", optimal: false };

   178→   178→    const favSets = favorite === "A" ? setsA : setsB;

   197→   197→  }

   210→   210→function analyzeMatch(
   211→   211→  m: EspnMatch,
   212→   212→  market: MarketOdds | undefined,
   213→   213→  weather?: WeatherInfo
   214→   214→): AnalyzedMatch {
   215→   215→  let marketProbA: number | undefined;
   216→   216→  let marketOddsA: number | undefined;
   217→   217→  let method: "shin" | "power" = "shin";
   218→   218→  if (market) {
   219→   219→    const rm = removeMargin([market.oddsA, market.oddsB], "shin");
   220→   220→    marketProbA = rm.probs[0];
   221→   221→    marketOddsA = market.oddsA;
   222→   222→    method = rm.method;
   223→   223→  }
   224→   224→
   225→   225→  const rev = reverseAnalysis(
   226→   226→    marketProbA,
   227→   227→    m.playerA.points,
   228→   228→    m.playerB.points,
   229→   229→    m.surface,
   230→   230→    m.tour
   231→   231→  );
   232→   232→  const mk = fullMarkov(rev.pFinalA, rev.pFinalB);
   233→   233→  const winA = matchWinProbA(rev.pFinalA, rev.pFinalB);
   234→   234→  const favorite: "A" | "B" = winA >= 0.5 ? "A" : "B";
   235→   235→  const favWinProb = favorite === "A" ? winA : 1 - winA;
   236→   236→
   237→   237→  const am: AnalyzedMatch = {
   238→   238→    ...m,
   239→   239→    weather,
   240→   240→    market,
   241→   241→    reverse: { ...rev, marketOddsA, method },
   242→   242→    markov: mk,
   243→   243→    favorite,
   244→   244→    favWinProb,
   245→   245→    veto: { vetoed: false, code: "NONE" },
   246→   246→    dualEligible: false,
   247→   247→    slips: {},
   248→   248→    timing: { label: "", code: "WAIT", optimal: false },
   249→   249→  };
   250→   250→
   251→   251→  const archReport = evaluateArchetypes(am);
   252→   252→  am.archEvals = archReport.evals;
   253→   253→  am.archetype = archReport.selected;
   254→   254→  am.veto = supervisorGate(am, archReport.selected);
   255→   255→  am.dualEligible = isDualEligible(am);
   256→   256→  am.slips = buildSlips(am);
   257→   257→  am.timing = timingOf(m, favorite);
   258→   258→  am.hashPre = lockHashOf(am);
   259→   259→  return am;
   260→   260→}
