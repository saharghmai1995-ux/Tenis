    39→export function tehranFields(ms = Date.now()): {
    40→  y: number; mon: number; dom: number; h: number; min: number; s: number; dow: number;
    41→} {
    42→  const parts = partsFmt.formatToParts(new Date(ms));
    43→  const get = (t: string) => Number(parts.find((p) => p.type === t)?.value ?? 0);
    44→  const wd = dowFmt.format(new Date(ms)).toUpperCase();
    45→  return {
    46→    y: get("year"),
    47→    mon: get("month"),
    48→    dom: get("day"),
    49→    h: get("hour") % 24,
    50→    min: get("minute"),
    51→    s: get("second"),
    52→    dow: { SUN: 0, MON: 1, TUE: 2, WED: 3, THU: 4, FRI: 5, SAT: 6 }[wd] ?? 0,
    53→  };
    54→}

    63→type Field = { wildcard: boolean; any: Set<number>; step?: number };
    64→
    65→function parseField(raw: string, min: number, max: number): Field {
    66→  const any = new Set<number>();
    67→  let wildcard = false;
    68→  let step: number | undefined;
    69→  for (const chunk of raw.split(",")) {
    70→    if (chunk === "*") {
    71→      wildcard = true;
    72→      for (let i = min; i <= max; i++) any.add(i);
    73→    } else if (chunk.startsWith("*/")) {
    74→      wildcard = true;
    75→      step = Number(chunk.slice(2));
    76→      for (let i = min; i <= max; i += step) any.add(i);
    77→    } else if (chunk.includes("-")) {
    78→      const [a, b] = chunk.split("-").map(Number);
    79→      for (let i = a; i <= b; i++) any.add(i);
    80→    } else {
    81→      any.add(Number(chunk));
    82→    }
    83→  }
    84→  return { wildcard, any, step };
    85→}
    86→
    87→interface Cron {
    88→  sec: Field; min: Field; hour: Field; dom: Field; mon: Field; dow: Field;
    89→}
    90→
    91→function parseCron(expr: string): Cron {
    92→  const [sec, min, hour, dom, mon, dow] = expr.trim().split(/\s+/);
    93→  return {
    94→    sec: parseField(sec, 0, 59),
    95→    min: parseField(min, 0, 59),
    96→    hour: parseField(hour, 0, 23),
    97→    dom: parseField(dom, 1, 31),
    98→    mon: parseField(mon, 1, 12),
    99→    dow: parseField(dow, 0, 6),
   100→  };
   101→}

   103→function cronMatches(c: Cron, f: ReturnType<typeof tehranFields>): boolean {
   104→  return (
   105→    c.sec.any.has(f.s) &&
   106→    c.min.any.has(f.min) &&
   107→    c.hour.any.has(f.h) &&
   108→    c.mon.any.has(f.mon) &&
   109→    (c.dom.wildcard || c.dow.wildcard
   110→      ? c.dom.any.has(f.dom) || c.dow.any.has(f.dow)
   111→      : c.dom.any.has(f.dom) && c.dow.any.has(f.dow))
   112→  );
   113→}

   115→/** next run strictly after `after` (ms epoch) for a Tehran-anchored cron */
   116→export function nextCronRun(expr: string, after = Date.now()): number {
   117→  const c = parseCron(expr);
   118→  // iterate the next 2 days at second resolution starting from `after + 1s`
   119→  let t = after + 1000;
   120→  const limit = after + 2 * 86_400_000;
   121→  while (t <= limit) {
   122→    if (cronMatches(c, tehranFields(t))) return t;
   123→    // jump smartly: skip to next minute boundary when seconds won't match soon
   124→    const f = tehranFields(t);
   125→    if (!c.hour.any.has(f.h) || !c.min.any.has(f.min)) {
   126→      t += (59 - f.s) * 1000 + 1000; // finish current minute
   127→      if (!c.hour.any.has(f.h)) t += (59 - f.min) * 60_000; // finish current hour
   128→    } else {
   129→      t += 1000;
   130→    }
   131→  }
   132→  return after + 60_000;
   133→}

   179→async function deliver(
   180→  jobId: JobId,
   181→  target: string,
   182→  kind: PosterKind,
   183→  caption: string,
   184→  png: Buffer
   185→): Promise<PostOutcome> {
   186→  saveOutbox(jobId, png);
   187→  if (!isLive()) {
   188→    store.logPost({
   189→      at: new Date().toISOString(),
   190→      jobId,
   191→      target,
   192→      posterKind: kind,
   193→      mode: "DRY-RUN",
   194→      caption,
   195→      status: "DRY-RUN",
   196→    });
   197→    console.log(`[job:${jobId}] DRY-RUN ${kind} → ${target} (${Math.round(png.length / 1024)}KB)`);
   198→    return { status: "DRY-RUN" };
   199→  }
   200→  const chatId = target === "vip" ? vipChatId() : CONFIG.publicChatId;
   201→  if (!chatId) {
   202→    store.logPost({
   203→      at: new Date().toISOString(),
   204→      jobId,
   205→      target,
   206→      posterKind: kind,
   207→      mode: "LIVE",
   208→      caption,
   209→      status: "SKIPPED",
   210→      error: "chat id not configured",
   211→    });
   212→    return { status: "SKIPPED", error: "chat id not configured" };
   213→  }
   214→  const res = await telegram.sendPhoto(chatId, png, caption);
   215→  store.logPost({
   216→    at: new Date().toISOString(),
   217→    jobId,
   218→    target,
   219→    posterKind: kind,
   220→    mode: "LIVE",
   221→    messageId: res.messageId,
   222→    caption,
   223→    status: res.ok ? "SENT" : "ERROR",
   224→    error: res.description,
   225→  });
   226→  console.log(
   227→    `[job:${jobId}] ${res.ok ? "SENT" : "ERROR"} ${kind} → ${target}${res.messageId ? ` #${res.messageId}` : ""}${res.description ? ` (${res.description})` : ""}`
   228→  );
   229→  return { status: res.ok ? "SENT" : "ERROR", error: res.description, messageId: res.messageId };
   230→}

   626→  async flashcards() {
   627→    await ensureWarm();
   628→    const dateKey = tehranDateKey();
   629→    store.kvSet("flashDateKey", dateKey);
   630→    const pool = dataHub.approvedPool();
   631→    if (pool.length === 0) return { status: "SKIPPED", error: "flash pool empty" };
   632→    const hist = loadFlashHist(dateKey);
   633→    // ONE-POST POLICY: a match is eligible only if it has NEVER been posted
   634→    // today — exactly one card per game, then it never repeats this day.
   635→    const eligible = pool.filter((c) => !hist[c.eventKey]);
   636→    if (eligible.length === 0) {
   637→      return {
   638→        status: "SKIPPED",
   639→        error: `one-post policy - all ${pool.length} match(es) already have their card today`,
   640→      };
   641→    }
   642→    // soonest kick-off first → the most urgent match gets its single card first
   643→    eligible.sort((a, b) => {
   644→      const ta = a.startTime ? new Date(a.startTime).getTime() : Number.MAX_SAFE_INTEGER;
   645→      const tb = b.startTime ? new Date(b.startTime).getTime() : Number.MAX_SAFE_INTEGER;
   646→      if (ta !== tb) return ta - tb;
   647→      return (b.coverage ?? 0) - (a.coverage ?? 0);
   648→    });
   649→    const card = eligible[0];
   650→    // DELIVERY-FIRST POLICY (revival fix): the one-post mark is written only
   651→    // AFTER a confirmed SENT (or in dry-run mode). A failed Telegram delivery
   652→    // (network / auth / 404) does NOT burn the card — the next 90s tick
   653→    // retries the same match until the card is actually on the channel.
   654→    const angle = angleFor(card, 0);
   655→    const png = await render(flashcardRequest(card, "FLASHCARD", angle, 1));
   656→    const caption = cap.captionFlashcard(card, !!card.adaptive, angle);
   657→    const outcome = await deliver("flashcards", "vip", "FLASHCARD", caption, png);
   658→    if (outcome.status === "SENT" || outcome.status === "DRY-RUN") {
   659→      hist[card.eventKey] = { c: 1, at: Date.now() };
   660→      saveFlashHist(dateKey, hist);
   661→      store.kvSet("flashCursor", String(Object.values(hist).reduce((s, e) => s + e.c, 0)));
   662→    } else {
   663→      console.warn(
   664→        `[job:flashcards] delivery ${outcome.status} for ${card.eventKey} — card NOT marked, retry next tick (${outcome.error ?? "unknown"})`
   665→      );
   666→    }
   667→    return outcome;
   668→  },

   782→let running = new Set<JobId>();
   783→
   784→async function runJob(jobId: JobId, reason: "cron" | "manual" = "cron"): Promise<PostOutcome> {
   785→  if (running.has(jobId)) return { status: "SKIPPED", error: "already running" };
   786→  running.add(jobId);
   787→  try {
   788→    const out = await handlers[jobId]();
   789→    store.recordRun(jobId, out.status, { error: out.error, messageId: out.messageId });
   790→    return out;
   791→  } catch (e) {
   792→    const error = e instanceof Error ? e.message : String(e);
   793→    console.error(`[job:${jobId}] ${reason} error:`, error);
   794→    store.recordRun(jobId, "ERROR", { error });
   795→    return { status: "ERROR", error };
   796→  } finally {
   797→    running.delete(jobId);
   798→  }
   799→}

   801→function scheduleAll(): void {
   802→  for (const job of JOB_SPECS) {
   803→    if (job.cron) {
   804→      nextRunAt.set(job.id, nextCronRun(job.cron));
   805→    } else if (job.everySeconds) {
   806→      nextRunAt.set(job.id, Date.now() + (job.everySeconds * 1000));
   807→    }
   808→  }
   809→}
   810→
   811→function tick(): void {
   812→  const now = Date.now();
   813→  for (const job of JOB_SPECS) {
   814→    const next = nextRunAt.get(job.id);
   815→    if (next === undefined || now < next) continue;
   816→    if (job.cron) {
   817→      nextRunAt.set(job.id, nextCronRun(job.cron, now));
   818→    } else if (job.everySeconds) {
   819→      nextRunAt.set(job.id, now + job.everySeconds * 1000);
   820→    }
   821→    void runJob(job.id);
   822→  }
   823→}

   825→export function startScheduler(): void {
   826→  registerFonts();
   827→  restoreLive();
   828→  void refreshMe().catch(() => null);
   829→  scheduleAll();
   830→  // warm immediately so the first flashcard is never cold
   831→  void dataHub.warm().then((ok) => {
   832→    if (ok) store.kvSet("lastFeedWarmMs", String(Date.now()));
   833→  });
   834→  setInterval(tick, 1000);
   835→  // engine keep-warm loop — the core must never go cold
   836→  setInterval(() => {
   837→    void ensureWarm();
   838→  }, 5 * 60_000);
   839→  // tennis radar keep-warm (news/live/fitness cache stays hot)
   840→  setInterval(() => {
   841→    void fetchTennisRadar().catch(() => null);
   842→  }, 4 * 60_000);
   843→  // VIP channel auto-discovery — the moment the bot becomes admin of (or
   844→  // receives any update from) the VIP channel, its numeric id is captured.
   845→  setInterval(() => {
   846→    void discoverVipLoop();
   847→  }, 30_000);
   848→  console.log(
   849→    `[scheduler] started - ${JOB_SPECS.length} jobs - live=${isLive()} - tz=${TZ}`
   850→  );
   851→  for (const job of JOB_SPECS) {
   852→    const next = nextRunAt.get(job.id);
   853→    console.log(`[scheduler] ${job.id.padEnd(18)} next=${next ? new Date(next).toISOString() : "-"}`);
   854→  }
   855→  void discoverVipLoop();
   856→}

   858→/** capture the VIP channel id from any chat the bot can see */
   859→async function discoverVipLoop(): Promise<void> {
   860→  try {
   861→    const chats = await telegram.discoverChats();
   862→    for (const ch of chats) {
   863→      const isPublic = !!ch.username && CONFIG.publicChatId.includes(ch.username);
   864→      const isChannelish = ch.type === "channel" || ch.type === "supergroup";
   865→      if (isChannelish && ch.id) {
   866→        store.kvSet(`chatSeen:${ch.id}`, JSON.stringify(ch));
   867→      }
   868→      if (isChannelish && !isPublic && ch.id) {
   869→        const idStr = String(ch.id);
   870→        if (vipChatId() !== idStr) {
   871→          store.kvSet("vipChatId", idStr);
   872→          store.kvSet("vipChatTitle", ch.title ?? "");
   873→          console.log(`[discover] VIP chat captured: ${idStr} (${ch.title ?? ch.type})`);
   874→        }
   875→      }
   876→    }
   877→  } catch {
   878→    /* discovery is best-effort */
   879→  }
   880→}
