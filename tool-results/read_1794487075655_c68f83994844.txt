    90→    90→
    91→    91→function hashStringToSeed(input: string): number {
    92→    92→  let h = 1779033703 ^ input.length;
    93→    93→  for (let i = 0; i < input.length; i++) {
    94→    94→    h = Math.imul(h ^ input.charCodeAt(i), 3432918353);
    95→    95→    h = (h << 13) | (h >>> 19);
    96→    96→  }
    97→    97→  return h >>> 0;
    98→    98→}
    99→    99→
   100→   100→function mulberry32(seed: number): () => number {
   101→   101→  let a = seed >>> 0;
   102→   102→  return () => {
   103→   103→    a = (a + 0x6d2b79f5) | 0;
   104→   104→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   105→   105→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   106→   106→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   107→   107→  };
   108→   108→}

   112→   112→function font(weight: string, px: number, mono = false): string {
   113→   113→  return `${weight} ${px}px ${mono ? MONO_STACK : FONT_STACK}`;

   116→   116→/** Shrink font size until `text` fits `maxWidth`. Returns the final px size. */
   117→   117→function fitText(
   118→   118→  ctx: CanvasRenderingContext2D,
   119→   119→  text: string,
   120→   120→  maxWidth: number,
   121→   121→  startPx: number,
   122→   122→  weight: string,
   123→   123→  mono = false,
   124→   124→  minPx = 13
   125→   125→): number {
   126→   126→  let px = startPx;
   127→   127→  for (;;) {
   128→   128→    ctx.font = font(weight, px, mono);
   129→   129→    if (ctx.measureText(text).width <= maxWidth || px <= minPx) break;
   130→   130→    px -= 2;
   131→   131→  }
   132→   132→  return px;
   133→   133→}

   150→   150→    px -= 2;

   155→   155→function measureTracked(ctx: CanvasRenderingContext2D, text: string, tracking: number): number {
   156→   156→  if (!text) return 0;
   157→   157→  let w = 0;
   158→   158→  for (const ch of text) w += ctx.measureText(ch).width + tracking;
   159→   159→  return w - tracking;
   160→   160→}

   197→   197→function roundRectPath(
   198→   198→  ctx: CanvasRenderingContext2D,
   199→   199→  x: number,
   200→   200→  y: number,
   201→   201→  w: number,
   202→   202→  h: number,
   203→   203→  r: number
   204→   204→): void {
   205→   205→  const rr = Math.min(r, w / 2, h / 2);
   206→   206→  ctx.beginPath();
   207→   207→  ctx.moveTo(x + rr, y);
   208→   208→  ctx.lineTo(x + w - rr, y);
   209→   209→  ctx.arcTo(x + w, y, x + w, y + rr, rr);
   210→   210→  ctx.lineTo(x + w, y + h - rr);
   211→   211→  ctx.arcTo(x + w, y + h, x + w - rr, y + h, rr);
   212→   212→  ctx.lineTo(x + rr, y + h);
   213→   213→  ctx.arcTo(x, y + h, x, y + h - rr, rr);
   214→   214→  ctx.lineTo(x, y + rr);
   215→   215→  ctx.arcTo(x, y, x + rr, y, rr);
   216→   216→  ctx.closePath();
   217→   217→}

   231→   231→function loadFlag(url: string): Promise<HTMLImageElement | null> {
   232→   232→  return new Promise((resolve) => {
   233→   233→    const img = new Image();
   234→   234→    let settled = false;
   235→   235→    let timer = 0;
   236→   236→    const finish = (result: HTMLImageElement | null): void => {
   237→   237→      if (settled) return;
   238→   238→      settled = true;
   239→   239→      if (timer) window.clearTimeout(timer);
   240→   240→      resolve(result);
   241→   241→    };
   242→   242→    timer = window.setTimeout(() => finish(null), FLAG_TIMEOUT_MS);
   243→   243→    img.onload = () => finish(img);
   244→   244→    img.onerror = () => finish(null);
   245→   245→    img.src = `/api/tennis/img?url=${encodeURIComponent(url)}`;
   246→   246→  });
   247→   247→}

   251→   251→function drawCourt(ctx: CanvasRenderingContext2D, pal: CourtPalette): void {
   252→   252→  const cx = W / 2;
   253→   253→  const topY = 520;
   254→   254→  const botY = 1436;
   255→   255→  const topHW = 200; // half width at top baseline
   256→   256→  const botHW = 530; // half width at bottom baseline
   257→   257→  const span = botY - topY;
   258→   258→  const netFrac = 0.4;
   259→   259→  const singlesFrac = 0.75;
   260→   260→  const serviceGapFrac = 18 / 39; // net→service line : service line→baseline
   261→   261→
   262→   262→  const halfWAt = (y: number): number => topHW + (botHW - topHW) * ((y - topY) / span);
   263→   263→  const xAt = (side: 1 | -1, y: number, frac: number): number => cx + side * halfWAt(y) * frac;
   264→   264→
   265→   265→  // Surround
   266→   266→  const bg = ctx.createLinearGradient(0, 0, 0, H);
   267→   267→  bg.addColorStop(0, pal.surroundTop);
   268→   268→  bg.addColorStop(1, pal.surroundBottom);
   269→   269→  ctx.fillStyle = bg;
   270→   270→  ctx.fillRect(0, 0, W, H);
   271→   271→
   272→   272→  // Court trapezoid (wider at bottom → broadcast overhead perspective)
   273→   273→  const netY = topY + span * netFrac;
   274→   274→  const serviceTopY = netY - (netY - topY) * serviceGapFrac;
   275→   275→  const serviceBotY = netY + (botY - netY) * serviceGapFrac;
   276→   276→
   277→   277→  ctx.beginPath();
   278→   278→  ctx.moveTo(cx - topHW, topY);
   279→   279→  ctx.lineTo(cx + topHW, topY);
   280→   280→  ctx.lineTo(cx + botHW, botY);
   281→   281→  ctx.lineTo(cx - botHW, botY);
   282→   282→  ctx.closePath();
   283→   283→  const cg = ctx.createLinearGradient(0, topY, 0, botY);
   284→   284→  cg.addColorStop(0, pal.courtTop);
   285→   285→  cg.addColorStop(1, pal.courtBottom);
   286→   286→  ctx.fillStyle = cg;
   287→   287→  ctx.fill();
   288→   288→
   289→   289→  // Court shadow edge (subtle 3D depth along top edge)
   290→   290→  const edge = ctx.createLinearGradient(0, topY - 26, 0, topY + 30);
   291→   291→  edge.addColorStop(0, "rgba(0,0,0,0)");
   292→   292→  edge.addColorStop(0.75, "rgba(0,0,0,0.22)");
   293→   293→  edge.addColorStop(1, "rgba(0,0,0,0)");
   294→   294→  ctx.fillStyle = edge;
   295→   295→  ctx.fillRect(cx - topHW - 60, topY - 26, topHW * 2 + 120, 56);
   296→   296→
   297→   297→  // Lines — neon white, width grows toward the viewer
   298→   298→  ctx.save();
   299→   299→  ctx.strokeStyle = pal.line;
   300→   300→  ctx.shadowColor = "rgba(255,255,255,0.55)";
   301→   301→  ctx.shadowBlur = 7;
   302→   302→  ctx.lineCap = "butt";
   303→   303→
   304→   304→  const lwAt = (y: number): number => 3 + 3.6 * ((y - topY) / span);
   305→   305→  const seg = (x1: number, y1: number, x2: number, y2: number, yRef: number): void => {
   306→   306→    ctx.lineWidth = lwAt(yRef);
   307→   307→    ctx.beginPath();
   308→   308→    ctx.moveTo(x1, y1);
   309→   309→    ctx.lineTo(x2, y2);
   310→   310→    ctx.stroke();
   311→   311→  };
   312→   312→
   313→   313→  // Doubles sidelines + baselines
   314→   314→  seg(xAt(-1, topY, 1), topY, xAt(-1, botY, 1), botY, (topY + botY) / 2);
   315→   315→  seg(xAt(1, topY, 1), topY, xAt(1, botY, 1), botY, (topY + botY) / 2);
   316→   316→  seg(xAt(-1, topY, 1), topY, xAt(1, topY, 1), topY, topY);
   317→   317→  seg(xAt(-1, botY, 1), botY, xAt(1, botY, 1), botY, botY);
   318→   318→
   319→   319→  // Singles sidelines
   320→   320→  seg(xAt(-1, topY, singlesFrac), topY, xAt(-1, botY, singlesFrac), botY, (topY + botY) / 2);
   321→   321→  seg(xAt(1, topY, singlesFrac), topY, xAt(1, botY, singlesFrac), botY, (topY + botY) / 2);
   322→   322→
   323→   323→  // Service lines
   324→   324→  seg(xAt(-1, serviceTopY, singlesFrac), serviceTopY, xAt(1, serviceTopY, singlesFrac), serviceTopY, serviceTopY);
   325→   325→  seg(xAt(-1, serviceBotY, singlesFrac), serviceBotY, xAt(1, serviceBotY, singlesFrac), serviceBotY, serviceBotY);
   326→   326→
   327→   327→  // Center service line (net → service line, both halves)
   328→   328→  seg(cx, serviceTopY, cx, serviceBotY, netY);
   329→   329→
   330→   330→  // Net: dark band + bright cord, slightly wider than the court
   331→   331→  const netHalfW = halfWAt(netY) * 1.12;
   332→   332→  ctx.shadowBlur = 0;
   333→   333→  ctx.fillStyle = "rgba(2,8,20,0.35)";
   334→   334→  ctx.fillRect(cx - netHalfW, netY - 6, netHalfW * 2, 14);
   335→   335→  ctx.shadowColor = "rgba(255,255,255,0.8)";
   336→   336→  ctx.shadowBlur = 10;
   337→   337→  ctx.strokeStyle = "rgba(255,255,255,0.95)";
   338→   338→  ctx.lineWidth = 4;
   339→   339→  ctx.beginPath();
   340→   340→  ctx.moveTo(cx - netHalfW, netY);
   341→   341→  ctx.lineTo(cx + netHalfW, netY);
   342→   342→  ctx.stroke();
   343→   343→
   344→   344→  // Center marks on both baselines
   345→   345→  ctx.shadowBlur = 4;
   346→   346→  ctx.strokeStyle = pal.line;
   347→   347→  ctx.lineWidth = 5;
   348→   348→  ctx.beginPath();
   349→   349→  ctx.moveTo(cx, topY + 4);
   350→   350→  ctx.lineTo(cx, topY + 20);
   351→   351→  ctx.moveTo(cx, botY - 20);
   352→   352→  ctx.lineTo(cx, botY - 4);
   353→   353→  ctx.stroke();
   354→   354→  ctx.restore();
   355→   355→}

   838→   838→export async function drawPoster(canvas: HTMLCanvasElement, data: PosterData): Promise<void> {
   839→   839→  canvas.width = W;
   840→   840→  canvas.height = H;
   841→   841→  const ctx = canvas.getContext("2d");
   842→   842→  if (!ctx) throw new Error("Canvas 2D context is unavailable");
   843→   843→
   844→   844→  // Reset any inherited state
   845→   845→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   846→   846→  ctx.globalAlpha = 1;
   847→   847→  ctx.globalCompositeOperation = "source-over";
   848→   848→  ctx.filter = "none";
   849→   849→  ctx.textBaseline = "middle";
   850→   850→
   851→   851→  // Flags load in parallel through the same-origin proxy; failures fall back
