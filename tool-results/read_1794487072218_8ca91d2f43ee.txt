    89→// ─── Deterministic RNG (mulberry32) ────────────────────────────────────────
    90→
    91→function hashStringToSeed(input: string): number {
    92→  let h = 1779033703 ^ input.length;
    93→  for (let i = 0; i < input.length; i++) {
    94→    h = Math.imul(h ^ input.charCodeAt(i), 3432918353);
    95→    h = (h << 13) | (h >>> 19);
    96→  }
    97→  return h >>> 0;
    98→}
    99→
   100→function mulberry32(seed: number): () => number {
   101→  let a = seed >>> 0;
   102→  return () => {
   103→    a = (a + 0x6d2b79f5) | 0;
   104→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   105→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   106→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   107→  };
   108→}

   112→function font(weight: string, px: number, mono = false): string {
   113→  return `${weight} ${px}px ${mono ? MONO_STACK : FONT_STACK}`;
   114→}
   115→
   116→/** Shrink font size until `text` fits `maxWidth`. Returns the final px size. */
   117→function fitText(
   118→  ctx: CanvasRenderingContext2D,
   119→  text: string,
   120→  maxWidth: number,
   121→  startPx: number,
   122→  weight: string,
   123→  mono = false,
   124→  minPx = 13
   125→): number {
   126→  let px = startPx;
   127→  for (;;) {
   128→    ctx.font = font(weight, px, mono);
   129→    if (ctx.measureText(text).width <= maxWidth || px <= minPx) break;
   130→    px -= 2;
   131→  }
   132→  return px;
   133→}
   134→
   135→/** Same as fitText but accounts for letter tracking. */
   136→function fitTracked(
   137→  ctx: CanvasRenderingContext2D,
   138→  text: string,
   139→  maxWidth: number,
   140→  startPx: number,
   141→  weight: string,
   142→  tracking: number,
   143→  mono = false,
   144→  minPx = 12
   145→): number {
   146→  let px = startPx;
   147→  for (;;) {
   148→    ctx.font = font(weight, px, mono);
   149→    if (measureTracked(ctx, text, tracking) <= maxWidth || px <= minPx) break;
   150→    px -= 2;
   151→  }
   152→  return px;
   153→}
   154→
   155→function measureTracked(ctx: CanvasRenderingContext2D, text: string, tracking: number): number {
   156→  if (!text) return 0;
   157→  let w = 0;
   158→  for (const ch of text) w += ctx.measureText(ch).width + tracking;
   159→  return w - tracking;
   160→}

   162→function drawTracked(
   163→  ctx: CanvasRenderingContext2D,
   164→  text: string,
   165→  cx: number,
   166→  cy: number,
   167→  tracking: number
   168→): void {
   169→  const total = measureTracked(ctx, text, tracking);
   170→  let x = cx - total / 2;
   171→  const prevAlign = ctx.textAlign;
   172→  ctx.textAlign = "left";
   173→  for (const ch of text) {
   174→    ctx.fillText(ch, x, cy);
   175→    x += ctx.measureText(ch).width + tracking;
   176→  }
   177→  ctx.textAlign = prevAlign;
   178→}

   197→function roundRectPath(
   198→  ctx: CanvasRenderingContext2D,
   199→  x: number,
   200→  y: number,
   201→  w: number,
   202→  h: number,
   203→  r: number
   204→): void {
   205→  const rr = Math.min(r, w / 2, h / 2);
   206→  ctx.beginPath();
   207→  ctx.moveTo(x + rr, y);
   208→  ctx.lineTo(x + w - rr, y);
   209→  ctx.arcTo(x + w, y, x + w, y + rr, rr);
   210→  ctx.lineTo(x + w, y + h - rr);
   211→  ctx.arcTo(x + w, y + h, x + w - rr, y + h, rr);
   212→  ctx.lineTo(x + rr, y + h);
   213→  ctx.arcTo(x, y + h, x, y + h - rr, rr);
   214→  ctx.lineTo(x, y + rr);
   215→  ctx.arcTo(x, y, x + rr, y, rr);
   216→  ctx.closePath();
   217→}

   219→function getInitials(name: string): string {
   220→  const parts = name.trim().split(/\s+/).filter(Boolean);
   221→  if (parts.length === 0) return "?";
   222→  const initials = parts
   223→    .slice(0, 2)
   224→    .map((p) => p.charAt(0))
   225→    .join("");
   226→  return initials.toUpperCase() || "?";
   227→}

   229→// ─── Flag loading (same-origin proxy, parallel, timeout fallback) ──────────

   231→function loadFlag(url: string): Promise<HTMLImageElement | null> {
   232→  return new Promise((resolve) => {
   233→    const img = new Image();
   234→    let settled = false;
   235→    let timer = 0;
   236→    const finish = (result: HTMLImageElement | null): void => {
   237→      if (settled) return;
   238→      settled = true;
   239→      if (timer) window.clearTimeout(timer);
   240→      resolve(result);
   241→    };
   242→    timer = window.setTimeout(() => finish(null), FLAG_TIMEOUT_MS);
   243→    img.onload = () => finish(img);
   244→    img.onerror = () => finish(null);
   245→    img.src = `/api/tennis/img?url=${encodeURIComponent(url)}`;
   246→  });
   247→}

   251→function drawCourt(ctx: CanvasRenderingContext2D, pal: CourtPalette): void {
   252→  const cx = W / 2;
   253→  const topY = 520;
   254→  const botY = 1436;
   255→  const topHW = 200; // half width at top baseline
   256→  const botHW = 530; // half width at bottom baseline
   257→  const span = botY - topY;
   258→  const netFrac = 0.4;
   259→  const singlesFrac = 0.75;
   260→  const serviceGapFrac = 18 / 39; // net→service line : service line→baseline
   261→
   262→  const halfWAt = (y: number): number => topHW + (botHW - topHW) * ((y - topY) / span);
   263→  const xAt = (side: 1 | -1, y: number, frac: number): number => cx + side * halfWAt(y) * frac;
   264→
   265→  // Surround
   266→  const bg = ctx.createLinearGradient(0, 0, 0, H);
   267→  bg.addColorStop(0, pal.surroundTop);
   268→  bg.addColorStop(1, pal.surroundBottom);
   269→  ctx.fillStyle = bg;
   270→  ctx.fillRect(0, 0, W, H);
   271→
   272→  // Court trapezoid (wider at bottom → broadcast overhead perspective)
   273→  const netY = topY + span * netFrac;
   274→  const serviceTopY = netY - (netY - topY) * serviceGapFrac;
   275→  const serviceBotY = netY + (botY - netY) * serviceGapFrac;
   276→
   277→  ctx.beginPath();
   278→  ctx.moveTo(cx - topHW, topY);
   279→  ctx.lineTo(cx + topHW, topY);
   280→  ctx.lineTo(cx + botHW, botY);
   281→  ctx.lineTo(cx - botHW, botY);
   282→  ctx.closePath();
   283→  const cg = ctx.createLinearGradient(0, topY, 0, botY);
   284→  cg.addColorStop(0, pal.courtTop);
   285→  cg.addColorStop(1, pal.courtBottom);
   286→  ctx.fillStyle = cg;
   287→  ctx.fill();
   288→
   289→  // Court shadow edge (subtle 3D depth along top edge)
   290→  const edge = ctx.createLinearGradient(0, topY - 26, 0, topY + 30);
   291→  edge.addColorStop(0, "rgba(0,0,0,0)");
   292→  edge.addColorStop(0.75, "rgba(0,0,0,0.22)");
   293→  edge.addColorStop(1, "rgba(0,0,0,0)");
   294→  ctx.fillStyle = edge;
   295→  ctx.fillRect(cx - topHW - 60, topY - 26, topHW * 2 + 120, 56);
   296→
   297→  // Lines — neon white, width grows toward the viewer
   298→  ctx.save();
   299→  ctx.strokeStyle = pal.line;
   300→  ctx.shadowColor = "rgba(255,255,255,0.55)";
   301→  ctx.shadowBlur = 7;
   302→  ctx.lineCap = "butt";
   303→
   304→  const lwAt = (y: number): number => 3 + 3.6 * ((y - topY) / span);
   305→  const seg = (x1: number, y1: number, x2: number, y2: number, yRef: number): void => {
   306→    ctx.lineWidth = lwAt(yRef);
   307→    ctx.beginPath();
   308→    ctx.moveTo(x1, y1);
   309→    ctx.lineTo(x2, y2);
   310→    ctx.stroke();
   311→  };
   312→
   313→  // Doubles sidelines + baselines
   314→  seg(xAt(-1, topY, 1), topY, xAt(-1, botY, 1), botY, (topY + botY) / 2);
   315→  seg(xAt(1, topY, 1), topY, xAt(1, botY, 1), botY, (topY + botY) / 2);
   316→  seg(xAt(-1, topY, 1), topY, xAt(1, topY, 1), topY, topY);
   317→  seg(xAt(-1, botY, 1), botY, xAt(1, botY, 1), botY, botY);
   318→
   319→  // Singles sidelines
   320→  seg(xAt(-1, topY, singlesFrac), topY, xAt(-1, botY, singlesFrac), botY, (topY + botY) / 2);
   321→  seg(xAt(1, topY, singlesFrac), topY, xAt(1, botY, singlesFrac), botY, (topY + botY) / 2);
   322→
   323→  // Service lines
   324→  seg(xAt(-1, serviceTopY, singlesFrac), serviceTopY, xAt(1, serviceTopY, singlesFrac), serviceTopY, serviceTopY);
   325→  seg(xAt(-1, serviceBotY, singlesFrac), serviceBotY, xAt(1, serviceBotY, singlesFrac), serviceBotY, serviceBotY);
   326→
   327→  // Center service line (net → service line, both halves)
   328→  seg(cx, serviceTopY, cx, serviceBotY, netY);
   329→
   330→  // Net: dark band + bright cord, slightly wider than the court
   331→  const netHalfW = halfWAt(netY) * 1.12;
   332→  ctx.shadowBlur = 0;
   333→  ctx.fillStyle = "rgba(2,8,20,0.35)";
   334→  ctx.fillRect(cx - netHalfW, netY - 6, netHalfW * 2, 14);
   335→  ctx.shadowColor = "rgba(255,255,255,0.8)";
   336→  ctx.shadowBlur = 10;
   337→  ctx.strokeStyle = "rgba(255,255,255,0.95)";
   338→  ctx.lineWidth = 4;
   339→  ctx.beginPath();
   340→  ctx.moveTo(cx - netHalfW, netY);
   341→  ctx.lineTo(cx + netHalfW, netY);
   342→  ctx.stroke();
   343→
   344→  // Center marks on both baselines
   345→  ctx.shadowBlur = 4;
   346→  ctx.strokeStyle = pal.line;
   347→  ctx.lineWidth = 5;
   348→  ctx.beginPath();
   349→  ctx.moveTo(cx, topY + 4);
   350→  ctx.lineTo(cx, topY + 20);
   351→  ctx.moveTo(cx, botY - 20);
   352→  ctx.lineTo(cx, botY - 4);
   353→  ctx.stroke();
   354→  ctx.restore();
   355→}

   357→function drawFloodBeams(ctx: CanvasRenderingContext2D, intensity: number): void {
   358→  ctx.save();
   359→  ctx.globalCompositeOperation = "screen";
   360→
   361→  const beam = (x0: number, y0: number, x1: number, x2: number, y1: number): void => {
   362→    const g = ctx.createLinearGradient(x0, y0, (x1 + x2) / 2, y1);
   363→    g.addColorStop(0, `rgba(103,232,249,${(0.2 * intensity).toFixed(3)})`);
   364→    g.addColorStop(1, "rgba(103,232,249,0)");
   365→    ctx.fillStyle = g;
   366→    ctx.beginPath();
   367→    ctx.moveTo(x0, y0);
   368→    ctx.lineTo(x1, y1);
   369→    ctx.lineTo(x2, y1);
   370→    ctx.closePath();
   371→    ctx.fill();
   372→  };
   373→  beam(-40, -40, 170, 640, 780);
   374→  beam(W + 40, -40, W - 170, W - 640, 780);
   375→
   376→  const lamp = (x: number, y: number): void => {
   377→    const g = ctx.createRadialGradient(x, y, 0, x, y, 280);
   378→    g.addColorStop(0, `rgba(224,247,255,${(0.55 * intensity).toFixed(3)})`);
   379→    g.addColorStop(1, "rgba(224,247,255,0)");
   380→    ctx.fillStyle = g;
   381→    ctx.fillRect(x - 280, y - 280, 560, 560);
   382→  };
   383→  lamp(24, 10);
   384→  lamp(W - 24, 10);
   385→  ctx.restore();
   386→}

   442→function drawFlagOrInitials(
   443→  ctx: CanvasRenderingContext2D,
   444→  img: HTMLImageElement | null,
   445→  cx: number,
   446→  cy: number,
   447→  r: number,
   448→  name: string,
   449→  accent: string
   450→): void {
   451→  ctx.save();
   452→  if (img && img.width > 0 && img.height > 0) {
   453→    ctx.beginPath();
   454→    ctx.arc(cx, cy, r, 0, Math.PI * 2);
   455→    ctx.clip();
   456→    const scale = Math.max((2 * r) / img.width, (2 * r) / img.height);
   457→    const dw = img.width * scale;
   458→    const dh = img.height * scale;
   459→    ctx.drawImage(img, cx - dw / 2, cy - dh / 2, dw, dh);
   460→  } else {
   461→    // Fallback: dark disc + neon ring + initials
   462→    ctx.beginPath();
   463→    ctx.arc(cx, cy, r, 0, Math.PI * 2);
   464→    ctx.fillStyle = "rgba(4,10,24,0.85)";
   465→    ctx.fill();
   466→    ctx.shadowColor = accent;
   467→    ctx.shadowBlur = 18;
   468→    ctx.strokeStyle = accent;
   469→    ctx.lineWidth = 4;
   470→    ctx.beginPath();
   471→    ctx.arc(cx, cy, r - 7, 0, Math.PI * 2);
   472→    ctx.stroke();
   473→    ctx.shadowBlur = 0;
   474→    const initials = getInitials(name);
   475→    fitText(ctx, initials, r * 1.25, 46, "800");
   476→    ctx.fillStyle = "#ffffff";
   477→    ctx.textAlign = "center";
   478→    ctx.textBaseline = "middle";
   479→    ctx.fillText(initials, cx, cy + 2);
   480→  }
   481→  ctx.restore();
   482→}

   611→function drawBall(ctx: CanvasRenderingContext2D, cx: number, cy: number, r: number, glow: number): void {
   612→  // Soft glow halo
   613→  ctx.save();
   614→  ctx.globalCompositeOperation = "screen";
   615→  const halo = ctx.createRadialGradient(cx, cy, r * 0.55, cx, cy, r * 2.1);
   616→  halo.addColorStop(0, `rgba(223,255,79,${(0.35 * glow).toFixed(3)})`);
   617→  halo.addColorStop(1, "rgba(223,255,79,0)");
   618→  ctx.fillStyle = halo;
   619→  ctx.fillRect(cx - r * 2.2, cy - r * 2.2, r * 4.4, r * 4.4);
   620→  ctx.restore();
   621→
   622→  // Sphere body
   623→  ctx.save();
   624→  ctx.shadowColor = "rgba(223,255,79,0.65)";
   625→  ctx.shadowBlur = 42;
   626→  const g = ctx.createRadialGradient(cx - r * 0.35, cy - r * 0.4, r * 0.1, cx, cy, r);
   627→  g.addColorStop(0, "#f7ffbe");
   628→  g.addColorStop(0.35, OPTIC);
   629→  g.addColorStop(0.75, "#c3e02e");
   630→  g.addColorStop(1, "#8fb31c");
   631→  ctx.beginPath();
   632→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
   633→  ctx.fillStyle = g;
   634→  ctx.fill();
   635→  ctx.shadowBlur = 0;
   636→
   637→  // White seam arcs (clipped to sphere)
   638→  ctx.beginPath();
   639→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
   640→  ctx.clip();
   641→  ctx.strokeStyle = "rgba(255,255,255,0.92)";
   642→  ctx.lineWidth = r * 0.09;
   643→  ctx.beginPath();
   644→  ctx.arc(cx - r * 1.18, cy, r * 0.95, -0.62, 0.62);
   645→  ctx.stroke();
   646→  ctx.beginPath();
   647→  ctx.arc(cx + r * 1.18, cy, r * 0.95, Math.PI - 0.62, Math.PI + 0.62);
   648→  ctx.stroke();
   649→
   650→  // Sphere edge shading
   651→  const sh = ctx.createRadialGradient(cx, cy, r * 0.55, cx, cy, r);
   652→  sh.addColorStop(0, "rgba(0,0,0,0)");
   653→  sh.addColorStop(1, "rgba(30,50,0,0.35)");
   654→  ctx.fillStyle = sh;
   655→  ctx.fillRect(cx - r, cy - r, 2 * r, 2 * r);
   656→  ctx.restore();
   657→}

   728→function drawSlipBox(ctx: CanvasRenderingContext2D, opts: SlipBoxOpts): void {
   729→  const { x, y, w, h, label, slip, blurred } = opts;
   730→  const pad = 30;
   731→
   732→  // Glass panel
   733→  roundRectPath(ctx, x, y, w, h, 26);

   735→  ctx.fill();
   736→  ctx.strokeStyle = "rgba(255,255,255,0.25)";
   737→  ctx.lineWidth = 2;
   738→  ctx.stroke();
   739→
   740→  ctx.textBaseline = "middle";
   741→

   743→  ctx.font = font("700", 19, true);
   744→  ctx.fillStyle = "rgba(255,255,255,0.72)";
   745→  ctx.textAlign = "left";
   746→  drawTrackedLeft(ctx, label, x + pad, y + 28, 4);
   747→
   748→  // Content (blurred in teaser mode)
   749→  if (blurred) ctx.filter = "blur(7px)";
   750→
   751→  const maxTextW = w - pad * 2 - 210;
   752→  fitText(ctx, slip.textEn, maxTextW, 38, "800");
   753→  ctx.fillStyle = "#ffffff";
   754→  ctx.fillText(slip.textEn, x + pad, y + 72);

   757→  ctx.textAlign = "right";
   758→  ctx.fillStyle = GOLD;

   760→  ctx.textAlign = "left";
   761→
   762→  const impliedPct = Math.max(0, Math.min(100, 100 / Math.max(slip.odds, 1.01)));
   763→  ctx.font = font("600", 20, true);
   764→  ctx.fillStyle = "rgba(255,255,255,0.68)";
   765→  ctx.fillText(`P ≈ ${impliedPct.toFixed(1)}%`, x + pad, y + 106);
   766→
   767→  ctx.filter = "none";
   768→}

   838→export async function drawPoster(canvas: HTMLCanvasElement, data: PosterData): Promise<void> {
   839→  canvas.width = W;
   840→  canvas.height = H;
   841→  const ctx = canvas.getContext("2d");
   842→  if (!ctx) throw new Error("Canvas 2D context is unavailable");
   843→
   844→  // Reset any inherited state
   845→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   846→  ctx.globalAlpha = 1;
   847→  ctx.globalCompositeOperation = "source-over";
   848→  ctx.filter = "none";
   849→  ctx.textBaseline = "middle";
   850→
   851→  // Flags load in parallel through the same-origin proxy; failures fall back
   852→  // to neon-initials rings after the timeout.
   853→  const [flagA, flagB] = await Promise.all([
   854→    data.playerA.flagUrl ? loadFlag(data.playerA.flagUrl) : Promise.resolve(null),
   855→    data.playerB.flagUrl ? loadFlag(data.playerB.flagUrl) : Promise.resolve(null),
   856→  ]);
   857→
   858→  // Deterministic per-match RNG: particles / glow variation are seeded from
   859→  // the player names so the same match renders pixel-identically.
   860→  const rng = mulberry32(hashStringToSeed(`${data.playerA.name}${data.playerB.name}`));
   861→  const glowJitter = 0.85 + rng() * 0.3; // 0.85..1.15
   862→  const glow2 = 0.85 + rng() * 0.3;
   863→
   864→  const pal = PALETTES[data.surface];
   865→
   866→  // 1. Procedural court background
   867→  drawCourt(ctx, pal);
   868→  if (data.surface === "HARD") drawFloodBeams(ctx, glowJitter);
   869→  if (data.surface === "CLAY") {
   870→    // Dust particles are seeded from hashPre so the same match renders identically
   871→    drawClayDust(ctx, mulberry32(hashStringToSeed(data.hashPre ?? `${data.playerA.name} vs ${data.playerB.name}`)));
   872→  }
   873→  if (data.surface === "GRASS") drawGrassHalo(ctx, W / 2, 690, glowJitter);
   874→  drawVignette(ctx);
   875→
   876→  // 2. Player rings + names + badges
   877→  const ringY = 190;
   878→  const ringR = 88;
   879→  const cxA = 268;
   880→  const cxB = W - 268;
   881→  drawPlayerRing(ctx, data.playerA, flagA, cxA, ringY, ringR, data.favorite === "A");
   882→  drawPlayerRing(ctx, data.playerB, flagB, cxB, ringY, ringR, data.favorite === "B");
   883→  drawPlayerLabel(ctx, data.playerA, cxA, 320, data.favorite === "A");
   884→  drawPlayerLabel(ctx, data.playerB, cxB, 320, data.favorite === "B");
   885→
   886→  // 3. Tournament header strip
   887→  drawTournamentStrip(ctx, data);
   888→
   889→  // 4. Center glowing ball
   890→  drawBall(ctx, W / 2, 690, 86, glow2);
   891→
   892→  // 5. MODEL P chip
   893→  if (typeof data.marketProbAFav === "number" && Number.isFinite(data.marketProbAFav)) {
   894→    drawModelChip(ctx, Math.max(0, Math.min(1, data.marketProbAFav)), W / 2, 818);
   895→  }
   896→
   897→  // 6. Slip boxes
   898→  const boxX = 90;
   899→  const boxW = W - 180;
   900→  const boxH = 140;
   901→  if (data.slipA) {
   902→    drawSlipBox(ctx, { x: boxX, y: 868, w: boxW, h: boxH, label: "SLIP A", slip: data.slipA, blurred: false });
   903→  }
   904→  if (data.slipB) {
   905→    const teaser = data.mode === "teaser";
   906→    const by = 1028;
   907→    drawSlipBox(ctx, { x: boxX, y: by, w: boxW, h: boxH, label: "SLIP B", slip: data.slipB, blurred: teaser });
   908→    if (teaser) {
   909→      // Dark backing panel + padlock + "UNLOCK IN VIP"
   910→      const panelW = 250;
   911→      const panelH = 112;
   912→      const panelCy = by + boxH / 2 - 2;
   913→      roundRectPath(ctx, W / 2 - panelW / 2, panelCy - panelH / 2, panelW, panelH, 20);
   914→      ctx.fillStyle = "rgba(6,10,22,0.6)";
   915→      ctx.fill();
   916→      ctx.strokeStyle = "rgba(255,209,102,0.4)";
   917→      ctx.lineWidth = 1.5;
   918→      ctx.stroke();
   919→      drawPadlock(ctx, W / 2, panelCy - 12);
   920→      ctx.font = font("800", 19, true);
   921→      ctx.fillStyle = GOLD;
   922→      ctx.textBaseline = "middle";
   923→      drawTracked(ctx, "UNLOCK IN VIP", W / 2, panelCy + 38, 4);
   924→    }
   925→  }
   926→
   927→  // 7. Timing chip
   928→  if (data.timingLabel) drawTimingChip(ctx, data.timingLabel, W / 2, 1212);
   929→
   930→  // 8. Footer telemetry
   931→  drawFooter(ctx, data);
   932→}
