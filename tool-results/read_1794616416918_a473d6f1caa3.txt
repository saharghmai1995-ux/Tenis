    63→    63→type Field = { wildcard: boolean; any: Set<number>; step?: number };
    64→    64→
    65→    65→function parseField(raw: string, min: number, max: number): Field {
    66→    66→  const any = new Set<number>();
    67→    67→  let wildcard = false;
    68→    68→  let step: number | undefined;
    69→    69→  for (const chunk of raw.split(",")) {
    70→    70→    if (chunk === "*") {
    71→    71→      wildcard = true;
    72→    72→      for (let i = min; i <= max; i++) any.add(i);
    73→    73→    } else if (chunk.startsWith("*/")) {
    74→    74→      wildcard = true;
    75→    75→      step = Number(chunk.slice(2));
    76→    76→      for (let i = min; i <= max; i += step) any.add(i);
    77→    77→    } else if (chunk.includes("-")) {
    78→    78→      const [a, b] = chunk.split("-").map(Number);
    79→    79→      for (let i = a; i <= b; i++) any.add(i);
    80→    80→    } else {
    81→    81→      any.add(Number(chunk));
    82→    82→    }
    83→    83→  }
    84→    84→  return { wildcard, any, step };
    85→    85→}

    87→    87→interface Cron {
    88→    88→  sec: Field; min: Field; hour: Field; dom: Field; mon: Field; dow: Field;
    89→    89→}
    90→    90→
    91→    91→function parseCron(expr: string): Cron {
    92→    92→  const [sec, min, hour, dom, mon, dow] = expr.trim().split(/\s+/);
    93→    93→  return {
    94→    94→    sec: parseField(sec, 0, 59),
    95→    95→    min: parseField(min, 0, 59),
    96→    96→    hour: parseField(hour, 0, 23),
    97→    97→    dom: parseField(dom, 1, 31),
    98→    98→    mon: parseField(mon, 1, 12),
    99→    99→    dow: parseField(dow, 0, 6),
   100→   100→  };
   101→   101→}

   103→   103→function cronMatches(c: Cron, f: ReturnType<typeof tehranFields>): boolean {
   104→   104→  return (
   105→   105→    c.sec.any.has(f.s) &&
   106→   106→    c.min.any.has(f.min) &&
   107→   107→    c.hour.any.has(f.h) &&
   108→   108→    c.mon.any.has(f.mon) &&
   109→   109→    (c.dom.wildcard || c.dow.wildcard
   110→   110→      ? c.dom.any.has(f.dom) || c.dow.any.has(f.dow)
   111→   111→      : c.dom.any.has(f.dom) && c.dow.any.has(f.dow))
   112→   112→  );
   113→   113→}

   115→   115→/** next run strictly after `after` (ms epoch) for a Tehran-anchored cron */
   116→   116→export function nextCronRun(expr: string, after = Date.now()): number {
   117→   117→  const c = parseCron(expr);
   118→   118→  // iterate the next 2 days at second resolution starting from `after + 1s`
   119→   119→  let t = after + 1000;
   120→   120→  const limit = after + 2 * 86_400_000;
   121→   121→  while (t <= limit) {
   122→   122→    if (cronMatches(c, tehranFields(t))) return t;
   123→   123→    // jump smartly: skip to next minute boundary when seconds won't match soon
   124→   124→    const f = tehranFields(t);
   125→   125→    if (!c.hour.any.has(f.h) || !c.min.any.has(f.min)) {
   126→   126→      t += (59 - f.s) * 1000 + 1000; // finish current minute
   127→   127→      if (!c.hour.any.has(f.h)) t += (59 - f.min) * 60_000; // finish current hour
   128→   128→    } else {
   129→   129→      t += 1000;
   130→   130→    }
   131→   131→  }
   132→   132→  return after + 60_000;
   133→   133→}

   612→   612→
   613→   613→async function runJob(jobId: JobId, reason: "cron" | "manual" = "cron"): Promise<PostOutcome> {
   614→   614→  if (running.has(jobId)) return { status: "SKIPPED", error: "already running" };
   615→   615→  running.add(jobId);
   616→   616→  try {
   617→   617→    const out = await handlers[jobId]();
   618→   618→    store.recordRun(jobId, out.status, { error: out.error, messageId: out.messageId });
   619→   619→    return out;
   620→   620→  } catch (e) {
   621→   621→    const error = e instanceof Error ? e.message : String(e);
   622→   622→    console.error(`[job:${jobId}] ${reason} error:`, error);
   623→   623→    store.recordRun(jobId, "ERROR", { error });
   624→   624→    return { status: "ERROR", error };
   625→   625→  } finally {
   626→   626→    running.delete(jobId);
   627→   627→  }
   628→   628→}

   630→   630→function scheduleAll(): void {
   631→   631→  for (const job of JOB_SPECS) {
   632→   632→    if (job.cron) {
   633→   633→      nextRunAt.set(job.id, nextCronRun(job.cron));
   634→   634→    } else if (job.everySeconds) {
   635→   635→      nextRunAt.set(job.id, Date.now() + (job.everySeconds * 1000));
   636→   636→    }
   637→   637→  }
   638→   638→}
   639→   639→
   640→   640→function tick(): void {
   641→   641→  const now = Date.now();
   642→   642→  for (const job of JOB_SPECS) {
   643→   643→    const next = nextRunAt.get(job.id);
   644→   644→    if (next === undefined || now < next) continue;
   645→   645→    if (job.cron) {
   646→   646→      nextRunAt.set(job.id, nextCronRun(job.cron, now));
   647→   647→    } else if (job.everySeconds) {
   648→   648→      nextRunAt.set(job.id, now + job.everySeconds * 1000);
   649→   649→    }
   650→   650→    void runJob(job.id);
   651→   651→  }
   652→   652→}

   654→   654→export function startScheduler(): void {
   655→   655→  registerFonts();
   656→   656→  restoreLive();
   657→   657→  scheduleAll();
   658→   658→  // warm immediately so the first flashcard is never cold
   659→   659→  void dataHub.warm().then((ok) => {
   660→   660→    if (ok) store.kvSet("lastFeedWarmMs", String(Date.now()));
   661→   661→  });
   662→   662→  setInterval(tick, 1000);
   663→   663→  // engine keep-warm loop — the core must never go cold
   664→   664→  setInterval(() => {
   665→   665→    void ensureWarm();
   666→   666→  }, 5 * 60_000);
   667→   667→  // tennis radar keep-warm (news/live/fitness cache stays hot)
   668→   668→  setInterval(() => {
   669→   669→    void fetchTennisRadar().catch(() => null);
   670→   670→  }, 4 * 60_000);
   671→   671→  // VIP channel auto-discovery — the moment the bot becomes admin of (or
   672→   672→  // receives any update from) the VIP channel, its numeric id is captured.
   673→   673→  setInterval(() => {
   674→   674→    void discoverVipLoop();
   675→   675→  }, 30_000);
   676→   676→  console.log(
   677→   677→    `[scheduler] started - ${JOB_SPECS.length} jobs - live=${isLive()} - tz=${TZ}`
   678→   678→  );
   679→   679→  for (const job of JOB_SPECS) {
   680→   680→    const next = nextRunAt.get(job.id);
   681→   681→    console.log(`[scheduler] ${job.id.padEnd(18)} next=${next ? new Date(next).toISOString() : "-"}`);
   682→   682→  }
   683→   683→  void discoverVipLoop();
   684→   684→}

   686→   686→/** capture the VIP channel id from any chat the bot can see */
   687→   687→async function discoverVipLoop(): Promise<void> {
   688→   688→  try {
   689→   689→    const chats = await telegram.discoverChats();
   690→   690→    for (const ch of chats) {
   691→   691→      const isPublic = !!ch.username && CONFIG.publicChatId.includes(ch.username);
   692→   692→      const isChannelish = ch.type === "channel" || ch.type === "supergroup";
   693→   693→      if (isChannelish && ch.id) {
   694→   694→        store.kvSet(`chatSeen:${ch.id}`, JSON.stringify(ch));
   695→   695→      }
   696→   696→      if (isChannelish && !isPublic && ch.id) {
   697→   697→        const idStr = String(ch.id);
   698→   698→        if (vipChatId() !== idStr) {
   699→   699→          store.kvSet("vipChatId", idStr);
   700→   700→          store.kvSet("vipChatTitle", ch.title ?? "");
   701→   701→          console.log(`[discover] VIP chat captured: ${idStr} (${ch.title ?? ch.type})`);
   702→   702→        }
   703→   703→      }
   704→   704→    }
   705→   705→  } catch {
   706→   706→    /* discovery is best-effort */
   707→   707→  }
