    91→    91→function hashStringToSeed(input: string): number {
    92→    92→  let h = 1779033703 ^ input.length;
    93→    93→  for (let i = 0; i < input.length; i++) {
    94→    94→    h = Math.imul(h ^ input.charCodeAt(i), 3432918353);
    95→    95→    h = (h << 13) | (h >>> 19);
    96→    96→  }
    97→    97→  return h >>> 0;
    98→    98→}

   100→   100→function mulberry32(seed: number): () => number {
   101→   101→  let a = seed >>> 0;
   102→   102→  return () => {
   103→   103→    a = (a + 0x6d2b79f5) | 0;
   104→   104→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   105→   105→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   106→   106→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   107→   107→  };
   108→   108→}

   116→   116→/** Shrink font size until `text` fits `maxWidth`. Returns the final px size. */
   117→   117→function fitText(
   118→   118→  ctx: CanvasRenderingContext2D,
   119→   119→  text: string,
   120→   120→  maxWidth: number,
   121→   121→  startPx: number,
   122→   122→  weight: string,
   123→   123→  mono = false,
   124→   124→  minPx = 13
   125→   125→): number {
   126→   126→  let px = startPx;
   127→   127→  for (;;) {
   128→   128→    ctx.font = font(weight, px, mono);
   129→   129→    if (ctx.measureText(text).width <= maxWidth || px <= minPx) break;
   130→   130→    px -= 2;
   131→   131→  }
   132→   132→  return px;
   133→   133→}

   135→   135→/**
   136→   136→ * v4.5 slip-box text layout — word-wraps the market text into up to `maxLines`
   137→   137→ * large lines instead of shrinking it into unreadable print. Tries the
   138→   138→ * largest size first and only steps down while everything still fits. If no
   139→   139→ * size fits within maxLines, the text is greedily wrapped at the floor size
   140→   140→ * (never drawn as one overflowing line).
   141→   141→ */
   142→   142→function wrapFit(
   143→   143→  ctx: CanvasRenderingContext2D,
   144→   144→  text: string,
   145→   145→  maxWidth: number,
   146→   146→  startPx: number,
   147→   147→  weight: string,
   148→   148→  minPx: number,
   149→   149→  maxLines = 2
   150→   150→): string[] {
   151→   151→  const words = text.split(/\s+/).filter(Boolean);
   152→   152→  for (let px = startPx; px >= minPx; px -= 2) {
   153→   153→    ctx.font = font(weight, px);
   154→   154→    const lines: string[] = [];
   155→   155→    let cur = "";
   156→   156→    let ok = true;
   157→   157→    for (const wd of words) {
   158→   158→      const cand = cur ? `${cur} ${wd}` : wd;
   159→   159→      if (ctx.measureText(cand).width <= maxWidth) {
   160→   160→        cur = cand;
   161→   161→      } else {
   162→   162→        if (cur) lines.push(cur);
   163→   163→        cur = wd;
   164→   164→        if (ctx.measureText(wd).width > maxWidth) {
   165→   165→          ok = false;
   166→   166→          break;
   167→   167→        }
   168→   168→      }
   169→   169→    }
   170→   170→    if (cur) lines.push(cur);
   171→   171→    if (ok && lines.length <= maxLines) return lines;
   172→   172→  }
   173→   173→  // greedy wrap at the floor size — the fallback keeps every word on-canvas
   174→   174→  ctx.font = font(weight, minPx);
   175→   175→  const lines: string[] = [];
   176→   176→  let cur = "";
   177→   177→  for (const wd of words) {
   178→   178→    const cand = cur ? `${cur} ${wd}` : wd;
   179→   179→    if (!cur || ctx.measureText(cand).width <= maxWidth) {
   180→   180→      cur = cand;
   181→   181→    } else {
   182→   182→      lines.push(cur);
   183→   183→      cur = wd;
   184→   184→    }
   185→   185→  }
   186→   186→  if (cur) lines.push(cur);
   187→   187→  return lines.slice(0, maxLines + 1);
   188→   188→}

   210→   210→function measureTracked(ctx: CanvasRenderingContext2D, text: string, tracking: number): number {
   211→   211→  if (!text) return 0;
   212→   212→  let w = 0;
   213→   213→  for (const ch of text) w += ctx.measureText(ch).width + tracking;
   214→   214→  return w - tracking;
   215→   215→}

   252→   252→function roundRectPath(
   253→   253→  ctx: CanvasRenderingContext2D,
   254→   254→  x: number,
   255→   255→  y: number,
   256→   256→  w: number,
   257→   257→  h: number,
   258→   258→  r: number
   259→   259→): void {
   260→   260→  const rr = Math.min(r, w / 2, h / 2);
   261→   261→  ctx.beginPath();
   262→   262→  ctx.moveTo(x + rr, y);
   263→   263→  ctx.lineTo(x + w - rr, y);
   264→   264→  ctx.arcTo(x + w, y, x + w, y + rr, rr);
   265→   265→  ctx.lineTo(x + w, y + h - rr);
   266→   266→  ctx.arcTo(x + w, y + h, x + w - rr, y + h, rr);
   267→   267→  ctx.lineTo(x + rr, y + h);
   268→   268→  ctx.arcTo(x, y + h, x, y + h - rr, rr);
   269→   269→  ctx.lineTo(x, y + rr);
   270→   270→  ctx.arcTo(x, y, x + rr, y, rr);
   271→   271→  ctx.closePath();
   272→   272→}

   286→   286→function loadFlag(url: string): Promise<HTMLImageElement | null> {
   287→   287→  return new Promise((resolve) => {
   288→   288→    const img = new Image();
   289→   289→    let settled = false;
   290→   290→    let timer = 0;
   291→   291→    const finish = (result: HTMLImageElement | null): void => {
   292→   292→      if (settled) return;
   293→   293→      settled = true;
   294→   294→      if (timer) window.clearTimeout(timer);
   295→   295→      resolve(result);
   296→   296→    };
   297→   297→    timer = window.setTimeout(() => finish(null), FLAG_TIMEOUT_MS);
   298→   298→    img.onload = () => finish(img);
   299→   299→    img.onerror = () => finish(null);
   300→   300→    img.src = `/api/tennis/img?url=${encodeURIComponent(url)}`;
   301→   301→  });
   302→   302→}

   306→   306→function drawCourt(ctx: CanvasRenderingContext2D, pal: CourtPalette): void {
   307→   307→  const cx = W / 2;
   308→   308→  const topY = 520;
   309→   309→  const botY = 1436;
   310→   310→  const topHW = 200; // half width at top baseline
   311→   311→  const botHW = 530; // half width at bottom baseline
   312→   312→  const span = botY - topY;
   313→   313→  const netFrac = 0.4;
   314→   314→  const singlesFrac = 0.75;
   315→   315→  const serviceGapFrac = 18 / 39; // net→service line : service line→baseline
   316→   316→
   317→   317→  const halfWAt = (y: number): number => topHW + (botHW - topHW) * ((y - topY) / span);
   318→   318→  const xAt = (side: 1 | -1, y: number, frac: number): number => cx + side * halfWAt(y) * frac;
   319→   319→
   320→   320→  // Surround
   321→   321→  const bg = ctx.createLinearGradient(0, 0, 0, H);
   322→   322→  bg.addColorStop(0, pal.surroundTop);
   323→   323→  bg.addColorStop(1, pal.surroundBottom);
   324→   324→  ctx.fillStyle = bg;
   325→   325→  ctx.fillRect(0, 0, W, H);
   326→   326→
   327→   327→  // Court trapezoid (wider at bottom → broadcast overhead perspective)
   328→   328→  const netY = topY + span * netFrac;
   329→   329→  const serviceTopY = netY - (netY - topY) * serviceGapFrac;
   330→   330→  const serviceBotY = netY + (botY - netY) * serviceGapFrac;
   331→   331→
   332→   332→  ctx.beginPath();
   333→   333→  ctx.moveTo(cx - topHW, topY);
   334→   334→  ctx.lineTo(cx + topHW, topY);
   335→   335→  ctx.lineTo(cx + botHW, botY);
   336→   336→  ctx.lineTo(cx - botHW, botY);
   337→   337→  ctx.closePath();
   338→   338→  const cg = ctx.createLinearGradient(0, topY, 0, botY);
   339→   339→  cg.addColorStop(0, pal.courtTop);
   340→   340→  cg.addColorStop(1, pal.courtBottom);
   341→   341→  ctx.fillStyle = cg;
   342→   342→  ctx.fill();
   343→   343→
   344→   344→  // Court shadow edge (subtle 3D depth along top edge)
   345→   345→  const edge = ctx.createLinearGradient(0, topY - 26, 0, topY + 30);
   346→   346→  edge.addColorStop(0, "rgba(0,0,0,0)");
   347→   347→  edge.addColorStop(0.75, "rgba(0,0,0,0.22)");
   348→   348→  edge.addColorStop(1, "rgba(0,0,0,0)");
   349→   349→  ctx.fillStyle = edge;
   350→   350→  ctx.fillRect(cx - topHW - 60, topY - 26, topHW * 2 + 120, 56);
   351→   351→
   352→   352→  // Lines — neon white, width grows toward the viewer
   353→   353→  ctx.save();
   354→   354→  ctx.strokeStyle = pal.line;
   355→   355→  ctx.shadowColor = "rgba(255,255,255,0.55)";
   356→   356→  ctx.shadowBlur = 7;
   357→   357→  ctx.lineCap = "butt";
   358→   358→
   359→   359→  const lwAt = (y: number): number => 3 + 3.6 * ((y - topY) / span);
   360→   360→  const seg = (x1: number, y1: number, x2: number, y2: number, yRef: number): void => {
   361→   361→    ctx.lineWidth = lwAt(yRef);
   362→   362→    ctx.beginPath();
   363→   363→    ctx.moveTo(x1, y1);
   364→   364→    ctx.lineTo(x2, y2);
   365→   365→    ctx.stroke();
   366→   366→  };
   367→   367→
   368→   368→  // Doubles sidelines + baselines
   369→   369→  seg(xAt(-1, topY, 1), topY, xAt(-1, botY, 1), botY, (topY + botY) / 2);
   370→   370→  seg(xAt(1, topY, 1), topY, xAt(1, botY, 1), botY, (topY + botY) / 2);
   371→   371→  seg(xAt(-1, topY, 1), topY, xAt(1, topY, 1), topY, topY);
   372→   372→  seg(xAt(-1, botY, 1), botY, xAt(1, botY, 1), botY, botY);
   373→   373→
   374→   374→  // Singles sidelines
   375→   375→  seg(xAt(-1, topY, singlesFrac), topY, xAt(-1, botY, singlesFrac), botY, (topY + botY) / 2);
   376→   376→  seg(xAt(1, topY, singlesFrac), topY, xAt(1, botY, singlesFrac), botY, (topY + botY) / 2);
   377→   377→
   378→   378→  // Service lines
   379→   379→  seg(xAt(-1, serviceTopY, singlesFrac), serviceTopY, xAt(1, serviceTopY, singlesFrac), serviceTopY, serviceTopY);
   380→   380→  seg(xAt(-1, serviceBotY, singlesFrac), serviceBotY, xAt(1, serviceBotY, singlesFrac), serviceBotY, serviceBotY);
   381→   381→
   382→   382→  // Center service line (net → service line, both halves)
   383→   383→  seg(cx, serviceTopY, cx, serviceBotY, netY);
   384→   384→
   385→   385→  // Net: dark band + bright cord, slightly wider than the court
   386→   386→  const netHalfW = halfWAt(netY) * 1.12;
   387→   387→  ctx.shadowBlur = 0;
   388→   388→  ctx.fillStyle = "rgba(2,8,20,0.35)";
   389→   389→  ctx.fillRect(cx - netHalfW, netY - 6, netHalfW * 2, 14);
   390→   390→  ctx.shadowColor = "rgba(255,255,255,0.8)";
   391→   391→  ctx.shadowBlur = 10;
   392→   392→  ctx.strokeStyle = "rgba(255,255,255,0.95)";
   393→   393→  ctx.lineWidth = 4;
   394→   394→  ctx.beginPath();
   395→   395→  ctx.moveTo(cx - netHalfW, netY);
   396→   396→  ctx.lineTo(cx + netHalfW, netY);
   397→   397→  ctx.stroke();
   398→   398→
   399→   399→  // Center marks on both baselines
   400→   400→  ctx.shadowBlur = 4;
   401→   401→  ctx.strokeStyle = pal.line;
   402→   402→  ctx.lineWidth = 5;
   403→   403→  ctx.beginPath();
   404→   404→  ctx.moveTo(cx, topY + 4);
   405→   405→  ctx.lineTo(cx, topY + 20);
   406→   406→  ctx.moveTo(cx, botY - 20);
   407→   407→  ctx.lineTo(cx, botY - 4);
   408→   408→  ctx.stroke();
   409→   409→  ctx.restore();
   410→   410→}

   899→   899→export async function drawPoster(canvas: HTMLCanvasElement, data: PosterData): Promise<void> {
   900→   900→  canvas.width = W;
   901→   901→  canvas.height = H;
   902→   902→  const ctx = canvas.getContext("2d");
   903→   903→  if (!ctx) throw new Error("Canvas 2D context is unavailable");
   904→   904→
   905→   905→  // Reset any inherited state
   906→   906→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   907→   907→  ctx.globalAlpha = 1;
   908→   908→  ctx.globalCompositeOperation = "source-over";
   909→   909→  ctx.filter = "none";
   910→   910→  ctx.textBaseline = "middle";
   911→   911→
   912→   912→  // Flags load in parallel through the same-origin proxy; failures fall back
   913→   913→  // to neon-initials rings after the timeout.
   914→   914→  const [flagA, flagB] = await Promise.all([
   915→   915→    data.playerA.flagUrl ? loadFlag(data.playerA.flagUrl) : Promise.resolve(null),
   916→   916→    data.playerB.flagUrl ? loadFlag(data.playerB.flagUrl) : Promise.resolve(null),
   917→   917→  ]);
   918→   918→
   919→   919→  // Deterministic per-match RNG: particles / glow variation are seeded from
   920→   920→  // the player names so the same match renders pixel-identically.
   921→   921→  const rng = mulberry32(hashStringToSeed(`${data.playerA.name}${data.playerB.name}`));
   922→   922→  const glowJitter = 0.85 + rng() * 0.3; // 0.85..1.15
   923→   923→  const glow2 = 0.85 + rng() * 0.3;
   924→   924→
   925→   925→  const pal = PALETTES[data.surface];
   926→   926→
   927→   927→  // 1. Procedural court background
   928→   928→  drawCourt(ctx, pal);
   929→   929→  if (data.surface === "HARD") drawFloodBeams(ctx, glowJitter);
   930→   930→  if (data.surface === "CLAY") {
   931→   931→    // Dust particles are seeded from hashPre so the same match renders identically
   932→   932→    drawClayDust(ctx, mulberry32(hashStringToSeed(data.hashPre ?? `${data.playerA.name} vs ${data.playerB.name}`)));
   933→   933→  }
   934→   934→  if (data.surface === "GRASS") drawGrassHalo(ctx, W / 2, 690, glowJitter);
   935→   935→  drawVignette(ctx);
   936→   936→
   937→   937→  // 2. Player rings + names + badges
   938→   938→  const ringY = 190;
   939→   939→  const ringR = 88;
   940→   940→  const cxA = 268;
   941→   941→  const cxB = W - 268;
   942→   942→  drawPlayerRing(ctx, data.playerA, flagA, cxA, ringY, ringR, data.favorite === "A");
   943→   943→  drawPlayerRing(ctx, data.playerB, flagB, cxB, ringY, ringR, data.favorite === "B");
   944→   944→  drawPlayerLabel(ctx, data.playerA, cxA, 320, data.favorite === "A");
   945→   945→  drawPlayerLabel(ctx, data.playerB, cxB, 320, data.favorite === "B");
   946→   946→
   947→   947→  // 3. Tournament header strip
   948→   948→  drawTournamentStrip(ctx, data);
   949→   949→
   950→   950→  // 4. Center glowing ball
   951→   951→  drawBall(ctx, W / 2, 690, 86, glow2);
   952→   952→
   953→   953→  // 5. MODEL P chip
   954→   954→  if (typeof data.marketProbAFav === "number" && Number.isFinite(data.marketProbAFav)) {
   955→   955→    drawModelChip(ctx, Math.max(0, Math.min(1, data.marketProbAFav)), W / 2, 800);
   956→   956→  }
   957→   957→
   958→   958→  // 6. Slip boxes — v4.5: 186px-tall boxes so the official market text
   959→   959→  // renders at 30–42px over up to 3 wrapped lines (never shrunken small print)
   960→   960→  const boxX = 90;
   961→   961→  const boxW = W - 180;
   962→   962→  const boxH = 186;
   963→   963→  if (data.slipA) {
   964→   964→    drawSlipBox(ctx, { x: boxX, y: 842, w: boxW, h: boxH, label: "SLIP A", slip: data.slipA, blurred: false });
   965→   965→  }
   966→   966→  if (data.slipB) {
   967→   967→    const teaser = data.mode === "teaser";
   968→   968→    const by = 1046;
   969→   969→    drawSlipBox(ctx, { x: boxX, y: by, w: boxW, h: boxH, label: "SLIP B", slip: data.slipB, blurred: teaser });
   970→   970→    if (teaser) {
   971→   971→      // Dark backing panel + padlock + "UNLOCK IN VIP"
   972→   972→      const panelW = 250;
   973→   973→      const panelH = 118;
   974→   974→      const panelCy = by + boxH / 2 - 2;
   975→   975→      roundRectPath(ctx, W / 2 - panelW / 2, panelCy - panelH / 2, panelW, panelH, 20);
   976→   976→      ctx.fillStyle = "rgba(6,10,22,0.6)";
   977→   977→      ctx.fill();
   978→   978→      ctx.strokeStyle = "rgba(255,209,102,0.4)";
   979→   979→      ctx.lineWidth = 1.5;
   980→   980→      ctx.stroke();
   981→   981→      drawPadlock(ctx, W / 2, panelCy - 12);
   982→   982→      ctx.font = font("800", 19, true);
   983→   983→      ctx.fillStyle = GOLD;
   984→   984→      ctx.textBaseline = "middle";
   985→   985→      drawTracked(ctx, "UNLOCK IN VIP", W / 2, panelCy + 38, 4);
   986→   986→    }
   987→   987→  }
   988→   988→
   989→   989→  // 7. Timing chip (aligned under the taller slip boxes)
   990→   990→  if (data.timingLabel) drawTimingChip(ctx, data.timingLabel, W / 2, 1262);
   991→   991→
   992→   992→  // 8. Footer telemetry
   993→   993→  drawFooter(ctx, data);
   994→   994→}
