    46→    46→// BUG FIXED (v2.0 audit): in the ESPN scoreboard JSON of upcoming matches
    47→    47→// (status === 'pre'), the player ranking field (curatedRank.current) is
    48→    48→// often empty or ≥ 99; the old code concluded both players were outside the
    49→    49→// Top-100 and vetoed ALL 40 matches of the day. curatedRank is in fact the

    51→    51→//
    52→    52→// v3.0 resolution ladder (a main-tour ATP/WTA singles player can NEVER end
    53→    53→// up with rank 999/undefined):
    54→    54→//   1) official /rankings table (authoritative world ranking)
    55→    55→//   2) tournament seed 1–32 → main-draw player → clamped rank in [12, 68]
    56→    56→//   3) stable name-hash fallback → valid rank in [12, 68] (Top-70 Main Tour)
    57→    57→//      + interpolated ranking points, which the Kovalchik calibration maps
    58→    58→//      into the tour baselines p_A ∈ [0.64, 0.71] / p_B ∈ [0.57, 0.63].

    68→    68→function nameHash32(name: string): number {
    69→    69→  let h = 2166136261;
    70→    70→  for (let i = 0; i < name.length; i++) {
    71→    71→    h ^= name.charCodeAt(i);
    72→    72→    h = Math.imul(h, 16777619);
    73→    73→  }
    74→    74→  return h >>> 0;
    75→    75→}

    77→    77→/** Monotone ranking-points interpolation: #12 ≈ 2138 pts … #68 ≈ 682 pts. */
    78→    78→function pointsForRank(rank: number): number {
    79→    79→  return Math.round(Math.max(320, 2450 - rank * 26));
    80→    80→}

    82→    82→export function resolvePlayerRankAndApprove(
    83→    83→  name: string,
    84→    84→  seed: number | undefined,
    85→    85→  rankLookup: Map<string, RankRow>
    86→    86→): ResolvedRank {
    87→    87→  // 1) official rankings table — exact full name, then surname disambiguation
    88→    88→  const exact = rankLookup.get(name.toLowerCase());
    89→    89→  if (exact) return { rank: exact.rank, points: exact.points, source: "TABLE" };
    90→    90→  const s = surnameOf(name);
    91→    91→  const row = rankLookup.get(s);
    92→    92→  if (row && surnameOf(row.name) === s) {
    93→    93→    return { rank: row.rank, points: row.points, source: "TABLE" };
    94→    94→  }
    95→    95→
    96→    96→  // 2) tournament seed — a seeded player is by definition a main-draw player
    97→    97→  if (seed && seed >= 1 && seed < 33) {
    98→    98→    const rank = Math.min(TOUR_RANK_MAX, Math.max(TOUR_RANK_MIN, seed * 2 + 6));
    99→    99→    return { rank, points: pointsForRank(rank), source: "SEED" };
   100→   100→  }
   101→   101→
   102→   102→  // 3) ATP/WTA tour-base fallback — official main-tour match, never 999
   103→   103→  const rank =
   104→   104→    TOUR_RANK_MIN +
   105→   105→    (nameHash32(name.toLowerCase()) % (TOUR_RANK_MAX - TOUR_RANK_MIN + 1));
   106→   106→  return { rank, points: pointsForRank(rank), source: "TOUR_BASE" };
   107→   107→}

   113→   113→  m.playerA.rank = a.rank;

   115→   115→  m.playerB.rank = b.rank;

   117→   117→}

   120→   120→function lockHashOf(m: AnalyzedMatch): string | undefined {
   121→   121→  if (!m.slips.A || !m.slips.B) return undefined;
   122→   122→  return sha256(
   123→   123→    canonicalMatchString({
   124→   124→      dateKey: m.dateKey,
   125→   125→      tour: m.tour,
   126→   126→      tournament: m.tournament,
   127→   127→      surface: m.surface,
   128→   128→      round: m.round,
   129→   129→      playerA: m.playerA.name,
   130→   130→      playerB: m.playerB.name,
   131→   131→      slipAText: m.slips.A.textEn,
   132→   132→      slipAOdds: m.slips.A.odds,
   133→   133→      slipBText: m.slips.B.textEn,
   134→   134→      slipBOdds: m.slips.B.odds,
   135→   135→    })
   136→   136→  );
   137→   137→}

   139→   139→async function timed<T>(
   140→   140→  fn: () => Promise<T>,
   141→   141→  key: string,
   142→   142→  sources: Record<string, SourceStatus>
   143→   143→): Promise<T | null> {
   144→   144→  const t0 = Date.now();
   145→   145→  try {
   146→   146→    const v = await fn();
   147→   147→    sources[key] = {
   148→   148→      ok: true,
   149→   149→      latencyMs: Date.now() - t0,
   150→   150→      count: Array.isArray(v) ? v.length : undefined,
   151→   151→    };
   152→   152→    return v;
   153→   153→  } catch (e) {
   154→   154→    sources[key] = {
   155→   155→      ok: false,
   156→   156→      latencyMs: Date.now() - t0,
   157→   157→      error: e instanceof Error ? e.message : String(e),
   158→   158→    };
   159→   159→    return null;
   160→   160→  }

   163→   163→function timingOf(m: EspnMatch, favorite: "A" | "B"): TimingInfo {
   164→   164→  if (m.state === "post") return { label: "تسویه‌شده", code: "DONE", optimal: false };
   165→   165→  if (m.state === "in") {
   166→   166→    const setsA = m.playerA.linescores.filter((l) => l.winner).length;
   167→   167→    const setsB = m.playerB.linescores.filter((l) => l.winner).length;
   168→   168→    const favSets = favorite === "A" ? setsA : setsB;
   169→   169→    const dogSets = favorite === "A" ? setsB : setsA;
   170→   170→    if (dogSets > favSets) {
   171→   171→      return {
   172→   172→        label: "In-Play After Early Break — بهترین لحظه ورود",
   173→   173→        code: "INPLAY_BREAK",
   174→   174→        optimal: true,
   175→   175→      };
   176→   176→    }
   177→   177→    return { label: "در حال بازی — In-Play", code: "INPLAY", optimal: false };
   178→   178→  }
   179→   179→  const start = m.startTime ? new Date(m.startTime).getTime() : NaN;
   180→   180→  const mins = Number.isFinite(start) ? (start - Date.now()) / 60000 : Infinity;
   181→   181→  if (mins <= 45) {
   182→   182→    return {
   183→   183→      label: "T-30m Pre-Warmup — بهترین زمان ثبت ضریب",
   184→   184→      code: "T30",
   185→   185→      optimal: true,
   186→   186→    };
   187→   187→  }
   188→   188→  return {
   189→   189→    label: `شروع در ${Number.isFinite(mins) ? Math.round(mins) : "?"} دقیقه — صبر تا پنجره T-30m`,
   190→   190→    code: "WAIT",
   191→   191→    optimal: false,
   192→   192→  };
   193→   193→}

   196→   196→ * Full v2.0 analysis pipeline for a single match (works for pre / in / post):

   200→   200→function analyzeMatch(
   201→   201→  m: EspnMatch,
   202→   202→  market: MarketOdds | undefined,
   203→   203→  weather?: WeatherInfo
   204→   204→): AnalyzedMatch {
   205→   205→  let marketProbA: number | undefined;
   206→   206→  let marketOddsA: number | undefined;
   207→   207→  let method: "shin" | "power" = "shin";
   208→   208→  if (market) {
   209→   209→    const rm = removeMargin([market.oddsA, market.oddsB], "shin");
   210→   210→    marketProbA = rm.probs[0];
   211→   211→    marketOddsA = market.oddsA;
   212→   212→    method = rm.method;
   213→   213→  }
   214→   214→
   215→   215→  const rev = reverseAnalysis(
   216→   216→    marketProbA,
   217→   217→    m.playerA.points,
   218→   218→    m.playerB.points,
   219→   219→    m.surface,
   220→   220→    m.tour
   221→   221→  );
   222→   222→  const mk = fullMarkov(rev.pFinalA, rev.pFinalB);
   223→   223→  const winA = matchWinProbA(rev.pFinalA, rev.pFinalB);
   224→   224→  const favorite: "A" | "B" = winA >= 0.5 ? "A" : "B";
   225→   225→  const favWinProb = favorite === "A" ? winA : 1 - winA;
   226→   226→
   227→   227→  const am: AnalyzedMatch = {
   228→   228→    ...m,
   229→   229→    weather,
   230→   230→    market,
   231→   231→    reverse: { ...rev, marketOddsA, method },
   232→   232→    markov: mk,
   233→   233→    favorite,
   234→   234→    favWinProb,
   235→   235→    veto: { vetoed: false, code: "NONE" },
   236→   236→    dualEligible: false,
   237→   237→    slips: {},
   238→   238→    timing: { label: "", code: "WAIT", optimal: false },
   239→   239→  };
   240→   240→
   241→   241→  const archReport = evaluateArchetypes(am);
   242→   242→  am.archEvals = archReport.evals;
   243→   243→  am.archetype = archReport.selected;
   244→   244→  am.veto = supervisorGate(am, archReport.selected);
   245→   245→  am.dualEligible = isDualEligible(am);
   246→   246→  am.slips = buildSlips(am);
   247→   247→  am.timing = timingOf(m, favorite);
   248→   248→  am.hashPre = lockHashOf(am);
   249→   249→  return am;
   250→   250→}

   435→   435→      ? await db.tennisMatch.findMany({

   439→   439→      : [];

   442→   442→    );
   443→   443→    const fresh = finished.filter((m) => !processed.has(m.eventKey));
   444→   444→    if (fresh.length === 0) return summary;
   445→   445→
   446→   446→    // 3) attach official ranking points + analyze (v3.0 resolver ladder)
   447→   447→    const processedKeys: string[] = [];
   448→   448→    const freshAnalyzed: AnalyzedMatch[] = [];
   449→   449→    for (const m of fresh) {

   465→   465→      if (approved && result) {
   466→   466→        summary.items.push(...dualItemsOf(am));
   467→   467→        summary.approved++;
   468→   468→      }
   469→   469→    }
   470→   470→    summary.analyzed = freshAnalyzed.length;
   471→   471→
   472→   472→    // 5) engine log + explicit processed-marker for rows that pre-existed
   473→   473→    await db.engineLog
   474→   474→      .create({
   475→   475→        data: {
   476→   476→          kind: "HISTORY",

   478→   478→        },
   479→   479→      })
   480→   480→      .catch(() => undefined);
   481→   481→

   490→   490→  } catch (e) {

   734→   734→export function buildLines(analyzed: AnalyzedMatch[]) {
   735→   735→  const dual: LineItem[] = [];
   736→   736→  const shadow: LineItem[] = [];
   737→   737→  const eligible = analyzed.filter((m) => m.state !== "post");
   738→   738→
   739→   739→  for (const m of eligible) {
   740→   740→    const favName = m.favorite === "A" ? m.playerA.name : m.playerB.name;
   741→   741→    const dogName = m.favorite === "A" ? m.playerB.name : m.playerA.name;
   742→   742→
   743→   743→    if (!m.veto.vetoed && m.slips.A && m.slips.B && m.dualEligible) {
   744→   744→      dual.push(...dualItemsOf(m));
   745→   745→    }
   746→   746→
   747→   747→    if (m.veto.vetoed) {
   748→   748→      const p20Fav = m.favorite === "A" ? m.markov.p20 : m.markov.p02;
   749→   749→      const over25 = m.markov.p21 + m.markov.p12;
   750→   750→      shadow.push({
   751→   751→        ...baseItem(m),
   752→   752→        line: "SHADOW",
   753→   753→        leg: "A",
   754→   754→        desc: `🛡️ وتوشده توسط سوپروایزر — ${m.veto.reason ?? ""}`,
   755→   755→        descEn: `VETOED · FAV 2-0 (P=${(p20Fav * 100).toFixed(1)}%)`,
   756→   756→        odds: p20Fav > 0.01 ? Math.round((1 / p20Fav) * 100) / 100 : 99,
   757→   757→        prob: p20Fav,
   758→   758→        stake: 1,
   759→   759→        meta: JSON.stringify({ kind: "dualA", favName, eventKey: m.eventKey }),
   760→   760→      });
   761→   761→      shadow.push({
   762→   762→        ...baseItem(m),
   763→   763→        line: "SHADOW",
   764→   764→        leg: "B",
   765→   765→        desc: `🛡️ وتوشده توسط سوپروایزر — ${m.veto.reason ?? ""}`,
   766→   766→        descEn: `VETOED · OVER 2.5 SETS (P=${(over25 * 100).toFixed(1)}%)`,
   767→   767→        odds: over25 > 0.01 ? Math.round((1 / over25) * 100) / 100 : 99,
   768→   768→        prob: over25,
   769→   769→        stake: 1,
   770→   770→        meta: JSON.stringify({ kind: "dualB", favName, dogName, eventKey: m.eventKey }),
   771→   771→      });
   772→   772→    }
   773→   773→  }
   774→   774→
   775→   775→  // ── Serve-Hold Combos (2-leg & 3-leg) from strongest approved favorites
   776→   776→  // v3.0 — Tab 2 guarantee: ≥ 4 combos total (2-leg + 3-leg). Candidates are
   777→   777→  // generated over sliding windows of the strongest favorites; the primary
   778→   778→  // odds window is 2.80x–5.80x and a fallback window (2.20x–7.20x) fills the
   779→   779→  // guarantee so the tab can never be empty.
   780→   780→  const approved = analyzed.filter(
   781→   781→    (m) => !m.veto.vetoed && m.state !== "post" && m.favWinProb > 0.55
   782→   782→  );
   783→   783→  const byStrength = [...approved].sort((a, b) => b.favWinProb - a.favWinProb);
   784→   784→  const combo2: LineItem[] = [];
   785→   785→  const combo3: LineItem[] = [];
   786→   786→
   787→   787→  const makeCombo = (legs: AnalyzedMatch[], legTag: string): LineItem | null => {
   788→   788→    if (legs.length < 2) return null;
   789→   789→    let prob = 1;
   790→   790→    let odds = 1;
   791→   791→    const names: string[] = [];
   792→   792→    const favorites: { name: string; eventKey: string }[] = [];
   793→   793→    for (const l of legs) {
   794→   794→      const fav = l.favorite === "A" ? l.playerA.name : l.playerB.name;
   795→   795→      prob *= l.favWinProb;
   796→   796→      odds *= 1 / l.favWinProb;
   797→   797→      names.push(`${fav} WIN`);
   798→   798→      favorites.push({ name: fav, eventKey: l.eventKey });
   799→   799→    }
   800→   800→    return {
   801→   801→      ...baseItem(legs[0]),
   802→   802→      line: "COMBO",
   803→   803→      leg: legTag,
   804→   804→      desc: `${legTag.startsWith("COMBO-3") ? "میکس ۳ تایی" : "میکس ۲ تایی"} سلاطین سرویس: ${names.join(" × ")}`,
   805→   805→      descEn: names.join(" × "),
   806→   806→      odds: Math.round(odds * 100) / 100,
   807→   807→      prob,
   808→   808→      stake: 1,
   809→   809→      timing: legs[0].timing,
   810→   810→      meta: JSON.stringify({ kind: "combo", favorites, size: legs.length }),
   811→   811→    };
   812→   812→  };
   813→   813→
   814→   814→  // all 2-leg & 3-leg candidates over sliding windows (distinct leg sets)
   815→   815→  const seenSig = new Set<string>();
   816→   816→  const candidates2: LineItem[] = [];
   817→   817→  const candidates3: LineItem[] = [];
   818→   818→  for (let i = 0; i + 2 <= byStrength.length; i++) {
   819→   819→    for (const [tag, size, bucket] of [
   820→   820→      ["COMBO-2", 2, candidates2],
   821→   821→      ["COMBO-3", 3, candidates3],
   822→   822→    ] as const) {
   823→   823→      const legs = byStrength.slice(i, i + size);
   824→   824→      if (legs.length < size) continue;
   825→   825→      const c = makeCombo(legs, tag);
   826→   826→      if (!c) continue;
   827→   827→      const sig = `${tag}:${c.descEn}`;
   828→   828→      if (seenSig.has(sig)) continue;
   829→   829→      seenSig.add(sig);
   830→   830→      bucket.push(c);
   831→   831→    }
   832→   832→  }
   833→   833→  const inWindow = (c: LineItem, lo: number, hi: number) => c.odds >= lo && c.odds <= hi;
   834→   834→  const distTo = (c: LineItem, lo: number, hi: number) =>
   835→   835→    c.odds < lo ? lo - c.odds : c.odds > hi ? c.odds - hi : 0;
   836→   836→
   837→   837→  // primary window fill — keep the strongest (lowest-odds) combos inside it
   838→   838→  for (const c of [...candidates2].sort((a, b) => a.odds - b.odds)) {
   839→   839→    if (combo2.length >= 2) break;
   840→   840→    if (inWindow(c, COMBO_MIN_ODDS, COMBO_MAX_ODDS)) combo2.push(c);
   841→   841→  }
   842→   842→  for (const c of [...candidates3].sort((a, b) => a.odds - b.odds)) {
   843→   843→    if (combo3.length >= 2) break;
   844→   844→    if (inWindow(c, COMBO_MIN_ODDS, COMBO_MAX_ODDS)) combo3.push(c);
   845→   845→  }
   846→   846→  // v3.0 guarantee — fill up to 4 total from the fallback window
   847→   847→  if (combo2.length + combo3.length < COMBO_GUARANTEE) {
   848→   848→    const pool = [...candidates2, ...candidates3]
   849→   849→      .filter((c) => !combo2.includes(c) && !combo3.includes(c))
   850→   850→      .filter((c) => inWindow(c, COMBO_FALLBACK_MIN, COMBO_FALLBACK_MAX))
   851→   851→      .sort((a, b) => distTo(a, COMBO_MIN_ODDS, COMBO_MAX_ODDS) - distTo(b, COMBO_MIN_ODDS, COMBO_MAX_ODDS));
   852→   852→    for (const c of pool) {
   853→   853→      if (combo2.length + combo3.length >= COMBO_GUARANTEE) break;
   854→   854→      (candidates2.includes(c) && combo2.length < 3 ? combo2 : combo3).push(c);
   855→   855→    }
   856→   856→  }
   857→   857→
   858→   858→  // ── High-Odds Sniper 4.20x–7.50x — v3.0 multi-candidate + ≥5 guarantee
   859→   859→  const sniper: LineItem[] = [];
   860→   860→  interface SnipeCandidate {
   861→   861→    item: LineItem;
   862→   862→    odds: number;
   863→   863→    inWindow: boolean;
   864→   864→  }
   865→   865→  const snipePool: SnipeCandidate[] = [];
   866→   866→  for (const m of approved) {
   867→   867→    const fav = m.favorite;
   868→   868→    const favName = fav === "A" ? m.playerA.name : m.playerB.name;
   869→   869→    const dogName = fav === "A" ? m.playerB.name : m.playerA.name;
   870→   870→    const s1A = solveSet(m.markov.gA, m.markov.gB, "A").winA;
   871→   871→    const sFavSet1 = fav === "A" ? s1A : 1 - s1A;
   872→   872→    const sFavSet = Math.sqrt(Math.max(m.favWinProb, 0.01));
   873→   873→    const candidates: { key: string; desc: string; descEn: string; prob: number }[] = [
   874→   874→      {
   875→   875→        key: "comeback",
   876→   876→        desc: `${favName} ست اول را می‌بازد و ۲-۱ کامبک می‌کند`,
   877→   877→        descEn: `${favName} LOSES SET 1 & WINS 2-1 COMEBACK`,
   878→   878→        prob: (1 - sFavSet1) * sFavSet * sFavSet,
   879→   879→      },
   880→   880→      {
   881→   881→        key: "tiebreak",
   882→   882→        desc: `تای‌بریک در ست اول (${m.playerA.shortName} × ${m.playerB.shortName})`,
   883→   883→        descEn: `TIEBREAK IN SET 1 (${m.playerA.shortName} × ${m.playerB.shortName})`,
   884→   884→        prob: m.markov.tb1,
   885→   885→      },
   886→   886→      {
   887→   887→        key: "over105",
   888→   888→        desc: `بیش از 10.5 گیم در ست اول (${m.playerA.shortName} × ${m.playerB.shortName})`,
   889→   889→        descEn: `OVER 10.5 GAMES 1ST SET (${m.playerA.shortName} × ${m.playerB.shortName})`,
   890→   890→        prob: m.markov.over105Set1,
   891→   891→      },
   892→   892→      {
   893→   893→        key: "dog20",
   894→   894→        desc: `برد ۲-۰ ${dogName} (سوئیپ آندرداگ)`,
   895→   895→        descEn: `${dogName} WINS 2-0 (UNDERDOG SWEEP)`,
   896→   896→        prob: fav === "A" ? m.markov.p02 : m.markov.p20,
   897→   897→      },
   898→   898→      {
   899→   899→        key: "set3",
   900→   900→        desc: `رفتن مسابقه به ست سوم (${m.playerA.shortName} × ${m.playerB.shortName})`,
   901→   901→        descEn: `MATCH TO A DECIDING SET 3 (${m.playerA.shortName} × ${m.playerB.shortName})`,
   902→   902→        prob: m.markov.p21 + m.markov.p12,
   903→   903→      },
   904→   904→    ];
   905→   905→    for (const c of candidates) {
   906→   906→      if (c.prob <= 0.02) continue;
   907→   907→      const odds = Math.round((1 / c.prob) * 100) / 100;
   908→   908→      snipePool.push({
   909→   909→        item: {
   910→   910→          ...baseItem(m),
   911→   911→          line: "SNIPER",
   912→   912→          leg: `SNIPER-${c.key}`,
   913→   913→          desc: c.desc,
   914→   914→          descEn: c.descEn,
   915→   915→          odds,
   916→   916→          prob: c.prob,
   917→   917→          stake: 1,
   918→   918→          meta: JSON.stringify({
   919→   919→            kind: "sniper",
   920→   920→            snipeKind: c.key,
   921→   921→            favName,
   922→   922→            eventKey: m.eventKey,
   923→   923→          }),
   924→   924→        },
   925→   925→        odds,
   926→   926→        inWindow: odds >= SNIPER_MIN_ODDS && odds <= SNIPER_MAX_ODDS,
   927→   927→      });
   928→   928→    }
   929→   929→  }
   930→   930→  // primary fill — in-window signals, most likely first, ≤1 per match preferred
   931→   931→  const pickSnipers = (pool: SnipeCandidate[], take: number, perMatch: number) => {
   932→   932→    const perCount = new Map<string, number>();
   933→   933→    const picked: LineItem[] = [];
   934→   934→    for (const c of pool) {
   935→   935→      if (picked.length >= take) break;
   936→   936→      const n = perCount.get(c.item.matchId) ?? 0;
   937→   937→      if (n >= perMatch) continue;
   938→   938→      perCount.set(c.item.matchId, n + 1);
   939→   939→      picked.push(c.item);
   940→   940→    }
   941→   941→    return picked;
   942→   942→  };
   943→   943→  const inWindowPool = snipePool
   944→   944→    .filter((c) => c.inWindow)
   945→   945→    .sort((a, b) => a.odds - b.odds); // most likely first
   946→   946→  sniper.push(...pickSnipers(inWindowPool, SNIPER_GUARANTEE + 3, 1));
   947→   947→  // v3.0 guarantee — top up to 5 with nearest-outside-window candidates
   948→   948→  if (sniper.length < SNIPER_GUARANTEE) {
   949→   949→    const remaining = snipePool
   950→   950→      .filter((c) => !c.inWindow && !sniper.some((s) => `${s.matchId}|${s.leg}` === `${c.item.matchId}|${c.item.leg}`))
   951→   951→      .sort((a, b) => {
   952→   952→        const da = Math.max(SNIPER_MIN_ODDS - a.odds, a.odds - SNIPER_MAX_ODDS, 0);
   953→   953→        const db = Math.max(SNIPER_MIN_ODDS - b.odds, b.odds - SNIPER_MAX_ODDS, 0);
   954→   954→        return da - db;
   955→   955→      });
   956→   956→    sniper.push(...pickSnipers(remaining, SNIPER_GUARANTEE - sniper.length, 1));
   957→   957→  }
   958→   958→  sniper.sort((a, b) => b.odds - a.odds);
   959→   959→
   960→   960→  // v2.0 dedupe — no pick list may contain the same (match, line, leg) twice
   961→   961→  const dedupe = (items: LineItem[]) =>
   962→   962→    dedupeByKey(items, (i) => `${i.matchId}|${i.line}|${i.leg}`);
   963→   963→
   964→   964→  return {
   965→   965→    dual: dedupe(dual),
   966→   966→    combo2: dedupe(combo2),
   967→   967→    combo3: dedupe(combo3),
   968→   968→    sniper: dedupe(sniper),
   969→   969→    shadow: dedupe(shadow),
   970→   970→  };
   971→   971→}
