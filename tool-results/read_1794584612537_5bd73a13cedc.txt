    91→function hashStringToSeed(input: string): number {
    92→  let h = 1779033703 ^ input.length;
    93→  for (let i = 0; i < input.length; i++) {
    94→    h = Math.imul(h ^ input.charCodeAt(i), 3432918353);
    95→    h = (h << 13) | (h >>> 19);
    96→  }
    97→  return h >>> 0;
    98→}
    99→
   100→function mulberry32(seed: number): () => number {
   101→  let a = seed >>> 0;
   102→  return () => {
   103→    a = (a + 0x6d2b79f5) | 0;
   104→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   105→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   106→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   107→  };
   108→}

   112→function font(weight: string, px: number, mono = false): string {
   113→  return `${weight} ${px}px ${mono ? MONO_STACK : FONT_STACK}`;

   116→/** Shrink font size until `text` fits `maxWidth`. Returns the final px size. */
   117→function fitText(
   118→  ctx: CanvasRenderingContext2D,
   119→  text: string,
   120→  maxWidth: number,
   121→  startPx: number,
   122→  weight: string,
   123→  mono = false,
   124→  minPx = 13
   125→): number {
   126→  let px = startPx;
   127→  for (;;) {
   128→    ctx.font = font(weight, px, mono);
   129→    if (ctx.measureText(text).width <= maxWidth || px <= minPx) break;
   130→    px -= 2;
   131→  }
   132→  return px;
   133→}
   134→
   135→/**
   136→ * v4.5 slip-box text layout — word-wraps the market text into up to `maxLines`
   137→ * large lines instead of shrinking it into unreadable print. Tries the
   138→ * largest size first and only steps down while everything still fits. If no
   139→ * size fits within maxLines, the text is greedily wrapped at the floor size
   140→ * (never drawn as one overflowing line).
   141→ */
   142→function wrapFit(
   143→  ctx: CanvasRenderingContext2D,
   144→  text: string,
   145→  maxWidth: number,
   146→  startPx: number,
   147→  weight: string,
   148→  minPx: number,
   149→  maxLines = 2
   150→): string[] {
   151→  const words = text.split(/\s+/).filter(Boolean);
   152→  for (let px = startPx; px >= minPx; px -= 2) {
   153→    ctx.font = font(weight, px);
   154→    const lines: string[] = [];
   155→    let cur = "";
   156→    let ok = true;
   157→    for (const wd of words) {
   158→      const cand = cur ? `${cur} ${wd}` : wd;
   159→      if (ctx.measureText(cand).width <= maxWidth) {
   160→        cur = cand;
   161→      } else {
   162→        if (cur) lines.push(cur);
   163→        cur = wd;
   164→        if (ctx.measureText(wd).width > maxWidth) {
   165→          ok = false;
   166→          break;
   167→        }
   168→      }
   169→    }
   170→    if (cur) lines.push(cur);
   171→    if (ok && lines.length <= maxLines) return lines;
   172→  }
   173→  // greedy wrap at the floor size — the fallback keeps every word on-canvas
   174→  ctx.font = font(weight, minPx);
   175→  const lines: string[] = [];
   176→  let cur = "";
   177→  for (const wd of words) {
   178→    const cand = cur ? `${cur} ${wd}` : wd;
   179→    if (!cur || ctx.measureText(cand).width <= maxWidth) {
   180→      cur = cand;
   181→    } else {
   182→      lines.push(cur);
   183→      cur = wd;
   184→    }
   185→  }
   186→  if (cur) lines.push(cur);
   187→  return lines.slice(0, maxLines + 1);
   188→}

   190→/** Same as fitText but accounts for letter tracking. */
   191→function fitTracked(
   192→  ctx: CanvasRenderingContext2D,
   193→  text: string,
   194→  maxWidth: number,
   195→  startPx: number,
   196→  weight: string,
   197→  tracking: number,
   198→  mono = false,
   199→  minPx = 12
   200→): number {
   201→  let px = startPx;
   202→  for (;;) {
   203→    ctx.font = font(weight, px, mono);
   204→    if (measureTracked(ctx, text, tracking) <= maxWidth || px <= minPx) break;
   205→    px -= 2;
   206→  }
   207→  return px;
   208→}
   209→
   210→function measureTracked(ctx: CanvasRenderingContext2D, text: string, tracking: number): number {
   211→  if (!text) return 0;
   212→  let w = 0;
   213→  for (const ch of text) w += ctx.measureText(ch).width + tracking;
   214→  return w - tracking;
   215→}

   217→function drawTracked(
   218→  ctx: CanvasRenderingContext2D,
   219→  text: string,
   220→  cx: number,
   221→  cy: number,
   222→  tracking: number
   223→): void {
   224→  const total = measureTracked(ctx, text, tracking);
   225→  let x = cx - total / 2;
   226→  const prevAlign = ctx.textAlign;
   227→  ctx.textAlign = "left";
   228→  for (const ch of text) {
   229→    ctx.fillText(ch, x, cy);
   230→    x += ctx.measureText(ch).width + tracking;
   231→  }
   232→  ctx.textAlign = prevAlign;
   233→}

   252→function roundRectPath(
   253→  ctx: CanvasRenderingContext2D,
   254→  x: number,
   255→  y: number,
   256→  w: number,
   257→  h: number,
   258→  r: number
   259→): void {
   260→  const rr = Math.min(r, w / 2, h / 2);
   261→  ctx.beginPath();
   262→  ctx.moveTo(x + rr, y);
   263→  ctx.lineTo(x + w - rr, y);
   264→  ctx.arcTo(x + w, y, x + w, y + rr, rr);
   265→  ctx.lineTo(x + w, y + h - rr);
   266→  ctx.arcTo(x + w, y + h, x + w - rr, y + h, rr);
   267→  ctx.lineTo(x + rr, y + h);
   268→  ctx.arcTo(x, y + h, x, y + h - rr, rr);
   269→  ctx.lineTo(x, y + rr);
   270→  ctx.arcTo(x, y, x + rr, y, rr);
   271→  ctx.closePath();
   272→}

   274→function getInitials(name: string): string {
   275→  const parts = name.trim().split(/\s+/).filter(Boolean);
   276→  if (parts.length === 0) return "?";
   277→  const initials = parts
   278→    .slice(0, 2)
   279→    .map((p) => p.charAt(0))
   280→    .join("");
   281→  return initials.toUpperCase() || "?";
   282→}

   284→// ─── Flag loading (same-origin proxy, parallel, timeout fallback) ──────────

   286→function loadFlag(url: string): Promise<HTMLImageElement | null> {
   287→  return new Promise((resolve) => {
   288→    const img = new Image();
   289→    let settled = false;
   290→    let timer = 0;
   291→    const finish = (result: HTMLImageElement | null): void => {
   292→      if (settled) return;
   293→      settled = true;
   294→      if (timer) window.clearTimeout(timer);
   295→      resolve(result);
   296→    };
   297→    timer = window.setTimeout(() => finish(null), FLAG_TIMEOUT_MS);
   298→    img.onload = () => finish(img);
   299→    img.onerror = () => finish(null);
   300→    img.src = `/api/tennis/img?url=${encodeURIComponent(url)}`;
   301→  });
   302→}

   306→function drawCourt(ctx: CanvasRenderingContext2D, pal: CourtPalette): void {
   307→  const cx = W / 2;
   308→  const topY = 520;
   309→  const botY = 1436;
   310→  const topHW = 200; // half width at top baseline
   311→  const botHW = 530; // half width at bottom baseline
   312→  const span = botY - topY;
   313→  const netFrac = 0.4;
   314→  const singlesFrac = 0.75;
   315→  const serviceGapFrac = 18 / 39; // net→service line : service line→baseline
   316→
   317→  const halfWAt = (y: number): number => topHW + (botHW - topHW) * ((y - topY) / span);
   318→  const xAt = (side: 1 | -1, y: number, frac: number): number => cx + side * halfWAt(y) * frac;
   319→
   320→  // Surround
   321→  const bg = ctx.createLinearGradient(0, 0, 0, H);
   322→  bg.addColorStop(0, pal.surroundTop);
   323→  bg.addColorStop(1, pal.surroundBottom);
   324→  ctx.fillStyle = bg;
   325→  ctx.fillRect(0, 0, W, H);
   326→
   327→  // Court trapezoid (wider at bottom → broadcast overhead perspective)
   328→  const netY = topY + span * netFrac;
   329→  const serviceTopY = netY - (netY - topY) * serviceGapFrac;
   330→  const serviceBotY = netY + (botY - netY) * serviceGapFrac;
   331→
   332→  ctx.beginPath();
   333→  ctx.moveTo(cx - topHW, topY);
   334→  ctx.lineTo(cx + topHW, topY);
   335→  ctx.lineTo(cx + botHW, botY);
   336→  ctx.lineTo(cx - botHW, botY);
   337→  ctx.closePath();
   338→  const cg = ctx.createLinearGradient(0, topY, 0, botY);
   339→  cg.addColorStop(0, pal.courtTop);
   340→  cg.addColorStop(1, pal.courtBottom);
   341→  ctx.fillStyle = cg;
   342→  ctx.fill();
   343→
   344→  // Court shadow edge (subtle 3D depth along top edge)
   345→  const edge = ctx.createLinearGradient(0, topY - 26, 0, topY + 30);
   346→  edge.addColorStop(0, "rgba(0,0,0,0)");
   347→  edge.addColorStop(0.75, "rgba(0,0,0,0.22)");
   348→  edge.addColorStop(1, "rgba(0,0,0,0)");
   349→  ctx.fillStyle = edge;
   350→  ctx.fillRect(cx - topHW - 60, topY - 26, topHW * 2 + 120, 56);
   351→
   352→  // Lines — neon white, width grows toward the viewer
   353→  ctx.save();
   354→  ctx.strokeStyle = pal.line;
   355→  ctx.shadowColor = "rgba(255,255,255,0.55)";
   356→  ctx.shadowBlur = 7;
   357→  ctx.lineCap = "butt";
   358→
   359→  const lwAt = (y: number): number => 3 + 3.6 * ((y - topY) / span);
   360→  const seg = (x1: number, y1: number, x2: number, y2: number, yRef: number): void => {
   361→    ctx.lineWidth = lwAt(yRef);
   362→    ctx.beginPath();
   363→    ctx.moveTo(x1, y1);
   364→    ctx.lineTo(x2, y2);
   365→    ctx.stroke();
   366→  };
   367→
   368→  // Doubles sidelines + baselines
   369→  seg(xAt(-1, topY, 1), topY, xAt(-1, botY, 1), botY, (topY + botY) / 2);
   370→  seg(xAt(1, topY, 1), topY, xAt(1, botY, 1), botY, (topY + botY) / 2);
   371→  seg(xAt(-1, topY, 1), topY, xAt(1, topY, 1), topY, topY);
   372→  seg(xAt(-1, botY, 1), botY, xAt(1, botY, 1), botY, botY);
   373→
   374→  // Singles sidelines
   375→  seg(xAt(-1, topY, singlesFrac), topY, xAt(-1, botY, singlesFrac), botY, (topY + botY) / 2);
   376→  seg(xAt(1, topY, singlesFrac), topY, xAt(1, botY, singlesFrac), botY, (topY + botY) / 2);
   377→
   378→  // Service lines
   379→  seg(xAt(-1, serviceTopY, singlesFrac), serviceTopY, xAt(1, serviceTopY, singlesFrac), serviceTopY, serviceTopY);
   380→  seg(xAt(-1, serviceBotY, singlesFrac), serviceBotY, xAt(1, serviceBotY, singlesFrac), serviceBotY, serviceBotY);
   381→
   382→  // Center service line (net → service line, both halves)
   383→  seg(cx, serviceTopY, cx, serviceBotY, netY);
   384→
   385→  // Net: dark band + bright cord, slightly wider than the court
   386→  const netHalfW = halfWAt(netY) * 1.12;
   387→  ctx.shadowBlur = 0;
   388→  ctx.fillStyle = "rgba(2,8,20,0.35)";
   389→  ctx.fillRect(cx - netHalfW, netY - 6, netHalfW * 2, 14);
   390→  ctx.shadowColor = "rgba(255,255,255,0.8)";
   391→  ctx.shadowBlur = 10;
   392→  ctx.strokeStyle = "rgba(255,255,255,0.95)";
   393→  ctx.lineWidth = 4;
   394→  ctx.beginPath();
   395→  ctx.moveTo(cx - netHalfW, netY);
   396→  ctx.lineTo(cx + netHalfW, netY);
   397→  ctx.stroke();
   398→
   399→  // Center marks on both baselines
   400→  ctx.shadowBlur = 4;
   401→  ctx.strokeStyle = pal.line;
   402→  ctx.lineWidth = 5;
   403→  ctx.beginPath();
   404→  ctx.moveTo(cx, topY + 4);
   405→  ctx.lineTo(cx, topY + 20);
   406→  ctx.moveTo(cx, botY - 20);
   407→  ctx.lineTo(cx, botY - 4);
   408→  ctx.stroke();
   409→  ctx.restore();
   410→}

   413→  ctx.save();
   414→  ctx.globalCompositeOperation = "screen";
   415→
   416→  const beam = (x0: number, y0: number, x1: number, x2: number, y1: number): void => {
   417→    const g = ctx.createLinearGradient(x0, y0, (x1 + x2) / 2, y1);
   418→    g.addColorStop(0, `rgba(103,232,249,${(0.2 * intensity).toFixed(3)})`);
   419→    g.addColorStop(1, "rgba(103,232,249,0)");
   420→    ctx.fillStyle = g;
   421→    ctx.beginPath();
   422→    ctx.moveTo(x0, y0);
   423→    ctx.lineTo(x1, y1);
   424→    ctx.lineTo(x2, y1);
   425→    ctx.closePath();
   426→    ctx.fill();
   427→  };
   428→  beam(-40, -40, 170, 640, 780);
   429→  beam(W + 40, -40, W - 170, W - 640, 780);
   430→
   431→  const lamp = (x: number, y: number): void => {
   432→    const g = ctx.createRadialGradient(x, y, 0, x, y, 280);
   433→    g.addColorStop(0, `rgba(224,247,255,${(0.55 * intensity).toFixed(3)})`);

   439→  lamp(W - 24, 10);

   495→// ─── Players: neon rings, flags, names, badges ─────────────────────────────
   496→
   497→function drawFlagOrInitials(
   498→  ctx: CanvasRenderingContext2D,
   499→  img: HTMLImageElement | null,
   500→  cx: number,
   501→  cy: number,
   502→  r: number,
   503→  name: string,
   504→  accent: string
   505→): void {
   506→  ctx.save();
   507→  if (img && img.width > 0 && img.height > 0) {
   508→    ctx.beginPath();
   509→    ctx.arc(cx, cy, r, 0, Math.PI * 2);
   510→    ctx.clip();
   511→    const scale = Math.max((2 * r) / img.width, (2 * r) / img.height);
   512→    const dw = img.width * scale;
   513→    const dh = img.height * scale;
   514→    ctx.drawImage(img, cx - dw / 2, cy - dh / 2, dw, dh);
   515→  } else {
   516→    // Fallback: dark disc + neon ring + initials
   517→    ctx.beginPath();

   666→function drawBall(ctx: CanvasRenderingContext2D, cx: number, cy: number, r: number, glow: number): void {
   667→  // Soft glow halo
   668→  ctx.save();
   669→  ctx.globalCompositeOperation = "screen";
   670→  const halo = ctx.createRadialGradient(cx, cy, r * 0.55, cx, cy, r * 2.1);
   671→  halo.addColorStop(0, `rgba(223,255,79,${(0.35 * glow).toFixed(3)})`);
   672→  halo.addColorStop(1, "rgba(223,255,79,0)");
   673→  ctx.fillStyle = halo;
   674→  ctx.fillRect(cx - r * 2.2, cy - r * 2.2, r * 4.4, r * 4.4);
   675→  ctx.restore();
   676→
   677→  // Sphere body
   678→  ctx.save();
   679→  ctx.shadowColor = "rgba(223,255,79,0.65)";
   680→  ctx.shadowBlur = 42;
   681→  const g = ctx.createRadialGradient(cx - r * 0.35, cy - r * 0.4, r * 0.1, cx, cy, r);
   682→  g.addColorStop(0, "#f7ffbe");
   683→  g.addColorStop(0.35, OPTIC);
   684→  g.addColorStop(0.75, "#c3e02e");
   685→  g.addColorStop(1, "#8fb31c");
   686→  ctx.beginPath();
   687→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
   688→  ctx.fillStyle = g;
   689→  ctx.fill();
   690→  ctx.shadowBlur = 0;
   691→
   692→  // White seam arcs (clipped to sphere)
   693→  ctx.beginPath();
   694→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
   695→  ctx.clip();
   696→  ctx.strokeStyle = "rgba(255,255,255,0.92)";
   697→  ctx.lineWidth = r * 0.09;
   698→  ctx.beginPath();
   699→  ctx.arc(cx - r * 1.18, cy, r * 0.95, -0.62, 0.62);
   700→  ctx.stroke();
   701→  ctx.beginPath();
   702→  ctx.arc(cx + r * 1.18, cy, r * 0.95, Math.PI - 0.62, Math.PI + 0.62);
   703→  ctx.stroke();
   704→
   705→  // Sphere edge shading
   706→  const sh = ctx.createRadialGradient(cx, cy, r * 0.55, cx, cy, r);
   707→  sh.addColorStop(0, "rgba(0,0,0,0)");
   708→  sh.addColorStop(1, "rgba(30,50,0,0.35)");
   709→  ctx.fillStyle = sh;
   710→  ctx.fillRect(cx - r, cy - r, 2 * r, 2 * r);
   711→  ctx.restore();
   712→}

   783→function drawSlipBox(ctx: CanvasRenderingContext2D, opts: SlipBoxOpts): void {
   784→  const { x, y, w, h, label, slip, blurred } = opts;
   785→  const pad = 32;
   786→
   787→  // Glass panel
   788→  roundRectPath(ctx, x, y, w, h, 26);
   789→  ctx.fillStyle = "rgba(255,255,255,0.08)";
   790→  ctx.fill();
   791→  ctx.strokeStyle = "rgba(255,255,255,0.25)";
   792→  ctx.lineWidth = 2;
   793→  ctx.stroke();
   794→
   795→  ctx.textBaseline = "middle";
   796→
   797→  // Label (always sharp) — v4.5: 24px tracked label
   798→  ctx.font = font("700", 24, true);
   799→  ctx.fillStyle = "rgba(255,255,255,0.72)";
   800→  ctx.textAlign = "left";
   801→  drawTrackedLeft(ctx, label, x + pad, y + 34, 4);
   802→
   803→  // Content (blurred in teaser mode)
   804→  if (blurred) ctx.filter = "blur(7px)";
   805→
   806→  // v4.5 FIX — the slip market text renders BIG: up to 3 wrapped lines at
   807→  // 42px → 30px floor (v4.0 capped it at a 26px two-line floor that still
   808→  // read as tiny print on long official market names).
   809→  const maxTextW = w - pad * 2 - 250;
   810→  const lines = wrapFit(ctx, slip.textEn, maxTextW, 42, "800", 30, 3);
   811→  const lineYs = [y + 76, y + 120, y + 164];
   812→  ctx.fillStyle = "#ffffff";
   813→  ctx.textAlign = "left";
   814→  lines.slice(0, 3).forEach((ln, i) => ctx.fillText(ln, x + pad, lineYs[i]));
   815→
   816→  // Odds — v4.5: 52px bold gold in the right column
   817→  ctx.font = font("800", 52, true);
   818→  ctx.textAlign = "right";
   819→  ctx.fillStyle = GOLD;
   820→  ctx.fillText(`@${slip.odds.toFixed(2)}`, x + w - pad, y + 84);
   821→  // implied probability directly under the odds (right column)
   822→  const impliedPct = Math.max(0, Math.min(100, 100 / Math.max(slip.odds, 1.01)));
   823→  ctx.font = font("600", 26, true);
   824→  ctx.fillStyle = "rgba(255,255,255,0.72)";
   825→  ctx.fillText(`P ≈ ${impliedPct.toFixed(1)}%`, x + w - pad, y + 136);
   826→  ctx.textAlign = "left";
   827→
   828→  ctx.filter = "none";
   829→}

   899→export async function drawPoster(canvas: HTMLCanvasElement, data: PosterData): Promise<void> {
   900→  canvas.width = W;
   901→  canvas.height = H;
   902→  const ctx = canvas.getContext("2d");
   903→  if (!ctx) throw new Error("Canvas 2D context is unavailable");
   904→
   905→  // Reset any inherited state
   906→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   907→  ctx.globalAlpha = 1;
   908→  ctx.globalCompositeOperation = "source-over";
   909→  ctx.filter = "none";
   910→  ctx.textBaseline = "middle";
   911→
   912→  // Flags load in parallel through the same-origin proxy; failures fall back
   913→  // to neon-initials rings after the timeout.
   914→  const [flagA, flagB] = await Promise.all([
   915→    data.playerA.flagUrl ? loadFlag(data.playerA.flagUrl) : Promise.resolve(null),
   916→    data.playerB.flagUrl ? loadFlag(data.playerB.flagUrl) : Promise.resolve(null),
   917→  ]);
   918→
   919→  // Deterministic per-match RNG: particles / glow variation are seeded from
   920→  // the player names so the same match renders pixel-identically.
   921→  const rng = mulberry32(hashStringToSeed(`${data.playerA.name}${data.playerB.name}`));
   922→  const glowJitter = 0.85 + rng() * 0.3; // 0.85..1.15
   923→  const glow2 = 0.85 + rng() * 0.3;
   924→
   925→  const pal = PALETTES[data.surface];
   926→
   927→  // 1. Procedural court background
   928→  drawCourt(ctx, pal);
   929→  if (data.surface === "HARD") drawFloodBeams(ctx, glowJitter);
   930→  if (data.surface === "CLAY") {
   931→    // Dust particles are seeded from hashPre so the same match renders identically
   932→    drawClayDust(ctx, mulberry32(hashStringToSeed(data.hashPre ?? `${data.playerA.name} vs ${data.playerB.name}`)));
   933→  }
   934→  if (data.surface === "GRASS") drawGrassHalo(ctx, W / 2, 690, glowJitter);
   935→  drawVignette(ctx);
   936→
   937→  // 2. Player rings + names + badges
   938→  const ringY = 190;
   939→  const ringR = 88;
   940→  const cxA = 268;
   941→  const cxB = W - 268;
   942→  drawPlayerRing(ctx, data.playerA, flagA, cxA, ringY, ringR, data.favorite === "A");
   943→  drawPlayerRing(ctx, data.playerB, flagB, cxB, ringY, ringR, data.favorite === "B");
   944→  drawPlayerLabel(ctx, data.playerA, cxA, 320, data.favorite === "A");
   945→  drawPlayerLabel(ctx, data.playerB, cxB, 320, data.favorite === "B");
   946→
   947→  // 3. Tournament header strip
   948→  drawTournamentStrip(ctx, data);
   949→
   950→  // 4. Center glowing ball
   951→  drawBall(ctx, W / 2, 690, 86, glow2);
   952→
   953→  // 5. MODEL P chip
   954→  if (typeof data.marketProbAFav === "number" && Number.isFinite(data.marketProbAFav)) {
   955→    drawModelChip(ctx, Math.max(0, Math.min(1, data.marketProbAFav)), W / 2, 800);
   956→  }
   957→
   958→  // 6. Slip boxes — v4.5: 186px-tall boxes so the official market text
   959→  // renders at 30–42px over up to 3 wrapped lines (never shrunken small print)
   960→  const boxX = 90;
   961→  const boxW = W - 180;
   962→  const boxH = 186;
   963→  if (data.slipA) {
   964→    drawSlipBox(ctx, { x: boxX, y: 842, w: boxW, h: boxH, label: "SLIP A", slip: data.slipA, blurred: false });
   965→  }
   966→  if (data.slipB) {
   967→    const teaser = data.mode === "teaser";
   968→    const by = 1046;
   969→    drawSlipBox(ctx, { x: boxX, y: by, w: boxW, h: boxH, label: "SLIP B", slip: data.slipB, blurred: teaser });
   970→    if (teaser) {
   971→      // Dark backing panel + padlock + "UNLOCK IN VIP"
   972→      const panelW = 250;
   973→      const panelH = 118;
   974→      const panelCy = by + boxH / 2 - 2;
   975→      roundRectPath(ctx, W / 2 - panelW / 2, panelCy - panelH / 2, panelW, panelH, 20);
   976→      ctx.fillStyle = "rgba(6,10,22,0.6)";
   977→      ctx.fill();
   978→      ctx.strokeStyle = "rgba(255,209,102,0.4)";
   979→      ctx.lineWidth = 1.5;
   980→      ctx.stroke();
   981→      drawPadlock(ctx, W / 2, panelCy - 12);
   982→      ctx.font = font("800", 19, true);
   983→      ctx.fillStyle = GOLD;
   984→      ctx.textBaseline = "middle";
   985→      drawTracked(ctx, "UNLOCK IN VIP", W / 2, panelCy + 38, 4);
   986→    }
   987→  }
   988→
   989→  // 7. Timing chip (aligned under the taller slip boxes)
   990→  if (data.timingLabel) drawTimingChip(ctx, data.timingLabel, W / 2, 1262);
   991→
   992→  // 8. Footer telemetry
   993→  drawFooter(ctx, data);
   994→}
