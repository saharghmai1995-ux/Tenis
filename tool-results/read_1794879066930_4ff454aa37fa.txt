    57→    57→  const [dStr, mStr, yStr] = TEHRAN_DATE_FMT.format(new Date(iso)).split("/");

    62→    62→  const [sd, sm, sy] = TEHRAN_DATE_FMT.format(d).split("/");

    71→    71→function bucketize(
    72→    72→  points: BankPoint[],
    73→    73→  keyOf: (iso: string) => { key: string; label: string }
    74→    74→): BankPeriodBucket[] {
    75→    75→  const order: string[] = [];
    76→    76→  const map = new Map<string, BankPeriodBucket>();
    77→    77→  for (const p of points) {
    78→    78→    if (!p.at || p.index === 0) continue; // skip the START point
    79→    79→    const { key, label } = keyOf(p.at);
    80→    80→    let b = map.get(key);
    81→    81→    if (!b) {
    82→    82→      b = { period: key, label, pnl: 0, wins: 0, losses: 0, matches: 0, winRate: 0, bankEnd: 0, staked: 0, roi: 0 };
    83→    83→      map.set(key, b);
    84→    84→      order.push(key);
    85→    85→    }
    86→    86→    b.pnl += p.delta ?? 0;
    87→    87→    b.matches += 1;
    88→    88→    if (p.status === "WON") b.wins += 1;
    89→    89→    else if (p.status === "LOST") b.losses += 1;
    90→    90→    b.bankEnd = p.bank;
    91→    91→  }
    92→    92→  return order.map((k) => {
    93→    93→    const b = map.get(k)!;
    94→    94→    b.pnl = Math.round(b.pnl * 100) / 100;
    95→    95→    const decided = b.wins + b.losses;
    96→    96→    b.winRate = decided > 0 ? b.wins / decided : 0;
    97→    97→    b.roi = b.staked > 0 ? b.pnl / b.staked : b.pnl > 0 ? 1 : b.pnl < 0 ? -1 : 0;
    98→    98→    return b;
    99→    99→  });

   104→   104→function buildBankHistory(bankCurve: BankPoint[], chronological: LedgerMatchRow[]): BankHistory {
   105→   105→  const daily = bucketize(bankCurve, (iso) => {
   106→   106→    const key = tehranDayKey(iso);
   107→   107→    return { key, label: `${key.slice(6, 8)}/${key.slice(4, 6)}` };
   108→   108→  });
   109→   109→  // stake totals per daily bucket from chronological rows
   110→   110→  const dayKeyOfRow = (r: LedgerMatchRow) => tehranDayKey(r.settledAt ?? r.createdAt);
   111→   111→  const dayStakes = new Map<string, number>();
   112→   112→  for (const r of chronological) {
   113→   113→    const k = dayKeyOfRow(r);
   114→   114→    dayStakes.set(k, (dayStakes.get(k) ?? 0) + r.stake);
   115→   115→  }
   116→   116→  for (const b of daily) b.staked = Math.round((dayStakes.get(b.period) ?? 0) * 100) / 100;
   117→   117→
   118→   118→  const weekly = bucketize(bankCurve, (iso) => tehranWeekKey(iso));
   119→   119→  const weekStakes = new Map<string, number>();
   120→   120→  for (const r of chronological) {
   121→   121→    const { key } = tehranWeekKey(r.settledAt ?? r.createdAt);
   122→   122→    weekStakes.set(key, (weekStakes.get(key) ?? 0) + r.stake);
   123→   123→  }
   124→   124→  for (const b of weekly) b.staked = Math.round((weekStakes.get(b.period) ?? 0) * 100) / 100;
   125→   125→
   126→   126→  const monthly = bucketize(bankCurve, (iso) => {
   127→   127→    const key = tehranDayKey(iso);
   128→   128→    const mk = `${key.slice(0, 6)}`;
   129→   129→    return { key: mk, label: `${FA_MONTHS[Number(key.slice(4, 6)) - 1] ?? mk} ${key.slice(0, 4)}` };
   130→   130→  });
   131→   131→  const monthStakes = new Map<string, number>();
   132→   132→  for (const r of chronological) {
   133→   133→    const k = tehranDayKey(r.settledAt ?? r.createdAt).slice(0, 6);
   134→   134→    monthStakes.set(k, (monthStakes.get(k) ?? 0) + r.stake);
   135→   135→  }
   136→   136→  for (const b of monthly) b.staked = Math.round((monthStakes.get(b.period) ?? 0) * 100) / 100;
   137→   137→
   138→   138→  const decidedDaily = daily.filter((b) => b.matches > 0);
   139→   139→  const bestDay = [...decidedDaily].sort((a, b) => b.pnl - a.pnl)[0];
   140→   140→  const worstDay = [...decidedDaily].sort((a, b) => a.pnl - b.pnl)[0];
   141→   141→
   142→   142→  const todayKey = tehranDayKey(new Date().toISOString());
   143→   143→  const todayRows = chronological.filter((r) => dayKeyOfRow(r) === todayKey);
   144→   144→  const todayWins = todayRows.filter((r) => r.status === "WON").length;
   145→   145→  const todayLosses = todayRows.filter((r) => r.status === "LOST").length;
   146→   146→  const todayMatches = todayRows.length;
   147→   147→
   148→   148→  // peak & max drawdown over the full point series
   149→   149→  let peak: { bank: number; at: string } = { bank: BANK_START, at: bankCurve[0]?.at ?? new Date().toISOString() };
   150→   150→  let peakBank = BANK_START;
   151→   151→  let maxDd = { units: 0, pct: 0, from: "", to: "" };
   152→   152→  let ddPeakAt = bankCurve[0]?.at ?? new Date().toISOString();
   153→   153→  for (const p of bankCurve) {
   154→   154→    if (p.bank >= peakBank) {
   155→   155→      peakBank = p.bank;
   156→   156→      peak = { bank: p.bank, at: p.at ?? ddPeakAt };
   157→   157→      ddPeakAt = p.at ?? ddPeakAt;
   158→   158→    } else {
   159→   159→      const ddUnits = Math.round((peakBank - p.bank) * 100) / 100;
   160→   160→      const ddPct = peakBank > 0 ? Math.round((ddUnits / peakBank) * 10000) / 100 : 0;
   161→   161→      if (ddUnits > maxDd.units) {
   162→   162→        maxDd = { units: ddUnits, pct: ddPct, from: ddPeakAt, to: p.at ?? ddPeakAt };
   163→   163→      }
   164→   164→    }
   165→   165→  }
   166→   166→
   167→   167→  return {
   168→   168→    daily,
   169→   169→    weekly,
   170→   170→    monthly,
   171→   171→    bestDay,
   172→   172→    worstDay,
   173→   173→    today: {
   174→   174→      matches: todayMatches,
   175→   175→      wins: todayWins,
   176→   176→      losses: todayLosses,
   177→   177→      pnl: Math.round(todayRows.reduce((s, r) => s + r.profit, 0) * 100) / 100,
   178→   178→      winRate: todayWins + todayLosses > 0 ? todayWins / (todayWins + todayLosses) : 0,
   179→   179→    },
   180→   180→    peak,
   181→   181→    maxDrawdown: maxDd,
   182→   182→  };
   183→   183→}

   317→   317→ * v5.5 — the 100-unit stake law (master prompt §1, mandatory in ledger.ts).
   318→   318→ * Idempotent migration that rewrites every legacy stake in the database:
   319→   319→ *   1) single-leg cards (BANKER 60u / mix 25u / accumulator 25u / shadow 1u)
   320→   320→ *      → exactly STANDARD_MATCH_STAKE (100u);
   321→   321→ *   2) multi-wing positions (DUAL / REGCOMBO / CLUSTER / SNIPER) → every
   322→   322→ *      match group rescaled proportionally so the wings sum EXACTLY to 100u
   323→   323→ *      (silver-era 50u halves and legacy 50/50 splits included);
   324→   324→ *   3) 1u / 2u naked sniper stakes of the bleeding era are gone by the same
   325→   325→ *      rescale — no ledger row can ever carry them again.
   326→   326→ * Wing profits are re-derived from (odds × stake) at read time, so the
   327→   327→ * accounting identity stays intact after the rewrite.
   328→   328→ */
   329→   329→export async function normalizeStakeUnits(): Promise<void> {
   330→   330→  try {
   331→   331→    await db.tennisPick.updateMany({
   332→   332→      where: {
   333→   333→        line: { in: ["BANKER", "COMBO", "ACCUMULATOR", "SHADOW"] },
   334→   334→        stake: { not: STANDARD_MATCH_STAKE },
   335→   335→      },
   336→   336→      data: { stake: STANDARD_MATCH_STAKE },
   337→   337→    });
   338→   338→
   339→   339→    const legs = await db.tennisPick.findMany({
   340→   340→      where: { line: { in: ["DUAL", "REGCOMBO", "CLUSTER", "SNIPER"] } },
   341→   341→      select: { id: true, matchId: true, line: true, stake: true },
   342→   342→    });
   343→   343→    const groups = new Map<string, { id: string; stake: number }[]>();
   344→   344→    for (const l of legs) {
   345→   345→      const key = `${l.line}|${l.matchId}`;
   346→   346→      const arr = groups.get(key) ?? [];
   347→   347→      arr.push({ id: l.id, stake: l.stake });
   348→   348→      groups.set(key, arr);
   349→   349→    }
   350→   350→    for (const group of groups.values()) {
   351→   351→      if (group.length === 0) continue;
   352→   352→      const sum = group.reduce((s, g) => s + g.stake, 0);
   353→   353→      if (sum <= 0) continue;
   354→   354→      const scaled = group.map((g) => ({
   355→   355→        id: g.id,
   356→   356→        stake: Math.round((g.stake / sum) * STANDARD_MATCH_STAKE * 10) / 10,
   357→   357→      }));
   358→   358→      const drift =
   359→   359→        Math.round((STANDARD_MATCH_STAKE - scaled.reduce((s, g) => s + g.stake, 0)) * 10) / 10;
   360→   360→      const biggest = scaled.reduce((a, b) => (b.stake > a.stake ? b : a));
   361→   361→      biggest.stake = Math.round((biggest.stake + drift) * 10) / 10;
   362→   362→      for (const g of scaled) {
   363→   363→        const before = group.find((x) => x.id === g.id)!.stake;
   364→   364→        if (Math.abs(g.stake - before) < 0.05) continue;
   365→   365→        await db.tennisPick.update({ where: { id: g.id }, data: { stake: g.stake } });
   366→   366→      }
   367→   367→    }
   368→   368→  } catch (e) {
   369→   369→    console.error("[ledger] 100-unit stake normalization failed", e);
   370→   370→  }
   371→   371→}
