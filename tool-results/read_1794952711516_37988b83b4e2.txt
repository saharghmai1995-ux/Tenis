   329→   329→  oddsA: number,
   330→   330→  oddsB: number
   331→   331→): { stakeA: number; stakeB: number } {
   332→   332→  const invA = 1 / Math.max(oddsA, 1.01);
   333→   333→  const invB = 1 / Math.max(oddsB, 1.01);
   334→   334→  const stakeA = Math.round(((100 * invA) / (invA + invB)) * 10) / 10;
   335→   335→  const stakeB = Math.round((100 - stakeA) * 10) / 10;
   336→   336→  return { stakeA, stakeB };
   337→   337→}

   968→   968→/** Per-unit betting Sharpe of a single-leg position: EV / σ of the P&L. */
   969→   969→export function bettingSharpe(prob: number, odds: number): number {
   970→   970→  const ev = prob * odds - 1;
   971→   971→  const secondMoment = prob * Math.pow(odds - 1, 2) + (1 - prob);
   972→   972→  const variance = secondMoment - ev * ev;
   973→   973→  return variance > 1e-9 ? ev / Math.sqrt(variance) : 0;
   974→   974→}

   976→   976→/** Equal-profit dutching stakes for N legs (sum = total). */
   977→   977→function dutchN(oddsList: number[], total = 100): number[] {
   978→   978→  const invs = oddsList.map((o) => 1 / Math.max(o, 1.01));
   979→   979→  const s = invs.reduce((a, b) => a + b, 0);
   980→   980→  return invs.map((i) => Math.round(((total * i) / s) * 10) / 10);
   981→   981→}
   982→   982→
   983→   983→/**
   984→   984→ * Per-unit Sharpe of a dutched multi-leg position over disjoint outcomes:
   985→   985→ * winProbs[i] → leg i pays; else the whole position loses.
   986→   986→ */
   987→   987→function dutchedSharpe(winProbs: number[], oddsList: number[]): { sharpe: number; ev: number; stakes: number[] } {
   988→   988→  const stakes = dutchN(oddsList, 100);
   989→   989→  const payoffs = stakes.map((st, i) => (st * oddsList[i] - 100) / 100); // per-unit net
   990→   990→  const pLose = Math.max(0, 1 - winProbs.reduce((a, b) => a + b, 0));
   991→   991→  const ev = winProbs.reduce((s, p, i) => s + p * payoffs[i], 0) - pLose * 1;
   992→   992→  const secondMoment =
   993→   993→    winProbs.reduce((s, p, i) => s + p * payoffs[i] * payoffs[i], 0) + pLose;
   994→   994→  const variance = secondMoment - ev * ev;
   995→   995→  return {
   996→   996→    sharpe: variance > 1e-9 ? ev / Math.sqrt(variance) : 0,
   997→   997→    ev,
   998→   998→    stakes,
   999→   999→  };
