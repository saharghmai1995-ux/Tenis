    66→    66→    66→/**
    67→    67→    67→ * v10 — THE positive-overlap gate (master-prompt v10 §2, verbatim algebra):
    68→    68→    68→ *   impliedOverlapSum = (1/oddsA) + (1/oddsB)
    69→    69→    69→ *   effectiveOverlapOdds = 1 / impliedOverlapSum
    70→    70→    70→ *   netOverlapReturnPct = (effectiveOverlapOdds − 1) × 100
    71→    71→    71→ *   isApprovedPositiveOverlap =
    72→    72→    72→ *     impliedOverlapSum ≤ 0.865 ∧ netOverlapReturnPct ≥ 15.5 ∧
    73→    73→    73→ *     combinedCoverageProb ≥ 0.895 ∧ sharpeRatio ≥ 0.25
    74→    74→    74→ */
    75→    75→    75→export function positiveOverlapGate(
    76→    76→    76→  oddsA: number,
    77→    77→    77→  oddsB: number,
    78→    78→    78→  combinedCoverageProb: number,
    79→    79→    79→  sharpeRatio: number
    80→    80→    80→): OverlapGateMetrics {
    81→    81→    81→  const impliedOverlapSum =
    82→    82→    82→    1 / Math.max(oddsA, 1.01) + 1 / Math.max(oddsB, 1.01);
    83→    83→    83→  const effectiveOverlapOdds = 1 / impliedOverlapSum;
    84→    84→    84→  const netOverlapReturnPct = (effectiveOverlapOdds - 1) * 100;
    85→    85→    85→  const checks: { ok: boolean; why: string }[] = [
    86→    86→    86→    {
    87→    87→    87→      ok: impliedOverlapSum <= OVERLAP_MAX_IMPLIED_SUM,
    88→    88→    88→      why: `مجموع معکوس ضرایب ${impliedOverlapSum.toFixed(3)} بالای سقف 0.865`,
    89→    89→    89→    },
    90→    90→    90→    {
    91→    91→    91→      ok: netOverlapReturnPct >= OVERLAP_MIN_NET_RETURN_PCT,
    92→    92→    92→      why: `سود خالص همپوشانی +${netOverlapReturnPct.toFixed(1)}% زیر کف +15.5%`,
    93→    93→    93→    },
    94→    94→    94→    {
    95→    95→    95→      ok: combinedCoverageProb >= OVERLAP_MIN_COVERAGE,
    96→    96→    96→      why: `پوشش زنجیره مارکوف ${(combinedCoverageProb * 100).toFixed(1)}% زیر کف 89.5%`,
    97→    97→    97→    },
    98→    98→    98→    {
    99→    99→    99→      ok: sharpeRatio >= OVERLAP_MIN_SHARPE,
   100→   100→   100→      why: `نسبت شارپ ${sharpeRatio.toFixed(2)} زیر کف 0.25`,
   101→   101→   101→    },
   102→   102→   102→  ];
   103→   103→   103→  const failed = checks.filter((c) => !c.ok);
   104→   104→   104→  return {
   105→   105→   105→    impliedOverlapSum,
   106→   106→   106→    effectiveOverlapOdds,
   107→   107→   107→    netOverlapReturnPct,
   108→   108→   108→    combinedCoverageProb,
   109→   109→   109→    sharpe: sharpeRatio,
   110→   110→   110→    approved: failed.length === 0,
   111→   111→   111→    whyNot: failed.length > 0 ? failed.map((f) => f.why).join(" · ") : undefined,
   112→   112→   112→  };
   113→   113→   113→}

   328→   328→   328→export function dutchStakes(
   329→   329→   329→  oddsA: number,
   330→   330→   330→  oddsB: number
   331→   331→   331→): { stakeA: number; stakeB: number } {
   332→   332→   332→  const invA = 1 / Math.max(oddsA, 1.01);
   333→   333→   333→  const invB = 1 / Math.max(oddsB, 1.01);
   334→   334→   334→  const stakeA = Math.round(((100 * invA) / (invA + invB)) * 10) / 10;
   335→   335→   335→  const stakeB = Math.round((100 - stakeA) * 10) / 10;
   336→   336→   336→  return { stakeA, stakeB };
   337→   337→   337→}
