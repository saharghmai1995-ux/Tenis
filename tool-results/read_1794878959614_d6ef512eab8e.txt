   213→   213→/** v4.0 — equal-profit dutching stakes from the two leg prices (sum = 100). */
   214→   214→export function dutchStakes(
   215→   215→  oddsA: number,
   216→   216→  oddsB: number
   217→   217→): { stakeA: number; stakeB: number } {
   218→   218→  const invA = 1 / Math.max(oddsA, 1.01);
   219→   219→  const invB = 1 / Math.max(oddsB, 1.01);
   220→   220→  const stakeA = Math.round(((100 * invA) / (invA + invB)) * 10) / 10;
   221→   221→  const stakeB = Math.round((100 - stakeA) * 10) / 10;
   222→   222→  return { stakeA, stakeB };
   223→   223→}

   246→   246→/**
   247→   247→ * Evaluates the 3 official atomic archetypes and selects the eligible one
   248→   248→ * with the lowest exact simultaneous-loss probability. If none qualifies the
   249→   249→ * match is vetoed to the shadow ledger (no forced approvals).
   250→   250→ */
   251→   251→export function evaluateArchetypes(m: AnalyzedMatch): ArchetypeReport {
   252→   252→  const a = m.markov.arch;
   253→   253→  const sFav = m.favorite === "A" ? m.markov.setWinA : 1 - m.markov.setWinA;
   254→   254→  const holdFav = m.favorite === "A" ? m.markov.gA : m.markov.gB;
   255→   255→  const holdDog = m.favorite === "A" ? m.markov.gB : m.markov.gA;
   256→   256→
   257→   257→  const mkEval = (
   258→   258→    code: "ARCH1" | "ARCH2" | "ARCH3",
   259→   259→    pairLoss: number,
   260→   260→    windowOk: boolean,
   261→   261→    windowWhy: string
   262→   262→  ): ArchetypeEval => {
   263→   263→    const max =
   264→   264→      code === "ARCH1"
   265→   265→        ? ARCH1_MAX_PAIR_LOSS
   266→   266→        : code === "ARCH2"
   267→   267→          ? ARCH2_MAX_PAIR_LOSS
   268→   268→          : ARCH3_MAX_PAIR_LOSS;
   269→   269→    const riskOk = pairLoss <= max;
   270→   270→    const eligible = windowOk && riskOk;
   271→   271→    return {
   272→   272→      code,
   273→   273→      labelFa: ARCH_LABELS[code].fa,
   274→   274→      labelEn: ARCH_LABELS[code].en,
   275→   275→      coverage: 1 - pairLoss,
   276→   276→      pairLoss,
   277→   277→      eligible,
   278→   278→      whyNot: eligible
   279→   279→        ? undefined
   280→   280→        : !windowOk
   281→   281→          ? `${windowWhy} (${code === "ARCH2" ? "پنجره الگو ۲: 54% ≤ S_A < 64%" : code === "ARCH1" ? "پنجره الگو ۱: S_fav ≥ 64%" : "پنجره الگو ۳: Hold ≥ 84%"})`
   282→   282→          : `ریسک باخت همزمان ${(pairLoss * 100).toFixed(1)}% بالاتر از سقف ${(max * 100).toFixed(1)}%`,
   283→   283→    };
   284→   284→  };
   285→   285→
   286→   286→  const arch1 = mkEval(
   287→   287→    "ARCH1",
   288→   288→    a.pairLoss1,
   289→   289→    sFav >= ARCH1_MIN_S,
   290→   290→    `S_fav = ${(sFav * 100).toFixed(1)}% زیر آستانه تسلط 64%`
   291→   291→  );
   292→   292→  const arch2 = mkEval(
   293→   293→    "ARCH2",
   294→   294→    a.pairLoss2,
   295→   295→    m.markov.setWinA >= ARCH2_MIN_S && m.markov.setWinA < ARCH2_MAX_S,
   296→   296→    m.markov.setWinA < ARCH2_MIN_S
   297→   297→      ? `S_A = ${(m.markov.setWinA * 100).toFixed(1)}% زیر کف پنجره رقابتی 54%`
   298→   298→      : `S_A = ${(m.markov.setWinA * 100).toFixed(1)}% بالاتر از سقف پنجره رقابتی 64%`
   299→   299→  );
   300→   300→  const arch3 = mkEval(
   301→   301→    "ARCH3",
   302→   302→    a.pairLoss3,
   303→   303→    holdFav >= ARCH3_MIN_HOLD_FAV && holdDog >= ARCH3_MIN_HOLD_DOG,
   304→   304→    `Hold فاوریت ${(holdFav * 100).toFixed(1)}% / حریف ${(holdDog * 100).toFixed(1)}% — زیر آستانه دوئل سرویس (84%/82%)`
   305→   305→  );
   306→   306→
   307→   307→  const evals = [arch1, arch2, arch3];
   308→   308→  const eligible = evals.filter((e) => e.eligible);
   309→   309→  if (eligible.length === 0) return { evals };
   310→   310→
   311→   311→  // most robust coverage wins the tie
   312→   312→  eligible.sort((x, y) => x.pairLoss - y.pairLoss);
   313→   313→  const win = eligible[0];
   314→   314→  return {
   315→   315→    evals,
   316→   316→    selected: {
   317→   317→      code: win.code,
   318→   318→      labelFa: win.labelFa,
   319→   319→      labelEn: win.labelEn,
   320→   320→      coverage: win.coverage,
   321→   321→      pairLoss: win.pairLoss,
   322→   322→    },
   323→   323→  };
   324→   324→}

   327→   327→export function supervisorGate(m: AnalyzedMatch, selected?: ArchetypeInfo): VetoInfo {
   328→   328→  const rankA = m.playerA.rank;
   329→   329→  const rankB = m.playerB.rank;
   330→   330→  const badRankA = !rankA || rankA > TOP100_LIMIT;
   331→   331→  const badRankB = !rankB || rankB > TOP100_LIMIT;
   332→   332→  if (badRankA || badRankB) {
   333→   333→    return {
   334→   334→      vetoed: true,
   335→   335→      code: "TOP100",
   336→   336→      reason: `فیلتر تاپ-۱۰۰ سوپروایزر — ${badRankA ? `${m.playerA.name}${rankA ? ` (#${rankA})` : " (بدون رتبه)"}` : ""}${badRankA && badRankB ? " و " : ""}${badRankB ? `${m.playerB.name}${rankB ? ` (#${rankB})` : " (بدون رتبه)"}` : ""} خارج از دامنه دقت مدل (Kovalchik 2016)`,
   337→   337→    };
   338→   338→  }
   339→   339→
   340→   340→  if (!selected) {
   341→   341→    const sFav = m.favorite === "A" ? m.markov.setWinA : 1 - m.markov.setWinA;
   342→   342→    const best = [...(m.archEvals ?? [])].sort((x, y) => x.pairLoss - y.pairLoss)[0];
   343→   343→    const bestTxt = best
   344→   344→      ? `کم‌ریسک‌ترین الگو (${best.labelFa}) ریسک ${(best.pairLoss * 100).toFixed(1)}% دارد`
   345→   345→      : "هیچ الگویی ارزیابی نشد";
   346→   346→    return {
   347→   347→      vetoed: true,
   348→   348→      code: "P02_RISK",
   349→   349→      reason: `هیچ الگوی مکملی صلاحیت ندارد — S_fav = ${(sFav * 100).toFixed(1)}% خارج از پنجره الگوها و ${bestTxt}؛ انتقال به دفتر سایه`,
   350→   350→    };
   351→   351→  }
   352→   352→
   353→   353→  return { vetoed: false, code: "NONE" };
   354→   354→}

   360→   360→/**
   361→   361→ * v6.0 — Monte-Carlo P(total games > 20.5) from the simulated games histogram.
   362→   362→ * Feeds the heavy-favorite line switch (Bug-1 fix, Tab 1).
   363→   363→ */
   364→   364→function mcOverGames(m: AnalyzedMatch, line: number): number | undefined {
   365→   365→  const mc = m.monteCarlo;
   366→   366→  if (!mc || !mc.gamesDist || mc.runs <= 0) return undefined;
   367→   367→  let hits = 0;
   368→   368→  for (let g = Math.ceil(line); g < mc.gamesDist.length; g++) hits += mc.gamesDist[g];
   369→   369→  return hits / mc.runs;
   370→   370→}

   648→   648→/** Per-unit betting Sharpe of a single-leg position: EV / σ of the P&L. */
   649→   649→export function bettingSharpe(prob: number, odds: number): number {
   650→   650→  const ev = prob * odds - 1;
   651→   651→  const secondMoment = prob * Math.pow(odds - 1, 2) + (1 - prob);
   652→   652→  const variance = secondMoment - ev * ev;
   653→   653→  return variance > 1e-9 ? ev / Math.sqrt(variance) : 0;
   654→   654→}

   656→   656→/** Equal-profit dutching stakes for N legs (sum = total). */
   657→   657→function dutchN(oddsList: number[], total = 100): number[] {
   658→   658→  const invs = oddsList.map((o) => 1 / Math.max(o, 1.01));
   659→   659→  const s = invs.reduce((a, b) => a + b, 0);
   660→   660→  return invs.map((i) => Math.round(((total * i) / s) * 10) / 10);
   661→   661→}

   663→   663→/**
   664→   664→ * Per-unit Sharpe of a dutched multi-leg position over disjoint outcomes:
   665→   665→ * winProbs[i] → leg i pays; else the whole position loses.
   666→   666→ */
   667→   667→function dutchedSharpe(winProbs: number[], oddsList: number[]): { sharpe: number; ev: number; stakes: number[] } {
   668→   668→  const stakes = dutchN(oddsList, 100);
   669→   669→  const payoffs = stakes.map((st, i) => (st * oddsList[i] - 100) / 100); // per-unit net
   670→   670→  const pLose = Math.max(0, 1 - winProbs.reduce((a, b) => a + b, 0));
   671→   671→  const ev = winProbs.reduce((s, p, i) => s + p * payoffs[i], 0) - pLose * 1;
   672→   672→  const secondMoment =
   673→   673→    winProbs.reduce((s, p, i) => s + p * payoffs[i] * payoffs[i], 0) + pLose;
   674→   674→  const variance = secondMoment - ev * ev;
   675→   675→  return {
   676→   676→    sharpe: variance > 1e-9 ? ev / Math.sqrt(variance) : 0,
   677→   677→    ev,
   678→   678→    stakes,
   679→   679→  };

   693→   693→  selected?: RegimeEval;
   694→   694→}

   696→   696→interface RegimeContext {
   697→   697→  sFav: number;
   698→   698→  holdFav: number;
   699→   699→  holdDog: number;
   700→   700→  pFavWin: number;
   701→   701→  pFav20: number;
   702→   702→  pDog20: number;
   703→   703→  surfaceIsSlow: boolean; // CLAY/HARD → longer rallies, games-grind
   704→   704→  serveBattle: boolean;
   705→   705→}
   706→   706→
   707→   707→function regimeContext(m: AnalyzedMatch, mc: MonteCarloResult): RegimeContext {
   708→   708→  const fav = m.favorite;
   709→   709→  const sFav = fav === "A" ? m.markov.setWinA : 1 - m.markov.setWinA;
   710→   710→  const holdFav = fav === "A" ? m.markov.gA : m.markov.gB;
   711→   711→  const holdDog = fav === "A" ? m.markov.gB : m.markov.gA;
   712→   712→  const pFavWin = fav === "A" ? mc.outcomes.p20 + mc.outcomes.p21 : mc.outcomes.p12 + mc.outcomes.p02;
   713→   713→  const pFav20 = fav === "A" ? mc.outcomes.p20 : mc.outcomes.p02;
   714→   714→  const pDog20 = fav === "A" ? mc.outcomes.p02 : mc.outcomes.p20;
   715→   715→  return {
   716→   716→    sFav,
   717→   717→    holdFav,
   718→   718→    holdDog,
   719→   719→    pFavWin,
   720→   720→    pFav20,
   721→   721→    pDog20,
   722→   722→    surfaceIsSlow: m.surface !== "GRASS",
   723→   723→    serveBattle: holdFav >= 0.8 && holdDog >= 0.78,
   724→   724→  };
   725→   725→}

   729→   729→  return m.favorite === "A" ? m.playerA.name : m.playerB.name;

   732→   732→/**
   733→   733→ * STATION 4 — non-linear 5-regime router (master-prompt architecture):
   734→   734→ * EVERY regime is priced from the 10k Monte-Carlo pipeline, the non-linear
   735→   735→ * Sharpe of each position is computed, and the match is routed to the single
   736→   736→ * OPTIMAL regime (max Sharpe among the eligible ones). No regime qualifies →
   737→   737→ * the match is vetoed to the shadow ledger.
   738→   738→ */

   744→   744→  const ctx = regimeContext(m, mc);

   749→   749→
   750→   750→  // ── 1 · 🏦 TENNIS BANKER ─────────────────────────────────────────────────
   751→   751→  {

   754→   754→    const cands: { marketEn: string; marketFa: string; prob: number; window: [number, number] }[] = [
   755→   755→      {

   774→   774→      const price = fairPrice(c.prob, ONEXBET_OVERROUND_MAIN);
   775→   775→      const sh = bettingSharpe(c.prob, price);
   776→   776→      if (sh > bestSharpe) {
   777→   777→        bestSharpe = sh;

   779→   779→        bestPrice = price;
   780→   780→      }
   781→   781→    }
   782→   782→    const eligible = windowOk && pairLoss <= BANKER_MAX_PAIR_LOSS;
   783→   783→    evals.push({
   784→   784→      code: "BANKER",
   785→   785→      eligible,
   786→   786→      sharpe: bestSharpe,
   787→   787→      pairLoss,
   788→   788→      whyNot: windowOk
   789→   789→        ? `ریسک باخت فاوریت ${(pairLoss * 100).toFixed(1)}% بالای سقف ۲۲٪`
   790→   790→        : `S_fav = ${(ctx.sFav * 100).toFixed(1)}% / Hold = ${(ctx.holdFav * 100).toFixed(1)}% — زیر آستانه بانکر (70%/78%)`,
   791→   791→    });
   792→   792→    if (eligible && best) {
   793→   793→      candidates.push({
   794→   794→        sharpe: bestSharpe,
   795→   795→        pairLoss,
   796→   796→        decision: {
   797→   797→          code: "BANKER",
   798→   798→          ...REGIME_META.BANKER,

   804→   804→          whyEn: `SAFE SINGLE-PICK · S_fav ${(ctx.sFav * 100).toFixed(0)}% · HOLD ${(ctx.holdFav * 100).toFixed(0)}% · DIRECT ${STANDARD_MATCH_STAKE}u STAKE`,
   805→   805→          marketEn: best.marketEn,
   806→   806→          marketFa: best.marketFa,
   807→   807→          targetWindow: best.window,
   808→   808→          stakePlanFa: `استیک مستقیم ${STANDARD_MATCH_STAKE} واحد + تغذیه آکومولاتور سلاطین سرویس`,
   809→   809→          price: bestPrice,
   810→   810→          prob: best.prob,

   812→   812→        },
   813→   813→      });
   814→   814→    }

   816→   816→
   817→   817→  // ── 4 · ⚖️ DUAL-HEDGE (champion golden pair) ─────────────────────────────

   828→   828→    evals.push({
   829→   829→      code: "DUAL-HEDGE",
   830→   830→      eligible: windowOk,
   831→   831→      sharpe,
   832→   832→      pairLoss,
   833→   833→      whyNot: `P(باخت ۲-۰ فاوریت) = ${(pairLoss * 100).toFixed(1)}% بالای سقف الگوی قهرمان ۸.۸%`,
   834→   834→    });
   835→   835→    if (windowOk) {
   836→   836→      candidates.push({
   837→   837→        sharpe,
   838→   838→        pairLoss,
   839→   839→        decision: {

   847→   847→          whyEn: `CHAMPION PATTERN · P(0-2) ${(pairLoss * 100).toFixed(1)}% ≤ 8.8% · 88.2% PROVEN WIN RATE`,
   848→   848→          marketEn: `SET BETTING — ${favName} 2-0 + TOTAL SETS — OVER 2.5`,
   849→   849→          marketFa: `زوج‌تیپ: برد ۲-۰ ${favNameRaw} + مجموع ست‌ها بالای ۲.۵`,

   858→   858→  {

   860→   860→    const dominant = ctx.surfaceIsSlow && ctx.sFav >= 0.6;
   861→   861→    // The master-prompt «copula correlation > 76%» gate reads as the copula
   862→   862→    // JOINT mass of the dominant scenario: for slow-surface dominators it is
   863→   863→    // P(fav win ∧ under 21.5 games); for serve battles it is the share of
   864→   864→    // favorite wins that stay over 20.5 games.
   865→   865→    const jointMass = dominant ? mc.favWinUnder215 : mc.favWinShareOver205;
   866→   866→    const gate = dominant ? COMBO_MIN_CORR : COMBO_MIN_SHARE_SRV;
   867→   867→    const windowOk = (dominant || ctx.serveBattle) && jointMass > gate;
   868→   868→    let pMain: number;
   869→   869→    let pCover: number;
   870→   870→    let mainEn: string;
   871→   871→    let coverEn: string;
   872→   872→    let mainFa: string;
   873→   873→    let coverFa: string;
   874→   874→    if (dominant) {
   875→   875→      pMain = mc.favWinUnder215;
   876→   876→      // cover = TOTAL GAMES — OVER 21.5 (exact Monte-Carlo mass); main ∧ cover
   877→   877→      // are mutually exclusive → uncovered = 1 − pMain − pCover
   878→   878→      pCover = Math.max(0.02, mc.coverOver215);
   879→   879→      mainEn = `${favName} WIN & UNDER 21.5 GAMES`;
   880→   880→      coverEn = "TOTAL GAMES — OVER 21.5";
   881→   881→      mainFa = `برد ${favNameRaw} و زیر ۲۱.۵ گیم`;
   882→   882→      coverFa = "مجموع گیم‌ها بالای ۲۱.۵ (پوشش)";
   883→   883→    } else {

   885→   885→      // cover = under 20.5 games OR the underdog wins the match — DISJOINT
   886→   886→      // from the main leg; both lose only when the favorite LOSES ∧ over 20.5

   889→   889→      mainEn = `${favName} WIN & OVER 20.5 GAMES`;

   893→   893→    }
   894→   894→    const coverage = Math.min(1, pMain + pCover);
   895→   895→    const pairLoss = Math.max(0, 1 - coverage);
   896→   896→    // v6.0 Bug-3 fix — the 70.8% Over-21.5 wing now prints its REAL 1xBet
   897→   897→    // price (≈ 1.34x) instead of the old clampPrice(...,2.7,3.3) lie of 2.70x.

   899→   899→    const oddsCover = fairPrice(pCover, ONEXBET_OVERROUND_PROP);

   902→   902→    const eligible = windowOk && coverage >= minCoverage;
   903→   903→    evals.push({
   904→   904→      code: "COMBO",
   905→   905→      eligible,
   906→   906→      sharpe,
   907→   907→      pairLoss,
   908→   908→      whyNot: `جرم مشترک کاپولا برد×گیم = ${(jointMass * 100).toFixed(1)}% (آستانه ${(gate * 100).toFixed(0)}%) یا پوشش ${(coverage * 100).toFixed(1)}% زیر سقف ${(minCoverage * 100).toFixed(0)}%`,
   909→   909→    });
   910→   910→    if (eligible) {
   911→   911→      const variant = dominant ? "DOM" : "SRV";
   912→   912→      candidates.push({
   913→   913→        sharpe,
   914→   914→        pairLoss,
   915→   915→        decision: {
   916→   916→          code: "COMBO",
   917→   917→          ...REGIME_META.COMBO,
   918→   918→          sharpe,
   919→   919→          mcScore: 0,
   920→   920→          pairLoss,
   921→   921→          coverage,
   922→   922→          why: `جرم مشترک کاپولای برد فاوریت × خط گیم = ${(jointMass * 100).toFixed(1)}% بالای ${(gate * 100).toFixed(0)}% — کمبوی رسمی تک‌دکمه + پوشش با پوشش ${(coverage * 100).toFixed(1)}% (همبستگی نهفته ${(corr * 100).toFixed(0)}%)`,
   923→   923→          whyEn: `COPULA JOINT MASS ${(jointMass * 100).toFixed(0)}% > ${(gate * 100).toFixed(0)}% · COMBO + COVER · ${(coverage * 100).toFixed(1)}% COVERAGE`,
   924→   924→          marketEn: `${mainEn} + ${coverEn}`,
   925→   925→          marketFa: `${mainFa} + ${coverFa}`,
   926→   926→          targetWindow: [1.85, 2.4],
   927→   927→          stakePlanFa: `داچینگ ۱۰۰ واحدی: ${mainEn} @ ${oddsMain} + پوشش @ ${oddsCover}`,
   928→   928→          legCodes: { main: `RC-MAIN-${variant}`, cover: `RC-COVER-${variant}` },
   929→   929→          comboLegs: {
   930→   930→            mainEn,
   931→   931→            mainFa,
   932→   932→            mainProb: pMain,
   933→   933→            mainPrice: oddsMain,
   934→   934→            coverEn,
   935→   935→            coverFa,
   936→   936→            coverProb: pCover,
   937→   937→            coverPrice: oddsCover,
   938→   938→          },
   939→   939→        },
   940→   940→      });
   941→   941→    }
   942→   942→  }
   943→   943→
   944→   944→  // ── 3 · 🎯 EXACT SET & GAME CLUSTER ──────────────────────────────────────
   945→   945→  {
   946→   946→    const windowOk = mc.entropy < CLUSTER_MAX_ENTROPY && mc.topScenariosMass >= CLUSTER_MIN_MASS;
   947→   947→    const pairLoss = Math.max(0, 1 - mc.topScenariosMass);
   948→   948→    // v6.0 Bug-3 fix — exact-scenario odds are the REAL 1xBet exotic prices of
   949→   949→    // their own probabilities; the old clampPrice(...,2.8,4.2) lock that froze
   950→   950→    // scenarios 2 & 3 at 4.20x is removed.

   952→   952→    const probs = mc.topScenarios.map((s) => s.prob);

   956→   956→      eligible: windowOk,
   957→   957→      sharpe,
   958→   958→      pairLoss,
   959→   959→      whyNot: `آنتروپی شانون = ${mc.entropy.toFixed(2)} (سقف ۱.۲۵) یا جرم ۳ سناریو = ${(mc.topScenariosMass * 100).toFixed(1)}% زیر ۶۴%`,

   961→   961→    if (windowOk) {
   962→   962→      const stakes = dutchN(odds, 100);
   963→   963→      candidates.push({
   964→   964→        sharpe,
   965→   965→        pairLoss,

   968→   968→          ...REGIME_META.CLUSTER,

   972→   972→          coverage: mc.topScenariosMass,
   973→   973→          why: `آنتروپی شانون ${mc.entropy.toFixed(2)} پایین و ۳ سناریوی گیم‌به‌گیم ${(mc.topScenariosMass * 100).toFixed(1)}% — داچینگ ۱۰۰ واحدی روی ۳ سناریوی دقیق`,

   977→   977→          targetWindow: [2.8, 4.2],
   978→   978→          stakePlanFa: `داچینگ ۱۰۰ واحدی: ${mc.topScenarios.map((s, i) => `${s.label} @ ${odds[i]}`).join(" + ")}`,
   979→   979→          legCodes: { main: "CLU-1" },
   980→   980→          scenarios: mc.topScenarios.map((sc, i) => ({
   981→   981→            label: sc.label,
   982→   982→            descFa: sc.descFa,

   995→   995→

   997→   997→  evals.push({

   999→   999→    eligible: false,

  1003→  1003→  });
  1004→  1004→
  1005→  1005→  // ── selection: the OPTIMAL regime = max non-linear Sharpe ────────────────
  1006→  1006→  if (candidates.length > 0) {
  1007→  1007→    candidates.sort((a, b) => b.sharpe - a.sharpe);
  1008→  1008→    const win = candidates[0];
  1009→  1009→    return { evals, selected: evals.find((e) => e.code === win.decision.code), decision: win.decision };
  1010→  1010→  }
  1011→  1011→  // no regime qualified → vetoed to the shadow ledger
  1012→  1012→  const best = [...evals].filter((e) => e.code !== "ACCUMULATOR").sort((a, b) => a.pairLoss - b.pairLoss)[0];
  1013→  1013→  return {
  1014→  1014→    evals,
  1015→  1015→    decision: {
  1016→  1016→      code: "VETOED",
  1017→  1017→      ...REGIME_META.VETOED,
  1018→  1018→      sharpe: 0,
  1019→  1019→      mcScore: 0,
  1020→  1020→      why: `هیچ رژیمی صلاحیت ندارد — ${best ? best.whyNot : "بدون رژیم"}؛ انتقال به دفتر سایه`,
  1021→  1021→      whyEn: "NO REGIME QUALIFIED · SHADOW LEDGER",
  1022→  1022→    },
  1023→  1023→  };
  1024→  1024→}
  1025→  1025→
  1026→  1026→/**
  1027→  1027→ * Adaptive routing for the 65% coverage quota — ignores the window gates and
  1028→  1028→ * grants the least-risk regime (max coverage) stamped as a SILVER crown.
  1029→  1029→ * v5.5 — adaptive grants carry the SAME full 100-unit stake (no halving).
  1030→  1030→ */
  1031→  1031→export function routeRegimeAdaptive(
  1032→  1032→  m: AnalyzedMatch,
  1033→  1033→  mc: MonteCarloResult
  1034→  1034→): { decision: RegimeDecision; pairLoss: number } {
  1035→  1035→  const ctx = regimeContext(m, mc);
  1036→  1036→  // least-risk = smallest simultaneous-loss space among the three core shapes
