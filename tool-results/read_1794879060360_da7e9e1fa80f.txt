    49→/** Tehran calendar day key (YYYYMMDD) of an ISO timestamp */
    50→function tehranDayKey(iso: string): string {
    51→  const [d, m, y] = TEHRAN_DATE_FMT.format(new Date(iso)).split("/");
    52→  return `${y}${m}${d}`;
    53→}

    55→/** Iranian week key (weeks start SATURDAY in Tehran) + label */
    56→function tehranWeekKey(iso: string): { key: string; label: string } {
    57→  const [dStr, mStr, yStr] = TEHRAN_DATE_FMT.format(new Date(iso)).split("/");
    58→  const d = new Date(Number(yStr), Number(mStr) - 1, Number(dStr));
    59→  // JS dow: 6=Saturday → shift so Saturday = 0 (Sat=0, Sun=1, … Fri=6)
    60→  const offset = (d.getDay() + 1) % 7;
    61→  d.setDate(d.getDate() - offset); // back to the Saturday week-start
    62→  const [sd, sm, sy] = TEHRAN_DATE_FMT.format(d).split("/");
    63→  return { key: `${sy}${sm}${sd}`, label: `هفته ${sd}/${sm}` };
    64→}

    71→function bucketize(
    72→  points: BankPoint[],
    73→  keyOf: (iso: string) => { key: string; label: string }
    74→): BankPeriodBucket[] {
    75→  const order: string[] = [];
    76→  const map = new Map<string, BankPeriodBucket>();
    77→  for (const p of points) {
    78→    if (!p.at || p.index === 0) continue; // skip the START point
    79→    const { key, label } = keyOf(p.at);
    80→    let b = map.get(key);
    81→    if (!b) {
    82→      b = { period: key, label, pnl: 0, wins: 0, losses: 0, matches: 0, winRate: 0, bankEnd: 0, staked: 0, roi: 0 };
    83→      map.set(key, b);
    84→      order.push(key);
    85→    }
    86→    b.pnl += p.delta ?? 0;
    87→    b.matches += 1;
    88→    if (p.status === "WON") b.wins += 1;
    89→    else if (p.status === "LOST") b.losses += 1;
    90→    b.bankEnd = p.bank;
    91→  }
    92→  return order.map((k) => {
    93→    const b = map.get(k)!;
    94→    b.pnl = Math.round(b.pnl * 100) / 100;
    95→    const decided = b.wins + b.losses;
    96→    b.winRate = decided > 0 ? b.wins / decided : 0;
    97→    b.roi = b.staked > 0 ? b.pnl / b.staked : b.pnl > 0 ? 1 : b.pnl < 0 ? -1 : 0;
    98→    return b;
    99→  });
   100→}

   104→function buildBankHistory(bankCurve: BankPoint[], chronological: LedgerMatchRow[]): BankHistory {
   105→  const daily = bucketize(bankCurve, (iso) => {
   106→    const key = tehranDayKey(iso);
   107→    return { key, label: `${key.slice(6, 8)}/${key.slice(4, 6)}` };
   108→  });
   109→  // stake totals per daily bucket from chronological rows
   110→  const dayKeyOfRow = (r: LedgerMatchRow) => tehranDayKey(r.settledAt ?? r.createdAt);
   111→  const dayStakes = new Map<string, number>();
   112→  for (const r of chronological) {
   113→    const k = dayKeyOfRow(r);
   114→    dayStakes.set(k, (dayStakes.get(k) ?? 0) + r.stake);
   115→  }
   116→  for (const b of daily) b.staked = Math.round((dayStakes.get(b.period) ?? 0) * 100) / 100;
   117→
   118→  const weekly = bucketize(bankCurve, (iso) => tehranWeekKey(iso));
   119→  const weekStakes = new Map<string, number>();
   120→  for (const r of chronological) {
   121→    const { key } = tehranWeekKey(r.settledAt ?? r.createdAt);
   122→    weekStakes.set(key, (weekStakes.get(key) ?? 0) + r.stake);
   123→  }
   124→  for (const b of weekly) b.staked = Math.round((weekStakes.get(b.period) ?? 0) * 100) / 100;
   125→
   126→  const monthly = bucketize(bankCurve, (iso) => {
   127→    const key = tehranDayKey(iso);
   128→    const mk = `${key.slice(0, 6)}`;
   129→    return { key: mk, label: `${FA_MONTHS[Number(key.slice(4, 6)) - 1] ?? mk} ${key.slice(0, 4)}` };
   130→  });
   131→  const monthStakes = new Map<string, number>();
   132→  for (const r of chronological) {
   133→    const k = tehranDayKey(r.settledAt ?? r.createdAt).slice(0, 6);
   134→    monthStakes.set(k, (monthStakes.get(k) ?? 0) + r.stake);
   135→  }
   136→  for (const b of monthly) b.staked = Math.round((monthStakes.get(b.period) ?? 0) * 100) / 100;
   137→
   138→  const decidedDaily = daily.filter((b) => b.matches > 0);
   139→  const bestDay = [...decidedDaily].sort((a, b) => b.pnl - a.pnl)[0];
   140→  const worstDay = [...decidedDaily].sort((a, b) => a.pnl - b.pnl)[0];
   141→
   142→  const todayKey = tehranDayKey(new Date().toISOString());
   143→  const todayRows = chronological.filter((r) => dayKeyOfRow(r) === todayKey);
   144→  const todayWins = todayRows.filter((r) => r.status === "WON").length;
   145→  const todayLosses = todayRows.filter((r) => r.status === "LOST").length;
   146→  const todayMatches = todayRows.length;
   147→
   148→  // peak & max drawdown over the full point series
   149→  let peak: { bank: number; at: string } = { bank: BANK_START, at: bankCurve[0]?.at ?? new Date().toISOString() };
   150→  let peakBank = BANK_START;
   151→  let maxDd = { units: 0, pct: 0, from: "", to: "" };
   152→  let ddPeakAt = bankCurve[0]?.at ?? new Date().toISOString();
   153→  for (const p of bankCurve) {
   154→    if (p.bank >= peakBank) {
   155→      peakBank = p.bank;
   156→      peak = { bank: p.bank, at: p.at ?? ddPeakAt };
   157→      ddPeakAt = p.at ?? ddPeakAt;
   158→    } else {
   159→      const ddUnits = Math.round((peakBank - p.bank) * 100) / 100;
   160→      const ddPct = peakBank > 0 ? Math.round((ddUnits / peakBank) * 10000) / 100 : 0;
   161→      if (ddUnits > maxDd.units) {
   162→        maxDd = { units: ddUnits, pct: ddPct, from: ddPeakAt, to: p.at ?? ddPeakAt };
   163→      }
   164→    }
   165→  }
   166→
   167→  return {
   168→    daily,
   169→    weekly,
   170→    monthly,
   171→    bestDay,
   172→    worstDay,
   173→    today: {
   174→      matches: todayMatches,
   175→      wins: todayWins,
   176→      losses: todayLosses,
   177→      pnl: Math.round(todayRows.reduce((s, r) => s + r.profit, 0) * 100) / 100,
   178→      winRate: todayWins + todayLosses > 0 ? todayWins / (todayWins + todayLosses) : 0,
   179→    },
   180→    peak,
   181→    maxDrawdown: maxDd,
   182→  };
   183→}

   316→/**
   317→ * v5.5 — the 100-unit stake law (master prompt §1, mandatory in ledger.ts).
   318→ * Idempotent migration that rewrites every legacy stake in the database:
   319→ *   1) single-leg cards (BANKER 60u / mix 25u / accumulator 25u / shadow 1u)
   320→ *      → exactly STANDARD_MATCH_STAKE (100u);
   321→ *   2) multi-wing positions (DUAL / REGCOMBO / CLUSTER / SNIPER) → every
   322→ *      match group rescaled proportionally so the wings sum EXACTLY to 100u
   323→ *      (silver-era 50u halves and legacy 50/50 splits included);
   324→ *   3) 1u / 2u naked sniper stakes of the bleeding era are gone by the same
   325→ *      rescale — no ledger row can ever carry them again.
   326→ * Wing profits are re-derived from (odds × stake) at read time, so the
   327→ * accounting identity stays intact after the rewrite.
   328→ */
   329→export async function normalizeStakeUnits(): Promise<void> {
   330→  try {
   331→    await db.tennisPick.updateMany({
   332→      where: {
   333→        line: { in: ["BANKER", "COMBO", "ACCUMULATOR", "SHADOW"] },
   334→        stake: { not: STANDARD_MATCH_STAKE },
   335→      },
   336→      data: { stake: STANDARD_MATCH_STAKE },
   337→    });
   338→
   339→    const legs = await db.tennisPick.findMany({
   340→      where: { line: { in: ["DUAL", "REGCOMBO", "CLUSTER", "SNIPER"] } },
   341→      select: { id: true, matchId: true, line: true, stake: true },
   342→    });
   343→    const groups = new Map<string, { id: string; stake: number }[]>();
   344→    for (const l of legs) {
   345→      const key = `${l.line}|${l.matchId}`;
   346→      const arr = groups.get(key) ?? [];
   347→      arr.push({ id: l.id, stake: l.stake });
   348→      groups.set(key, arr);
   349→    }
   350→    for (const group of groups.values()) {
   351→      if (group.length === 0) continue;
   352→      const sum = group.reduce((s, g) => s + g.stake, 0);
   353→      if (sum <= 0) continue;
   354→      const scaled = group.map((g) => ({
   355→        id: g.id,
   356→        stake: Math.round((g.stake / sum) * STANDARD_MATCH_STAKE * 10) / 10,
   357→      }));
   358→      const drift =
   359→        Math.round((STANDARD_MATCH_STAKE - scaled.reduce((s, g) => s + g.stake, 0)) * 10) / 10;
   360→      const biggest = scaled.reduce((a, b) => (b.stake > a.stake ? b : a));
   361→      biggest.stake = Math.round((biggest.stake + drift) * 10) / 10;
   362→      for (const g of scaled) {
   363→        const before = group.find((x) => x.id === g.id)!.stake;
   364→        if (Math.abs(g.stake - before) < 0.05) continue;
   365→        await db.tennisPick.update({ where: { id: g.id }, data: { stake: g.stake } });
   366→      }
   367→    }
   368→  } catch (e) {
   369→    console.error("[ledger] 100-unit stake normalization failed", e);
   370→  }
   371→}

   373→export async function buildLedgerPayload(
   374→  quota?: QuotaTelemetry
   375→): Promise<LedgerPayload> {
   376→  const picks = (await db.tennisPick.findMany({
   377→    include: { match: true },
   378→    orderBy: { createdAt: "asc" },
   379→  })) as unknown as PickWithMatch[];
   380→
   381→  // de-id dupes defensively
   382→  const byId = new Map<string, PickWithMatch>();
   383→  for (const p of picks) if (!byId.has(p.id)) byId.set(p.id, p);
   384→  const all = [...byId.values()];
   385→
   386→  // ── v3.0 · group the DUAL hedge into ONE match row ────────────────────────
   387→  interface DualAgg {
   388→    matchId: string;
   389→    legs: LedgerLeg[];
   390→    match: PickWithMatch["match"];
   391→    meta: PickMeta;
   392→    stake: number;
   393→    anyPending: boolean;
   394→    winner?: LedgerLeg;
   395→    settledAt: string;
   396→    createdAt: string;
   397→  }
   398→  const dual = new Map<string, DualAgg>();
   399→  const cluster = new Map<DualAgg["matchId"], DualAgg>();
   400→  // v5.5 — REGCOMBO (main + cover) and the packaged SNIPER position
   401→  // (main + capital-protection cover) also group into ONE match row so the
   402→  // SHA-256 record keeps exactly one row per match across ALL regimes.
   403→  const regCombo = new Map<DualAgg["matchId"], DualAgg>();
   404→  const sniperAgg = new Map<DualAgg["matchId"], DualAgg>();
   405→  const singles: { p: PickWithMatch; leg: LedgerLeg }[] = [];
   406→
   407→  const aggFor = (map: Map<string, DualAgg>, p: PickWithMatch): DualAgg => {
   408→    let agg = map.get(p.matchId);
   409→    if (!agg) {
   410→      agg = {
   411→        matchId: p.matchId,
   412→        legs: [],
   413→        match: p.match,
   414→        meta: parseMeta(p.meta),
   415→        stake: 0,
   416→        anyPending: false,
   417→        settledAt: (p.settledAt ?? p.createdAt).toISOString(),
   418→        createdAt: p.createdAt.toISOString(),
   419→      };
   420→      map.set(p.matchId, agg);
   421→    }
   422→    return agg;
   423→  };
   424→
   425→  const pushAgg = (map: Map<string, DualAgg>, p: PickWithMatch, leg: LedgerLeg) => {
   426→    const agg = aggFor(map, p);
   427→    agg.legs.push(leg);
   428→    agg.stake += leg.stake;
   429→    if (leg.status === "PENDING") agg.anyPending = true;
   430→    if (leg.status === "WON") agg.winner = leg;
   431→    const t = (p.settledAt ?? p.createdAt).toISOString();
   432→    if (t > agg.settledAt) agg.settledAt = t;
   433→  };
   434→
   435→  for (const p of all) {
   436→    const leg = buildLeg(p);
   437→    if (p.line === "DUAL") {
   438→      const agg = aggFor(dual, p);
   439→      // prefer the meta of wing A (carries the archetype lock)
   440→      if (leg.leg.startsWith("A") && !agg.meta.arch && parseMeta(p.meta).arch) {
   441→        agg.meta = parseMeta(p.meta);
   442→      }
   443→      pushAgg(dual, p, leg);
   444→      dual.set(p.matchId, agg);
   445→    } else if (p.line === "REGCOMBO") {
   446→      pushAgg(regCombo, p, leg);
   447→    } else if (p.line === "SNIPER") {
   448→      pushAgg(sniperAgg, p, leg);
   449→    } else if (p.line === "CLUSTER") {
   450→      pushAgg(cluster, p, leg);
   451→    } else if (p.line === "COMBO" || p.line === "BANKER" || p.line === "ACCUMULATOR") {
   452→      // v4.5 — VOID rows (buggy-era settlements) stay out of the official record
   453→      if (p.status !== "VOID") singles.push({ p, leg });
   454→    }
   455→    // SHADOW rows stay in the disjoint shadow ledger — never in the record
   456→  }
   457→
   458→  // v5.0 — one shared mapper for match-level multi-wing positions (DUAL 2
   459→  // wings, CLUSTER 3 dutching scenarios)
   460→  const aggRowOf = (a: DualAgg, line: ProductionLine): LedgerMatchRow => {
   461→    const decided = !a.anyPending && a.legs.every((l) => l.status !== "PENDING");
   462→    const status: LedgerMatchRow["status"] = !decided
   463→      ? "PENDING"
   464→      : a.winner
   465→        ? "WON"
   466→        : "LOST";
   467→    // doc formula: netMatchProfit = Σ(winningSlipStake × winningSlipOdds) − total stake
   468→    const wonReturn = a.legs
   469→      .filter((l) => l.status === "WON")
   470→      .reduce((s, l) => s + l.odds * l.stake, 0);
   471→    const profit = !decided ? 0 : wonReturn > 0 ? wonReturn - a.stake : -a.stake;
   472→    const archCode = a.meta.arch;
   473→    const fallback = archCode ? ARCH_FALLBACK[archCode] : undefined;
   474→    return {
   475→      id: `${line}:${a.matchId}`,
   476→      matchId: a.matchId,
   477→      line,
   478→      matchLabel: `${a.match.playerA} × ${a.match.playerB}`,
   479→      tournament: a.match.tournament,
   480→      tour: a.match.tour as Tour,
   481→      surface: a.match.surface as Surface,
   482→      round: a.match.round,
   483→      archetype:
   484→        archCode && fallback
   485→          ? {
   486→              code: archCode as "ARCH1" | "ARCH2" | "ARCH3",
   487→              labelFa: a.meta.archLabelFa ?? fallback.labelFa,
   488→              labelEn: a.meta.archLabelEn ?? fallback.labelEn,
   489→              coverage: a.meta.archCoverage ?? 0,
   490→              adaptive: a.meta.archAdaptive,
   491→            }
   492→          : undefined,
   493→      legs: a.legs.sort((x, y) => x.label.localeCompare(y.label)),
   494→      winnerLeg: a.winner?.leg,
   495→      status,
   496→      stake: a.stake,
   497→      profit: Math.round(profit * 100) / 100,
   498→      result: a.match.resultCode ?? undefined,
   499→      setsA: a.match.resultSetsA ?? undefined,
   500→      setsB: a.match.resultSetsB ?? undefined,
   501→      hash: a.match.hashPre ?? undefined,
   502→      espnUrl: espnUrlOf(a.match.eventKey),
   503→      settledAt: decided ? a.settledAt : undefined,
   504→      createdAt: a.createdAt,
   505→    };
   506→  };
   507→
   508→  const dualRows: LedgerMatchRow[] = [...dual.values()].map((a) => aggRowOf(a, "DUAL"));
   509→  const clusterRows: LedgerMatchRow[] = [...cluster.values()].map((a) => aggRowOf(a, "CLUSTER"));
   510→  const regComboRows: LedgerMatchRow[] = [...regCombo.values()].map((a) => aggRowOf(a, "REGCOMBO"));
   511→  const sniperRows: LedgerMatchRow[] = [...sniperAgg.values()].map((a) => aggRowOf(a, "SNIPER"));
   512→
   513→  const singleRows: LedgerMatchRow[] = singles.map(({ p, leg }) => {
   514→    const decided = leg.status === "WON" || leg.status === "LOST";
   515→    return {
   516→      id: `${p.line}:${p.id}`,
   517→      matchId: p.matchId,
   518→      line: p.line as ProductionLine,
   519→      matchLabel: `${p.match.playerA} × ${p.match.playerB}`,
   520→      tournament: p.match.tournament,
   521→      tour: p.match.tour as Tour,
   522→      surface: p.match.surface as Surface,
   523→      round: p.match.round,
   524→      legs: [leg],
   525→      winnerLeg: leg.status === "WON" ? leg.leg : undefined,
   526→      status: decided ? (leg.status as "WON" | "LOST") : "PENDING",
   527→      stake: leg.stake,
   528→      profit: Math.round(leg.profit * 100) / 100,
   529→      result: p.match.resultCode ?? undefined,
   530→      setsA: p.match.resultSetsA ?? undefined,
   531→      setsB: p.match.resultSetsB ?? undefined,
   532→      hash: p.match.hashPre ?? undefined,
   533→      espnUrl: espnUrlOf(p.match.eventKey),
   534→      settledAt: (p.settledAt ?? undefined)?.toISOString(),
   535→      createdAt: p.createdAt.toISOString(),
   536→    };
   537→  });
   538→
   539→  // latest first, bounded
   540→  const rows = [...dualRows, ...clusterRows, ...regComboRows, ...sniperRows, ...singleRows]
   541→    .sort(
   542→      (a, b) =>
   543→        (b.settledAt ?? b.createdAt).localeCompare(a.settledAt ?? a.createdAt)
   544→    )
   545→    .slice(0, 80);
   546→
   547→  // ── disjoint line stats (DUAL at MATCH level, others per pick) ────────────
   548→  const lineStatsOf = (line: ProductionLine, rs: LedgerMatchRow[]): LineStats => {
   549→    const won = rs.filter((r) => r.status === "WON").length;
   550→    const lost = rs.filter((r) => r.status === "LOST").length;
   551→    const pending = rs.filter((r) => r.status === "PENDING").length;
   552→    const profit = rs.reduce((s, r) => s + r.profit, 0);
   553→    const stake = rs.reduce((s, r) => s + r.stake, 0);
   554→    const decided = won + lost;
   555→    return {
   556→      line,
   557→      picks: rs.length,
   558→      won,
   559→      lost,
   560→      pending,
   561→      winRate: decided > 0 ? won / decided : 0,
   562→      profit: Math.round(profit * 100) / 100,
   563→      roi: stake > 0 ? profit / stake : 0,
   564→    };
   565→  };
   566→
   567→  const groups: LineStats[] = [
   568→    lineStatsOf("DUAL", dualRows),
   569→    lineStatsOf("BANKER", singleRows.filter((r) => r.line === "BANKER")),
   570→    lineStatsOf("REGCOMBO", regComboRows),
   571→    lineStatsOf("CLUSTER", clusterRows),
   572→    lineStatsOf("ACCUMULATOR", singleRows.filter((r) => r.line === "ACCUMULATOR")),
   573→    lineStatsOf("COMBO", singleRows.filter((r) => r.line === "COMBO")),
   574→    lineStatsOf("SNIPER", sniperRows),
   575→    lineStatsOf(
   576→      "SHADOW",
   577→      all
   578→        .filter((p) => p.line === "SHADOW")
   579→        .map((p) => ({
   580→          id: `SHADOW:${p.id}`,
   581→          matchId: p.matchId,
   582→          line: "SHADOW" as ProductionLine,
   583→          matchLabel: `${p.match.playerA} × ${p.match.playerB}`,
   584→          tournament: p.match.tournament,
   585→          tour: p.match.tour as Tour,
   586→          surface: p.match.surface as Surface,
   587→          legs: [buildLeg(p)],
   588→          status: (p.status === "WON" ? "WON" : p.status === "LOST" ? "LOST" : "PENDING") as
   589→            | "PENDING"
   590→            | "WON"
   591→            | "LOST",
   592→          stake: p.stake,
   593→          profit: Math.round(wingProfitOf(p.status, p.odds, p.stake) * 100) / 100,
   594→          createdAt: p.createdAt.toISOString(),
   595→        }))
   596→    ),
   597→  ];
   598→
   599→  // ── combined stats (official record only — shadow excluded, match level) ──
   600→  const official = [...dualRows, ...clusterRows, ...regComboRows, ...sniperRows, ...singleRows];
   601→  const decidedRows = official.filter((r) => r.status !== "PENDING");
   602→  const combinedWonCount = decidedRows.filter((r) => r.status === "WON").length;
   603→  const combinedProfit = decidedRows.reduce((s, r) => s + r.profit, 0);
   604→  const combinedStake = decidedRows.reduce((s, r) => s + r.stake, 0);
   605→
   606→  // v4.5 — Core System Win Rate: the headline gauge shows the CORE lines only
   607→  // (complementary golden pairs + safe service-king mixes). The high-odds
   608→  // sniper line keeps its own disjoint stats and is never merged into the
   609→  // core number — its 1-unit misses used to drag 89.4% down to 60.6%.
   610→  const coreRows = [
   611→    ...dualRows,
   612→    ...singleRows.filter(
   613→      (r) => r.line === "COMBO" || r.line === "ACCUMULATOR" || r.line === "BANKER"
   614→    ),
   615→    ...regComboRows,
   616→    ...clusterRows,
   617→  ];
   618→  const coreDecided = coreRows.filter((r) => r.status !== "PENDING");
   619→  const coreWon = coreDecided.filter((r) => r.status === "WON").length;
   620→  // v4.5 — stake-weighted alternative: Σ won stake / Σ decided stake over the
   621→  // whole official record
   622→  const wonStake = decidedRows
   623→    .filter((r) => r.status === "WON")
   624→    .reduce((s, r) => s + r.stake, 0);
   625→
   626→  // bank curve over decided rows (chronological, match-level P&L for DUAL)
   627→  // v6.0 — every point now carries its settle timestamp, outcome and delta so
   628→  // the dashboard can bucket the curve into daily/weekly/monthly analytics.
   629→  const chronological = [...decidedRows].sort(
   630→    (a, b) => (a.settledAt ?? a.createdAt).localeCompare(b.settledAt ?? b.createdAt)
   631→  );
   632→  const bankCurve: BankPoint[] = [{ index: 0, bank: BANK_START, label: "START" }];
   633→  let bank = BANK_START;
   634→  chronological.forEach((r, i) => {
   635→    bank += r.profit;
   636→    bankCurve.push({
   637→      index: i + 1,
   638→      bank: Math.round(bank * 100) / 100,
   639→      label: `${r.matchLabel} · ${r.status}`,
   640→      at: r.settledAt ?? r.createdAt,
   641→      status: r.status === "WON" || r.status === "LOST" ? r.status : undefined,
   642→      delta: Math.round(r.profit * 100) / 100,
   643→    });
   644→  });
   645→
   646→  // ── v6.0 · Professional bank-growth analytics (Tehran-anchored buckets) ──
   647→  const bankHistory = buildBankHistory(bankCurve, chronological);
   648→
   649→  // current streak — one result event per match (DUAL collapsed)
   650→  const resultEvents = chronological.map((r) => ({
   651→    status: r.status as "WON" | "LOST",
   652→    at: r.settledAt ?? r.createdAt,
   653→  }));
   654→  let streak = 0;
   655→  for (let i = resultEvents.length - 1; i >= 0; i--) {
   656→    const s = resultEvents[i].status;
   657→    if (i === resultEvents.length - 1) {
   658→      streak = s === "WON" ? 1 : -1;
   659→    } else {
   660→      if (s === "WON" && streak > 0) streak++;
   661→      else if (s === "LOST" && streak < 0) streak--;
   662→      else break;
   663→    }
   664→  }
   665→
   666→  const bestWin = [...official]
   667→    .filter((r) => r.status === "WON")
   668→    .sort((a, b) => b.profit - a.profit)[0];
   669→
   670→  // ── v5.0 · Live Bankroll Audit Inspector ──────────────────────────────────
   671→  // the algebraic accounting box rendered above the SHA-256 record table:
   672→  //   bankStart + Σ dual wins − Σ dual losses + combo + cluster + banker +
   673→  //   regCombo + sniper + accumulator = current bank (every user can sum it
   674→  //   in 3s)
   675→  // v5.6 FIX (master prompt 1877.1u): cluster/sniper profits were computed
   676→  // from `singleRows` which NEVER contains CLUSTER/SNIPER rows (they live in
   677→  // their own aggregated arrays) — the audit printed 0.00 for both and hid a
   678→  // −876.72 residual in `otherProfit`. Every term now comes from its own row
   679→  // set, REGCOMBO is an explicit term, and the identity is exact by
   680→  // construction (otherProfit ≡ 0: right side == real sum of left side).
   681→  const lineProfit = (line: ProductionLine, rs: LedgerMatchRow[]) =>
   682→    Math.round(
   683→      rs.filter((r) => r.line === line).reduce((s, r) => s + r.profit, 0) * 100
   684→    ) / 100;
   685→  const dualWonProfit =
   686→    Math.round(dualRows.filter((r) => r.status === "WON").reduce((s, r) => s + r.profit, 0) * 100) / 100;
   687→  const dualLostStake =
   688→    Math.round(dualRows.filter((r) => r.status === "LOST").reduce((s, r) => s + r.profit, 0) * 100) / 100;
   689→  const comboProfit = lineProfit("COMBO", singleRows);
   690→  const clusterProfit = lineProfit("CLUSTER", clusterRows);
   691→  const bankerProfit = lineProfit("BANKER", singleRows);
   692→  const regComboProfit = lineProfit("REGCOMBO", regComboRows);
   693→  const sniperProfit = lineProfit("SNIPER", sniperRows);
   694→  const accumulatorProfit = lineProfit("ACCUMULATOR", singleRows);
   695→  const bankAudit: BankAudit = {
   696→    bankStart: BANK_START,
   697→    dualWonProfit,
   698→    dualLostStake,
   699→    comboProfit,
   700→    clusterProfit,
   701→    bankerProfit,
   702→    regComboProfit,
   703→    sniperProfit,
   704→    accumulatorProfit,
   705→    // exact-by-construction identity: bankStart + every explicit term = bank
   706→    otherProfit:
   707→      Math.round(
   708→        (bank -
   709→          (BANK_START +
   710→            dualWonProfit +
   711→            dualLostStake +
   712→            comboProfit +
   713→            clusterProfit +
   714→            bankerProfit +
   715→            regComboProfit +
   716→            sniperProfit +
   717→            accumulatorProfit)) *
   718→          100
   719→      ) / 100,
   720→    bank: Math.round(bank * 100) / 100,
   721→    wonCount: combinedWonCount,
   722→    lostCount: decidedRows.length - combinedWonCount,
   723→    auditHash: "",
   724→  };
   725→  bankAudit.auditHash = sha256(
   726→    `${bankAudit.bankStart}|${bankAudit.dualWonProfit}|${bankAudit.dualLostStake}|${bankAudit.comboProfit}|${bankAudit.clusterProfit}|${bankAudit.bankerProfit}|${bankAudit.regComboProfit}|${bankAudit.sniperProfit}|${bankAudit.accumulatorProfit}|${bankAudit.otherProfit}|=${bankAudit.bank}`
   727→  );
   728→
   729→  return {
   730→    stats: {
   731→      combinedWinRate: decidedRows.length > 0 ? combinedWonCount / decidedRows.length : 0,
   732→      // v5.0 — core system win rate (DUAL + BANKER + COMBO + CLUSTER; sniper excluded)
   733→      coreWinRate: coreDecided.length > 0 ? coreWon / coreDecided.length : 0,
   734→      // v4.5 — stake-weighted win rate over the official record
   735→      stakeWeightedWinRate: combinedStake > 0 ? wonStake / combinedStake : 0,
   736→      bank: Math.round(bank * 100) / 100,
   737→      bankStart: BANK_START,
   738→      profit: Math.round(combinedProfit * 100) / 100,
   739→      roi: combinedStake > 0 ? combinedProfit / combinedStake : 0,
   740→      picks: official.length,
   741→      settled: decidedRows.length,
   742→      streak,
   743→      bestWin,
   744→      // ─── v5.0 · Live Bankroll Audit Inspector — exact algebraic identity ────
   745→      // bank = bankStart + Σ(profit of WON positions) + Σ(profit of LOST positions)
   746→      winsGross:

   748→          decidedRows.filter((r) => r.status === "WON").reduce((s, r) => s + r.profit, 0) * 100
   749→        ) / 100,
   750→      lossesGross:
   751→        Math.round(
   752→          decidedRows.filter((r) => r.status === "LOST").reduce((s, r) => s + r.profit, 0) * 100
   753→        ) / 100,
   754→      winsCount: combinedWonCount,
   755→      lossesCount: decidedRows.length - combinedWonCount,
   756→    },
   757→    groups,
   758→    rows,
   759→    quota,
   760→    bankCurve,
   761→    bankHistory,
   762→    bankAudit,
   763→  };
