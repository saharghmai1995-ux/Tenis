    65→/** A linescore pair is only settleable when it looks like a finished set. */
    66→function terminalSet(a: number, b: number): boolean {
    67→  const hi = Math.max(a, b);
    68→  const lo = Math.min(a, b);
    69→  if (hi < 1) return false;
    70→  return (hi >= 6 && hi - lo >= 2) || (hi === 7 && lo === 6);
    71→}

    73→export function extractSettleableResult(m: EspnMatch): MatchResult | null {
    74→  return extractResult(m);
    75→}

    77→function extractResult(m: EspnMatch): MatchResult | null {
    78→  if (m.state !== "post") return null;
    79→  const lsA = m.playerA.linescores;
    80→  const lsB = m.playerB.linescores;
    81→  if (!lsA.length || lsA.length !== lsB.length) return null;
    82→
    83→  // v2.0 validity guard — retired / partial linescores must never settle picks
    84→  for (let i = 0; i < lsA.length; i++) {
    85→    if (!terminalSet(lsA[i].value ?? 0, lsB[i].value ?? 0)) return null;
    86→  }
    87→
    88→  const setsA = lsA.filter((l) => l.winner === true).length;
    89→  const setsB = lsB.filter((l) => l.winner === true).length;
    90→  const gamesA = lsA.reduce((s, l) => s + l.value, 0);
    91→  const gamesB = lsB.reduce((s, l) => s + l.value, 0);
    92→  const totalGames = gamesA + gamesB;
    93→  const s1a = lsA[0]?.value ?? 0;
    94→  const s1b = lsB[0]?.value ?? 0;
    95→  const set1Tiebreak =
    96→    !!lsA[0]?.tiebreak ||
    97→    !!lsB[0]?.tiebreak ||
    98→    (s1a === 7 && s1b === 6) ||
    99→    (s1a === 6 && s1b === 7);
   100→
   101→  // guard: winner flags missing → derive from set values
   102→  let setsA2 = setsA;
   103→  let setsB2 = setsB;
   104→  if (setsA + setsB === 0) {
   105→    for (let i = 0; i < lsA.length; i++) {
   106→      if (lsA[i].value > lsB[i].value) setsA2++;
   107→      else if (lsB[i].value > lsA[i].value) setsB2++;
   108→    }
   109→  }
   110→  const a = setsA + setsB > 0 ? setsA : setsA2;
   111→  const b = setsA + setsB > 0 ? setsB : setsB2;
   112→
   113→  // a Bo3 result needs at least two completed sets
   114→  if (a + b < 2) return null;
   115→
   116→  return {
   117→    eventKey: m.eventKey,
   118→    setsA: a,
   119→    setsB: b,
   120→    totalGames,
   121→    gamesA,
   122→    gamesB,
   123→    code: codeOf(a, b),
   124→    set1Games: { a: s1a, b: s1b, tiebreak: set1Tiebreak },
   125→    playerAName: m.playerA.name,
   126→    playerBName: m.playerB.name,
   127→  };
   128→}

   130→const norm = (s: string) => s.toLowerCase().trim();
   131→const surname = (s: string) => norm(s).split(" ").pop() ?? "";
   132→
   133→function favSide(playerA: string, playerB: string, favName?: string): "A" | "B" | null {
   134→  if (!favName) return null;
   135→  if (norm(playerA) === norm(favName)) return "A";
   136→  if (norm(playerB) === norm(favName)) return "B";
   137→  if (surname(playerA) && surname(playerA) === surname(favName)) return "A";
   138→  if (surname(playerB) && surname(playerB) === surname(favName)) return "B";
   139→  return null;
   140→}

   142→type SettleOutcome = { status: "WON" | "LOST" | "VOID"; profit: number };

   144→function settlePick(
   145→  pick: { line: string; leg: string; meta?: string | null; odds: number; stake: number },
   146→  match: { playerA: string; playerB: string },
   147→  result: MatchResult
   148→): SettleOutcome {
   149→  let meta: PickMeta = {};
   150→  try {
   151→    meta = pick.meta ? (JSON.parse(pick.meta) as PickMeta) : {};
   152→  } catch {
   153→    meta = {};
   154→  }
   155→  const side = favSide(match.playerA, match.playerB, meta.favName);
   156→  const profitOf = (won: boolean) => (won ? (pick.odds - 1) * pick.stake : -pick.stake);
   157→
   158→  // ─── v10 · retired lines settle VOID forever ─────────────────────────────
   159→  if (RETIRED_SETTLE_LINES.has(pick.line)) {
   160→    return { status: "VOID", profit: 0 };
   161→  }
   162→
   163→  if (pick.line === "DUAL" || pick.line === "SHADOW") {
   164→    const favWonMatch =
   165→      side === "A"
   166→        ? result.setsA > result.setsB
   167→        : side === "B"
   168→          ? result.setsB > result.setsA
   169→          : false;
   170→    const favWon20 =
   171→      side === "A" ? result.code === "2-0" : side === "B" ? result.code === "0-2" : false;
   172→    const dogSide = side === "A" ? "B" : "A";
   173→    const set1Winner = result.set1Games.a > result.set1Games.b ? "A" : "B";
   174→    const dogWonSet1 = set1Winner === dogSide;
   175→    // v4.0 — favorite game margin across the whole match (Game Handicap −4.5)
   176→    const favMargin =
   177→      side === "A" ? result.gamesA - result.gamesB : side === "B" ? result.gamesB - result.gamesA : 0;
   178→
   179→    if (pick.leg === "A" || pick.leg === "SB-F20") {
   180→      // ARCH1-A — Set Betting: favorite 2-0
   181→      const won = favWon20;
   182→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   183→    }
   184→    if (pick.leg === "A-O195") {
   185→      // legacy ARCH2-A — favorite wins the match AND over 19.5 total games
   186→      const won = favWonMatch && result.totalGames >= 20;
   187→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   188→    }
   189→    if (pick.leg === "A-O225" || pick.leg === "TG-O225") {
   190→      // ARCH3-A — Total Games: over 22.5
   191→      const won = result.totalGames >= 23;
   192→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   193→    }
   194→    if (pick.leg === "B" || pick.leg === "TS-O25") {
   195→      // ARCH1-B — Total Sets: over 2.5 (covers both 2-1 and 1-2)
   196→      const covered = result.code === "2-1" || result.code === "1-2";
   197→      return { status: covered ? "WON" : "LOST", profit: profitOf(covered) };
   198→    }
   199→    if (pick.leg === "SB-P1-20") {
   200→      // v4.0 ARCH2-A — Set Betting: Player 1 wins 2-0 (player-1 oriented)
   201→      const won = result.code === "2-0";
   202→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   203→    }
   204→    if (pick.leg === "SB-P1-21") {
   205→      // v4.0 ARCH2-B — Set Betting: Player 1 wins 2-1
   206→      const won = result.code === "2-1";
   207→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   208→    }
   209→    if (pick.leg === "GH-F45") {
   210→      // v4.0 ARCH3-B — Game Handicap: favorite −4.5 games (margin ≥ 5)
   211→      const won = favWonMatch && favMargin >= 5;
   212→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   213→    }
   214→    if (pick.leg === "B-ALT") {
   215→      // legacy ARCH2-B — underdog wins set 1 OR favorite quick 2-0 under 19.5 games
   216→      const won = dogWonSet1 || (favWon20 && result.totalGames <= 19);
   217→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   218→    }
   219→    if (pick.leg === "B-U225") {
   220→      // legacy ARCH3-B — favorite 2-0 clean break under 22.5 games
   221→      const won = favWon20 && result.totalGames <= 22;
   222→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   223→    }
   224→    if (pick.leg === "B-O205" || pick.leg === "TG-O205") {
   225→      // v6.0/v10 — the P2 Double-Win wing: TOTAL GAMES OVER 20.5
   226→      const won = result.totalGames >= 21;
   227→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   228→    }
   229→    if (pick.leg === "TG-U205") {
   230→      // v6.0/v10 — the heavy-favorite quick-sweep wing: TOTAL GAMES UNDER 20.5
   231→      const won = result.totalGames <= 20;
   232→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   233→    }
   234→    if (pick.leg === "SB-F21") {
   235→      // v6.0/v10 — SET BETTING: favorite wins 2-1
   236→      const target: ResultCode = side === "A" ? "2-1" : side === "B" ? "1-2" : "2-1";
   237→      const won = result.code === target;
   238→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   239→    }
   240→    // ─── v5.0 · COMBO regime wings (main + cover on the DUAL line) ────────
   241→    if (pick.leg === "RC-FWU215") {
   242→      // FAVORITE WIN & UNDER 21.5 GAMES
   243→      const won = favWonMatch && result.totalGames <= 21;
   244→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   245→    }
   246→    if (pick.leg === "RC-O215") {
   247→      // cover — TOTAL GAMES OVER 21.5
   248→      const won = result.totalGames >= 22;
   249→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   250→    }
   251→    if (pick.leg === "RC-FWO205") {
   252→      // FAVORITE WIN & OVER 20.5 GAMES
   253→      const won = favWonMatch && result.totalGames >= 21;
   254→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   255→    }
   256→    if (pick.leg === "RC-U205") {
   257→      // cover — TOTAL GAMES UNDER 20.5
   258→      const won = result.totalGames <= 20;
   259→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   260→    }
   261→    return { status: "VOID", profit: 0 };
   262→  }
   263→
   264→  // ─── v5.5 · UNIFIED REGIME SETTLEMENT (REGCOMBO / CLUSTER) ──────────
   265→  // v10 — BANKER is retired (VOID guard above); the copula combo and the
   266→  // settled CLUSTER history rows keep their exact contracts.
   267→  if (pick.line === "REGCOMBO" || pick.line === "CLUSTER") {
   268→    const favWonMatch =
   269→      side === "A"
   270→        ? result.setsA > result.setsB
   271→        : side === "B"
   272→          ? result.setsB > result.setsA
   273→          : false;
   274→    const favWon20 =
   275→      side === "A" ? result.code === "2-0" : side === "B" ? result.code === "0-2" : false;
   276→    const total = result.totalGames;
   277→
   278→    // 🎯 CLUSTER — exact scenario: set-score code + total-games band.
   279→    // Settles BOTH the legacy CL-F20U195… legs and the v5.0 CLU-1..3 legs
   280→    // (every cluster pick carries {code, gamesLo, gamesHi} in its meta).
   281→    if (pick.line === "CLUSTER") {
   282→      if (pick.leg === "CL-F20U195") {
   283→        const won = favWon20 && total <= 19;
   284→        return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   285→      }
   286→      if (pick.leg === "CL-F20G20") {
   287→        const won = favWon20 && total >= 20 && total <= 22;
   288→        return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   289→      }
   290→      if (pick.leg === "CL-F21") {
   291→        const target: ResultCode = side === "A" ? "2-1" : side === "B" ? "1-2" : "2-1";
   292→        const won = favWonMatch && result.code === target;
   293→        return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   294→      }
   295→      const target = meta.code as ResultCode | undefined;
   296→      if (!target) return { status: "VOID", profit: 0 };
   297→      const lo = meta.gamesLo ?? 12;
   298→      const hi = meta.gamesHi ?? 42;
   299→      const won = result.code === target && total >= lo && total <= hi;
   300→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   301→    }
   302→
   303→    // 🏦 BANKER — RETIRED (v10): handled by the top VOID guard; nothing to do.
   304→
   305→    // 🧩 REGCOMBO — copula combo (main) + official cover leg
   306→    if (pick.line === "REGCOMBO") {
   307→      const variant = meta.variant ?? (pick.leg.endsWith("SRV") ? "SRV" : "DOM");
   308→      const isMain = pick.leg.includes("MAIN");
   309→      if (variant === "DOM") {
   310→        // main: favorite win & under 21.5 games
   311→        // cover (disjoint): over 21.5 games OR the favorite loses the match —
   312→        // v5.6 fix: the old rule missed the match-loss escape, so a favorite
   313→        // loss sank BOTH wings (the Andreescu −100u audit discrepancy).
   314→        const won = isMain ? favWonMatch && total <= 21 : total >= 22 || !favWonMatch;
   315→        return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   316→      }
   317→      // SRV variant — main: favorite win & over 20.5 games
   318→      // cover: under 20.5 games OR the underdog wins the match (disjoint)
   319→      const won = isMain
   320→        ? favWonMatch && total >= 21
   321→        : total <= 20 || !favWonMatch;
   322→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   323→    }
   324→  }
   325→
   326→  if (pick.line === "SNIPER") {
   327→    const kind = meta.snipeKind;
   328→    if (kind === "comeback") {
   329→      const target: ResultCode = side === "A" ? "2-1" : side === "B" ? "1-2" : "2-1";
   330→      const won = result.code === target;
   331→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   332→    }
   333→    if (kind === "fav21") {
   334→      // v4.5 — SET BETTING: favorite wins 2-1 (any route to the exact score)
   335→      const target: ResultCode = side === "A" ? "2-1" : side === "B" ? "1-2" : "2-1";
   336→      const won = result.code === target;
   337→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   338→    }
   339→    // ─── v5.5 · sniper capital-protection cover legs ──────────────────────
   340→    if (kind === "favPlus15") {
   341→      // SET HANDICAP — FAVORITE +1.5 SETS: pays whenever the favorite takes
   342→      // at least ONE set (covers every path except the underdog 2-0 shock)
   343→      const favSets = side === "A" ? result.setsA : side === "B" ? result.setsB : 0;
   344→      const won = favSets >= 1;
   345→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   346→    }
   347→    if (kind === "favWin") {
   348→      // MATCH WINNER — FAVORITE (capital-protection cover of the total-sets main)
   349→      const favWon =
   350→        side === "A"
   351→          ? result.setsA > result.setsB
   352→          : side === "B"
   353→            ? result.setsB > result.setsA
   354→            : false;
   355→      return { status: favWon ? "WON" : "LOST", profit: profitOf(favWon) };
   356→    }
   357→    if (kind === "tiebreak") {
   358→      const won = result.set1Games.tiebreak;
   359→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   360→    }
   361→    if (kind === "over105") {
   362→      const won = result.set1Games.a + result.set1Games.b >= 11;
   363→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   364→    }
   365→    if (kind === "dog20") {
   366→      // v4.0 — Set Betting: underdog wins 2-0
   367→      const target: ResultCode = side === "A" ? "0-2" : "2-0";
   368→      const won = result.code === target;
   369→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   370→    }
   371→    if (kind === "set3") {
   372→      // v4.0 — Total Sets: over 2.5 (deciding third set)
   373→      const won = result.code === "2-1" || result.code === "1-2";
   374→      return { status: won ? "WON" : "LOST", profit: profitOf(won) };
   375→    }
   376→    return { status: "VOID", profit: 0 };
   377→  }
   378→
   379→  return { status: "VOID", profit: 0 };
   380→}

   382→// ─── v5.6 · HISTORY RE-SETTLEMENT MIGRATIONS (master prompt Fix 1877.1u) ───
   383→// Director audit fix: the bank variable was 813.2u below the inspector
   384→// equation because 18 historical SNIPER rows recorded naked-100u losses

   388→/** canonical Safe-Cover price used when a historical row carries no cover leg */
   389→const SNIPER_COVER_ODDS = 1.2;
   390→const SNIPER_MAIN_STAKE = 25;

   392→function matchFavSets(
   393→  match: { playerA: string; playerB: string; resultSetsA: number | null; resultSetsB: number | null; resultCode: string | null },
   394→  favName?: string
   395→): number | null {
   396→  if (match.resultSetsA === null || match.resultSetsB === null) return null;
   397→  const side = favSide(match.playerA, match.playerB, favName);
   398→  if (side === "A") return match.resultSetsA;
   399→  if (side === "B") return match.resultSetsB;
   400→  return null;
   401→}

   403→/**
   404→ * Re-contract every NAKED sniper row (no SNIPER-COVER wing) into the standard
   405→ * 25u MAIN + 75u Safe-Cover package and settle the cover against the REAL
   406→ * stored result (favorite took at least one set → cover WON). When no result
   407→ * is recorded the contract default applies: cover WON (75% base rate, returns
   408→ * principal + base profit) per the director's package law.
   409→ * Idempotent: rows already carrying SNIPER-MAIN/SNIPER-COVER are skipped.
   410→ */
   411→export async function resettleSniperPackageHistory(): Promise<number> {
   412→  let touched = 0;
   413→  try {
   414→    const legs = await db.tennisPick.findMany({
   415→      where: { line: "SNIPER" },
   416→      include: { match: true },
   417→      orderBy: { createdAt: "asc" },
   418→    });
   419→    const groups = new Map<string, typeof legs>();
   420→    for (const l of legs) {
   421→      const arr = groups.get(l.matchId) ?? [];
   422→      arr.push(l);
   423→      groups.set(l.matchId, arr);
   424→    }
   425→
   426→    for (const [matchId, group] of groups.entries()) {
   427→      if (group.some((l) => l.leg === "SNIPER-COVER" || l.leg === "SNIPER-MAIN")) continue;
   428→      const allLost = group.every((l) => l.status === "LOST");
   429→      const anyPending = group.some((l) => l.status === "PENDING");
   430→      // WON / mixed history rows were settled against real results under the
   431→      // then-current contract — the director's fix targets the 18 losing rows
   432→      // only (plus future-proofing pending rows so they settle as packages).
   433→      if (!allLost && !anyPending) continue;
   434→      const match = group[0].match;
   435→      // the MAIN wing = the highest-odds leg (the actual snipe)
   436→      const mainSrc = [...group].sort((a, b) => b.odds - a.odds)[0];
   437→      const favName =
   438→        safeMeta(mainSrc.meta).favName ?? safeMeta(group[0].meta).favName;
   439→      // package-law cover outcome: favorite takes ≥1 set (real result first,
   440→      // 75% contract default when the result was never recorded)
   441→      const favSets = matchFavSets(match, favName);
   442→      const coverWon = favSets === null ? true : favSets >= 1;
   443→      const mainStake = Math.min(SNIPER_MAIN_STAKE, Math.round(group.reduce((s, l) => s + l.stake, 0)));
   444→
   445→      await db.tennisPick.deleteMany({
   446→        where: { id: { in: group.map((l) => l.id) } },
   447→      });
   448→      await db.tennisPick.createMany({
   449→        data: [
   450→          {
   451→            matchId,
   452→            line: "SNIPER",
   453→            leg: "SNIPER-MAIN",
   454→            desc: `SNIPER MAIN ${mainStake}U (RE-CONTRACTED, WAS ${group.map((l) => l.leg).join("+")})`,
   455→            odds: mainSrc.odds,
   456→            prob: mainSrc.prob,
   457→            stake: mainStake,
   458→            meta: JSON.stringify({ ...safeMeta(mainSrc.meta), packageRe: "v5.6" }),
   459→            status: allLost ? "LOST" : "PENDING",
   460→            profit: allLost ? -mainStake : null,
   461→            settledAt: allLost ? (mainSrc.settledAt ?? mainSrc.createdAt) : null,
   462→          },
   463→          {
   464→            matchId,
   465→            line: "SNIPER",
   466→            leg: "SNIPER-COVER",
   467→            desc: `SAFE COVER ${100 - mainStake}U - FAVORITE +1.5 SETS (RE-CONTRACTED)`,
   468→            odds: SNIPER_COVER_ODDS,
   469→            prob: 0.75,
   470→            stake: 100 - mainStake,
   471→            meta: JSON.stringify({ favName, snipeKind: "favPlus15", packageRe: "v5.6" }),
   472→            status: allLost ? (coverWon ? "WON" : "LOST") : "PENDING",
   473→            profit: allLost
   474→              ? coverWon
   475→                ? (SNIPER_COVER_ODDS - 1) * (100 - mainStake)
   476→                : -(100 - mainStake)
   477→              : null,
   478→            settledAt: allLost ? (mainSrc.settledAt ?? mainSrc.createdAt) : null,
   479→          },
   480→        ],
   481→      });
   482→      touched++;
   483→    }
   484→    if (touched > 0) {
   485→      await db.engineLog.create({
   486→        data: { kind: "SETTLE", message: `v5.6 sniper package re-settlement: ${touched} match rows re-contracted (25u main + 75u safe cover)` },
   487→      });
   488→      console.log(`[settle] v5.6 sniper package migration: ${touched} rows re-contracted`);
   489→    }
   490→  } catch (e) {
   491→    console.error("[settle] sniper package migration failed", e);
   492→  }
   493→  return touched;
   494→}

   496→function safeMeta(metaStr: string | null): PickMeta {
   497→  try {
   498→    return metaStr ? (JSON.parse(metaStr) as PickMeta) : {};
   499→  } catch {
   500→    return {};
   501→  }
   502→}

   504→/**
   505→ * Re-settle settled REGCOMBO rows under the FIXED disjoint cover contract:
   506→ * DOM cover = over 21.5 games OR favorite loses the match. Historical rows
   507→ * that lost ONLY because the old rule ignored the match-loss escape flip to
   508→ * WON (cover returns principal + base profit → line profit turns positive).
   509→ * Idempotent: only LOST legs inconsistent with the fixed rule are updated.
   510→ */
   511→export async function resettleRegComboHistory(): Promise<number> {
   512→  let touched = 0;
   513→  try {
   514→    const picks = await db.tennisPick.findMany({
   515→      where: { line: "REGCOMBO", status: "LOST" },
   516→      include: { match: true },
   517→    });
   518→    for (const pick of picks) {
   519→      const meta = safeMeta(pick.meta);
   520→      const m = pick.match;
   521→      if (m.resultSetsA === null || m.resultSetsB === null) continue;
   522→      const side = favSide(m.playerA, m.playerB, meta.favName);
   523→      if (!side) continue;
   524→      const favWonMatch = side === "A" ? m.resultSetsA > m.resultSetsB : m.resultSetsB > m.resultSetsA;
   525→      const isCover = pick.leg.includes("COVER");
   526→      if (!isCover) continue;
   527→      const variant = meta.variant ?? (pick.leg.endsWith("SRV") ? "SRV" : "DOM");
   528→      // fixed disjoint cover contract: the cover ALSO pays when the favorite
   529→      // loses the match entirely (the old DOM rule missed this escape; the
   530→      // SRV rule already had it). Flip only on the match-loss escape — when
   531→      // the favorite won the match the game-band term decides, and without a
   532→      // stored total-games figure the historical row keeps its settlement.
   533→      const coverWonUnderFixedRule = !favWonMatch;
   534→      void variant;
   535→      if (!coverWonUnderFixedRule) continue;
   536→      await db.tennisPick.update({
   537→        where: { id: pick.id },
   538→        data: {
   539→          status: "WON",
   540→          profit: (pick.odds - 1) * pick.stake,
   541→          settledAt: new Date(),
   542→        },
   543→      });
   544→      touched++;
   545→    }
   546→    if (touched > 0) {
   547→      await db.engineLog.create({
   548→        data: { kind: "SETTLE", message: `v5.6 REGCOMBO disjoint-cover re-settlement: ${touched} cover legs flipped to WON` },
   549→      });
   550→      console.log(`[settle] v5.6 REGCOMBO cover migration: ${touched} legs flipped to WON`);
   551→    }
   552→  } catch (e) {
   553→    console.error("[settle] REGCOMBO cover migration failed", e);
   554→  }
   555→  return touched;
   556→}

   558→export async function runSettlement(windowDays = 7): Promise<{
   559→  settled: number;
   560→  picked: number;
   561→  windowDays: number;
   562→}> {
   563→  // v4.5 migration — combos settled under the buggy v4.0 rule (0W/11L =
   564→  // −179u) carry no explicit leg side/market in their meta; their stored
   565→  // LOST results were produced by an inverted settlement check and by
   566→  // treating unresolved legs as losses. Those rows are VOIDed once so the
   567→  // official record only contains correctly-contracted slips.
   568→  try {
   569→    await db.tennisPick.updateMany({
   570→      where: {
   571→        line: { in: ["COMBO", "ACCUMULATOR"] },
   572→        status: { not: "VOID" },
   573→        OR: [{ meta: null }, { meta: { not: { contains: '"side"' } } }],
   574→      },
   575→      data: { status: "VOID", profit: 0 },
   576→    });
   577→  } catch (e) {
   578→    console.error("[settle] stale combo migration failed", e);
   579→  }
   580→
   581→  // ── v5.6 history re-settlement (master prompt Fix 1877.1u) ────────────────
   582→  // Idempotent migrations: re-contract naked sniper history into the 25u+75u
   583→  // package and flip REGCOMBO covers the fixed disjoint rule would have won.
   584→  await resettleSniperPackageHistory();
   585→  await resettleRegComboHistory();
   586→
   587→  // 1) live finished results across the settlement window (today + N past days)
   588→  const dateKeys = Array.from({ length: windowDays + 1 }, (_, i) =>
   589→    i === 0 ? todayKey() : dateKeyDaysAgo(i)
   590→  );
   591→  const tours: Tour[] = ["ATP", "WTA"];
   592→  const fetched: EspnMatch[] = [];
   593→  await Promise.all(
   594→    tours.flatMap((t) =>
   595→      dateKeys.map(async (d) => {
   596→        try {
   597→          const ms = await fetchScoreboard(t, d);
   598→          fetched.push(...ms);
   599→        } catch (e) {
   600→          console.error("[settle] scoreboard fetch failed", t, d, e);
   601→        }
   602→      })
   603→    )
   604→  );
   605→
   606→  const resultsByEvent = new Map<string, MatchResult>();
   607→  for (const m of fetched) {
   608→    const r = extractResult(m);
   609→    if (r) resultsByEvent.set(m.eventKey, r);
   610→  }
   611→
   612→  // 2) settle pending picks
   613→  const pendingPicks = await db.tennisPick.findMany({
   614→    where: { status: "PENDING" },
   615→    include: { match: true },
   616→  });
   617→
   618→  let settled = 0;
   619→
   620→  for (const pick of pendingPicks) {
   621→    let meta: PickMeta = {};
   622→    try {
   623→      meta = pick.meta ? (JSON.parse(pick.meta) as PickMeta) : {};
   624→    } catch {
   625→      meta = {};
   626→    }
   627→
   628→    if (pick.line === "COMBO" || pick.line === "ACCUMULATOR") {
   629→      const favorites = meta.favorites ?? [];
   630→      if (!favorites.length) continue;
   631→      // ── v4.5 FT Settlement Fix ────────────────────────────────────────
   632→      // Every leg checks the REAL winner of its own match: the stored
   633→      // favorite must have won (equivalent of competitor.winner === true —
   634→      // sets won > opponent sets) or, for SET HANDICAP +1.5 SETS legs, have
   635→      // won at least ONE set (sets won ≥ 1). An undecidable leg (match not
   636→      // finished / result not extractable / side unresolvable) keeps the
   637→      // whole slip PENDING — it is NEVER auto-destroyed as a loss (the v4.0
   638→      // bug that produced 0W/11L).
   639→      let allDecided = true;
   640→      let allWon = true;
   641→      for (const f of favorites) {
   642→        const ownMatch = fetched.find((x) => x.eventKey === f.eventKey);
   643→        const result = ownMatch ? extractResult(ownMatch) : undefined;
   644→        if (!result) {
   645→          allDecided = false;
   646→          break;
   647→        }
   648→        // explicit side first (v4.5 contract); legacy picks fall back to
   649→        // full-name/surname matching
   650→        const side: "A" | "B" | null = f.side ?? favSide(result.playerAName, result.playerBName, f.name);
   651→        if (!side) {
   652→          allDecided = false;
   653→          break;
   654→        }
   655→        const favSets = side === "A" ? result.setsA : result.setsB;
   656→        const oppSets = side === "A" ? result.setsB : result.setsA;
   657→        const legWon =
   658→          f.market === "HANDICAP"
   659→            ? favSets >= 1
   660→            : f.market === "HANDICAP15"
   661→              ? favSets === 2 // SET HANDICAP −1.5 SETS = the favorite swept 2-0
   662→              : f.market === "OVER195"
   663→                ? result.totalGames >= 20 // v7.0 — TOTAL GAMES OVER 19.5
   664→                : favSets > oppSets;
   665→        if (!legWon) {
   666→          allWon = false;
   667→          break;
   668→        }
   669→      }
   670→      if (!allDecided) continue; // stays PENDING until every leg is decidable
   671→      const profit = allWon ? pick.odds * pick.stake - pick.stake : -pick.stake;
   672→      await db.tennisPick.update({
   673→        where: { id: pick.id },
   674→        data: { status: allWon ? "WON" : "LOST", profit, settledAt: new Date() },
   675→      });
   676→      settled++;
   677→      continue;
   678→    }
   679→
   680→    const result = resultsByEvent.get(pick.match.eventKey);
   681→    if (!result) continue;
   682→
   683→    const outcome = settlePick(
   684→      pick,
   685→      { playerA: pick.match.playerA, playerB: pick.match.playerB },
   686→      result
   687→    );
   688→    if (outcome.status === "VOID") continue;
   689→    await db.tennisPick.update({
   690→      where: { id: pick.id },
   691→      data: { status: outcome.status, profit: outcome.profit, settledAt: new Date() },
   692→    });
   693→    settled++;
   694→  }
   695→
   696→  // 3) stamp match results on the match table
   697→  for (const [eventKey, r] of resultsByEvent.entries()) {
   698→    try {
   699→      await db.tennisMatch.updateMany({
   700→        where: { eventKey, settled: false },
   701→        data: {
   702→          resultSetsA: r.setsA,
   703→          resultSetsB: r.setsB,
   704→          resultCode: r.code,
   705→          settled: true,
   706→          status: "post",
   707→        },
   708→      });
   709→    } catch (e) {
   710→      console.error("[settle] stamp failed", eventKey, e);
   711→    }
   712→  }
   713→
   714→  await db.engineLog.create({
   715→    data: {
   716→      kind: "SETTLE",
   717→      message: `Auto-settlement window=${windowDays}d settled=${settled} pending=${pendingPicks.length} results=${resultsByEvent.size}`,
   718→    },
   719→  });
   720→
   721→  return { settled, picked: pendingPicks.length, windowDays };
   722→}
