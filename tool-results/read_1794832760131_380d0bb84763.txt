    63→    63→type Field = { wildcard: boolean; any: Set<number>; step?: number };
    64→    64→
    65→    65→function parseField(raw: string, min: number, max: number): Field {
    66→    66→  const any = new Set<number>();
    67→    67→  let wildcard = false;
    68→    68→  let step: number | undefined;
    69→    69→  for (const chunk of raw.split(",")) {
    70→    70→    if (chunk === "*") {
    71→    71→      wildcard = true;
    72→    72→      for (let i = min; i <= max; i++) any.add(i);
    73→    73→    } else if (chunk.startsWith("*/")) {
    74→    74→      wildcard = true;
    75→    75→      step = Number(chunk.slice(2));
    76→    76→      for (let i = min; i <= max; i += step) any.add(i);
    77→    77→    } else if (chunk.includes("-")) {
    78→    78→      const [a, b] = chunk.split("-").map(Number);
    79→    79→      for (let i = a; i <= b; i++) any.add(i);
    80→    80→    } else {
    81→    81→      any.add(Number(chunk));
    82→    82→    }
    83→    83→  }
    84→    84→  return { wildcard, any, step };
    85→    85→}

    87→    87→interface Cron {
    88→    88→  sec: Field; min: Field; hour: Field; dom: Field; mon: Field; dow: Field;
    89→    89→}

    91→    91→function parseCron(expr: string): Cron {
    92→    92→  const [sec, min, hour, dom, mon, dow] = expr.trim().split(/\s+/);
    93→    93→  return {
    94→    94→    sec: parseField(sec, 0, 59),
    95→    95→    min: parseField(min, 0, 59),
    96→    96→    hour: parseField(hour, 0, 23),
    97→    97→    dom: parseField(dom, 1, 31),
    98→    98→    mon: parseField(mon, 1, 12),
    99→    99→    dow: parseField(dow, 0, 6),
   100→   100→  };
   101→   101→}

   103→   103→function cronMatches(c: Cron, f: ReturnType<typeof tehranFields>): boolean {
   104→   104→  return (
   105→   105→    c.sec.any.has(f.s) &&
   106→   106→    c.min.any.has(f.min) &&
   107→   107→    c.hour.any.has(f.h) &&
   108→   108→    c.mon.any.has(f.mon) &&
   109→   109→    (c.dom.wildcard || c.dow.wildcard
   110→   110→      ? c.dom.any.has(f.dom) || c.dow.any.has(f.dow)
   111→   111→      : c.dom.any.has(f.dom) && c.dow.any.has(f.dow))
   112→   112→  );
   113→   113→}

   115→   115→/** next run strictly after `after` (ms epoch) for a Tehran-anchored cron */
   116→   116→export function nextCronRun(expr: string, after = Date.now()): number {
   117→   117→  const c = parseCron(expr);
   118→   118→  // iterate the next 2 days at second resolution starting from `after + 1s`
   119→   119→  let t = after + 1000;
   120→   120→  const limit = after + 2 * 86_400_000;
   121→   121→  while (t <= limit) {
   122→   122→    if (cronMatches(c, tehranFields(t))) return t;
   123→   123→    // jump smartly: skip to next minute boundary when seconds won't match soon
   124→   124→    const f = tehranFields(t);
   125→   125→    if (!c.hour.any.has(f.h) || !c.min.any.has(f.min)) {
   126→   126→      t += (59 - f.s) * 1000 + 1000; // finish current minute
   127→   127→      if (!c.hour.any.has(f.h)) t += (59 - f.min) * 60_000; // finish current hour
   128→   128→    } else {
   129→   129→      t += 1000;
   130→   130→    }
   131→   131→  }
   132→   132→  return after + 60_000;
   133→   133→}

   633→   633→    // ONE-POST POLICY: a match is eligible only if it has NEVER been posted

   652→   652→    // (network / auth / 404) does NOT burn the card — the next 90s tick

   784→   784→async function runJob(jobId: JobId, reason: "cron" | "manual" = "cron"): Promise<PostOutcome> {
   785→   785→  if (running.has(jobId)) return { status: "SKIPPED", error: "already running" };
   786→   786→  running.add(jobId);
   787→   787→  try {
   788→   788→    const out = await handlers[jobId]();
   789→   789→    store.recordRun(jobId, out.status, { error: out.error, messageId: out.messageId });
   790→   790→    return out;
   791→   791→  } catch (e) {
   792→   792→    const error = e instanceof Error ? e.message : String(e);
   793→   793→    console.error(`[job:${jobId}] ${reason} error:`, error);
   794→   794→    store.recordRun(jobId, "ERROR", { error });
   795→   795→    return { status: "ERROR", error };
   796→   796→  } finally {
   797→   797→    running.delete(jobId);
   798→   798→  }
   799→   799→}

   801→   801→function scheduleAll(): void {
   802→   802→  for (const job of JOB_SPECS) {
   803→   803→    if (job.cron) {
   804→   804→      nextRunAt.set(job.id, nextCronRun(job.cron));
   805→   805→    } else if (job.everySeconds) {
   806→   806→      nextRunAt.set(job.id, Date.now() + (job.everySeconds * 1000));
   807→   807→    }
   808→   808→  }
   809→   809→}
   810→   810→
   811→   811→function tick(): void {
   812→   812→  const now = Date.now();
   813→   813→  for (const job of JOB_SPECS) {
   814→   814→    const next = nextRunAt.get(job.id);
   815→   815→    if (next === undefined || now < next) continue;
   816→   816→    if (job.cron) {
   817→   817→      nextRunAt.set(job.id, nextCronRun(job.cron, now));
   818→   818→    } else if (job.everySeconds) {
   819→   819→      nextRunAt.set(job.id, now + job.everySeconds * 1000);
   820→   820→    }
   821→   821→    void runJob(job.id);
   822→   822→  }
   823→   823→}

   858→   858→/** capture the VIP channel id from any chat the bot can see */
   859→   859→async function discoverVipLoop(): Promise<void> {
   860→   860→  try {
   861→   861→    const chats = await telegram.discoverChats();
   862→   862→    for (const ch of chats) {
   863→   863→      const isPublic = !!ch.username && CONFIG.publicChatId.includes(ch.username);
   864→   864→      const isChannelish = ch.type === "channel" || ch.type === "supergroup";
   865→   865→      if (isChannelish && ch.id) {
   866→   866→        store.kvSet(`chatSeen:${ch.id}`, JSON.stringify(ch));
   867→   867→      }
   868→   868→      if (isChannelish && !isPublic && ch.id) {
   869→   869→        const idStr = String(ch.id);
   870→   870→        if (vipChatId() !== idStr) {
   871→   871→          store.kvSet("vipChatId", idStr);
   872→   872→          store.kvSet("vipChatTitle", ch.title ?? "");
   873→   873→          console.log(`[discover] VIP chat captured: ${idStr} (${ch.title ?? ch.type})`);
   874→   874→        }
   875→   875→      }
   876→   876→    }
   877→   877→  } catch {
   878→   878→    /* discovery is best-effort */
   879→   879→  }
   880→   880→}
